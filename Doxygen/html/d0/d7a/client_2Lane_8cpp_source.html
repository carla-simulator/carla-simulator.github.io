22 // Stepping away from the centre line: -1 -> -2, 1 -> 2.
23 static road::LaneId StepOutward(road::LaneId lane_id) {
24 return lane_id > 0 ? lane_id + 1 : lane_id - 1;
25 }
26
27 // Stepping toward the centre line. At +-1 the next lane across is on the
28 // other side of the road, so the sign flips and the neighbour runs the
29 // opposite way. road::Map::GetLeft does exactly this; kept identical here on
30 // purpose so Lane and Waypoint never disagree about who is to the left.
31 static road::LaneId StepInward(road::LaneId lane_id) {
32 if (std::abs(lane_id) == 1) {
33 return -lane_id;
34 }
35 return lane_id > 0 ? lane_id - 1 : lane_id + 1;
36 }

42 uint64_t Lane::GetUniqueId() const {
43 // road id and section id are unsigned, lane id is signed and small; pack
44 // them so the value is stable and collision-free within one map.
45 return (static_cast<uint64_t>(_road_id) << 32) ^
46 (static_cast<uint64_t>(_section_id) << 16) ^
47 static_cast<uint64_t>(static_cast<uint16_t>(_lane_id));
48 }

90 SharedPtr<Lane> Lane::GetRightLane() const {
91 if (_lane_id == 0) {
92 return nullptr;
93 }
94 const bool is_rht = GetLane().GetRoad()->IsRHT();
95 const road::LaneId neighbour = is_rht ? StepOutward(_lane_id) : StepInward(_lane_id);
96 return _parent->GetLane(_road_id, _section_id, neighbour);
97 }

99 SharedPtr<Lane> Lane::GetLeftLane() const {
100 if (_lane_id == 0) {
101 return nullptr;
102 }
103 const bool is_rht = GetLane().GetRoad()->IsRHT();
104 const road::LaneId neighbour = is_rht ? StepInward(_lane_id) : StepOutward(_lane_id);
105 return _parent->GetLane(_road_id, _section_id, neighbour);

132 SharedPtr<Waypoint> Lane::GetWaypoint(double s) const {
133 const road::Lane &lane = GetLane();
134 const double start = lane.GetDistance();
135 const double end = start + lane.GetLength();
136 if (s < start || s > end) {
137 return nullptr;
138 }
139 return _parent->MakeWaypoint(
140 road::element::Waypoint{_road_id, _section_id, _lane_id, s});
141 }

143 std::vector<SharedPtr<Waypoint>> Lane::GetWaypoints(double distance) const {
144 std::vector<SharedPtr<Waypoint>> result;
145 if (distance <= 0.0) {
146 return result;
147 }
148 const road::Lane &lane = GetLane();
149 const double start = lane.GetDistance();
150 const double end = start + lane.GetLength();
151 for (double s = start; s < end; s += distance) {
152 result.emplace_back(_parent->MakeWaypoint(
153 road::element::Waypoint{_road_id, _section_id, _lane_id, s}));
154 }
155 return result;
156 }

25 /// Euler angles of a rotation in **CARLA's / Unreal's left-handed frame**:
26 /// x forward, y right, z up, degrees.
27 ///
28 /// The three angles compose as
29 /// `R = Rz(+yaw) * Ry(-pitch) * Rx(-roll)` with the standard
30 /// (right-handed) elementary matrices, which is the same thing as saying:
31 ///
32 /// * `+yaw` turns the nose to the **right**;
33 /// * `+pitch` tilts the nose **up** -> `forward.z = +sin(pitch)`;
34 /// * `+roll` drops the **right** side -> `right.z = -cos(pitch)*sin(roll)`.
35 ///
36 /// This is CARLA 0.9.x's documented convention and it is what the engine
37 /// builds from the same `FRotator`, verified against rendered camera frames
38 /// on UE5.8. `RotateVector`, `Transform::GetMatrix` and
39 /// `Math::Get{Forward,Right,Up}Vector` all live in this frame and must not
40 /// be "corrected" towards a right-handed convention.
41 ///
42 /// For ROS / REP-103 consumers convert explicitly at the boundary with
43 /// `ToRightHanded()` / `FromRightHanded()`; see `RightHandedRotation` and
44 /// `Docs/coordinate_conventions.md`.

87 void RotateVector(Vector3D &in_point) const {
88 // Rotates Rz(yaw) * Ry(pitch) * Rx(roll) = first x, then y, then z,
89 // in Unreal's left-handed frame, matching what the engine does with
90 // the same FRotator and CARLA's documented 0.9.x semantics:
91 // +pitch tilts the forward axis UP -> z row has (+sp)
92 // +roll tilts the right axis DOWN -> z row has (-cp * sr)
93 // Empirically verified on the running UE5.8 server (2026-08-27): a
94 // camera spawned at Rotation(pitch=+20) fills the frame with sky and
95 // at Rotation(roll=+25) shows more ground on the right-hand side.
96 // #9751 briefly mirrored the pitch/roll signs here; that made this
97 // matrix disagree with the engine, with the ROS 2 bridge's own
98 // `ros2::TransformFromCarlaRotation`, and with every 0.9.x consumer.
99 const float cy = std::cos(Math::ToRadians(yaw));
100 const float sy = std::sin(Math::ToRadians(yaw));
101 const float cr = std::cos(Math::ToRadians(roll));
102 const float sr = std::sin(Math::ToRadians(roll));
103 const float cp = std::cos(Math::ToRadians(pitch));
104 const float sp = std::sin(Math::ToRadians(pitch));
105
106 Vector3D out_point;
107 out_point.x =
108 in_point.x * (cp * cy) +
109 in_point.y * (cy * sp * sr - sy * cr) +
110 in_point.z * (-cy * sp * cr - sy * sr);
111
112 out_point.y =
113 in_point.x * (cp * sy) +
114 in_point.y * (sy * sp * sr + cy * cr) +
115 in_point.z * (-sy * sp * cr + cy * sr);
116
117 out_point.z =
118 in_point.x * (sp) +
119 in_point.y * (-cp * sr) +
120 in_point.z * (cp * cr);
121
122 in_point = out_point;
123 }
124
125 Vector3D RotateVector(const Vector3D& in_point) const {
126 Vector3D out_point = in_point;
127 RotateVector(out_point);
128 return out_point;
129 }

131 void InverseRotateVector(Vector3D &in_point) const {
132 // Applies the transposed of the matrix used in RotateVector function,
133 // which is the rotation inverse.
134 const float cy = std::cos(Math::ToRadians(yaw));
135 const float sy = std::sin(Math::ToRadians(yaw));
136 const float cr = std::cos(Math::ToRadians(roll));
137 const float sr = std::sin(Math::ToRadians(roll));
138 const float cp = std::cos(Math::ToRadians(pitch));
139 const float sp = std::sin(Math::ToRadians(pitch));
140
141 Vector3D out_point;
142 out_point.x =
143 in_point.x * (cp * cy) +
144 in_point.y * (cp * sy) +
145 in_point.z * (sp);
146
147 out_point.y =
148 in_point.x * (cy * sp * sr - sy * cr) +
149 in_point.y * (sy * sp * sr + cy * cr) +
150 in_point.z * (-cp * sr);
151
152 out_point.z =
153 in_point.x * (-cy * sp * cr - sy * sr) +
154 in_point.y * (-sy * sp * cr + cy * sr) +
155 in_point.z * (cp * cr);
156
157 in_point = out_point;
158 }

164 Rotation Normalize() const {
165 Rotation V0 = Rotation(*this);
166
167 V0.pitch = V0.pitch - (360.0f * trunc(V0.pitch/360.0f));
168 V0.yaw = V0.yaw - (360.0f * trunc(V0.yaw/360.0f));
169 V0.roll = V0.roll - (360.0f * trunc(V0.roll/360.0f));
170
171 if( V0.pitch < 0.0f ){
172 V0.pitch += 360.0f;
173 }
174 if( V0.yaw < 0.0f ){
175 V0.yaw += 360.0f;
176 }
177 if( V0.roll < 0.0f ){
178 V0.roll += 360.0f;
179 }
180
181 if( V0.pitch >= 180.0f ){
182 V0.pitch -= 360.0f;
183 }
184 if( V0.yaw >= 180.0f ){
185 V0.yaw -= 360.0f;
186 }
187 if( V0.roll >= 180.0f ){
188 V0.roll -= 360.0f;
189 }
190
191 return V0;
192 }

198 /// This rotation expressed in a right-handed, x-forward / y-**left** /
199 /// z-up frame (ROS REP-103 "FLU"), as intrinsic Z-Y-X Euler angles under
200 /// the right-hand rule.
201 ///
202 /// Mirroring the Y axis maps `Rz(+yaw) * Ry(-pitch) * Rx(-roll)` onto
203 /// `Rz(-yaw) * Ry(+(-pitch)) * Rx(+roll)`, so pitch and yaw flip sign and
204 /// roll does not. Example: `Rotation(pitch=20, yaw=30, roll=10)` becomes
205 /// `RightHandedRotation(roll=10, pitch=-20, yaw=-30)`.
206 constexpr RightHandedRotation ToRightHanded() const {
207 return RightHandedRotation(roll, -pitch, -yaw);
208 }
209
210 /// Inverse of `ToRightHanded()`. The mapping is an involution, so the same
211 /// three sign flips apply in both directions.
212 static constexpr Rotation FromRightHanded(const RightHandedRotation &rhs) {
213 return Rotation(-rhs.pitch, -rhs.yaw, rhs.roll);
214 }

37 /// Runs @a task on a detached thread so that a reader that spins fails the
38 /// test instead of hanging the whole suite.
39 template <typename F>
40 auto RunWithWatchdog(F task) -> std::optional<decltype(task())> {
41 using Result = decltype(task());
42 auto done = std::make_shared<std::promise<Result>>();
43 auto future = done->get_future();
44 std::thread([done, task = std::move(task)]() mutable {
45 try {
46 done->set_value(task());
47 } catch (...) {
48 done->set_exception(std::current_exception());
49 }
50 }).detach();
51 if (future.wait_for(std::chrono::seconds(5)) != std::future_status::ready) {
52 return std::nullopt;
53 }
54 return future.get();
55 }

72 struct Parsed {
73 FrameDataReadResult result;
74 std::vector<uint32_t> values;
75 std::optional<FrameTime> time;
76 };

78 /// Parses @a bytes the way FFrameData does: value packets and FrameTime are
79 /// known, every other id is not.
80 Parsed Parse(std::string bytes) {
81 auto parsed = RunWithWatchdog([bytes = std::move(bytes)]() {
82 Parsed out;
83 out.result = ReadFrameData(
84 reinterpret_cast<const uint8_t *>(bytes.data()), bytes.size(),
85 [&](uint8_t id, std::istream &in, uint32_t size) {
86 switch (id) {
87 case kValuePacketId: {
88 uint32_t value = 0u;
89 in.read(reinterpret_cast<char *>(&value), sizeof(value));
90 if (!in.fail()) {
91 out.values.push_back(value);
92 }
93 return true;
94 }
95 case FrameTime::kPacketId: {
96 FrameTime time;
97 if (time.Read(in, size)) {
98 out.time = time;
99 }
100 return true;
101 }
102 default:
103 return false;
104 }
105 });
106 return out;
107 });
108 EXPECT_TRUE(parsed.has_value()) << "the frame reader did not terminate";
109 return parsed.value_or(Parsed{});
110 }

253TEST(MultiGpuFrameData, nested_counts_read_off_the_wire_terminate) {
254 // A recorder-style reader: an outer count of 65535 records, each with an
255 // inner count, over a 3-byte payload. Loops of the form (i < Total) && In
256 // must end as soon as the payload runs out.
257 std::ostringstream out;
258 WriteFrameDataHeader(out, kValuePacketId, 3u);
259 out.write("\xFF\xFF\x07", 3);
260 PutFrameCounter(out, 1u);
261 const std::string bytes = out.str();
262 const auto result = RunWithWatchdog([bytes]() {
263 return ReadFrameData(
264 reinterpret_cast<const uint8_t *>(bytes.data()), bytes.size(),
265 [](uint8_t, std::istream &in, uint32_t) {
266 uint16_t total = 0u;
267 in.read(reinterpret_cast<char *>(&total), sizeof(total));
268 for (uint16_t i = 0u; (i < total) && in; ++i) {
269 uint32_t inner = 0u;
270 in.read(reinterpret_cast<char *>(&inner), sizeof(inner));
271 for (uint32_t j = 0u; (j < inner) && in; ++j) {
272 uint64_t value = 0u;
273 in.read(reinterpret_cast<char *>(&value), sizeof(value));
274 }
275 }
276 return true;
277 });
278 });
279 ASSERT_TRUE(result.has_value()) << "the nested reader did not terminate";
280 EXPECT_EQ(result->status, FrameDataStatus::ShortPacket);
281}

483 /// Feeds @a message to SecondaryCommands::process_command and records what
484 /// reaches the engine's callback.
485 Delivered Process(const std::string &message) {
486 Delivered delivered;
487 carla::multigpu::SecondaryCommands commands;
488 commands.set_callback([&](carla::multigpu::MultiGPUCommand id, carla::Buffer data) {
489 delivered.ids.push_back(id);
490 delivered.payloads.emplace_back(reinterpret_cast<const char *>(data.data()), data.size());
491 });
492 commands.process_command(carla::Buffer(
493 reinterpret_cast<const unsigned char *>(message.data()), message.size()));
494 return delivered;
495 }

497 std::string Command(carla::multigpu::MultiGPUCommand id, uint32_t declared, const std::string &payload) {
498 carla::multigpu::CommandHeader header{id, declared};
499 std::string message(reinterpret_cast<const char *>(&header), sizeof(header));
500 return message + payload;
501 }

70/// Retrieve the definitions of all the sensors registered in the
71/// SensorRegistry by calling their static method
72/// SensorType::GetSensorDefinition().
73///
74/// @note To make this class ignore a given sensor, define a public member
75/// "not_spawnable" that defines a type. If so, the sensor won't be spawned by
76/// this factory.
77class FSensorDefinitionGatherer
78{
79 using Registry = carla::sensor::SensorRegistry;
80
81public:
82
83 static auto GetSensorDefinitions()
84 {
85 TArray<FActorDefinition> Definitions;
86 Definitions.Reserve(Registry::size());
87 AppendDefinitions(Definitions, std::make_index_sequence<Registry::size()>());
88 return Definitions;
89 }
90
91private:
92
93 // Type traits for detecting if a sensor is spawnable.
94
95 template<typename T>
96 struct void_type { typedef void type; };
97
98 template<typename T, typename = void>
99 struct is_spawnable : std::true_type {};
100
101 template<typename T>
102 struct is_spawnable<T, typename void_type<typename T::not_spawnable>::type > : std::false_type {};
103
104 // AppendDefinitions implementations.
105
106 template <typename SensorType>
107 static typename std::enable_if<is_spawnable<SensorType>::value, void>::type
108 AppendDefinitions(TArray<FActorDefinition> &Definitions)
109 {
110 auto Def = SensorType::GetSensorDefinition();
111 // Make sure the class matches the sensor type.
112 Def.Class = SensorType::StaticClass();
113 Definitions.Emplace(Def);
114 }
115
116 template <typename SensorType>
117 static typename std::enable_if<!is_spawnable<SensorType>::value, void>::type
118 AppendDefinitions(TArray<FActorDefinition> &) {}
119
120 template <size_t Index>
121 static void AppendDefinitions(TArray<FActorDefinition> &Definitions)
122 {
123 using SensorPtrType = typename Registry::get_by_index<Index>::key;
124 using SensorType = typename std::remove_pointer<SensorPtrType>::type;
125 AppendDefinitions<SensorType>(Definitions);
126 }
127
128 template <size_t... Is>
129 static void AppendDefinitions(
130 TArray<FActorDefinition> &Definitions,
131 std::index_sequence<Is...>)
132 {
133 std::initializer_list<int> ({(AppendDefinitions<Is>(Definitions), 0)...});
134 }
135};

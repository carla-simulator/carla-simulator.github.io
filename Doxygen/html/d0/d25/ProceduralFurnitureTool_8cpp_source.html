260bool AProceduralFurnitureTool::Bake()
261{
262 if (IsGenerating() || !bPreviewReady || PreviewInstanceCount() != Points.Num())
263 { Status = TEXT("Generate a complete preview before baking."); return false; }
264 Modify();
265 // Upgrade existing tool actors too: static instances cannot attach to a
266 // movable root, and must follow the region actor's transform after baking.
267 RootComponent->Modify();
268 RootComponent->SetMobility(EComponentMobility::Static);
269 TArray<UInstancedStaticMeshComponent*> PreviewComponents;
270 GetComponents(PreviewComponents);
271 TArray<TObjectPtr<UInstancedStaticMeshComponent>> NewBaked;
272 int32 Count = 0;
273 for (auto* Source : PreviewComponents)
274 {
275 if (BakedComponents.Contains(Source) || Source->GetInstanceCount() == 0) continue;
276 auto* Dest = NewObject<UInstancedStaticMeshComponent>(this, NAME_None, RF_Transactional);
277 Dest->SetStaticMesh(Source->GetStaticMesh());
278 Dest->SetMobility(EComponentMobility::Static);
279 Dest->SetupAttachment(RootComponent);
280 Dest->SetRelativeTransform(FTransform::Identity);
281 Dest->SetCollisionProfileName(TEXT("BlockAll"));
282 Dest->SetGenerateOverlapEvents(false);
283 Dest->SetCanEverAffectNavigation(true);
284 for (int32 M=0; M<Source->GetNumMaterials(); ++M) Dest->SetMaterial(M, Source->GetMaterial(M));
285 AddInstanceComponent(Dest);
286 Dest->RegisterComponent();
287 for (int32 I=0; I<Source->GetInstanceCount(); ++I)
288 {
289 FTransform Transform;
290 Source->GetInstanceTransform(I, Transform, true);
291 Dest->AddInstance(Transform, true);
292 ++Count;
293 }
294 NewBaked.Add(Dest);
295 }
296 for (UInstancedStaticMeshComponent* Old : BakedComponents) if (Old) { RemoveInstanceComponent(Old); Old->DestroyComponent(); }
297 BakedComponents = MoveTemp(NewBaked);
298 BakedInstanceCount = Count;
299 PCG->CleanupLocalImmediate(true);
300 bPreviewReady = false;
301 ATagger::TagActor(*this, true);
302 MarkPackageDirty();
303 Status = FString::Printf(TEXT("Baked %d furniture instances. Save the level."), Count);
304 return Count == Points.Num();
305}

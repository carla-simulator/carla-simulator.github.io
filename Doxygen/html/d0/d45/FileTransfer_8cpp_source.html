17 static fs::path CachePath = [] {
18 constexpr char OverridePathEV[] = "CARLA_CACHE_DIR";
19 constexpr char HomePathEV[] =
20#ifdef _WIN32
21 "USERPROFILE";
22#else
23 "HOME";
24#endif
25 const char* override_path = std::getenv(OverridePathEV);
26 if (override_path != NULL)
27 return fs::path(override_path);
28 fs::path path = std::getenv(HomePathEV);
29 path /= "carlaCache";

33 // The path arriving here comes from the server, so it must never be trusted
34 // to stay inside the cache. Normalize it and drop any root, drive or leading
35 // ".." components: a hostile or buggy server (e.g. one sending
36 // "../../../CarlaUnreal/Content/...") can then neither write nor read
37 // outside the cache root, while legitimate nested paths such as
38 // "Config/Town10HD_Opt/Vehicles.xml" resolve unchanged. Escaping paths are
39 // remapped instead of rejected so that both WriteFile and ReadFile agree on
40 // the same in-cache location.
41 static fs::path SanitizeRelativePath(std::string_view file) {
42 auto normalized = fs::path(file).lexically_normal().relative_path();
43 fs::path clean;
44 for (const auto &part : normalized)
45 if (part != ".." && part != ".")
46 clean /= part;
47 return clean;
48 }

68 bool FileTransfer::WriteFile(std::string_view path, std::vector<uint8_t> content) {
69 auto writePath = CachePath;
70 writePath /= carla::version();
71 writePath /= SanitizeRelativePath(path);
72
73 // Validate and create the file path
74 {
75 auto parent = writePath.parent_path();
76 if (fs::exists(parent))
77 {
78 if (!fs::is_directory(parent))
79 return false;
80 }
81 else
82 {
83 if (!fs::create_directories(parent))
84 return false;
85 }
86 }
87
88 // Open the file to truncate it in binary mode
89 std::ofstream out(
90 writePath,
91 std::ios::trunc | std::ios::binary);
92
93 if (!out.good())
94 return false;
95
96 // Write the content on and close it
97 for (auto file : content) {
98 out << file;
99 }
100 out.close();
101
102 return true;
103 }

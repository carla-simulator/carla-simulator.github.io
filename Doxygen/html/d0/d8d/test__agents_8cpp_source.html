106TEST(agents_misc, unit_vector_is_normalized) {
107 Location a(0.0f, 0.0f, 0.0f);
108 Location b(3.0f, 4.0f, 0.0f);
109 const auto u = UnitVector(a, b);
110 // 3-4-5 triangle: expect (0.6, 0.8, 0.0).

113 EXPECT_NEAR(u[2], 0.0f, 1e-3f);
114 // Magnitude is unit.

128TEST(agents_misc, is_within_distance_too_close_returns_true) {

131 // distance/angle (avoids divide-by-zero on the angle calculation).
132 Transform ref(Location(10, 10, 0), Rotation(0, 90, 0));

135}

137TEST(agents_misc, is_within_distance_with_forward_arc_filter) {
138 // Reference looks down +x. Python uses a strict open interval
139 // (min < angle < max), so a target exactly on the forward axis (angle=0°)
140 // is excluded from the (0°, 90°) cone — matched here by placing the
141 // target diagonally forward-right at 45°.
142 Transform ref(Location(0, 0, 0), Rotation(0, 0, 0));
143 Transform front_diag(Location(5, 5, 0), Rotation(0, 0, 0)); // 45° from +x
144 Transform behind(Location(-5, 0, 0), Rotation(0, 0, 0)); // 180° (behind)
145
146 EXPECT_TRUE (IsWithinDistance(front_diag, ref, 10.0f, std::make_pair(0.0f, 90.0f)));
147 EXPECT_FALSE(IsWithinDistance(behind, ref, 10.0f, std::make_pair(0.0f, 90.0f)));
148
149 // Strict-open boundary: target exactly on the forward axis (angle=0°) is
150 // NOT inside the (0°, 90°) interval. Matches Python parity.
151 Transform front_axial(Location(5, 0, 0), Rotation(0, 0, 0));
152 EXPECT_FALSE(IsWithinDistance(front_axial, ref, 10.0f, std::make_pair(0.0f, 90.0f)));
153}

233TEST(agents_api, surface_pins) {
234 using PlanItem = LocalPlanner::PlanItem;
235 static_assert(std::is_same_v<PlanItem,
236 std::pair<carla::SharedPtr<carla::client::Waypoint>, RoadOption>>,
237 "LocalPlanner::PlanItem must be (Waypoint, RoadOption)");
238
239 using RouteItem = GlobalRoutePlanner::RouteItem;
240 static_assert(std::is_same_v<RouteItem,
241 std::pair<carla::SharedPtr<carla::client::Waypoint>, RoadOption>>,
242 "GlobalRoutePlanner::RouteItem must be (Waypoint, RoadOption)");
243
244 using Behavior = BehaviorAgent::Behavior;
245 static_assert(static_cast<int>(Behavior::Cautious) != static_cast<int>(Behavior::Normal),
246 "BehaviorAgent::Behavior values must be distinct");
247 static_assert(static_cast<int>(Behavior::Normal) != static_cast<int>(Behavior::Aggressive),
248 "BehaviorAgent::Behavior values must be distinct");
249
250 // BasicAgent::Options derives from LocalPlanner::Options.
251 static_assert(std::is_base_of_v<LocalPlanner::Options, BasicAgent::Options>,
252 "BasicAgent::Options must inherit LocalPlanner::Options");
253 // ConstantOptions derives from BasicAgent::Options.
254 static_assert(std::is_base_of_v<BasicAgent::Options,
255 ConstantVelocityAgent::ConstantOptions>,
256 "ConstantOptions must inherit BasicAgent::Options");
257
258 // BehaviorAgent / ConstantVelocityAgent inherit BasicAgent.
259 static_assert(std::is_base_of_v<BasicAgent, BehaviorAgent>,
260 "BehaviorAgent must inherit BasicAgent");
261 static_assert(std::is_base_of_v<BasicAgent, ConstantVelocityAgent>,
262 "ConstantVelocityAgent must inherit BasicAgent");
263
264 // BasicAgent has a virtual destructor (so subclasses are safely deletable
265 // through a base pointer).
266 static_assert(std::has_virtual_destructor_v<BasicAgent>,
267 "BasicAgent must have a virtual destructor");
268
269 SUCCEED();

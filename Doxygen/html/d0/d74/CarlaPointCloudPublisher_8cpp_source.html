27// Translates the FastDDS-free PointFieldDataType enum (defined in
28// PointCloudFieldsLayout.h) to the carla::ros2::msg::PointField::* numeric
29// constants the message struct expects on the wire.
30std::uint8_t ToPointFieldDatatype(PointFieldDataType datatype) noexcept {
31 switch (datatype) {
32 case PointFieldDataType::Int8: return msg::PointField::INT8;
33 case PointFieldDataType::UInt8: return msg::PointField::UINT8;
34 case PointFieldDataType::Int16: return msg::PointField::INT16;
35 case PointFieldDataType::UInt16: return msg::PointField::UINT16;
36 case PointFieldDataType::Int32: return msg::PointField::INT32;
37 case PointFieldDataType::UInt32: return msg::PointField::UINT32;
38 case PointFieldDataType::Float32: return msg::PointField::FLOAT32;
39 case PointFieldDataType::Float64: return msg::PointField::FLOAT64;
40 }
41 return msg::PointField::FLOAT32;
42}

44std::vector<msg::PointField> BuildPointFields(
45 const PointFieldDescriptor *descriptors, std::size_t count) {
46 std::vector<msg::PointField> fields;
47 fields.reserve(count);
48 for (std::size_t i = 0; i < count; ++i) {
49 const auto &descriptor = descriptors[i];
50 msg::PointField field{};
51 field.name = std::string{descriptor.name};
52 field.offset = descriptor.offset;
53 field.datatype = ToPointFieldDatatype(descriptor.datatype);
54 field.count = descriptor.count;
55 fields.push_back(std::move(field));
56 }
57 return fields;
58}

88bool CarlaPointCloudPublisher::WritePointCloud(
89 std::int32_t seconds,
90 std::uint32_t nanoseconds,
91 std::uint32_t height,
92 std::uint32_t width,
93 std::vector<std::uint8_t> data) {
94 auto *message = _impl->GetMessage();
95 message->header.stamp.sec = seconds;
96 message->header.stamp.nanosec = nanoseconds;
97 message->header.frame_id = GetFrameId();
98
99 const std::size_t point_size = GetPointSize();
100
101 message->width = width;
102 message->height = height;
103 message->is_bigendian = false;
104 message->fields = BuildPointFields(GetFieldDescriptors(), GetFieldDescriptorCount());
105 message->point_step = static_cast<std::uint32_t>(point_size);
106 message->row_step = static_cast<std::uint32_t>(width * point_size);
107 // CARLA's point-cloud sensors emit one point per actual return and never
108 // NaN/Inf placeholders, so every published cloud is dense.
109 message->is_dense = true;
110 message->data = std::move(data);
111
112 return true;
113}

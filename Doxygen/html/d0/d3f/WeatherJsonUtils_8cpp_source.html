17 TSharedRef<FJsonObject> LoadJsonRoot(const FString& Path)
18 {
19 FString InputString;
20 TSharedPtr<FJsonObject> Root;
21 if (FFileHelper::LoadFileToString(InputString, *Path))
22 {
23 TSharedRef<TJsonReader<>> Reader = TJsonReaderFactory<>::Create(InputString);
24 FJsonSerializer::Deserialize(Reader, Root);
25 }
26 return Root.IsValid() ? Root.ToSharedRef() : MakeShared<FJsonObject>();
27 }

29 bool SaveJsonRoot(const TSharedRef<FJsonObject>& Root, const FString& Path)
30 {
31 FString OutputString;
32 TSharedRef<TJsonWriter<>> Writer = TJsonWriterFactory<>::Create(&OutputString);
33 if (!FJsonSerializer::Serialize(Root, Writer))
34 return false;
35 IFileManager::Get().MakeDirectory(*FPaths::GetPath(Path), true);
36 return FFileHelper::SaveStringToFile(OutputString, *Path);

41 TSharedRef<FJsonObject> GetOrCreateCatalog(const TSharedRef<FJsonObject>& Root, const TCHAR* CatalogKey)
42 {
43 const TSharedPtr<FJsonObject>* Existing = nullptr;
44 if (Root->TryGetObjectField(CatalogKey, Existing) && Existing->IsValid())
45 return Existing->ToSharedRef();
46 TSharedRef<FJsonObject> Catalog = MakeShared<FJsonObject>();
47 Root->SetObjectField(CatalogKey, Catalog);
48 return Catalog;
49 }

51 template <typename TPresetStruct>
52 bool SavePreset(const TPresetStruct& Preset, const FString& Name, const TCHAR* CatalogKey)
53 {
54 TSharedRef<FJsonObject> Root = LoadJsonRoot(UWeatherJsonUtils::GetPresetsPath());
55 TSharedRef<FJsonObject> Catalog = GetOrCreateCatalog(Root, CatalogKey);
56
57 TSharedPtr<FJsonObject> PresetObject = FJsonObjectConverter::UStructToJsonObject(Preset);
58 if (!PresetObject.IsValid())
59 return false;
60 Catalog->SetObjectField(Name, PresetObject);
61
62 return SaveJsonRoot(Root, UWeatherJsonUtils::GetPresetsPath());
63 }

65 template <typename TPresetStruct>
66 bool LoadPreset(TPresetStruct& OutPreset, const FString& Name, const TCHAR* CatalogKey)
67 {
68 TSharedRef<FJsonObject> Root = LoadJsonRoot(UWeatherJsonUtils::GetPresetsPath());
69 const TSharedPtr<FJsonObject>* Catalog = nullptr;
70 if (!Root->TryGetObjectField(CatalogKey, Catalog) || !Catalog->IsValid())
71 return false;
72
73 const TSharedPtr<FJsonObject>* PresetObject = nullptr;
74 if (!(*Catalog)->TryGetObjectField(Name, PresetObject) || !PresetObject->IsValid())
75 return false;
76
77 return FJsonObjectConverter::JsonObjectToUStruct(PresetObject->ToSharedRef(), &OutPreset);
78 }

80 TArray<FString> GetCatalogNames(const TCHAR* CatalogKey)
81 {
82 TArray<FString> Names;
83 TSharedRef<FJsonObject> Root = LoadJsonRoot(UWeatherJsonUtils::GetPresetsPath());
84 const TSharedPtr<FJsonObject>* Catalog = nullptr;
85 if (Root->TryGetObjectField(CatalogKey, Catalog) && Catalog->IsValid())
86 {
87 for (const auto& Pair : (*Catalog)->Values)
88 Names.Add(FString(Pair.Key));
89 }
90 Names.Sort();
91 return Names;

173bool UWeatherJsonUtils::GetMapDefaultWeather(const FString& MapName, const FWeatherParameters& Base, FWeatherParameters& OutWeather)
174{
175 TSharedRef<FJsonObject> Root = LoadJsonRoot(GetMapDefaultsPath());
176 const TSharedPtr<FJsonObject>* MapEntry = nullptr;
177 if (!Root->TryGetObjectField(MapName, MapEntry) || !MapEntry->IsValid())
178 return false;
179
180 FString TimeName, ConditionName;
181 const bool bIsPresetPair = (*MapEntry)->TryGetStringField(TEXT("time"), TimeName)
182 && (*MapEntry)->TryGetStringField(TEXT("condition"), ConditionName);
183 if (bIsPresetPair)
184 {
185 FWeatherTimePreset TimePreset;
186 FWeatherConditionPreset ConditionPreset;
187 if (!LoadTimePreset(TimePreset, TimeName) || !LoadConditionPreset(ConditionPreset, ConditionName))
188 return false;
189 OutWeather = MergeWeatherPresets(Base, TimePreset, ConditionPreset);
190 return true;
191 }
192
193 // Not a {time, condition} pair -- the entry itself is a full inline
194 // FWeatherParameters snapshot (see SetMapDefaultWeather). Start from Base
195 // so a future field this snapshot predates keeps Base's value instead of
196 // silently zeroing.
197 OutWeather = Base;
198 return FJsonObjectConverter::JsonObjectToUStruct((*MapEntry).ToSharedRef(), &OutWeather);
199}

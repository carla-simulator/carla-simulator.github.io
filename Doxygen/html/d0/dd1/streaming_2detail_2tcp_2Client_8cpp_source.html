87 void Client::Connect() {
88 auto self = shared_from_this();
89 if (_done) {
90 return;
91 }
92
93 using boost::system::error_code;
94
95 if (_socket.is_open()) {
96 _socket.close();
97 }
98
99 DEBUG_ASSERT(_token.is_valid());
100 DEBUG_ASSERT(_token.protocol_is_tcp());
101 const auto ep = _token.to_tcp_endpoint();
102
103 auto handle_connect = [this, self, ep](error_code ec) {
104 if (!ec) {
105 if (_done) {
106 return;
107 }
108 // This forces not using Nagle's algorithm.
109 // Improves the sync mode velocity on Linux by a factor of ~3.
110 _socket.set_option(boost::asio::ip::tcp::no_delay(true));
111 log_debug("streaming client: connected to", ep);
112 // Send the stream id to subscribe to the stream.
113 const auto &stream_id = _token.get_stream_id();
114 log_debug("streaming client: sending stream id", stream_id);
115 boost::asio::async_write(
116 _socket,
117 boost::asio::buffer(&stream_id, sizeof(stream_id)),
118 boost::asio::bind_executor(_strand, [=](error_code ec, size_t DEBUG_ONLY(bytes)) {
119 // Ensures to stop the execution once the connection has been stopped.
120 if (_done) {
121 return;
122 }
123 if (!ec) {
124 DEBUG_ASSERT_EQ(bytes, sizeof(stream_id));
125 // If succeeded start reading data.
126 ReadData();
127 } else {
128 // Else try again.
129 log_debug("streaming client: failed to send stream id:", ec.message());
130 Connect();
131 }
132 }));
133 } else {
134 log_info("streaming client: connection failed:", ec.message());
135 Reconnect();
136 }
137 };
138
139 log_debug("streaming client: connecting to", ep);
140 _socket.async_connect(ep, boost::asio::bind_executor(_strand, handle_connect));
141 }

143 void Client::Stop() {
144 _connection_timer.cancel();
145 auto self = shared_from_this();
146 // Serialize the shutdown with the completion handlers: every read/connect
147 // handler runs on _strand, so mutating _done and closing the socket from
148 // the caller's thread (game thread via sensor destruction) races a
149 // concurrent async_read initiation on an io_context worker -- asio sockets
150 // are not thread-safe, and the race crashed the client inside ReadData
151 // when switching sensors. self keeps the client alive until this runs.
152 boost::asio::post(_strand, [this, self]() {
153 _done = true;
154 if (_socket.is_open()) {
155 _socket.close();
156 }
157 });
158 }

160 void Client::Reconnect() {
161 auto self = shared_from_this();
162 _connection_timer.expires_after(time_duration::seconds(1u).to_chrono());
163 _connection_timer.async_wait([this, self](boost::system::error_code ec) {
164 if (!ec) {
165 Connect();
166 }
167 });
168 }

170 void Client::ReadData() {
171 auto self = shared_from_this();
172 if (_done) {
173 return;
174 }
175
176 // log_debug("streaming client: Client::ReadData");
177
178 auto message = std::make_shared<IncomingMessage>(_buffer_pool->Pop());
179
180 auto handle_read_data = [this, self, message](boost::system::error_code ec, size_t DEBUG_ONLY(bytes)) {
181 DEBUG_ONLY(log_debug("streaming client: Client::ReadData.handle_read_data", bytes, "bytes"));
182 // A read that completed while Stop() was being processed must not
183 // deliver its payload: the subscriber that _callback points into is
184 // already being torn down (sensor destruction), and calling into it
185 // is a use-after-free (observed as a manual_control segfault when
186 // switching sensors).
187 if (_done) {
188 return;
189 }
190 if (!ec) {
191 DEBUG_ASSERT_EQ(bytes, message->size());
192 DEBUG_ASSERT_NE(bytes, 0u);
193 // Move the buffer to the callback function and start reading the next
194 // piece of data.
195 // log_debug("streaming client: success reading data, calling the callback");
196 // An exception leaving this handler would escape io_context::run()
197 // on a worker thread and call std::terminate, killing the client
198 // process; log it and keep reading the stream instead.
199#ifndef LIBCARLA_NO_EXCEPTIONS
200 try {
201#endif // LIBCARLA_NO_EXCEPTIONS
202 _callback(message->pop());
203#ifndef LIBCARLA_NO_EXCEPTIONS
204 } catch (const std::exception &e) {
205 log_error("streaming client: exception in stream callback:", e.what());
206 }
207#endif // LIBCARLA_NO_EXCEPTIONS
208 ReadData();
209 } else {
210 // As usual, if anything fails start over from the very top.
211 log_debug("streaming client: failed to read data:", ec.message());
212 Connect();
213 }
214 };
215
216 auto handle_read_header = [this, self, message, handle_read_data](
217 boost::system::error_code ec,
218 size_t DEBUG_ONLY(bytes)) {
219 DEBUG_ONLY(log_debug("streaming client: Client::ReadData.handle_read_header", bytes, "bytes"));
220 if (!ec && (message->size() > 0u)) {
221 DEBUG_ASSERT_EQ(bytes, sizeof(message_size_type));
222 if (_done) {
223 return;
224 }
225 // Now that we know the size of the coming buffer, we can allocate our
226 // buffer and start putting data into it.
227 boost::asio::async_read(
228 _socket,
229 message->buffer(),
230 boost::asio::bind_executor(_strand, handle_read_data));
231 } else if (!_done) {
232 log_debug("streaming client: failed to read header:", ec.message());
233 DEBUG_ONLY(log_debug("size = ", message->size()));
234 DEBUG_ONLY(log_debug("bytes = ", bytes));
235 Connect();
236 }
237 };
238
239 // Read the size of the buffer that is coming.
240 boost::asio::async_read(
241 _socket,
242 message->size_as_buffer(),
243 boost::asio::bind_executor(_strand, handle_read_header));
244 }

81bool AWalkerController::GoToNavLocation(const FVector &WorldLocation)
82{
83 if (!StartNavigation(bNavCrosser))
84 {
85 return false;
86 }
87 const EPathFollowingRequestResult::Type Result = MoveToLocation(
88 WorldLocation,
89 /*AcceptanceRadius=*/50.0f,
90 /*bStopOnOverlap=*/true,
91 /*bUsePathfinding=*/true,
92 /*bProjectDestinationToNavigation=*/true,
93 /*bCanStrafe=*/true,
94 // Non-crossers plan on a filter that excludes road/crosswalk areas;
95 // crossers use the areas' own costs (crosswalk cheap, road expensive).
96 /*FilterClass=*/bNavCrosser
97 ? TSubclassOf<UNavigationQueryFilter>(UCarlaWalkerCrosserNavFilter::StaticClass())
98 : TSubclassOf<UNavigationQueryFilter>(UCarlaWalkerNavFilter::StaticClass()));
99 if (Result == EPathFollowingRequestResult::Failed)
100 {
101 UE_LOG(LogCarla, Warning,
102 TEXT("Walker %s: MoveToLocation to (%s) failed (no navmesh under target?)"),
103 *GetName(), *WorldLocation.ToCompactString());
104 return false;
105 }
106 return true;
107}

137void AWalkerController::SetNavigationActive(bool bActive)
138{
139 bNavigationActive = bActive;
140 ACharacter *CurrentCharacter = GetCharacter();
141 if (CurrentCharacter == nullptr)
142 {
143 return;
144 }
145 if (auto *Movement = CurrentCharacter->GetCharacterMovement())
146 {
147 // While navigating, MaxWalkSpeed IS the walking speed (path following
148 // moves at full input scale). The manual path drives MaxWalkSpeed from
149 // the commanded speed each tick, so on leaving navigation the ceiling
150 // is just a safe starting point until the next manual control arrives.
151 Movement->MaxWalkSpeed = bActive ? NavMaxSpeed : GetMaximumWalkSpeed();
152 // A previously applied manual control (or the interim acceleration
153 // compensation) must not leak into navigation mode.
154 Movement->MaxAcceleration = Movement->GetClass()
155 ->GetDefaultObject<UCharacterMovementComponent>()->MaxAcceleration;
156 }
157 if (bActive)
158 {
159 // Drop any leftover manual input so it cannot fight the path following.
160 Control = FWalkerControl();
161 }
162}

164void AWalkerController::GetBonesTransform(FWalkerBoneControlOut &WalkerBones)
165{
166 ACharacter *CurrentCharacter = GetCharacter();
167 if (!CurrentCharacter) return;
168
169 TArray<USkeletalMeshComponent *> SkeletalMeshes;
170 CurrentCharacter->GetComponents<USkeletalMeshComponent>(SkeletalMeshes, false);
171 USkeletalMeshComponent *SkeletalMesh = SkeletalMeshes.IsValidIndex(0) ? SkeletalMeshes[0] : nullptr;
172 if (!SkeletalMesh) return;
173
174 // get the walker animation class
175 auto *AnimInst = SkeletalMesh->GetAnimInstance();
176 if (!AnimInst) return;
177 UWalkerAnim *WalkerAnim = Cast<UWalkerAnim>(AnimInst);
178 if (!WalkerAnim) return;
179
180 // get current pose
181 FPoseSnapshot TempSnapshot;
182 SkeletalMesh->SnapshotPose(TempSnapshot);
183
184 // copy pose
185 // WalkerAnim->Snap = TempSnapshot;
186
187 for (int i=0; i<TempSnapshot.BoneNames.Num(); ++i)
188 {
189 FWalkerBoneControlOutData Transforms;
190 Transforms.World = SkeletalMesh->GetSocketTransform(TempSnapshot.BoneNames[i], ERelativeTransformSpace::RTS_World);
191 Transforms.Component = SkeletalMesh->GetSocketTransform(TempSnapshot.BoneNames[i], ERelativeTransformSpace::RTS_Actor);
192 Transforms.Relative = SkeletalMesh->GetSocketTransform(TempSnapshot.BoneNames[i], ERelativeTransformSpace::RTS_ParentBoneSpace);
193 // Transforms.Relative = TempSnapshot.LocalTransforms[i];
194
195 WalkerBones.BoneTransforms.Add(TempSnapshot.BoneNames[i].ToString(), Transforms);
196 }
197}

199void AWalkerController::SetBonesTransform(const FWalkerBoneControlIn &WalkerBones)
200{
201 ACharacter* CurrentCharacter = GetCharacter();
202 if (!CurrentCharacter) return;
203
204 TArray<USkeletalMeshComponent *> SkeletalMeshes;
205 CurrentCharacter->GetComponents<USkeletalMeshComponent>(SkeletalMeshes, false);
206 USkeletalMeshComponent *SkeletalMesh = SkeletalMeshes.IsValidIndex(0) ? SkeletalMeshes[0] : nullptr;
207 if (!SkeletalMesh) return;
208
209 // get the walker animation class
210 auto *AnimInst = SkeletalMesh->GetAnimInstance();
211 if (!AnimInst) return;
212 UWalkerAnim *WalkerAnim = Cast<UWalkerAnim>(AnimInst);
213 if (!WalkerAnim) return;
214
215 // if pose is empty, then get a first version
216 if (WalkerAnim->Snap.BoneNames.Num() == 0)
217 {
218 // get current pose
219 SkeletalMesh->SnapshotPose(WalkerAnim->Snap);
220 }
221
222 TMap<FName, FTransform> BonesMap;
223 for (const TPair<FString, FTransform> &pair : WalkerBones.BoneTransforms)
224 {
225 FName BoneName = FName(*pair.Key);
226 BonesMap.Add(BoneName, pair.Value);
227 }
228
229 // assign common bones
230 for (int i=0; i<WalkerAnim->Snap.BoneNames.Num(); ++i)
231 {
232 FTransform *Trans = BonesMap.Find(WalkerAnim->Snap.BoneNames[i]);
233 if (Trans)
234 {
235 WalkerAnim->Snap.LocalTransforms[i] = *Trans;
236 }
237 }
238}

280void AWalkerController::Tick(float DeltaSeconds)
281{
282 TRACE_CPUPROFILER_EVENT_SCOPE(AWalkerController::Tick);
283 Super::Tick(DeltaSeconds);
284
285 ACharacter* CurrentCharacter = GetCharacter();
286 if (!CurrentCharacter) return;
287
288 // In navigation mode the crowd following component feeds the movement
289 // input; injecting the manual control here as well would double-drive
290 // the character.
291 if (bNavigationActive) return;
292
293 // Commanded speed becomes MaxWalkSpeed and the input is direction-only
294 // at full magnitude: encoding the speed in the input magnitude (the
295 // historical Control.Speed / 4096 scheme) also scaled the acceleration
296 // (UCharacterMovementComponent::ScaleInputAcceleration), so a 1.5 m/s
297 // walker accelerated at ~4% of MaxAcceleration and crawled for seconds
298 // after every spawn or retarget.
299 if (auto *Movement = CurrentCharacter->GetCharacterMovement())
300 {
301 Movement->MaxWalkSpeed = FMath::Clamp(Control.Speed, 0.0f, GetMaximumWalkSpeed());
302 }
303 if (Control.Speed >= 1.0f)
304 {
305 CurrentCharacter->AddMovementInput(Control.Direction, 1.0f);
306 }
307
308 if (Control.Jump)
309 {
310 CurrentCharacter->Jump();
311 }

22bool AShaderBasedSensor_WideAngleLens::AddPostProcessingMaterial(const FString& Path)
23{
24 // ConstructorHelpers::FObjectFinder is documented as constructor-only and
25 // has undefined behavior when used from runtime functions. FSoftObjectPath
26 // is the runtime-safe path resolver; it accepts both plain object paths
27 // and the class-prefixed export-text form ("Material'/Game/.../Foo'") that
28 // FObjectFinder previously normalised internally.
29 UMaterial* Material = Cast<UMaterial>(FSoftObjectPath(Path).TryLoad());
30 if (Material != nullptr)
31 {
32 MaterialsFound.Add(Material);
33 return true;
34 }
35 UE_LOG(LogCarla, Error,
36 TEXT("AShaderBasedSensor_WideAngleLens::AddPostProcessingMaterial: failed to load material '%s'"),
37 *Path);
38 return false;
39}

41void AShaderBasedSensor_WideAngleLens::SetUpSceneCaptureComponents(TArrayView<USceneCaptureComponent2D_CARLA*> SceneCaptures)
42{
43 // Create one dynamic instance per post-processing material. The six face
44 // captures of a wide-angle sensor must share an identical post-process
45 // chain, so the instances are created once and reused across all faces.
46 // Creating them inside the per-face loop would append 6x the materials to
47 // the shared Shaders array and give each face a different blendable set.
48 for (const auto& MaterialFound : MaterialsFound)
49 {
50 // Create a dynamic instance of the Material (Shader)
51 AddShader({ UMaterialInstanceDynamic::Create(MaterialFound, this), 1.0 });
52 }
53
54 for (auto SceneCapture : SceneCaptures)
55 {
56 for (const auto& Shader : Shaders)
57 {
58 // Attach the instance into the blendables
59 SceneCapture->PostProcessSettings.AddBlendable(Shader.PostProcessMaterial, Shader.Weight);
60 }
61 }
62
63 // Set the value for each Float parameter in the shader. The dynamic
64 // material instances are shared across faces, so each parameter only
65 // needs to be applied once.
66 if (Shaders.Num() != 0)
67 {
68 for (const auto& ParameterValue : FloatShaderParams)
69 {
70 Shaders[ParameterValue.ShaderIndex].PostProcessMaterial->SetScalarParameterValue(
71 ParameterValue.ParameterName,
72 ParameterValue.Value);
73 }
74 }
75}

16// Wire-compatible POD mirror of sensor::data::SemanticLidarDetection:

49std::vector<std::uint8_t> CarlaSemanticLidarPublisher::ComputePointCloud(
50 std::uint32_t height, std::uint32_t width, const std::uint8_t *data) const {
51 const std::size_t total_points =
52 static_cast<std::size_t>(height) * static_cast<std::size_t>(width);
53 const std::size_t total_bytes = total_points * sizeof(SemanticLidarPoint);
54
55 std::vector<std::uint8_t> bytes(total_bytes);
56 std::memcpy(bytes.data(), data, total_bytes);
57
58 // Mirror the Y axis for the ROS right-handed frame. The buffer is a
59 // contiguous array of SemanticLidarPoint PODs by contract.
60 auto *points = reinterpret_cast<SemanticLidarPoint *>(bytes.data());
61 for (std::size_t i = 0; i < total_points; ++i) {
62 points[i].y *= -1.0f;
63 }
64 return bytes;
65}

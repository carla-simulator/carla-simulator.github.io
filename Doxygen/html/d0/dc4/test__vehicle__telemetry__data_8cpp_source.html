18// VehicleTelemetryData and WheelTelemetryData are POD-style RPC types crossing
19// the simulator/client boundary via MsgPack. These tests pin two contracts:
20// 1) default construction zero-initializes every field, so a fresh instance
21// is safe to inspect or send;
22// 2) every field round-trips through MsgPack pack/unpack unchanged, including
23// the nested wheels vector.
24
25TEST(WheelTelemetryData, default_construction_zero_initialises_all_fields) {
26 WheelTelemetryData wheel{};
27 EXPECT_EQ(wheel.lat_slip, 0.0f);
28 EXPECT_EQ(wheel.long_slip, 0.0f);
29 EXPECT_EQ(wheel.omega, 0.0f);
30}
31
32TEST(WheelTelemetryData, msgpack_roundtrip_preserves_every_field) {
33 WheelTelemetryData wheel{1.25f, 0.42f, 31.7f};
34
35 auto buffer = carla::MsgPack::Pack(wheel);
36 auto restored = carla::MsgPack::UnPack<WheelTelemetryData>(buffer);
37
38 EXPECT_EQ(restored, wheel);
39}

52TEST(VehicleTelemetryData, msgpack_roundtrip_preserves_vehicle_and_wheel_fields) {
53 std::vector<WheelTelemetryData> wheels{
54 WheelTelemetryData{1.0f, 0.1f, 25.0f},
55 WheelTelemetryData{2.0f, 0.2f, 26.0f},
56 WheelTelemetryData{3.0f, 0.3f, 27.0f}};
57
58 VehicleTelemetryData telemetry{
59 14.5f, 0.12f, 0.8f, 0.0f, 3500.0f, 4, std::move(wheels)};
60
61 auto buffer = carla::MsgPack::Pack(telemetry);
62 auto restored = carla::MsgPack::UnPack<VehicleTelemetryData>(buffer);
63
64 EXPECT_EQ(restored.speed, telemetry.speed);
65 EXPECT_EQ(restored.steer, telemetry.steer);
66 EXPECT_EQ(restored.throttle, telemetry.throttle);
67 EXPECT_EQ(restored.brake, telemetry.brake);
68 EXPECT_EQ(restored.engine_rpm, telemetry.engine_rpm);
69 EXPECT_EQ(restored.gear, telemetry.gear);
70 ASSERT_EQ(restored.wheels.size(), telemetry.wheels.size());
71 for (size_t i = 0; i < telemetry.wheels.size(); ++i) {
72 EXPECT_EQ(restored.wheels[i], telemetry.wheels[i]);
73 }
74 EXPECT_EQ(restored, telemetry);
75}

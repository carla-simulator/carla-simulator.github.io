25 /// The payload of the single packet in @a bytes, and its declared size.
26 std::string Payload(const std::string &bytes, uint32_t &size) {
27 std::memcpy(&size, bytes.data() + 1, sizeof(size));
28 return bytes.substr(carla::multigpu::kFrameDataHeaderSize);

33TEST(MultiGpuSensorSync, ticks_round_trip_with_bit_exact_deltas) {
34 SensorTicks sent;
35 sent.ticks.push_back(SensorTick{7u, 0.05f});
36 sent.ticks.push_back(SensorTick{9u, 0.1f + 1e-7f});
37 std::ostringstream out;
38 sent.Write(out);
39 const std::string bytes = out.str();
40 ASSERT_EQ(static_cast<uint8_t>(bytes[0]), SensorTicks::kPacketId);
41
42 uint32_t size = 0u;
43 std::istringstream in(Payload(bytes, size));
44 SensorTicks got;
45 ASSERT_TRUE(got.Read(in, size));
46 ASSERT_EQ(got.ticks.size(), 2u);
47 EXPECT_EQ(got.ticks[0].actor_id, 7u);
48 EXPECT_EQ(got.ticks[0].delta_seconds, 0.05f);
49 EXPECT_EQ(got.ticks[1].delta_seconds, 0.1f + 1e-7f);
50}

64TEST(MultiGpuSensorSync, states_round_trip_every_field) {
65 SensorStates sent;
66 SensorState lidar;

68 lidar.random_engine = 123456789u;
69 lidar.lidar_horizontal_angle = 2.5f;
70 SensorState imu;
71 imu.actor_id = 4u;
72 imu.random_engine = 42u;
73 carla::multigpu::ImuHistory history;
74 history.older_location = {1.0, 2.0, 3.0};
75 history.last_location = {4.0, 5.0, 6.000001};
76 history.last_delta_seconds = 0.05f;
77 imu.imu_history = history;
78 imu.imu_local_rotation = std::array<double, 4>{0.0, -0.25881904510252074, 0.0, 0.96592582628906831};
79 SensorState radar;
80 radar.actor_id = 5u;
81 radar.radar_previous_location = std::array<double, 3>{-10.5, 20.25, 0.125};
82 sent.states = {lidar, imu, radar};
83
84 std::ostringstream out;
85 sent.Write(out);
86 uint32_t size = 0u;
87 std::istringstream in(Payload(out.str(), size));
88 SensorStates got;
89 ASSERT_TRUE(got.Read(in, size));
90 ASSERT_EQ(got.states.size(), 3u);
91 EXPECT_EQ(got.states[0].random_engine, lidar.random_engine);
92 EXPECT_EQ(got.states[0].lidar_horizontal_angle, lidar.lidar_horizontal_angle);
93 EXPECT_FALSE(got.states[0].imu_history.has_value());
94 ASSERT_TRUE(got.states[1].imu_history.has_value());
95 EXPECT_EQ(got.states[1].imu_history->last_location[2], 6.000001);
96 EXPECT_EQ(got.states[1].imu_history->last_delta_seconds, 0.05f);
97 ASSERT_TRUE(got.states[1].imu_local_rotation.has_value());
98 EXPECT_EQ(*got.states[1].imu_local_rotation, *imu.imu_local_rotation);
99 EXPECT_FALSE(got.states[0].imu_local_rotation.has_value());
100 ASSERT_TRUE(got.states[2].radar_previous_location.has_value());
101 EXPECT_EQ((*got.states[2].radar_previous_location)[1], 20.25);
102 EXPECT_FALSE(got.states[2].random_engine.has_value());
103}

105TEST(MultiGpuSensorSync, unknown_and_longer_fields_are_skipped) {
106 // One record: id 8, an unknown kind 9, then a random engine field that grew.
107 std::ostringstream record;
108 const uint32_t id = 8u;
109 record.write(reinterpret_cast<const char *>(&id), sizeof(id));
110 carla::multigpu::WriteField(record, 9u, std::string("future"));
111 std::string engine(sizeof(uint32_t) + 2u, '\0');
112 const uint32_t state = 777u;
113 std::memcpy(engine.data(), &state, sizeof(state));
114 carla::multigpu::WriteField(record, SensorStates::kRandomEngine, engine);
115 std::ostringstream payload;
116 const uint32_t count = 1u;
117 payload.write(reinterpret_cast<const char *>(&count), sizeof(count));
118 carla::multigpu::WriteSizedBytes(payload, record.str());
119
120 const std::string body = payload.str();
121 std::istringstream in(body);
122 SensorStates got;
123 ASSERT_TRUE(got.Read(in, static_cast<uint32_t>(body.size())));
124 ASSERT_EQ(got.states.size(), 1u);
125 EXPECT_EQ(got.states[0].actor_id, 8u);
126 EXPECT_EQ(got.states[0].random_engine, std::optional<uint32_t>(777u));
127}

145TEST(MultiGpuSensorSync, minstd_resumes_the_same_draws) {
146 // A mirror created later continues the primary sensor's noise exactly.
147 std::minstd_rand primary(1234u);
148 for (int i = 0; i < 17; ++i) {
149 (void)std::uniform_real_distribution<float>()(primary);
150 }
151 std::minstd_rand mirror(99u);
152 ASSERT_TRUE(SetEngineState(mirror, GetEngineState(primary)));
153 for (int i = 0; i < 50; ++i) {
154 EXPECT_EQ(std::uniform_real_distribution<float>()(mirror), std::uniform_real_distribution<float>()(primary));
155 EXPECT_EQ(std::normal_distribution<float>(0.0f, 2.0f)(mirror), std::normal_distribution<float>(0.0f, 2.0f)(primary));
156 EXPECT_EQ(std::uniform_int_distribution<int32_t>(0, 1000)(mirror), std::uniform_int_distribution<int32_t>(0, 1000)(primary));
157 }
158}

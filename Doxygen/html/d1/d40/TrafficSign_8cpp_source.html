22 std::vector<SharedPtr<Waypoint>> TrafficSign::GetAffectedLaneWaypoints() const {
23 std::vector<SharedPtr<Waypoint>> result;
24 SharedPtr<Map> carla_map = GetEpisode().Lock()->GetCurrentMap();
25 std::vector<SharedPtr<Landmark>> landmarks = carla_map->GetLandmarksFromId(GetSignId());
26 for (auto& landmark : landmarks) {
27 for (const road::LaneValidity& validity : landmark->GetValidities()) {
28 if (validity._from_lane < validity._to_lane) {
29 for (int lane_id = validity._from_lane; lane_id <= validity._to_lane; ++lane_id) {
30 if(lane_id == 0) continue;
31 result.emplace_back(
32 carla_map->GetWaypointXODR(
33 landmark->GetRoadId(), lane_id, static_cast<float>(landmark->GetS())));
34 }
35 } else {
36 for (int lane_id = validity._from_lane; lane_id >= validity._to_lane; --lane_id) {
37 if(lane_id == 0) continue;
38 result.emplace_back(
39 carla_map->GetWaypointXODR(
40 landmark->GetRoadId(), lane_id, static_cast<float>(landmark->GetS())));
41 }
42 }
43 }
44 }
45 return result;
46 }

48 std::vector<SharedPtr<Waypoint>> TrafficSign::GetStopWaypoints() const {
49 std::vector<SharedPtr<Waypoint>> result;
50 SharedPtr<Map> carla_map = GetEpisode().Lock()->GetCurrentMap();
51 geom::BoundingBox box = GetTriggerVolume();
52 geom::Transform transform = GetTransform();
53 geom::Location box_position = box.location;
54 transform.TransformPoint(box_position);
55 geom::Vector3D right_direction = transform.GetForwardVector();
56 float min_x = -0.9f*box.extent.x;
57 float max_x = 0.9f*box.extent.x;
58 float current_x = min_x;
59 std::unordered_map<road::RoadId, std::unordered_set<road::LaneId>> road_lanes_map;
60 while (current_x < max_x) {
61 geom::Location query_point = box_position + geom::Location(right_direction*current_x);
62 SharedPtr<Waypoint> waypoint = carla_map->GetWaypoint(query_point);
63 if (road_lanes_map[waypoint->GetRoadId()].count(waypoint->GetLaneId()) == 0) {
64 road_lanes_map[waypoint->GetRoadId()].insert(waypoint->GetLaneId());
65 result.emplace_back(waypoint);
66 }
67 current_x += 1.f;
68 }
69 return result;
70 }

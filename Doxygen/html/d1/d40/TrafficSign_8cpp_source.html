26 std::vector<SharedPtr<Waypoint>> TrafficSign::GetAffectedLaneWaypoints() const {
27 std::vector<SharedPtr<Waypoint>> result;
28 SharedPtr<Map> carla_map = GetEpisode().Lock()->GetCurrentMap();
29 std::vector<SharedPtr<Landmark>> landmarks = carla_map->GetLandmarksFromId(GetSignId());
30 for (auto& landmark : landmarks) {
31 for (const road::LaneValidity& validity : landmark->GetValidities()) {
32 if (validity._from_lane < validity._to_lane) {
33 for (int lane_id = validity._from_lane; lane_id <= validity._to_lane; ++lane_id) {
34 if(lane_id == 0) continue;
35 result.emplace_back(
36 carla_map->GetWaypointXODR(
37 landmark->GetRoadId(), lane_id, static_cast<float>(landmark->GetS())));
38 }
39 } else {
40 for (int lane_id = validity._from_lane; lane_id >= validity._to_lane; --lane_id) {
41 if(lane_id == 0) continue;
42 result.emplace_back(
43 carla_map->GetWaypointXODR(
44 landmark->GetRoadId(), lane_id, static_cast<float>(landmark->GetS())));
45 }
46 }
47 }
48 }
49 return result;
50 }

52 std::vector<SharedPtr<Waypoint>> TrafficSign::GetStopWaypoints() const {
53 std::vector<SharedPtr<Waypoint>> result;
54 SharedPtr<Map> carla_map = GetEpisode().Lock()->GetCurrentMap();
55 geom::BoundingBox box = GetTriggerVolume();
56 geom::Transform transform = GetTransform();
57 geom::Location box_position = box.location;
58 transform.TransformPoint(box_position);
59 geom::Vector3D right_direction = transform.GetForwardVector();
60 float min_x = -0.9f*box.extent.x;
61 float max_x = 0.9f*box.extent.x;
62 float current_x = min_x;
63 std::unordered_map<road::RoadId, std::unordered_set<road::LaneId>> road_lanes_map;
64 while (current_x < max_x) {
65 geom::Location query_point = box_position + geom::Location(right_direction*current_x);
66 SharedPtr<Waypoint> waypoint = carla_map->GetWaypoint(query_point);
67 if (road_lanes_map[waypoint->GetRoadId()].count(waypoint->GetLaneId()) == 0) {
68 road_lanes_map[waypoint->GetRoadId()].insert(waypoint->GetLaneId());
69 result.emplace_back(waypoint);
70 }
71 current_x += 1.f;
72 }
73 return result;
74 }

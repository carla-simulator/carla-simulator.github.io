15constexpr float kEps = 1e-5f;

17// Recover yaw (radians, REP-103) from a (w, x, y, z) quaternion that is known
18// to be yaw-only (x == y == 0).
19float YawFrom(const std::array<float, 4> &q) {
20 return 2.0f * std::atan2(q[3], q[0]);
21}

23} // namespace

25TEST(ImuMath, north_is_pi_over_two) {
26 // Compass = 0 means the vehicle faces magnetic North. REP-103 yaw measured
27 // counter-clockwise from East should be +pi/2.
28 const auto q = carla::ros2::OrientationFromCompass(0.0f);
29
30 ASSERT_NEAR(q[1], 0.0f, kEps);
31 ASSERT_NEAR(q[2], 0.0f, kEps);
32 ASSERT_NEAR(q[0], std::cos(static_cast<float>(M_PI) / 4.0f), kEps);
33 ASSERT_NEAR(q[3], std::sin(static_cast<float>(M_PI) / 4.0f), kEps);
34 ASSERT_NEAR(YawFrom(q), static_cast<float>(M_PI) / 2.0f, kEps);
35}

37TEST(ImuMath, east_is_zero_yaw) {
38 // Compass = pi/2 means the vehicle faces magnetic East. REP-103 yaw = 0.

57TEST(ImuMath, west_is_pi) {
58 // Compass = 3*pi/2 means the vehicle faces magnetic West. REP-103 yaw is
59 // pi/2 - 3*pi/2 = -pi (equivalently +pi); compare on |yaw| to avoid the
60 // atan2 wrap-around at the +/-pi seam.
61 const auto q = carla::ros2::OrientationFromCompass(
62 3.0f * static_cast<float>(M_PI) / 2.0f);
63
64 ASSERT_NEAR(q[1], 0.0f, kEps);
65 ASSERT_NEAR(q[2], 0.0f, kEps);
66 ASSERT_NEAR(std::abs(YawFrom(q)), static_cast<float>(M_PI), kEps);
67}

92TEST(ImuMath, is_not_the_regressed_pi_over_four) {
93 // Regression guard: the spelling shipped on ue5-dev before this fix
94 // returned pi/4 - compass instead of pi/2 - compass. With compass = 0 the
95 // buggy yaw is pi/4. Verify the helper does not produce that anymore.
96 const auto q = carla::ros2::OrientationFromCompass(0.0f);
97 ASSERT_GT(std::abs(YawFrom(q) - static_cast<float>(M_PI) / 4.0f), 1e-3f);

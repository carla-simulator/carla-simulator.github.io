204void FDenseTile::InitializeTile(uint32_t TextureSize, float AffectedRadius, float ParticleSize, float Depth,
205 FDVector TileOrigin, FDVector TileEnd,
206 const FString& InSavePath, const FHeightMapData &HeightMap)
207{
208
209 TileSize = (TileEnd.X - TileOrigin.X );
210 PartialHeightMapSize = TileSize * TextureSize / (2*AffectedRadius);
211 std::string FileName = std::string(TCHAR_TO_UTF8(*(InSavePath + TileOrigin.ToString() + ".tile" ) ) );
212
213 //UE_LOG(LogCarla, Log, TEXT("Tile origin %s"), *TileOrigin.ToString() );
214 if( FPaths::FileExists(FString(FileName.c_str())) )
215 {
216
217 TRACE_CPUPROFILER_EVENT_SCOPE(DenseTile::InitializeTile::Read);
218 std::ifstream ReadStream(FileName);
219 FVector VectorToRead;
220 ReadFVector(ReadStream, VectorToRead );
221 TilePosition = FDVector(VectorToRead);
222 ReadStdVector<FParticle> (ReadStream, Particles);
223 //UE_LOG(LogCarla, Log, TEXT("Reading data, got %d particles"), Particles.size());
224 }
225 else
226 {
227 TRACE_CPUPROFILER_EVENT_SCOPE(DenseTile::InitializeTile::Create);
228
229 TilePosition = TileOrigin;
230 uint32_t NumParticles_X = (TileEnd.X - TileOrigin.X) / ParticleSize;
231 uint32_t NumParticles_Y = FMath::Abs(TileEnd.Y - TileOrigin.Y) / ParticleSize;
232 uint32_t NumParticles_Z = (Depth) / ParticleSize;
233 Particles = std::vector<FParticle>(NumParticles_X*NumParticles_Y*NumParticles_Z);
234
235 //UE_LOG(LogCarla, Log, TEXT("Initializing Tile with (%d,%d,%d) particles at location %s, size %f, depth %f, HeightMap at tile origin %f"),
236 // NumParticles_X,NumParticles_Y,NumParticles_Z, *TileOrigin.ToString(), ParticleSize, Depth, HeightMap.GetHeight(TileOrigin) );
237
238 for(uint32_t i = 0; i < NumParticles_X; i++)
239 {
240 for(uint32_t j = 0; j < NumParticles_Y; j++)
241 {
242 FDVector ParticleLocalPosition = FDVector(i*ParticleSize, j*ParticleSize, 0.0f);
243 FDVector ParticlePosition = TileOrigin + ParticleLocalPosition;
244 float Height = HeightMap.GetHeight(ParticlePosition);
245 for(uint32_t k = 0; k < NumParticles_Z; k++)
246 {
247 ParticlePosition.Z = TileOrigin.Z + Height - k*ParticleSize;
248 Particles[k*NumParticles_X*NumParticles_Y + j*NumParticles_X + i] =
249 {ParticlePosition, FVector(0), ParticleSize/2.f};
250 }
251 }
252 }
253
254 //UE_LOG(LogCarla, Log, TEXT("Building local heightMap of %d pixels"), PartialHeightMapSize);
255
256 }
257 ParticlesZOrdered.resize( PartialHeightMapSize*PartialHeightMapSize );
258
259 for(float& Height : ParticlesHeightMap)
260 {
261 Height = 0;
262 }
263
264 bParticlesZOrderedInitialized = false;
265 bHeightmapNeedToUpdate = true;
266}

269{
270 {
271 TRACE_CPUPROFILER_EVENT_SCOPE(DenseTile::InitializeTile::ParticlesZOrdered);
272 ParticlesHeightMap.clear();
273 ParticlesHeightMap.resize( PartialHeightMapSize*PartialHeightMapSize );
274 float InverseTileSize = 1.f/TileSize;
275 float Transformation = InverseTileSize * PartialHeightMapSize;
276
277 for (size_t i = 0; i < Particles.size(); i++)
278 {
279 const FParticle& P = Particles[i];
280 FDVector ParticleLocalPosition = P.Position - TilePosition;
281 // Recalculate position to get it into heightmap coords
282 FIntVector HeightMapCoords = FIntVector(
283 ParticleLocalPosition.X * Transformation,
284 ParticleLocalPosition.Y * Transformation, 0);
285
286 uint32_t Index = HeightMapCoords.Y * PartialHeightMapSize + HeightMapCoords.X;
287 // Compare to the current value, if higher replace
288 if(Index < ParticlesZOrdered.size() ){
289 ParticlesZOrdered[Index].insert(P.Position.Z);
290 }
291 }
292
293 }
294 bParticlesZOrderedInitialized = true;
295 bHeightmapNeedToUpdate = true;

468std::vector<uint64_t> FSparseHighDetailMap::GetIntersectingTiles(
469 const FOrientedBox& OBox)
470{
471TRACE_CPUPROFILER_EVENT_SCOPE(FSparseHighDetailMap::GetIntersectingTiles);
472 std::vector<uint64_t> IntersectingTiles;
473
474 FVector BoxCenter = UEFrameToSI(OBox.Center);
475 float ExtentX = UEFrameToSI(OBox.ExtentX);
476 float ExtentXSqr = ExtentX*ExtentX;
477 FVector AxisX = UEFrameToSIDirection(OBox.AxisX);
478 float ExtentY = UEFrameToSI(OBox.ExtentY);
479 float ExtentYSqr = ExtentY*ExtentY;
480 FVector AxisY = UEFrameToSIDirection(OBox.AxisY);
481
482 uint64_t CenterTileId = GetTileId(BoxCenter);
483 uint32_t CenterTile_X = (uint32_t)(CenterTileId >> 32);
484 uint32_t CenterTile_Y = (uint32_t)(CenterTileId & (uint32_t)(~0));
485 uint32_t TileRange = 1;
486 IntersectingTiles.emplace_back(CenterTileId);
487 FVector2D BoxVert0 = FVector2D(BoxCenter - AxisX*ExtentX - AxisY*ExtentY);
488 FVector2D BoxVert1 = FVector2D(BoxCenter + AxisX*ExtentX - AxisY*ExtentY);
489 FVector2D BoxVert2 = FVector2D(BoxCenter + AxisX*ExtentX + AxisY*ExtentY);
490 FVector2D BoxVert3 = FVector2D(BoxCenter - AxisX*ExtentX + AxisY*ExtentY);
491
492 // Use separate axis theorem to detect intersecting tiles with OBox
493 struct F2DRectangle
494 {
495 FVector2D V1, V2, V3, V4;
496 std::vector<FVector2D> ComputeNormals() const
497 {
498 FVector2D V12 = (V2 - V1).GetSafeNormal();
499 FVector2D V23 = (V3 - V2).GetSafeNormal();
500 return {FVector2D(-V12.Y, V12.X), FVector2D(-V23.Y, V23.X)};
501 }
502 };
503 auto SATtest = [&](const FVector2D& Axis, const F2DRectangle &Shape,
504 float &MinAlong, float &MaxAlong)
505 {
506 for (const FVector2D& Vert : {Shape.V1, Shape.V2, Shape.V3, Shape.V4})
507 {
508 float DotVal = FVector2D::DotProduct(Vert, Axis);
509 if( DotVal < MinAlong )
510 MinAlong = DotVal;
511 if( DotVal > MaxAlong )
512 MaxAlong = DotVal;
513 }
514 };
515 auto IsBetweenOrdered = [&]( float Val, float LowerBound, float UpperBound ) -> bool
516 {
517 return LowerBound <= Val && Val <= UpperBound ;
518 };
519 auto Overlaps = [&]( float Min1, float Max1, float Min2, float Max2 ) -> bool
520 {
521 return IsBetweenOrdered( Min2, Min1, Max1 ) || IsBetweenOrdered( Min1, Min2, Max2 ) ;
522 };
523 auto RectangleIntersect = [&](
524 const F2DRectangle& Rectangle1,
525 const F2DRectangle& Rectangle2) -> bool
526 {
527 for (const FVector2D& Normal : Rectangle1.ComputeNormals())
528 {
529 constexpr float LargeNumber = 10000000;
530 float Shape1Min = LargeNumber, Shape1Max = -LargeNumber;
531 float Shape2Min = LargeNumber, Shape2Max = -LargeNumber;
532 SATtest(Normal, Rectangle1, Shape1Min, Shape1Max);
533 SATtest(Normal, Rectangle2, Shape2Min, Shape2Max);
534 if (!Overlaps(Shape1Min, Shape1Max, Shape2Min, Shape2Max))
535 {
536 return false;
537 }
538 }
539 for (const FVector2D& Normal : Rectangle2.ComputeNormals())
540 {
541 constexpr float LargeNumber = 10000000;
542 float Shape1Min = LargeNumber, Shape1Max = -LargeNumber;
543 float Shape2Min = LargeNumber, Shape2Max = -LargeNumber;
544 SATtest(Normal, Rectangle1, Shape1Min, Shape1Max);
545 SATtest(Normal, Rectangle2, Shape2Min, Shape2Max);
546 if (!Overlaps(Shape1Min, Shape1Max, Shape2Min, Shape2Max))
547 {
548 return false;
549 }
550 }
551 return true;
552 };
553 // check surrounding tiles except CenterTile (as it is always included)
554 std::vector<std::pair<uint32_t, uint32_t>> TilesToCheck = {
555 {CenterTile_X-1, CenterTile_Y-1}, {CenterTile_X, CenterTile_Y-1},
556 {CenterTile_X+1, CenterTile_Y-1}, {CenterTile_X-1, CenterTile_Y},
557 {CenterTile_X+1, CenterTile_Y}, {CenterTile_X-1, CenterTile_Y+1},
558 {CenterTile_X, CenterTile_Y+1}, {CenterTile_X+1, CenterTile_Y+1}};
559 for (auto& TileIdPair : TilesToCheck)
560 {
561 uint32_t &Tile_X = TileIdPair.first;
562 uint32_t &Tile_Y = TileIdPair.second;
563 FVector2D V1 = FVector2D(GetTilePosition(Tile_X, Tile_Y).ToFVector());
564 FVector2D V2 = FVector2D(GetTilePosition(Tile_X+1, Tile_Y).ToFVector());
565 FVector2D V3 = FVector2D(GetTilePosition(Tile_X+1, Tile_Y+1).ToFVector());
566 FVector2D V4 = FVector2D(GetTilePosition(Tile_X, Tile_Y+1).ToFVector());
567 if (RectangleIntersect(
568 F2DRectangle{BoxVert0,BoxVert1,BoxVert2,BoxVert3},
569 F2DRectangle{V1,V2,V3,V4}))
570 {
571 IntersectingTiles.emplace_back(GetTileId(Tile_X, Tile_Y));
572 }
573 }
574
575 return IntersectingTiles;
576}

587 uint32_t MinX = 0,MinY = 0,MaxX = 0,MaxY = 0;
588 {
589 TRACE_CPUPROFILER_EVENT_SCOPE(Comparisons);
590 if( Tile_X < RadiusInTiles){
591 MinX = 0;
592 }else{
593 MinX = Tile_X - RadiusInTiles;
594 }
595
596 if( Tile_Y < RadiusInTiles){
597 MinY = 0;
598 }else{
599 MinY = Tile_Y - RadiusInTiles;
600 }
601
602 MaxX = Tile_X + RadiusInTiles;
603 MaxY = Tile_Y + RadiusInTiles;
604 }
605
606 std::vector<uint64_t> LoadedTiles;
607 {
608 TRACE_CPUPROFILER_EVENT_SCOPE(Looping);
609 for( uint32_t X = MinX; X < MaxX; ++X )
610 {
611 for( uint32_t Y = MinY; Y < MaxY; ++Y )
612 {
613 uint64_t CurrentTileId = GetTileId(X,Y);
614 if( Map.find(CurrentTileId) != Map.end() )
615 {
616 LoadedTiles.emplace_back(CurrentTileId);
617 }
618 }
619 }
620 }
621 return LoadedTiles;

772void FSparseHighDetailMap::UpdateMaps(
773 FDVector Position, float RadiusX, float RadiusY, float CacheRadiusX, float CacheRadiusY)
774{
775 TRACE_CPUPROFILER_EVENT_SCOPE(FSparseHighDetailMap::UpdateMaps);
776 double MinX = Position.X - RadiusX;
777 double MinY = Position.Y - RadiusY;
778 double MaxX = Position.X + RadiusX;
779 double MaxY = Position.Y + RadiusY;
780
781 FIntVector MinVector = GetVectorTileId(FDVector(MinX, MinY, 0));
782 FIntVector MaxVector = GetVectorTileId(FDVector(MaxX, MaxY, 0));
783
784 FIntVector CacheMinVector = GetVectorTileId(
785 FDVector(Position.X - CacheRadiusX, Position.Y - CacheRadiusY, 0));
786 FIntVector CacheMaxVector = GetVectorTileId(
787 FDVector(Position.X + CacheRadiusX, Position.Y + CacheRadiusY, 0));
788
789 auto IsInCacheRange = [&](int32_t Tile_X, int32_t Tile_Y) -> bool
790 {
791 return Tile_X >= CacheMinVector.X && Tile_X <= CacheMaxVector.X &&
792 Tile_Y >= CacheMinVector.Y && Tile_Y <= CacheMaxVector.Y;
793 };
794 auto IsInMapRange = [&](int32_t Tile_X, int32_t Tile_Y) -> bool
795 {
796 return Tile_X >= MinVector.X && Tile_X <= MaxVector.X &&
797 Tile_Y >= MinVector.Y && Tile_Y <= MaxVector.Y;
798 };
799 {
800 FScopeLock ScopeLock(&Lock_Map);
801 FScopeLock ScopeCacheLock(&Lock_CacheMap);
802 TRACE_CPUPROFILER_EVENT_SCOPE(UpdateMap);
803 // unload extra tiles
804 std::vector<uint64_t> TilesToErase;
805 for (auto &Element : Map)
806 {
807 uint64_t TileId = Element.first;
808 FIntVector VectorTileId = GetVectorTileId(TileId);
809 if (!IsInMapRange(VectorTileId.X, VectorTileId.Y))
810 {
811 CacheMap.emplace(TileId, std::move(Element.second));
812 TilesToErase.emplace_back(TileId);
813 }
814 }
815 for (uint64_t TileId : TilesToErase)
816 {
817 Map.erase(TileId);
818 }
819 }
820 {
821 FScopeLock ScopeCacheLock(&Lock_CacheMap);
822 TRACE_CPUPROFILER_EVENT_SCOPE(UpdateCache);
823
824 // unload extra tiles
825 std::vector<uint64_t> TilesToErase;
826 {
827 TRACE_CPUPROFILER_EVENT_SCOPE(GetTilesToErase);
828 for (auto &Element : CacheMap)
829 {
830 uint64_t TileId = Element.first;
831 FIntVector VectorTileId = GetVectorTileId(TileId);
832 if (!IsInCacheRange(VectorTileId.X, VectorTileId.Y))
833 {
834 TilesToErase.emplace_back(TileId);
835 }
836 }
837 }
838
839 {
840 TRACE_CPUPROFILER_EVENT_SCOPE(EraseTiles);
841 ParallelFor(TilesToErase.size(), [&](int32 Idx)
842 {
843 TRACE_CPUPROFILER_EVENT_SCOPE(SaveData);
844 uint64_t TileId = TilesToErase[Idx];
845 auto& Tile = CacheMap[TileId];
846 std::string FileToSavePath = std::string(
847 TCHAR_TO_UTF8(*( SavePath + Tile.TilePosition.ToString() + ".tile")));
848 std::ofstream OutputStream(FileToSavePath.c_str());
849 WriteFVector(OutputStream, Tile.TilePosition.ToFVector());
850 WriteStdVector<FParticle> (OutputStream, Tile.Particles);
851 OutputStream.close();
852 });
853 {
854 TRACE_CPUPROFILER_EVENT_SCOPE(CacheMap.erase);
855 for (uint64_t TileId : TilesToErase)
856 {
857 CacheMap.erase(TileId);
858 }
859 }
860 }
861 }

970{

972 const int32_t TextureSizeX = TextureToUpdate->GetSizeX();
973 if ( TextureSizeX == 0) return;
974
975
976 FDVector TextureCenterPosition = UEFrameToSI(GetTileCenter(LastUpdatedPosition));
977
978 std::vector<uint64_t> LoadedTiles =
979 SparseMap.GetLoadedTilesInRange(TextureCenterPosition, TextureRadius );
980 FDVector TextureOrigin = TextureCenterPosition - FDVector(TextureRadius, TextureRadius, 0);
981 float GlobalTexelSize = (2.0f * TextureRadius) / TextureSizeX;
982 int32_t PartialHeightMapSize = std::floor( SparseMap.GetTileSize() * TextureSizeX / (2*TextureRadius) );
983
984 float LocalTexelSize = SparseMap.GetTileSize() / PartialHeightMapSize;
985 LocalTexelSize = std::floor( LocalTexelSize * 1000.0f ) / 1000.0f;
986
987 Data.Init( 128, Data.Num() );
988 float DisplacementRange = MaxDisplacement - MinDisplacement;
989 float InverseDisplacementRange = 1.0f / DisplacementRange;
990
991 for (uint64_t TileId : LoadedTiles)
992 {
993 FDenseTile& CurrentTile = SparseMap.GetTile(TileId);
994
995 if (!CurrentTile.bParticlesZOrderedInitialized)
996 {

998 }

1001 for (int32_t Local_Y = 0; Local_Y < PartialHeightMapSize; ++Local_Y)
1002 {
1003 for (int32_t Local_X = 0; Local_X < PartialHeightMapSize; ++Local_X)
1004 {
1005 int32_t LocalIndex = Local_Y * PartialHeightMapSize + Local_X;
1006 float Height = CurrentTile.ParticlesHeightMap[LocalIndex];

1008 TilePosition + FDVector(Local_X*LocalTexelSize, Local_Y*LocalTexelSize, 0);

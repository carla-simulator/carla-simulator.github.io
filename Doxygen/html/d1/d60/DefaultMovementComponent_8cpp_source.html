31 if (CarlaVehicle->GetVehicleControl().bReverse != Control.bReverse)

34 MovementComponent->SetTargetGear(Control.bReverse ? -1 : 1, true);
35 }
36 else
37 {
38 MovementComponent->SetUseAutomaticGears(!Control.bManualGearShift);
39 if (Control.bManualGearShift)

41 MovementComponent->SetTargetGear(Control.Gear, true);
42 }
43 }
44 Control.Gear = MovementComponent->GetCurrentGear();

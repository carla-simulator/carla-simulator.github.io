80// Road test
81static auto get_total_road_marks(
82 pugi::xml_object_range<pugi::xml_named_node_iterator> &lane_nodes,
83 LaneSection& lane_section) {
84 constexpr auto error = 1e-5;
85 auto total_road_mark = 0;
86 auto total_road_mark_parser = 0;
87 for (pugi::xml_node lane_node : lane_nodes) {
88 // Check Road Mark
89 auto road_mark_nodes = lane_node.children("roadMark");
90 total_road_mark_parser += std::distance(road_mark_nodes.begin(), road_mark_nodes.end());
91
92 const int lane_id = lane_node.attribute("id").as_int();
93 Lane* lane = nullptr;
94 lane = lane_section.GetLane(lane_id);
95 EXPECT_NE(lane, nullptr);
96
97 // <roadMark sOffset="0.0000000000000000e+0" type="none" material="standard" color="white" laneChange="none"/>
98 for (pugi::xml_node road_mark_node : road_mark_nodes) {
99 const auto s_offset = road_mark_node.attribute("sOffset").as_double();
100 const auto type = road_mark_node.attribute("type").as_string();
101 const auto material = road_mark_node.attribute("material").as_string();
102 const auto color = road_mark_node.attribute("color").as_string();
103 const auto road_mark = lane->GetInfo<RoadInfoMarkRecord>(lane->GetDistance() + s_offset);
104 if (road_mark != nullptr) {
105 EXPECT_NEAR(lane->GetDistance() + s_offset, road_mark->GetDistance(), error);
106 EXPECT_EQ(type, road_mark->GetType());
107 EXPECT_EQ(material, road_mark->GetMaterial());
108 EXPECT_EQ(color, road_mark->GetColor());
109 ++total_road_mark;
110 }
111 }
112 }
113 return std::make_pair(total_road_mark, total_road_mark_parser);
114}

116static void test_roads(const pugi::xml_document &xml, std::optional<Map>& map) {
117 pugi::xml_node open_drive_node = xml.child("OpenDRIVE");
118
119 // Check total Roads
120 auto roads_parser = open_drive_node.children("road");
121 auto total_roads_parser = std::distance(roads_parser.begin(), roads_parser.end());
122 auto total_roads = map->GetMap().GetRoads().size();
123 ASSERT_EQ(total_roads, total_roads_parser);
124
125 for (pugi::xml_node road_node : roads_parser) {
126 RoadId road_id = road_node.attribute("id").as_uint();
127
128 for (pugi::xml_node lanes_node : road_node.children("lanes")) {
129
130 // Check total Lane Sections
131 auto lane_sections_parser = lanes_node.children("laneSection");
132 auto total_lane_sections_parser = std::distance(lane_sections_parser.begin(), lane_sections_parser.end());
133 auto total_lane_sections = map->GetMap().GetRoad(road_id).GetLaneSections().size();
134 ASSERT_EQ(total_lane_sections, total_lane_sections_parser);
135
136 for (pugi::xml_node lane_section_node : lane_sections_parser) {
137
138 // Check total Lanes
139 const double s = lane_section_node.attribute("s").as_double();
140 auto lane_section = map->GetMap().GetRoad(road_id).GetLaneSectionsAt(s);
141 size_t total_lanes = 0u;
142 for (auto it = lane_section.begin(); it != lane_section.end(); ++it) {
143 total_lanes = it->GetLanes().size();
144 }
145 auto left_nodes = lane_section_node.child("left").children("lane");
146 auto center_nodes = lane_section_node.child("center").children("lane");
147 auto right_nodes = lane_section_node.child("right").children("lane");
148 auto total_lanes_parser = std::distance(left_nodes.begin(), left_nodes.end());
149 total_lanes_parser += std::distance(right_nodes.begin(), right_nodes.end());
150 total_lanes_parser += std::distance(center_nodes.begin(), center_nodes.end());
151
152 ASSERT_EQ(total_lanes, total_lanes_parser);
153
154
155 auto total_road_mark = 0;
156 auto total_road_mark_parser = 0;
157 for (auto it = lane_section.begin(); it != lane_section.end(); ++it) {
158 auto total_left = get_total_road_marks(left_nodes, *it);
159 auto total_center = get_total_road_marks(center_nodes, *it);
160 auto total_right = get_total_road_marks(right_nodes, *it);
161 total_road_mark = total_left.first + total_center.first + total_right.first;
162 total_road_mark_parser = total_left.first + total_center.first + total_right.first;
163 }
164 ASSERT_EQ(total_road_mark, total_road_mark_parser);
165 }
166 }
167 }

310TEST(road, iterate_waypoints) {
311 for (const auto& file : util::OpenDrive::GetAvailableFiles()) {
312 carla::logging::log("Parsing", file);
313 carla::StopWatch stop_watch;
314 auto m = OpenDriveParser::Load(util::OpenDrive::Load(file));
315 ASSERT_TRUE(m.has_value());
316 auto &map = *m;
317 if (IsRoadless(map)) {
318 carla::logging::log(" no roads, skipping:", file);
319 continue;
320 }
321 const auto topology = map.GenerateTopology();
322 ASSERT_FALSE(topology.empty());
323 auto count = 0u;
324 auto waypoints = map.GenerateWaypoints(0.5);
325 ASSERT_FALSE(waypoints.empty());
326 Random::Shuffle(waypoints);
327 const auto number_of_waypoints_to_explore =
328 std::min<size_t>(2000u, waypoints.size());
329 for (auto i = 0u; i < number_of_waypoints_to_explore; ++i) {
330 auto wp = waypoints[i];
331 map.ComputeTransform(wp);
332 if (i != 0u) {
333 ASSERT_NE(wp, waypoints[0u]);
334 }
335 for (auto &&successor : map.GetSuccessors(wp)) {
336 ASSERT_TRUE(
337 successor.road_id != wp.road_id ||
338 successor.section_id != wp.section_id ||
339 successor.lane_id != wp.lane_id ||
340 successor.s != wp.s);
341 }
342 auto origin = wp;
343 for (auto j = 0u; j < 200u; ++j) {
344 auto next_wps = map.GetNext(origin, Random::Uniform(0.0001, 150.0));
345 if (next_wps.empty()) {
346 break;
347 }
348 const auto number_of_next_wps_to_explore =
349 std::min<size_t>(10u, next_wps.size());
350 Random::Shuffle(next_wps);
351 for (auto k = 0u; k < number_of_next_wps_to_explore; ++k) {
352 auto next = next_wps[k];
353 ++count;
354 ASSERT_TRUE(
355 next.road_id != wp.road_id ||
356 next.section_id != wp.section_id ||
357 next.lane_id != wp.lane_id ||
358 next.s != wp.s);
359 auto right = map.GetRight(next);
360 if (right.has_value()) {
361 ASSERT_EQ(right->road_id, next.road_id);
362 ASSERT_EQ(right->section_id, next.section_id);
363 ASSERT_NE(right->lane_id, next.lane_id);
364 ASSERT_EQ(right->s, next.s);
365 }
366 auto left = map.GetLeft(next);
367 if (left.has_value()) {
368 ASSERT_EQ(left->road_id, next.road_id);
369 ASSERT_EQ(left->section_id, next.section_id);
370 ASSERT_NE(left->lane_id, next.lane_id);
371 ASSERT_EQ(left->s, next.s);
372 }
373 }
374 origin = next_wps[0u];
375 }
376 }
377 ASSERT_GT(count, 0u);
378 float seconds = 1e-3f * static_cast<float>(stop_watch.GetElapsedTime());
379 carla::logging::log(file, "done in", seconds, "seconds.");
380 }

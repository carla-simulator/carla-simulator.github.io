222namespace PID {
223static const float MAX_THROTTLE = 0.85f;
224static const float MAX_BRAKE = 0.7f;
225static const float MAX_STEERING = 0.8f;
226static const float MAX_STEERING_DIFF = 0.15f;
227// Speed-scaled steering envelope: |steer| <= STEER_LIMIT_GAIN / speed^2
228// (speed in m/s). Guards against a saturated PID output knifing the vehicle
229// across lanes at speed. Inactive below ~6 m/s (envelope > 0.8).
230// A/B note (Town10 matrix): raising the gain to 45 to let vehicles steer
231// through corners taken at 60 km/h regressed badly (saturated transients
232// became hard swerves into obstacles); the correct fix for running wide in
233// curves is slowing down before them (GetTurnTargetVelocity), not granting
234// more steering authority at speed.
235static const float STEER_LIMIT_GAIN = 30.0f;
236// Vehicle steering-authority normalization. A normalized steer command maps
237// to path curvature as kappa = tan(steer * max_steer_angle) / wheelbase, so
238// for the same command a long vehicle yields proportionally less curvature
239// than the car-class fleet the lateral gains and the STEER_LIMIT_GAIN
240// envelope were tuned on (measured: 8.5 m trucks pinned at the envelope on
241// 45% of junction ticks, ran R~13 m on R~11 m connecting roads and invaded
242// the adjacent lane on every junction turn while cars tracked cleanly).
243// Both the lateral P/D terms and the envelope are therefore scaled by
244// (bbox_length / REF_VEHICLE_LENGTH) * (REF_MAX_STEER_ANGLE / max_steer)
245// which equalizes commanded CURVATURE across the fleet; wheelbase tracks
246// bounding-box length closely, so the unknown wheelbase fraction cancels
247// against the car-anchored reference length. Clamped to
248// [1, MAX_STEER_AUTHORITY_CORRECTION] so car-class vehicles and the
249// physical lateral-acceleration guard stay exactly as tuned.
250static const float REF_VEHICLE_LENGTH = 4.9f; // m, tuning-fleet car bbox length
251static const float REF_MAX_STEER_ANGLE = 70.0f; // deg, tuning-fleet wheel lock
252static const float MAX_STEER_AUTHORITY_CORRECTION = 2.5f;
253// Ceiling for the COMBINED lateral P/D scale (pursuit-distance schedule x
254// steering-authority correction). Both factors approximate the pure-pursuit
255// equivalent gain, but their product on a long vehicle at a short pursuit
256// distance overshoots it (measured: 8.5 m trucks settled ~1.0 m INSIDE their
257// reference through junction turns at a composed scale of ~4.6). The ceiling
258// is the largest scale the schedule alone produces on the tuned car fleet
259// (MAX/MIN pursuit distance = 10/3), which validated cleanly; the authority
260// correction fills toward that ceiling for long vehicles instead of
261// exceeding it. The steering ENVELOPE keeps the full authority correction:
262// it is a physical curvature cap, not a loop gain.
263static const float MAX_LATERAL_GAIN_SCALE =
264 WaypointSelection::MAX_TARGET_WAYPOINT_DISTANCE /
265 WaypointSelection::MIN_TARGET_WAYPOINT_DISTANCE;
266// Bounding-box length to wheelbase ratio, stable across the CARLA fleet
267// (0.55-0.62 on the shipped vehicles); used with the physical wheel lock to
268// invert the steering-to-curvature map tan(steer * lock) / wheelbase exactly
269// instead of linearizing it. The linearized P law was tuned at car-scale
270// commands (|steer| ~ 0.1, where tan is linear); long vehicles need
271// |steer| ~ 0.4-0.5 through junctions, where the uncompensated tan() makes
272// the linear law over-curve and settle ~1 m inside its reference.
273static const float WHEELBASE_FRACTION = 0.6f;
274// Margin over the exact pure-pursuit curvature (kappa = 2 sin(alpha) / d).
275// Chosen so the anchor car at the cruise pursuit distance reproduces the
276// validated linear gain P = 2: P_pp = 2 * L_wb * pi / (d * lock_rad) = 1.49
277// at d = 10 m, and 1.32 * 1.49 * dev matches 2 * dev in the small-command
278// regime the cruise tuning was measured in.
279static const float PURSUIT_CURVATURE_MARGIN = 1.32f;
280// Nominal controller period: the sync-mode fixed_delta_seconds the gains were
281// tuned at. In asynchronous mode the real period is one server frame, which
282// under render load (e.g. a path-traced rt_lens sensor halving the frame
283// rate) can be 2-4x this. RunStep therefore takes the measured period and
284// compensates; DT is only the design point and the fallback when no valid
285// measurement exists (first tick after registration or a state reseed).
286static const float DT = 0.05f;
287static const float INV_DT = 1.0f / DT;
288// Valid range for the measured controller period. Below MIN the derivative
289// division gets noisy; above MAX the sim is hitching so badly that reacting
290// to the full elapsed time would command huge one-tick corrections.
291static constexpr float MIN_CONTROL_DT = 0.01f;
292static const float MAX_CONTROL_DT = 0.2f;
293// Steering slew budget, per second of simulation time (0.15 per 0.05 s tick
294// at the design rate). Applying it per second instead of per tick keeps the
295// physical steering rate constant when the tick rate changes.
296static const float MAX_STEERING_RATE = MAX_STEERING_DIFF / DT;
297// Max change in normalised angular deviation per tick accepted by the
298// lateral derivative term: 0.05 * 180 deg / 0.05 s = 180 deg/s, the upper
299// bound of real vehicle yaw. Larger jumps are target discontinuities.
300static const float MAX_DEVIATION_DELTA = 0.05f;
301// The same bound expressed as a rate, so it scales with the measured tick
302// period: 1.0 normalised units/s = 180 deg/s.
303static const float MAX_DEVIATION_RATE = MAX_DEVIATION_DELTA / DT;
304// Deadbands hold the previous command to remove per-frame dither around the
305// trim point. A slew limit is avoided on purpose: it must be asymmetric to stay
306// safe, and that biases the average command down. The throttle ramp applies
307// only near standstill (m/s); a higher threshold also catches corner exits.
308static const float LAUNCH_RAMP_SPEED = 2.0f;
309static constexpr float MAX_LAUNCH_THROTTLE_RISE_RATE = 1.7f;
310static constexpr float THROTTLE_DEADBAND = 0.01f;
311// Otherwise the deadband snaps every ramp step back and a vehicle at rest is
312// held at zero throttle for ever.
313static_assert(MAX_LAUNCH_THROTTLE_RISE_RATE * MIN_CONTROL_DT > THROTTLE_DEADBAND,
314 "the launch ramp must be able to step out of the throttle deadband");
315static const float BRAKE_DEADBAND = 0.01f;
316static const float STEER_DEADBAND = 0.002f;
317// m/s. The steering deadband is scaled by (ref / speed)^2 above this speed,
318// since the lateral acceleration it hides grows with the square of speed.
319static const float STEER_DEADBAND_REF_SPEED = 8.0f;
320// m/s2 at which the longitudinal reference closes on a stepping target. The
321// proportional gain saturates the throttle above a 7 per cent error (0.85 / 12),
322// so without it a target step commands full throttle in one frame.
323static const float COMFORT_ACCELERATION = 2.0f;
324// Max lead of the reference over the vehicle, as a fraction of the target, so
325// it cannot run away from a blocked vehicle. Must stay above the 7 per cent
326// that saturates the throttle.
327static const float REFERENCE_LEAD_FRACTION = 0.15f;
328static const std::vector<float> LONGITUDIAL_PARAM = {12.0f, 0.05f, 0.02f};
329static const std::vector<float> LONGITUDIAL_HIGHWAY_PARAM = {20.0f, 0.05f, 0.01f};
330// Lateral gains, step-response tuned against the LINEAR Chaos steering
331// mapping (the previous values were tuned while the engine still applied a
332// squared input curve, which hid a ~2-5x effective gain increase at small
333// commands). Key constraint: at cruise speeds the P term must stay inside
334// the STEER_LIMIT_GAIN/v^2 envelope for typical errors (~1.5 m offset ->
335// deviation ~0.05); P=2 keeps the loop linear at 60 km/h where P>=3
336// saturates the envelope and hunts around the lane center without settling
337// (measured: settle 1.1 s vs >6 s). D=0.24 damps the recovery to <=0.25 m
338// overshoot at 30-60 km/h.
339static const std::vector<float> LATERAL_PARAM = {2.0f, 0.03f, 0.24f};
340static const std::vector<float> LATERAL_HIGHWAY_PARAM = {2.0f, 0.02f, 0.24f};

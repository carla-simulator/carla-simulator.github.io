18 // Per-wheel runtime telemetry exposed to clients.
19 //
20 // Only the fields that the Chaos vehicle solver populates live are kept here.
21 // The PhysX-era surface (tire_friction, tire_load, normalized_tire_load,
22 // torque, long_force, lat_force, normalized_long_force, normalized_lat_force)
23 // is intentionally absent: those values were read from UVehicleWheel debug
24 // members that the Chaos plugin declares but never writes, so reporting them
25 // on UE5 would produce permanent zeros.
26 class WheelTelemetryData {
27 public:
28
29 WheelTelemetryData() = default;
30
31 WheelTelemetryData(
32 float lat_slip,
33 float long_slip,
34 float omega)
35 : lat_slip(lat_slip),
36 long_slip(long_slip),
37 omega(omega) {}
38
39 float lat_slip = 0.0f;
40 float long_slip = 0.0f;
41 float omega = 0.0f;
42
43#ifdef LIBCARLA_INCLUDED_FROM_UE4
44
45 WheelTelemetryData(const FWheelTelemetryData &Data)
46 : lat_slip(Data.LatSlip),
47 long_slip(Data.LongSlip),
48 omega(Data.Omega) {}
49
50 operator FWheelTelemetryData() const {
51 FWheelTelemetryData Data;
52 Data.LatSlip = lat_slip;
53 Data.LongSlip = long_slip;
54 Data.Omega = omega;
55 return Data;
56 }
57
58#endif // LIBCARLA_INCLUDED_FROM_UE4
59
60 bool operator!=(const WheelTelemetryData &rhs) const {

63 long_slip != rhs.long_slip ||
64 omega != rhs.omega;
65 }
66
67 bool operator==(const WheelTelemetryData &rhs) const {

69 }

74 omega

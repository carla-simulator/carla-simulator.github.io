113 static VehiclePhysicsControl FromFVehiclePhysicsControl(
114 const FVehiclePhysicsControl& Control) {
115 VehiclePhysicsControl Out = { };
116 // torque_curve and steering_curve carry non-empty defaults (see the
117 // member initializers above); clear them or the real keys get
118 // appended after the default ones, producing an unsorted curve.
119 Out.torque_curve.clear();
120 Out.torque_curve.reserve(Control.TorqueCurve.GetNumKeys());
121 for (auto& Key : Control.TorqueCurve.GetConstRefOfKeys())
122 Out.torque_curve.push_back(geom::Vector2D(Key.Time, Key.Value));
123 Out.max_torque = Control.MaxTorque;
124 Out.max_rpm = Control.MaxRPM;
125 Out.idle_rpm = Control.IdleRPM;
126 Out.brake_effect = Control.BrakeEffect;
127 Out.rev_up_moi = Control.RevUpMOI;
128 Out.rev_down_rate = Control.RevDownRate;
129 Out.differential_type = Control.DifferentialType;
130 Out.front_rear_split = Control.FrontRearSplit;
131 Out.use_automatic_gears = Control.bUseAutomaticGears;
132 Out.gear_change_time = Control.GearChangeTime;
133 Out.final_ratio = Control.FinalRatio;
134 Out.forward_gear_ratios.resize(Control.ForwardGearRatios.Num());
135 for (size_t i = 0; i != Out.forward_gear_ratios.size(); ++i)
136 Out.forward_gear_ratios[i] = Control.ForwardGearRatios[i];
137 Out.reverse_gear_ratios.resize(Control.ReverseGearRatios.Num());
138 for (size_t i = 0; i != Out.reverse_gear_ratios.size(); ++i)
139 Out.reverse_gear_ratios[i] = Control.ReverseGearRatios[i];
140 Out.change_up_rpm = Control.ChangeUpRPM;
141 Out.change_down_rpm = Control.ChangeDownRPM;
142 Out.transmission_efficiency = Control.TransmissionEfficiency;
143 Out.mass = Control.Mass;
144 Out.drag_coefficient = Control.DragCoefficient;
145 Out.center_of_mass = Control.CenterOfMass;
146 Out.chassis_width = Control.ChassisWidth;
147 Out.chassis_height = Control.ChassisHeight;
148 Out.downforce_coefficient = Control.DownforceCoefficient;
149 Out.drag_area = Control.DragArea;
150 Out.inertia_tensor_scale = geom::Vector3D(
151 Control.InertiaTensorScale.X,
152 Control.InertiaTensorScale.Y,
153 Control.InertiaTensorScale.Z);
154 Out.sleep_threshold = Control.SleepThreshold;
155 Out.sleep_slope_limit = Control.SleepSlopeLimit;
156 Out.steering_curve.clear();
157 Out.steering_curve.reserve(Control.SteeringCurve.GetNumKeys());
158 for (auto& Key : Control.SteeringCurve.GetConstRefOfKeys())
159 Out.steering_curve.push_back(geom::Vector2D(Key.Time, Key.Value));
160 Out.wheels.resize(Control.Wheels.Num());
161 for (size_t i = 0; i != Out.wheels.size(); ++i)
162 Out.wheels[i] = WheelPhysicsControl::FromFWheelPhysicsControl(Control.Wheels[i]);
163 Out.use_sweep_wheel_collision = Control.UseSweepWheelCollision;
164 return Out;
165 }

167 operator FVehiclePhysicsControl() const {
168 FVehiclePhysicsControl Control = { };
169 for (auto [x, y] : torque_curve)
170 Control.TorqueCurve.AddKey(x, y);
171 Control.MaxTorque = max_torque;
172 Control.MaxRPM = max_rpm;
173 Control.IdleRPM = idle_rpm;
174 Control.BrakeEffect = brake_effect;
175 Control.RevUpMOI = rev_up_moi;
176 Control.RevDownRate = rev_down_rate;
177 Control.DifferentialType = differential_type;
178 Control.FrontRearSplit = front_rear_split;
179 Control.bUseAutomaticGears = use_automatic_gears;
180 Control.GearChangeTime = gear_change_time;
181 Control.FinalRatio = final_ratio;
182 Control.ForwardGearRatios.SetNum(forward_gear_ratios.size());
183 for (size_t i = 0; i != Control.ForwardGearRatios.Num(); ++i)
184 Control.ForwardGearRatios[i] = forward_gear_ratios[i];
185 Control.ReverseGearRatios.SetNum(reverse_gear_ratios.size());
186 for (size_t i = 0; i != Control.ReverseGearRatios.Num(); ++i)
187 Control.ReverseGearRatios[i] = reverse_gear_ratios[i];
188 Control.ChangeUpRPM = change_up_rpm;
189 Control.ChangeDownRPM = change_down_rpm;
190 Control.TransmissionEfficiency = transmission_efficiency;
191 Control.Mass = mass;
192 Control.DragCoefficient = drag_coefficient;
193 Control.CenterOfMass = center_of_mass;
194 Control.ChassisWidth = chassis_width;
195 Control.ChassisHeight = chassis_height;
196 Control.DownforceCoefficient = downforce_coefficient;
197 Control.DragArea = drag_area;
198 Control.InertiaTensorScale = FVector(
199 inertia_tensor_scale.x,
200 inertia_tensor_scale.y,
201 inertia_tensor_scale.z);
202 Control.SleepThreshold = sleep_threshold;
203 Control.SleepSlopeLimit = sleep_slope_limit;
204 for (auto [x, y] : steering_curve)
205 Control.SteeringCurve.AddKey(x, y);
206 Control.Wheels.SetNum(wheels.size());
207 for (size_t i = 0; i != Control.Wheels.Num(); ++i)
208 Control.Wheels[i] = wheels[i];
209 Control.UseSweepWheelCollision = use_sweep_wheel_collision;
210 return Control;
211 }

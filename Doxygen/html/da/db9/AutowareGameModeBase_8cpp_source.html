27void AAutowareGameModeBase::LoadGeoReference()
28{
29 auto* WS = Cast<AAutowareWorldSettings>(GetWorld()->GetWorldSettings());
30
31 if (!IsValid(WS))
32 {
33 Super::LoadGeoReference(); // Fallback to ACarlaGameModeBase::LoadGeoReference default
34 return;
35 }
36
37 UE_LOG(LogCarla, Log, TEXT("Autoware Settings fetch succeeded."));
38 auto* Data = WS->MgrsDataAssetSoftPtr.LoadSynchronous();
39
40 if (!IsValid(Data))
41 {
42 return;
43 }
44
45 // tier4 stored a plain GeoLocation reference on the episode; our branch
46 // migrated the georeference to carla::geom::GeoProjection. A Transverse
47 // Mercator projection centered on the asset's reference point over the
48 // default (spherical) ellipsoid reproduces the legacy GeoLocation::Transform
49 // behavior.
50 Episode->MapGeoProjection = carla::geom::GeoProjection::Make(
51 carla::geom::TransverseMercatorParams(
52 Data->GeoReference.Latitude,
53 Data->GeoReference.Longitude,
54 1.0, 0.0, 0.0,
55 carla::geom::Ellipsoid()));
56
57 UE_LOG(LogCarla, Log, TEXT("MGRS georeference loaded successfully."));
58 StoreSpawnPoints();
59}

54void UBaseCarlaMovementComponent::DisableUE4VehiclePhysics()

71void UBaseCarlaMovementComponent::EnableUE4VehiclePhysics(bool bResetVelocity)
72{
73
74 FVector CurrentVelocity(0, 0, 0);
75 if (!bResetVelocity)
76 {
77 CurrentVelocity = GetVelocity();
78 }
79 CarlaVehicle->GetMesh()->SetPhysicsLinearVelocity(CurrentVelocity, false, "Vehicle_Base");
80 CarlaVehicle->GetVehicleMovementComponent()->SetComponentTickEnabled(true);
81 CarlaVehicle->GetVehicleMovementComponent()->Activate();
82 CarlaVehicle->GetMesh()->PhysicsTransformUpdateMode = EPhysicsTransformUpdateMode::SimulationUpatesComponentTransform;
83 auto * Bone = CarlaVehicle->GetMesh()->GetBodyInstance(NAME_None);
84 if (Bone)
85 {
86 Bone->SetInstanceSimulatePhysics(true);
87 }
88 else
89 {
90 carla::log_warning("No bone with name");
91 }
92 CarlaVehicle->GetMesh()->SetCollisionResponseToChannel(ECollisionChannel::ECC_WorldStatic, ECollisionResponse::ECR_Block);
93 CarlaVehicle->GetMesh()->SetCollisionProfileName("Vehicle");
94 CarlaVehicle->RestoreVehiclePhysicsControl();
95}

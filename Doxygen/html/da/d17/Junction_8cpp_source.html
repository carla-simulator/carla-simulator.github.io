33 std::vector<SharedPtr<Road>> Junction::GetConnectingRoads() const {
34 std::vector<SharedPtr<Road>> result;
35 std::unordered_set<road::RoadId> seen;
36 const road::Junction *junction = _parent->GetMap().GetJunction(_id);
37 if (junction == nullptr) {
38 return result;
39 }
40 for (const auto &pair : junction->GetConnections()) {
41 const road::RoadId road_id = pair.second.connecting_road;

73 // Entry and exit lanes are found by walking the lane graph out of each
74 // connecting lane rather than by reading Connection::lane_links, because the
75 // graph already resolves which end of the adjacent road the junction meets
76 // and which lane section applies there. Lanes still inside the junction are
77 // dropped, so what comes back is always on an adjacent road.
78 std::vector<SharedPtr<Lane>> Junction::GetNeighbourLanes(
79 road::Lane::LaneType type,
80 bool entering) const {
81 std::vector<SharedPtr<Lane>> result;
82 std::unordered_set<uint64_t> seen;
83 for (const auto &pair : GetWaypoints(type)) {
84 const SharedPtr<Waypoint> &waypoint = entering ? pair.first : pair.second;
85 if (waypoint == nullptr) {
86 continue;
87 }
88 SharedPtr<Lane> lane = waypoint->GetLane();
89 if (lane == nullptr) {
90 continue;
91 }
92 const auto neighbours = entering ? lane->GetPreviousLanes() : lane->GetNextLanes();
93 for (const SharedPtr<Lane> &neighbour : neighbours) {
94 if (neighbour == nullptr) {
95 continue;
96 }
97 SharedPtr<Road> road = neighbour->GetRoad();
98 // Skip anything still inside this junction.
99 if (road == nullptr || road->GetJunctionId() == _id) {
100 continue;
101 }
102 if (!seen.insert(neighbour->GetUniqueId()).second) {
103 continue;
104 }
105 result.emplace_back(neighbour);
106 }
107 }
108 return result;
109 }

119 // The end of a lane that touches the junction. An entry lane carries traffic
120 // into the junction, so the junction sits at its downstream end; an exit lane
121 // carries traffic away, so the junction sits at its upstream end. Which end
122 // that is in road coordinates depends on the lane's direction.
123 static double JunctionSideDistance(const Lane &lane, bool entering) {
124 // Kept a hair inside the lane so the waypoint is always valid rather than
125 // landing exactly on the boundary.
126 constexpr double kEpsilon = 1e-3;
127 const double start = lane.GetDistance();
128 const double end = start + lane.GetLength();
129 const bool downstream = entering;
130 const bool at_end = lane.IsPositiveDirection() == downstream;
131 if (at_end) {
132 return std::max(start, end - kEpsilon);
133 }
134 return std::min(end, start + kEpsilon);
135 }

137 std::vector<SharedPtr<Waypoint>> Junction::GetNeighbourWaypoints(
138 road::Lane::LaneType type,
139 bool entering) const {
140 std::vector<SharedPtr<Waypoint>> result;
141 for (const SharedPtr<Lane> &lane : GetNeighbourLanes(type, entering)) {
142 SharedPtr<Waypoint> waypoint = lane->GetWaypoint(JunctionSideDistance(*lane, entering));

146 }

36 /// True if a `LaneChange` value contains the `flag` bit
37 /// (mirrors Python `==` checks like `right_turn == LaneChange.Right` /
38 /// `right_turn == LaneChange.Both`).
39 bool HasLaneChange(LaneChange value, LaneChange flag) {
40 using U = std::underlying_type_t<LaneChange>;
41 return (static_cast<U>(value) & static_cast<U>(flag)) == static_cast<U>(flag);
42 }

74void BehaviorAgent::UpdateInformation() {
75 _speed = GetSpeed(_vehicle);
76 _speed_limit = _vehicle->GetSpeedLimit();
77 // Even when follow_speed_limits is on this seed mirrors the Python reference.
78 _local_planner->SetSpeed(_speed_limit);
79
80 _direction = _local_planner->TargetRoadOption();
81 if (_direction == RoadOption::Void) {
82 _direction = RoadOption::LaneFollow;
83 }
84
85 _look_ahead_steps = static_cast<int>(_speed_limit / 10.0f);
86
87 auto incoming = _local_planner->GetIncomingWaypointAndDirection(_look_ahead_steps);
88 _incoming_waypoint = incoming.first;
89 _incoming_direction = incoming.second;
90 if (_incoming_direction == RoadOption::Void) {
91 _incoming_direction = RoadOption::LaneFollow;
92 }
93}

110void BehaviorAgent::Tailgating(SharedPtr<client::Waypoint> waypoint) {
111 // Python passes the prefiltered vehicle list around; the base helper
112 // `VehicleObstacleDetected` accepts a `SharedPtr<ActorList>` and applies
113 // its own proximity gating via `max_distance`, so we hand it the
114 // world-wide vehicle filter result here.
115 auto left_marking = waypoint->GetLeftLaneMarking();
116 auto right_marking = waypoint->GetRightLaneMarking();
117 auto left_wpt = waypoint->GetLeft();
118 auto right_wpt = waypoint->GetRight();
119
120 // Build the world-vehicle ActorList passthrough (matches Python's reuse of
121 // the filtered list).
122 auto vehicles_alist = _world->GetActors()
123 ? _world->GetActors()->Filter("*vehicle*")
124 : nullptr;
125
126 const float behind_max =
127 std::max(_behavior.min_proximity_threshold, _speed_limit / 2.0f);
128
129 auto behind = VehicleObstacleDetected(vehicles_alist,
130 behind_max,
131 /*up_angle_th=*/180.0f,
132 /*low_angle_th=*/160.0f,
133 /*lane_offset=*/0);
134
135 if (!(behind.affected && behind.actor)) return;
136 if (_speed >= GetSpeed(behind.actor)) return;
137
138 const int wp_lane_id = waypoint->GetLaneId();
139
140 // Right-lane tailgate avoidance.
141 if (right_marking.has_value() && right_wpt &&
142 (HasLaneChange(right_marking->lane_change, LaneChange::Right) ||
143 HasLaneChange(right_marking->lane_change, LaneChange::Both)) &&
144 (wp_lane_id * right_wpt->GetLaneId() > 0) &&
145 right_wpt->GetType() == road::Lane::LaneType::Driving) {
146 auto check = VehicleObstacleDetected(vehicles_alist,
147 behind_max,
148 /*up_angle_th=*/180.0f,
149 /*low_angle_th=*/0.0f,
150 /*lane_offset=*/1);
151 if (!check.affected) {
152 std::cout << "Tailgating, moving to the right!" << std::endl;
153 auto end_waypoint = _local_planner->TargetWaypoint();
154 _behavior.tailgate_counter = 200;
155 if (end_waypoint) {
156 SetDestination(end_waypoint->GetTransform().location,
157 right_wpt->GetTransform().location);
158 }
159 }
160 return;
161 }
162
163 // Left-lane tailgate avoidance.
164 if (left_marking.has_value() && left_wpt &&
165 HasLaneChange(left_marking->lane_change, LaneChange::Left) &&
166 (wp_lane_id * left_wpt->GetLaneId() > 0) &&
167 left_wpt->GetType() == road::Lane::LaneType::Driving) {
168 auto check = VehicleObstacleDetected(vehicles_alist,
169 behind_max,
170 /*up_angle_th=*/180.0f,
171 /*low_angle_th=*/0.0f,
172 /*lane_offset=*/-1);
173 if (!check.affected) {
174 std::cout << "Tailgating, moving to the left!" << std::endl;
175 auto end_waypoint = _local_planner->TargetWaypoint();
176 _behavior.tailgate_counter = 200;
177 if (end_waypoint) {
178 SetDestination(end_waypoint->GetTransform().location,
179 left_wpt->GetTransform().location);
180 }
181 }
182 }
183}

189BasicAgent::VehicleObstacle BehaviorAgent::CollisionAndCarAvoidManager(
190 SharedPtr<client::Waypoint> waypoint) {
191 auto vehicles_alist = _world->GetActors()
192 ? _world->GetActors()->Filter("*vehicle*")
193 : nullptr;
194
195 // Python applies a 45 m prefilter around the waypoint here (excluding the
196 // ego). The base helper `VehicleObstacleDetected` honors `max_distance`
197 // internally, so we just route the world-wide vehicle list to it.
198 VehicleObstacle obs;
199 if (_direction == RoadOption::ChangeLaneLeft) {
200 obs = VehicleObstacleDetected(
201 vehicles_alist,
202 std::max(_behavior.min_proximity_threshold, _speed_limit / 2.0f),
203 /*up_angle_th=*/180.0f,
204 /*low_angle_th=*/0.0f,
205 /*lane_offset=*/-1);
206 } else if (_direction == RoadOption::ChangeLaneRight) {
207 obs = VehicleObstacleDetected(
208 vehicles_alist,
209 std::max(_behavior.min_proximity_threshold, _speed_limit / 2.0f),
210 /*up_angle_th=*/180.0f,
211 /*low_angle_th=*/0.0f,
212 /*lane_offset=*/1);
213 } else {
214 obs = VehicleObstacleDetected(
215 vehicles_alist,
216 std::max(_behavior.min_proximity_threshold, _speed_limit / 3.0f),
217 /*up_angle_th=*/30.0f,
218 /*low_angle_th=*/0.0f,
219 /*lane_offset=*/0);
220
221 // Tailgating check.
222 if (!obs.affected &&
223 _direction == RoadOption::LaneFollow &&
224 !waypoint->IsJunction() &&
225 _speed > 10.0f &&
226 _behavior.tailgate_counter == 0) {
227 Tailgating(waypoint);
228 }
229 }
230
231 return obs;
232}

234BasicAgent::VehicleObstacle BehaviorAgent::PedestrianAvoidManager(
235 SharedPtr<client::Waypoint> waypoint) {
236 auto walkers_alist = _world->GetActors()
237 ? _world->GetActors()->Filter("*walker.pedestrian*")
238 : nullptr;
239
240 // Python applies a 10 m prefilter around the waypoint here. The base
241 // `VehicleObstacleDetected` honors `max_distance` internally, so we let
242 // it do the proximity gating from the world-wide walker list.
243 (void)waypoint; // location consumed indirectly via VehicleObstacleDetected
244 VehicleObstacle obs;
245 if (_direction == RoadOption::ChangeLaneLeft) {
246 obs = VehicleObstacleDetected(
247 walkers_alist,
248 std::max(_behavior.min_proximity_threshold, _speed_limit / 2.0f),
249 /*up_angle_th=*/90.0f,
250 /*low_angle_th=*/0.0f,
251 /*lane_offset=*/-1);
252 } else if (_direction == RoadOption::ChangeLaneRight) {
253 obs = VehicleObstacleDetected(
254 walkers_alist,
255 std::max(_behavior.min_proximity_threshold, _speed_limit / 2.0f),
256 /*up_angle_th=*/90.0f,
257 /*low_angle_th=*/0.0f,
258 /*lane_offset=*/1);
259 } else {
260 obs = VehicleObstacleDetected(
261 walkers_alist,
262 std::max(_behavior.min_proximity_threshold, _speed_limit / 3.0f),
263 /*up_angle_th=*/60.0f,
264 /*low_angle_th=*/0.0f,
265 /*lane_offset=*/0);
266 }
267 return obs;

274rpc::VehicleControl BehaviorAgent::CarFollowingManager(
275 SharedPtr<client::Actor> vehicle, float distance, bool debug) {
276 const float vehicle_speed = GetSpeed(vehicle);
277 const float delta_v = std::max(1.0f, (_speed - vehicle_speed) / 3.6f);
278 const float ttc = (delta_v != 0.0f)
279 ? distance / delta_v
280 : distance / std::nextafter(0.0f, 1.0f);
281
282 float target_speed = 0.0f;
283 if (_behavior.safety_time > ttc && ttc > 0.0f) {
284 target_speed = std::min({
285 Positive(vehicle_speed - _behavior.speed_decrease),
286 _behavior.max_speed,
287 _speed_limit - _behavior.speed_lim_dist});
288 } else if (2.0f * _behavior.safety_time > ttc && ttc >= _behavior.safety_time) {
289 target_speed = std::min({
290 std::max(_min_speed, vehicle_speed),
291 _behavior.max_speed,
292 _speed_limit - _behavior.speed_lim_dist});
293 } else {
294 target_speed = std::min({
295 _behavior.max_speed,
296 _speed_limit - _behavior.speed_lim_dist});
297 }
298 _local_planner->SetSpeed(target_speed);
299 return _local_planner->RunStep(debug);
300}

302// ---------------------------------------------------------------------------
303// Main run-step
304// ---------------------------------------------------------------------------
305
306rpc::VehicleControl BehaviorAgent::RunStep(bool debug) {
307 UpdateInformation();
308
309 if (_behavior.tailgate_counter > 0) {
310 _behavior.tailgate_counter -= 1;
311 }
312
313 const auto ego_vehicle_loc = _vehicle->GetLocation();
314 auto ego_vehicle_wp = _map->GetWaypoint(ego_vehicle_loc);
315
316 // 1. Red lights / stops.
317 if (TrafficLightManager()) {
318 return EmergencyStop();
319 }
320
321 // 2.1: Pedestrian avoidance.
322 auto walker_obs = PedestrianAvoidManager(ego_vehicle_wp);
323 if (walker_obs.affected && walker_obs.actor) {
324 const auto &walker_extent = walker_obs.actor->GetBoundingBox().extent;
325 const auto &ego_extent = _vehicle->GetBoundingBox().extent;
326 const float distance = walker_obs.distance
327 - std::max(walker_extent.y, walker_extent.x)
328 - std::max(ego_extent.y, ego_extent.x);
329 if (distance < _behavior.braking_distance) {
330 return EmergencyStop();
331 }
332 }
333
334 // 2.2: Vehicle avoidance / car following.
335 auto vehicle_obs = CollisionAndCarAvoidManager(ego_vehicle_wp);
336 rpc::VehicleControl control{};
337 if (vehicle_obs.affected && vehicle_obs.actor) {
338 const auto &v_extent = vehicle_obs.actor->GetBoundingBox().extent;
339 const auto &ego_extent = _vehicle->GetBoundingBox().extent;
340 const float distance = vehicle_obs.distance
341 - std::max(v_extent.y, v_extent.x)
342 - std::max(ego_extent.y, ego_extent.x);
343 if (distance < _behavior.braking_distance) {
344 return EmergencyStop();
345 }
346 control = CarFollowingManager(vehicle_obs.actor, distance, debug);
347 }
348 // 3: Intersection behaviour (left/right turn).
349 else if (_incoming_waypoint && _incoming_waypoint->IsJunction() &&
350 (_incoming_direction == RoadOption::Left ||
351 _incoming_direction == RoadOption::Right)) {
352 const float target_speed = std::min(_behavior.max_speed,
353 _speed_limit - 5.0f);
354 _local_planner->SetSpeed(target_speed);
355 control = _local_planner->RunStep(debug);
356 }
357 // 4: Normal behaviour.
358 else {
359 const float target_speed = std::min(_behavior.max_speed,
360 _speed_limit - _behavior.speed_lim_dist);
361 _local_planner->SetSpeed(target_speed);
362 control = _local_planner->RunStep(debug);
363 }
364 return control;
365}

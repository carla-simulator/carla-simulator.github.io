119void TrafficManagerLocal::SetupLocalMap() {
120 const carla::SharedPtr<const cc::Map> world_map = world.GetMap();
121 local_map = std::make_shared<InMemoryMap>(world_map);
122
123 auto files = episode_proxy.Lock()->GetRequiredFiles("TM");
124 if (!files.empty()) {
125 auto content = episode_proxy.Lock()->GetCacheFile(files[0], true);
126 if (content.size() != 0) {
127 if (!local_map->Load(content)) {
128 // The copy in the client's file cache (~/carlaCache) is kept for ever once
129 // downloaded, so a rejected cache is most likely a stale local copy: fetch
130 // the server's current file once and try again before rebuilding from the
131 // OpenDRIVE.
132 log_warning("fetching the Traffic Manager cache", files[0], "from the server again");
133 episode_proxy.Lock()->RequestFile(files[0]);
134 content = episode_proxy.Lock()->GetCacheFile(files[0], false);
135 local_map = std::make_shared<InMemoryMap>(world_map);
136 if (content.size() == 0 || !local_map->Load(content)) {
137 local_map = std::make_shared<InMemoryMap>(world_map);
138 local_map->SetUp();
139 }
140 }
141 } else {
142 log_warning("No InMemoryMap cache found. Setting up local map. This may take a while...");
143 local_map->SetUp();
144 }
145 } else {
146 log_warning("No InMemoryMap cache found. Setting up local map. This may take a while...");
147 local_map->SetUp();
148 }
149}

166void TrafficManagerLocal::Step() {
167 bool synchronous_mode = parameters.GetSynchronousMode();
168 bool hybrid_physics_mode = parameters.GetHybridPhysicsMode();
169
170 // Skipping velocity update if elapsed time is less than 0.05s in asynchronous, hybrid mode.
171 if (!synchronous_mode && hybrid_physics_mode) {
172 TimePoint current_instance = chr::system_clock::now();
173 chr::duration<float> elapsed_time = current_instance - previous_update_instance;
174 chr::duration<float> time_to_wait = chr::duration<float>(HYBRID_MODE_DT) - elapsed_time;
175 if (time_to_wait > chr::duration<float>(0.0f)) {
176 std::this_thread::sleep_for(time_to_wait);
177 }
178 previous_update_instance = current_instance;
179 }
180
181 // Read from the episode state the client already holds, so no round trip.
182 const carla::client::Timestamp timestamp = world.GetSnapshot().GetTimestamp();
183
184 // Stop TM from processing the same frame more than once
185 if (!synchronous_mode) {
186 if (timestamp.frame == last_frame) {
187 std::this_thread::sleep_for(SNAPSHOT_POLL_PERIOD);
188 return;
189 }
190 last_frame = timestamp.frame;
191 }
192
193 if (synchronous_mode ||
194 IsRefreshDue(timestamp.elapsed_seconds, last_settings_update, EPISODE_SETTINGS_REFRESH_PERIOD)) {
195 episode_settings = episode_proxy.Lock()->GetEpisodeSettings();
196 last_settings_update = timestamp.elapsed_seconds;
197 }
198 parameters.SetMaxBoundaries(20.0f, episode_settings.actor_active_distance);
199
200 std::unique_lock<std::mutex> registration_lock(registration_mutex);
201 // Updating simulation state, actor life cycle and performing necessary cleanup.
202 alsm.Update();
203
204 // Re-allocating inter-stage communication frames based on changed number of registered vehicles.
205 int current_registered_vehicles_state = registered_vehicles.GetState();
206 unsigned long number_of_vehicles = vehicle_id_list.size();
207 if (registered_vehicles_state != current_registered_vehicles_state || number_of_vehicles != registered_vehicles.Size()) {
208 vehicle_id_list = registered_vehicles.GetIDList();
209 number_of_vehicles = vehicle_id_list.size();
210
211 // Reserve more space if needed.
212 uint64_t growth_factor = static_cast<uint64_t>(static_cast<float>(number_of_vehicles) * INV_GROWTH_STEP_SIZE);
213 uint64_t new_frame_capacity = INITIAL_SIZE + GROWTH_STEP_SIZE * growth_factor;
214 if (new_frame_capacity > current_reserved_capacity) {
215 localization_frame.reserve(new_frame_capacity);
216 collision_frame.reserve(new_frame_capacity);
217 tl_frame.reserve(new_frame_capacity);
218 control_frame.reserve(new_frame_capacity);
219 }
220
221 registered_vehicles_state = registered_vehicles.GetState();
222 }
223
224 // Reset frames for current cycle.
225 localization_frame.clear();
226 localization_frame.resize(number_of_vehicles);
227 collision_frame.clear();
228 collision_frame.resize(number_of_vehicles);
229 tl_frame.clear();
230 tl_frame.resize(number_of_vehicles);
231 control_frame.clear();
232 // Reserve two frames for each vehicle: one for the ApplyVehicleControl command,
233 // and one for the optional SetVehicleLightState command
234 control_frame.reserve(2 * number_of_vehicles);
235 // Resize to accomodate at least all ApplyVehicleControl commands,
236 // that will be inserted by the motion_plan_stage stage.
237 control_frame.resize(number_of_vehicles);
238
239 // Run core operation stages.
240 for (unsigned long index = 0u; index < vehicle_id_list.size(); ++index) {
241 localization_stage.Update(index);
242 }
243 for (unsigned long index = 0u; index < vehicle_id_list.size(); ++index) {
244 if (!localization_frame[index].localized) {
245 continue;
246 }
247 collision_stage.Update(index);
248 }
249 collision_stage.ClearCycleCache();
250 vehicle_light_stage.UpdateWorldInfo(timestamp.elapsed_seconds, synchronous_mode);
251 motion_plan_stage.SetCycleTimestamp(timestamp);
252 for (unsigned long index = 0u; index < vehicle_id_list.size(); ++index) {
253 if (!localization_frame[index].localized) {
254 continue;
255 }
256 traffic_light_stage.Update(index);
257 motion_plan_stage.Update(index);
258 vehicle_light_stage.Update(index);
259 }
260
261 // A vehicle skipped above never had its control command written, and the
262 // default-constructed command would reach the server as an empty spawn
263 // request ("Invalid ActorDescription '' (UId=0)"). Drop those slots; the
264 // commands appended past the per-vehicle range (vehicle lights) are kept.
265 const bool any_skipped = std::any_of(
266 localization_frame.begin(),
267 localization_frame.begin() + static_cast<long>(vehicle_id_list.size()),
268 [](const LocalizationData &data) { return !data.localized; });
269 if (any_skipped) {
270 ControlFrame filtered_frame;
271 filtered_frame.reserve(control_frame.size());
272 for (unsigned long index = 0u; index < control_frame.size(); ++index) {
273 if (index < vehicle_id_list.size() && !localization_frame[index].localized) {
274 continue;
275 }
276 filtered_frame.push_back(std::move(control_frame[index]));
277 }
278 control_frame = std::move(filtered_frame);
279 }
280
281 registration_lock.unlock();
282
283 // Sending the current cycle's batch command to the simulator. Synchronous
284 // mode must apply it before the next tick; async waiting costs a server frame.
285 if (synchronous_mode) {
286 episode_proxy.Lock()->ApplyBatchSync(control_frame, false);
287 } else if (!control_frame.empty()) {
288 // An unwaited batch is acknowledged on receipt, not on application, so
289 // periodically waiting is what bounds the queue ahead of the game thread.
290 if (IsBatchSyncDue(unwaited_control_batches, MAX_UNWAITED_CONTROL_BATCHES)) {
291 episode_proxy.Lock()->ApplyBatchSync(control_frame, false);
292 unwaited_control_batches = 0u;
293 } else {
294 episode_proxy.Lock()->ApplyBatch(control_frame, false);
295 ++unwaited_control_batches;
296 }
297 }
298}

313void TrafficManagerLocal::Stop() {
314 run_traffic_manger.store(false);
315
316 // Join regardless of mode, or Start() may overwrite a joinable thread.
317 if (worker_thread) {
318 if (worker_thread->joinable()) {
319 worker_thread->join();
320 }
321 worker_thread.reset();
322 }
323
324 std::unique_lock<std::mutex> registration_lock(registration_mutex);
325 vehicle_id_list.clear();
326 registered_vehicles.Clear();
327 registered_vehicles_state = -1;
328 track_traffic.Clear();
329 previous_update_instance = chr::system_clock::now();
330 current_reserved_capacity = 0u;
331
332 simulation_state.Reset();
333 localization_stage.Reset();
334 collision_stage.Reset();
335 traffic_light_stage.Reset();
336 motion_plan_stage.Reset();
337 vehicle_light_stage.Reset();
338
339 last_frame = 0;
340 episode_settings = rpc::EpisodeSettings{};
341 last_settings_update = -std::numeric_limits<double>::infinity();
342 unwaited_control_batches = MAX_UNWAITED_CONTROL_BATCHES;
343
344 buffer_map.clear();
345 localization_frame.clear();
346 collision_frame.clear();
347 tl_frame.clear();
348 control_frame.clear();
349 registration_lock.unlock();
350
351 run_traffic_manger.store(true);
352}

361void TrafficManagerLocal::Reset(uint64_t episode_id) {
362 // Every client alive in the process reports the change; reset once.
363 if (episode_proxy.GetId() == episode_id) {
364 return;
365 }
366 // Our client may see the change after another one; wait, or we'd reset onto the old episode.
367 auto simulator = episode_proxy.Lock();
368 const auto deadline = std::chrono::steady_clock::now() + simulator->GetNetworkingTimeout().to_chrono();
369 while (simulator->GetCurrentEpisodeId() != episode_id && std::chrono::steady_clock::now() < deadline) {
370 std::this_thread::sleep_for(std::chrono::milliseconds(10));
371 }
372 Release();
373 episode_proxy = simulator->GetCurrentEpisode();
374 world = cc::World(episode_proxy);
375 SetupLocalMap();
376 Start();
377}

542void TrafficManagerLocal::SetSynchronousMode(bool mode) {
543 const bool previously_synchronous = parameters.GetSynchronousMode();
544 parameters.SetSynchronousMode(mode);
545 if (mode == previously_synchronous) { // Same mode as before, do nothing.
546 return;
547 }
548
549 if (!previously_synchronous) { // Switch from async to synchronous
550 if (worker_thread) {
551 if (worker_thread->joinable()) {
552 worker_thread->join();
553 }
554 worker_thread.reset();
555 }
556 }
557 else { // Switch from synchronous to async, start worker thread again.
558 if (worker_thread) {
559 // A worker thread should not exist while in synchronous mode. Tear it
560 // down before spawning a fresh one so a buggy prior transition does
561 // not leak running threads.
562 log_error("TrafficManagerLocal: worker thread present in synchronous mode; joining before async restart.");
563 if (worker_thread->joinable()) {
564 worker_thread->join();
565 }
566 worker_thread.reset();
567 }
568 worker_thread = std::make_unique<std::thread>(&TrafficManagerLocal::Run, this);
569 }
570}

23bool AShaderBasedSensor::AddPostProcessingMaterial(const FString &Path)
24{
25 // ConstructorHelpers::FObjectFinder is documented as constructor-only and
26 // has undefined behavior when used from runtime functions. FSoftObjectPath
27 // is the runtime-safe path resolver; it accepts both plain object paths
28 // and the class-prefixed export-text form ("Material'/Game/.../Foo'") that
29 // FObjectFinder previously normalised internally.
30 UMaterial *Material = Cast<UMaterial>(FSoftObjectPath(Path).TryLoad());
31 if (Material != nullptr)
32 {
33 MaterialsFound.Add(Material);
34 return true;
35 }
36 UE_LOG(LogCarla, Error,
37 TEXT("AShaderBasedSensor::AddPostProcessingMaterial: failed to load material '%s'"),
38 *Path);
39 return false;
40}

42void AShaderBasedSensor::SetUpSceneCaptureComponent(USceneCaptureComponent2D &SceneCapture)
43{
44 for (const auto &MaterialFound : MaterialsFound)
45 {
46 // Create a dynamic instance of the Material (Shader)
47 AddShader({UMaterialInstanceDynamic::Create(MaterialFound, this), 1.0});
48 }
49
50 for (const auto &Shader : Shaders)
51 {
52 // Attach the instance into the blendables
53 SceneCapture.PostProcessSettings.AddBlendable(Shader.PostProcessMaterial, Shader.Weight);
54 }
55
56 // Set the value for each Float parameter in the shader
57 for (const auto &ParameterValue : FloatShaderParams)
58 {
59 Shaders[ParameterValue.ShaderIndex].PostProcessMaterial->SetScalarParameterValue(
60 ParameterValue.ParameterName,
61 ParameterValue.Value);
62 }
63}

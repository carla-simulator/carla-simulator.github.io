32/// Data stored in a road map pixel. See URoadMap.
33struct FRoadMapPixelData
34{
35 friend class URoadMap;
36
37 constexpr static int IsRoadRow = 15;
38
39 constexpr static int HasDirectionRow = 14;
40
41 constexpr static uint16 MaximumEncodedAngle = (1 << 14) - 1;
42
43 constexpr static uint16 AngleMask = (0xFFFF >> 2);
44
45public:
46
47 explicit FRoadMapPixelData(uint16 inValue) : Value(inValue) {}
48
49 /// Whether this pixel lies in-road.
50 bool IsRoad() const
51 {
52 return (Value & (1 << IsRoadRow)) != 0;
53 }
54
55 /// Whether this pixel has a direction defined (e.g. road intersections are
56 /// not off-road but neither have defined direction).
57 bool HasDirection() const
58 {
59 return (Value & (1 << HasDirectionRow)) != 0;
60 }
61
62 /// Get the azimuth angle [-PI, PI] of the road direction (in spherical
63 /// coordinates) at this pixel.
64 ///
65 /// Undefined if !HasDirection().
66 float GetDirectionAzimuthalAngle() const
67 {
68 const float Angle = AngleMask & Value;
69 // Internally the angle is rotated by PI.
70 return (Angle * 2.0f * PI / MaximumEncodedAngle) - PI;
71 }
72
73 /// Get the road direction at this pixel.
74 ///
75 /// Undefined if !HasDirection().
76 FVector GetDirection() const
77 {
78 const FVector2D SphericalCoords(HALF_PI, GetDirectionAzimuthalAngle());
79 return SphericalCoords.SphericalToUnitCartesian();
80 }
81
82 FColor EncodeAsColor() const;
83
84private:
85
86 static uint16 Encode(bool IsRoad, bool HasDirection, const FVector &Direction);
87
88 uint16 Value;
89};

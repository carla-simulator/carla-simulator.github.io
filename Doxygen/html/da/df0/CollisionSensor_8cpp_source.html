41void ACollisionSensor::OnCollisionEvent(
42 AActor* Actor,
43 AActor* OtherActor,
44 FVector NormalImpulse,
45 const FHitResult& Hit)
46{
47 if (Actor == nullptr || OtherActor == nullptr)
48 {
49 return;
50 }
51
52 uint64_t CurrentFrame = FCarlaEngine::GetFrameCounter();
53
54 // remove all items from previous frames
55 CollisionRegistry.erase(
56 std::remove_if(
57 CollisionRegistry.begin(),
58 CollisionRegistry.end(),
59 [CurrentFrame](std::tuple<uint64_t, AActor*, AActor*> Item)
60 {
61 return std::get<0>(Item) < CurrentFrame;
62 }),
63 CollisionRegistry.end());
64
65 // check if this collision has been procesed already in this frame
66 for (auto& Collision : CollisionRegistry)
67 {
68 if (std::get<0>(Collision) == CurrentFrame &&
69 std::get<1>(Collision) == Actor &&
70 std::get<2>(Collision) == OtherActor)
71 {
72 return;
73 }
74 }
75
76 const auto& CurrentEpisode = GetEpisode();
77 constexpr float TO_METERS = 1e-2;
78 NormalImpulse *= TO_METERS;
79 GetDataStream(*this).SerializeAndSend(
80 *this,
81 CurrentEpisode.SerializeActor(Actor),
82 CurrentEpisode.SerializeActor(OtherActor),
83 carla::geom::Vector3D(
84 (float)NormalImpulse.X,
85 (float)NormalImpulse.Y,
86 (float)NormalImpulse.Z));
87 // record the collision event
88 if (CurrentEpisode.GetRecorder()->IsEnabled()) {
89 CurrentEpisode.GetRecorder()->AddCollision(Actor, OtherActor);
90 }
91
92 CollisionRegistry.emplace_back(CurrentFrame, Actor, OtherActor);
93
94 // ROS2
95#if defined(WITH_ROS2)
96 auto ROS2 = carla::ros2::ROS2::GetInstance();
97 if (ROS2->IsEnabled())
98 {
99 TRACE_CPUPROFILER_EVENT_SCOPE_STR("ROS2 Send");
100 auto StreamId = carla::streaming::detail::token_type(GetToken()).get_stream_id();
101 auto NormalImpulseVector = carla::geom::Vector3D(
102 (float)NormalImpulse.X,
103 (float)NormalImpulse.Y,
104 (float)NormalImpulse.Z);
105 AActor* ParentActor = GetAttachParentActor();
106 if (ParentActor)
107 {
108 FTransform LocalTransformRelativeToParent = GetActorTransform().GetRelativeTransform(ParentActor->GetActorTransform());
109 ROS2->ProcessDataFromCollisionSensor(
110 0, StreamId,
111 LocalTransformRelativeToParent,
112 OtherActor->GetUniqueID(),
113 NormalImpulseVector,
114 this);
115 }
116 else
117 {
118 ROS2->ProcessDataFromCollisionSensor(
119 0, StreamId,
120 GetActorTransform(),
121 OtherActor->GetUniqueID(),
122 NormalImpulseVector,
123 this);
124 }
125 }
126#endif
127}

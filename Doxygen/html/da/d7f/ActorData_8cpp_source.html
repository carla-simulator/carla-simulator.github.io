26FActorRigSnapshot FActorRigSnapshot::FromActor(const AActor *Actor)
27{
28 FActorRigSnapshot Rig;
29 if (Actor == nullptr)
30 {
31 return Rig;
32 }
33 const FTransform ActorToWorld = Actor->GetActorTransform();
34
35 TArray<UActorComponent *> Components;
36 Actor->GetComponents(Components);
37 for (UActorComponent *ActorComponent : Components)
38 {
39 // Scene components only: the others have no transform to report.
40 USceneComponent *SceneComponent = Cast<USceneComponent>(ActorComponent);
41 if (SceneComponent == nullptr)
42 {
43 continue;
44 }
45 Rig.ComponentNames.Add(SceneComponent->GetName());
46 Rig.ComponentTransforms.Add(
47 SceneComponent->GetComponentTransform().GetRelativeTransform(ActorToWorld));
48
49 // Authored sockets only. A skinned mesh also reports each of its bones as a
50 // socket, which duplicates the bone lists and can repeat a name.
51 TArray<FComponentSocketDescription> Sockets;
52 SceneComponent->QuerySupportedSockets(Sockets);
53 for (const FComponentSocketDescription &Socket : Sockets)
54 {
55 if (Socket.Type != EComponentSocketType::Socket)
56 {
57 continue;
58 }
59 Rig.SocketNames.Add(Socket.Name.ToString());
60 Rig.SocketTransforms.Add(
61 SceneComponent->GetSocketTransform(Socket.Name, ERelativeTransformSpace::RTS_Actor));
62 }
63 }
64
65 TArray<USkinnedMeshComponent *> SkinnedMeshComponents;
66 Actor->GetComponents<USkinnedMeshComponent>(SkinnedMeshComponents);
67 for (USkinnedMeshComponent *SkinnedMeshComponent : SkinnedMeshComponents)
68 {
69 const FTransform ComponentToWorld = SkinnedMeshComponent->GetComponentTransform();
70 TArray<FName> BoneNames;
71 SkinnedMeshComponent->GetBoneNames(BoneNames);
72 // GetBoneNames walks the reference skeleton in bone-index order, the same
73 // order as GetBoneTransform, so names and transforms stay index-aligned.
74 const int32 NumBones = FMath::Min(BoneNames.Num(), SkinnedMeshComponent->GetNumBones());
75 for (int32 BoneIndex = 0; BoneIndex < NumBones; ++BoneIndex)
76 {
77 Rig.BoneNames.Add(BoneNames[BoneIndex].ToString());
78 Rig.BoneTransforms.Add(
79 SkinnedMeshComponent->GetBoneTransform(BoneIndex, ComponentToWorld)
80 .GetRelativeTransform(ActorToWorld));
81 }
82 }
83 return Rig;
84}

94void FActorData::RecordActorData(FCarlaActor* CarlaActor, UCarlaEpisode* CarlaEpisode)
95{
96 AActor* Actor = CarlaActor->GetActor();
97 FTransform Transform = Actor->GetTransform();
98 Location = FDVector(Transform.GetLocation()) + CarlaEpisode->GetCurrentMapOrigin();
99 Rotation = Transform.GetRotation();
100 Scale = Transform.GetScale3D();
101 UPrimitiveComponent* Component = Cast<UPrimitiveComponent>(Actor->GetRootComponent());
102 if (Component)
103 {
104 bSimulatePhysics = Component->IsSimulatingPhysics();
105 AngularVelocity = Component->GetPhysicsAngularVelocityInDegrees();
106 }
107 Velocity = Actor->GetVelocity();
108 // The AActor is destroyed right after this, so the rig has to be kept here or
109 // every component, bone and socket query fails for as long as it is dormant.
110 Rig = FActorRigSnapshot::FromActor(Actor);
111}

113void FActorData::RestoreActorData(FCarlaActor* CarlaActor, UCarlaEpisode* CarlaEpisode)
114{
115 AActor* Actor = CarlaActor->GetActor();
116 Actor->SetActorTransform(GetLocalTransform(CarlaEpisode));
117 UPrimitiveComponent* Component = Cast<UPrimitiveComponent>(Actor->GetRootComponent());
118 if (Component)
119 {
120 Component->SetPhysicsLinearVelocity(
121 Velocity, false, "None");
122 Component->SetPhysicsAngularVelocityInDegrees(
123 AngularVelocity, false, "None");
124 }
125 auto* Character = Cast<ACharacter>(Actor);
126 auto* CarlaVehicle = Cast<ACarlaWheeledVehicle>(Actor);
127 // The physics in the vehicles works in a different way so to disable them.
128 if (CarlaVehicle != nullptr){
129 CarlaVehicle->SetSimulatePhysics(bSimulatePhysics);
130 }
131 // The physics in the walkers also works in a different way so to disable them,
132 // we need to do it in the UCharacterMovementComponent.
133 else if (Character != nullptr)
134 {
135
136 }
137 // In the rest of actors, the physics is controlled with the UPrimitiveComponent, so we use
138 // that for disable it.
139 else
140 {
141 auto RootComponent = Component;
142 if (RootComponent != nullptr)
143 {
144 RootComponent->SetSimulatePhysics(bSimulatePhysics);
145 RootComponent->SetCollisionEnabled(ECollisionEnabled::QueryAndPhysics);
146 }
147 }
148}

156void FVehicleData::RecordActorData(FCarlaActor* CarlaActor, UCarlaEpisode* CarlaEpisode)
157{
158 FActorData::RecordActorData(CarlaActor, CarlaEpisode);
159 AActor* Actor = CarlaActor->GetActor();
160 ACarlaWheeledVehicle* Vehicle = Cast<ACarlaWheeledVehicle>(Actor);
161 if (bSimulatePhysics)
162 {
163 PhysicsControl = Vehicle->GetVehiclePhysicsControl();
164 }
165 Control = Vehicle->GetVehicleControl();
166 AckermannControl = Vehicle->GetVehicleAckermannControl();
167 bAckermannControlActive = Vehicle->IsAckermannControlActive();
168 AckermannControllerSettings = Vehicle->GetAckermannControllerSettings();
169 LightState = Vehicle->GetVehicleLightState();
170 auto Controller = Cast<AWheeledVehicleAIController>(Vehicle->GetController());
171 if (Controller)
172 {
173 SpeedLimit = Controller->GetSpeedLimit();
174 }
175 FailureState = Vehicle->GetFailureState();
176}

178void FVehicleData::RestoreActorData(FCarlaActor* CarlaActor, UCarlaEpisode* CarlaEpisode)
179{
180 FActorData::RestoreActorData(CarlaActor, CarlaEpisode);
181 AActor* Actor = CarlaActor->GetActor();
182 ACarlaWheeledVehicle* Vehicle = Cast<ACarlaWheeledVehicle>(Actor);
183 Vehicle->SetSimulatePhysics(bSimulatePhysics);
184 if (bSimulatePhysics)
185 {
186 Vehicle->ApplyVehiclePhysicsControl(PhysicsControl);
187 }
188 Vehicle->ApplyAckermannControllerSettings(AckermannControllerSettings);
189 if (!bAckermannControlActive)
190 {
191 Vehicle->ApplyVehicleControl(Control, EVehicleInputPriority::Client);
192 }
193 else
194 {
195 Vehicle->ApplyVehicleAckermannControl(AckermannControl, EVehicleInputPriority::Client);
196 }
197 Vehicle->SetVehicleLightState(LightState);
198 auto Controller = Cast<AWheeledVehicleAIController>(Vehicle->GetController());
199 if (Controller)
200 {
201 Controller->SetSpeedLimit(SpeedLimit);
202 }
203 Vehicle->SetFailureState(FailureState);
204}

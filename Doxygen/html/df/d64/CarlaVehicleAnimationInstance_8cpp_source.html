16static float RollWheel(float SpinAngle, float Travel, float Radius, float MaxTravel)
17{
18 const bool bHasRadius = Radius > UE_KINDA_SMALL_NUMBER;
19 const bool bTeleported = FMath::Abs(Travel) > MaxTravel;
20 if (!bHasRadius || bTeleported)
21 {
22 return SpinAngle;
23 }
24 return FMath::UnwindDegrees(SpinAngle - FMath::RadiansToDegrees(Travel / Radius));
25}

27void UCarlaVehicleAnimationInstance::NativeUpdateAnimation(float DeltaSeconds)
28{
29 Super::NativeUpdateAnimation(DeltaSeconds);
30
31 const ACarlaWheeledVehicle *Vehicle = Cast<ACarlaWheeledVehicle>(GetOwningActor());
32 if (Vehicle == nullptr)
33 {
34 return;
35 }
36 UpdateWheelAnimationOverride(*Vehicle);
37 if (!ShouldRollWheels(*Vehicle))
38 {
39 bRollingWheels = false;
40 // So the first frame of rolling turns the wheels by that frame's travel.
41 TrackWheelLocations();
42 return;
43 }
44 if (!bRollingWheels)
45 {
46 BeginRolling();
47 bRollingWheels = true;
48 }
49 RollWheels(*Vehicle, DeltaSeconds);
50}

59void UCarlaVehicleAnimationInstance::InitWheels()
60{
61 const UChaosWheeledVehicleMovementComponent *Movement = GetWheeledVehicleComponent();
62 const FTransform MeshTransform = GetSkelMeshComponent()->GetComponentTransform();
63 const USkeletalMesh *SkeletalMesh = GetSkelMeshComponent()->GetSkeletalMeshAsset();
64
65 Wheels.Init(FWheel(), Movement->WheelSetups.Num());
66 for (int32 i = 0; i < Wheels.Num(); ++i)
67 {
68 FWheel &Wheel = Wheels[i];
69 if (SkeletalMesh != nullptr)
70 {
71 const FReferenceSkeleton &RefSkeleton = SkeletalMesh->GetRefSkeleton();
72 const int32 BoneIndex = RefSkeleton.FindBoneIndex(Movement->WheelSetups[i].BoneName);
73 if (BoneIndex != INDEX_NONE)
74 {
75 Wheel.RestLocation =
76 FAnimationRuntime::GetComponentSpaceTransformRefPose(RefSkeleton, BoneIndex).GetLocation();
77 }
78 }
79 Wheel.LastLocation = MeshTransform.TransformPosition(Wheel.RestLocation);
80 }
81}

86 if (Movement == nullptr)

102void UCarlaVehicleAnimationInstance::BeginRolling()
103{
104 if (Wheels.Num() != GetWheeledVehicleComponent()->WheelSetups.Num())
105 {
106 InitWheels();
107 }
108 const TArray<FWheelAnimationData> &WheelPoses = GetWheelPoses();
109 for (int32 i = 0; i < Wheels.Num(); ++i)
110 {
111 // Carry on from the angle Chaos left.
112 Wheels[i].SpinAngle = WheelPoses.IsValidIndex(i) ? WheelPoses[i].RotOffset.Pitch : 0.0f;
113 }

116void UCarlaVehicleAnimationInstance::RollWheels(const ACarlaWheeledVehicle &Vehicle, float DeltaSeconds)
117{
118 TArray<FWheelAnimationData> &WheelPoses = GetWheelPoses();
119 const FTransform MeshTransform = GetSkelMeshComponent()->GetComponentTransform();
120 const FVector Forward = MeshTransform.GetUnitAxis(EAxis::X);
121 const float MaxTravel = MaxTravelSpeed * DeltaSeconds;
122
123 for (int32 i = 0; i < Wheels.Num() && i < WheelPoses.Num(); ++i)
124 {
125 FWheel &Wheel = Wheels[i];
126 const FVector Location = MeshTransform.TransformPosition(Wheel.RestLocation);
127 const float Travel = FVector::DotProduct(Location - Wheel.LastLocation, Forward);
128 Wheel.SpinAngle = RollWheel(Wheel.SpinAngle, Travel, Vehicle.GetWheelRadius(i), MaxTravel);
129 Wheel.LastLocation = Location;
130 WheelPoses[i].RotOffset.Pitch = Wheel.SpinAngle;
131 }
132}

134void UCarlaVehicleAnimationInstance::UpdateWheelAnimationOverride(const ACarlaWheeledVehicle &Vehicle)
135{
136 const bool bOverridden = Vehicle.IsWheelAnimationOverridden();
137 TArray<FWheelAnimationData> &WheelPoses = GetWheelPoses();
138 if (bOverridden)
139 {
140 const TArray<FWheelAnimationData> &OverriddenPoses = Vehicle.GetOverriddenWheelPoses();
141 for (int32 i = 0; i < WheelPoses.Num() && i < OverriddenPoses.Num(); ++i)
142 {
143 WheelPoses[i].RotOffset = OverriddenPoses[i].RotOffset;
144 WheelPoses[i].LocOffset = OverriddenPoses[i].LocOffset;
145 }
146 }
147 else if (bWasWheelAnimationOverridden && !Vehicle.IsSimulatedByChaos())
148 {
149 // Nothing else steers or compresses the wheels without Chaos. The roll
150 // carries on from the pitch left.
151 for (FWheelAnimationData &Pose : WheelPoses)
152 {
153 Pose.RotOffset.Yaw = 0.0f;
154 Pose.LocOffset = FVector::ZeroVector;
155 }
156 }
157 bWasWheelAnimationOverridden = bOverridden;
158}

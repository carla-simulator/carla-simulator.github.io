52 void Primary::Open(
53 Listener::callback_function_type on_opened,
54 Listener::callback_function_type on_closed,
55 Listener::callback_function_type_response on_response) {
56 DEBUG_ASSERT(on_opened && on_closed);
57
58 // This forces not using Nagle's algorithm.
59 // Improves the sync mode velocity on Linux by a factor of ~3.
60 const boost::asio::ip::tcp::no_delay option(true);
61 _socket.set_option(option);
62 EnableKeepAlive(_socket);
63
64 // callbacks
65 _on_closed = std::move(on_closed);
66 _on_response = std::move(on_response);
67 on_opened(shared_from_this());
68
69 ReadData();
70 }

72 void Primary::Write(std::shared_ptr<const carla::streaming::detail::tcp::Message> message) {
73 DEBUG_ASSERT(message != nullptr);
74 DEBUG_ASSERT(!message->empty());
75 std::weak_ptr<Primary> weak = shared_from_this();
76 boost::asio::post(_strand, [weak, message]() {
77 auto self = weak.lock();
78 if (!self) return;
79 if (!self->_socket.is_open()) {
80 return;
81 }
82 self->_write_queue.push_back(message);
83 if (!self->_is_writing) {
84 self->WriteNext();
85 }
86 });
87 }

89 void Primary::WriteNext() {
90 if (_write_queue.empty() || !_socket.is_open()) {
91 _write_queue.clear();
92 _is_writing = false;
93 return;
94 }
95 _is_writing = true;
96 const message_type message = _write_queue.front();
97 std::weak_ptr<Primary> weak = shared_from_this();
98 auto handle_sent = [weak, message](const boost::system::error_code &ec, size_t) {
99 auto self = weak.lock();
100 if (!self) return;
101 if (ec) {
102 self->_write_queue.clear();
103 self->_is_writing = false;
104 // A write aborted by our own close needs no second close. Otherwise
105 // close the socket too, so that later writes are dropped instead of
106 // failing (and closing the session) again.
107 if (self->_socket.is_open()) {
108 log_error("session ", self->_session_id, ": error sending data: ", ec.message());
109 self->CloseNow();
110 }
111 return;
112 }
113 self->_write_queue.pop_front();
114 self->WriteNext();
115 };
116
117 _deadline.expires_after(_timeout.to_chrono());
118 boost::asio::async_write(
119 _socket,
120 message->GetBufferSequence(),
121 boost::asio::bind_executor(_strand, handle_sent));
122 }

130 void Primary::ReadData() {
131 std::weak_ptr<Primary> weak = shared_from_this();
132 boost::asio::post(_strand, [weak]() {
133 auto self = weak.lock();
134 if (!self) return;
135
136 auto message = std::make_shared<IncomingMessage>(self->_buffer_pool->Pop());
137
138 auto handle_read_data = [weak, message](boost::system::error_code ec, size_t DEBUG_ONLY(bytes)) {
139 auto self = weak.lock();
140 if (!self) return;
141 if (!ec) {
142 DEBUG_ASSERT_EQ(bytes, message->size());
143 DEBUG_ASSERT_NE(bytes, 0u);
144 // Move the buffer to the callback function and start reading the next
145 // piece of data.
146 self->_on_response(self, message->pop());
147 std::cout << "Getting data on listener\n";
148 self->ReadData();
149 } else {
150 // As usual, if anything fails start over from the very top.
151 log_error("primary server: failed to read data: ", ec.message());
152 }
153 };
154
155 auto handle_read_header = [weak, message, handle_read_data](
156 boost::system::error_code ec,
157 size_t) {
158 auto self = weak.lock();
159 if (!self) return;
160 if (!ec && (message->size() > kMaxMultiGpuMessageBytes)) {
161 log_error("primary server: secondary announced a", message->size(),
162 "byte message, larger than any valid one; closing the session");
163 self->Close();
164 } else if (!ec && (message->size() > 0u)) {
165 // Now that we know the size of the coming buffer, we can allocate our
166 // buffer and start putting data into it.
167 boost::asio::async_read(
168 self->_socket,
169 message->buffer(),
170 boost::asio::bind_executor(self->_strand, handle_read_data));
171 } else {
172 if (ec) {
173 log_error("Primary server: failed to read header: ", ec.message());
174 }
175 // Connect();
176 self->Close();
177 }
178 };
179
180 // Read the size of the buffer that is coming.
181 boost::asio::async_read(
182 self->_socket,
183 message->size_as_buffer(),
184 boost::asio::bind_executor(self->_strand, handle_read_header));
185 });
186 }

192 if (!self) return;
193 self->CloseNow();

197 void Primary::StartTimer() {
198 if (_deadline.expiry() <= std::chrono::steady_clock::now()) {
199 log_debug("session ", _session_id, " time out");
200 Close();
201 } else {
202 std::weak_ptr<Primary> weak = shared_from_this();
203 _deadline.async_wait([weak](boost::system::error_code ec) {
204 auto self = weak.lock();
205 if (!self) return;
206 if (!ec) {
207 self->StartTimer();
208 } else {
209 log_error("session ", self->_session_id, " timed out error: ", ec.message());
210 }
211 });
212 }
213 }
214
215 void Primary::CloseNow(boost::system::error_code ec) {
216 _deadline.cancel();
217 if (!ec)
218 {
219 if (_socket.is_open()) {
220 boost::system::error_code ec2;
221 _socket.shutdown(boost::asio::socket_base::shutdown_both, ec2);
222 _socket.close();
223 }
224 }
225 _on_closed(shared_from_this());
226 log_debug("session", _session_id, "closed");
227 }

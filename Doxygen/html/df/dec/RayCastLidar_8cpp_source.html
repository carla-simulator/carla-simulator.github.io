46void ARayCastLidar::Set(const FLidarDescription &LidarDescription)
47{
48 Description = LidarDescription;
49 SetSeed(Description.RandomSeed);
50 LidarData = FLidarData(Description.Channels);
51 CreateLasers();
52 PointsPerChannel.resize(Description.Channels);
53
54 // Compute drop off model parameters
55 DropOffBeta = 1.0f - Description.DropOffAtZeroIntensity;
56 DropOffAlpha = Description.DropOffAtZeroIntensity / Description.DropOffIntensityLimit;
57 DropOffGenActive = Description.DropOffGenRate > std::numeric_limits<float>::epsilon();
58}

96float ARayCastLidar::ComputeIntensity(const FSemanticDetection& RawDetection) const
97{
98 const carla::geom::Location HitPoint = RawDetection.point;
99 const float Distance = HitPoint.Length();
100
101 const float AttenAtm = Description.AtmospAttenRate;
102 const float AbsAtm = exp(-AttenAtm * Distance);
103
104 const float IntRec = AbsAtm;
105
106 return IntRec;
107}

109ARayCastLidar::FDetection ARayCastLidar::ComputeDetection(const FHitResult& HitInfo, const FTransform& InverseSensorTransform) const
110{
111 FDetection Detection;
112 const FVector HitPoint = HitInfo.ImpactPoint;
113 Detection.point = InverseSensorTransform.TransformPosition(HitPoint);
114
115 const float Distance = Detection.point.Length();
116
117 const float AttenAtm = Description.AtmospAttenRate;
118 const float AbsAtm = exp(-AttenAtm * Distance);
119
120 const float IntRec = AbsAtm;
121
122 Detection.intensity = IntRec;
123
124 return Detection;
125}

127 void ARayCastLidar::PreprocessRays(uint32_t Channels, uint32_t MaxPointsPerChannel) {
128 Super::PreprocessRays(Channels, MaxPointsPerChannel);
129
130 if (!DropOffGenActive)
131 {
132 return;
133 }
134
135 for (auto ch = 0u; ch < Channels; ch++) {
136 for (auto p = 0u; p < MaxPointsPerChannel; p++) {
137 RayPreprocessCondition[ch][p] = !(DropOffGenActive && RandomEngine->GetUniformFloat() < Description.DropOffGenRate);
138 }
139 }
140 }

142 bool ARayCastLidar::PostprocessDetection(FDetection& Detection) const
143 {
144 if (Description.NoiseStdDev > std::numeric_limits<float>::epsilon()) {
145 const auto ForwardVector = Detection.point.MakeUnitVector();
146 const auto Noise = ForwardVector * RandomEngine->GetNormalDistribution(0.0f, Description.NoiseStdDev);
147 Detection.point += Noise;
148 }
149
150 const float Intensity = Detection.intensity;
151 if(Intensity > Description.DropOffIntensityLimit)
152 return true;
153 else
154 return RandomEngine->GetUniformFloat() < DropOffAlpha * Intensity + DropOffBeta;
155 }

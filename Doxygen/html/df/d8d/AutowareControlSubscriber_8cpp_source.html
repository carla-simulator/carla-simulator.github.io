49AutowareControlSubscriber::AutowareControlSubscriber(
50 void *vehicle, std::string frame_id)
51 : BaseSubscriber(vehicle, "rt/control/command", std::move(frame_id)),
52 _impl(std::make_shared<Impl>()) {
53 // QoS per tier4's AutowareController: RELIABLE / TRANSIENT_LOCAL /
54 // KEEP_LAST 1 so the latest latched command survives a late join.
55 const QosProfile qos = QosProfile::ReliableTransientLocal();
56 bool ok = true;
57 ok &= _impl->control.Init("rt/control/command/control_cmd", qos);
58 ok &= _impl->gear.Init("rt/control/command/gear_cmd", qos);
59 ok &= _impl->turn_indicators.Init("rt/control/command/turn_indicators_cmd", qos);
60 ok &= _impl->hazard_lights.Init("rt/control/command/hazard_lights_cmd", qos);
61 ok &= _impl->emergency.Init("rt/control/command/emergency_cmd", qos);
62 ok &= _impl->engage.Init("rt/vehicle/engage", qos);
63 if (!ok) {
64 log_error("AutowareControlSubscriber: failed to initialize one or more command readers");
65 }
66}

70ROS2CallbackData AutowareControlSubscriber::GetMessage() {
71 // Drain every subscriber (clearing its new-message flag) even though only
72 // control_cmd feeds the vehicle today — tier4 AutowareController::GetControl
73 // does the same; gear/turn/hazard/emergency/engage application is a tier4
74 // TODO deferred to the Unreal phase.
75 const auto control_in = _impl->control.GetMessage();
76 (void)_impl->gear.GetMessage();
77 (void)_impl->turn_indicators.GetMessage();
78 (void)_impl->hazard_lights.GetMessage();
79 (void)_impl->emergency.GetMessage();
80 (void)_impl->engage.GetMessage();
81
82 VehicleAccelerationControl control_out{};
83
84 // Longitudinal: use ONLY acceleration from /control/command/control_cmd to
85 // drive the vehicle. Velocity and jerk from the message are intentionally
86 // ignored to keep pure acceleration control.
87 control_out.acceleration = control_in.longitudinal.acceleration;
88
89 // Set lateral negative, because CARLA treats positive as right and Autoware
90 // expects positive to represent left (all when moving forward).
91 const float raw_steering = -control_in.lateral.steering_tire_angle;
92
93 // Apply steering compensation lookup table (measured on Lincoln MKZ). This
94 // reduces understeer / outward drift by mapping Autoware's target tire angle
95 // to a vehicle input that better matches the requested actual tire angle.
96 control_out.steer = autoware_steering_compensation::GetSteeringInput(raw_steering);
97 control_out.steer_speed = 0.0f;
98 if (control_in.lateral.is_defined_steering_tire_rotation_rate) {
99 control_out.steer_speed = -control_in.lateral.steering_tire_rotation_rate;
100 }
101
102 return ROS2CallbackData(control_out);
103}

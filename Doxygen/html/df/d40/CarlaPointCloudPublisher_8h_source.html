29// Abstract base for ROS 2 publishers that ship sensor_msgs::msg::PointCloud2:
30// lidar, semantic-lidar, radar, and the point-cloud side of the DVS sensor.
31// Each subclass declares its PointField layout (via PointCloudFieldsLayout.h)
32// and implements ComputePointCloud(), which is responsible for any per-point
33// transform (handedness flip, polar-to-Cartesian conversion, etc.) and
34// returns the bytes that get written to the wire. The Init() call on the
35// underlying PublisherImpl happens in the ctor, so a subclass only has to
36// pass the (base_topic_name, frame_id) pair through.
37class CarlaPointCloudPublisher : public BasePublisher {
38public:
39 // exact_topic: when true the writer publishes on base_topic_name verbatim
40 // (no "/point_cloud" suffix). Used by single-topic point-cloud sensors when
41 // the actor carries a ros_topic_name override — tier4's custom topic names
42 // replaced the whole topic for those publishers (e.g. Autoware expects
43 // exactly /sensing/lidar/top/pointcloud_raw_ex). The DVS point-cloud lane
44 // keeps the suffix: it shares its base with the image/camera_info topics.
45 CarlaPointCloudPublisher(
46 std::string base_topic_name, std::string frame_id, bool exact_topic = false);
47 ~CarlaPointCloudPublisher() override;

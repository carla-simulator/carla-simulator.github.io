40 std::optional<Map> MapBuilder::Build() {
41
42 CreatePointersBetweenRoadSegments();
43 RemoveZeroLaneValiditySignalReferences();
44
45 for (auto &&info : _temp_road_info_container) {
46 DEBUG_ASSERT(info.first != nullptr);
47 info.first->_info = InformationSet(std::move(info.second));
48 }
49
50 for (auto &&info : _temp_lane_info_container) {
51 DEBUG_ASSERT(info.first != nullptr);
52 info.first->_info = InformationSet(std::move(info.second));
53 }
54
55 // compute transform requires the roads to have the RoadInfo
56 SolveSignalReferencesAndTransforms();
57 SolveStencilReferencesAndTransforms();
58
59 SolveControllerAndJuntionReferences();
60
61 // remove temporal already used information
62 _temp_road_info_container.clear();
63 _temp_lane_info_container.clear();
64
65 // _map_data is a memeber of MapBuilder so you must especify if
66 // you want to keep it (will return copy -> Map(const Map &))
67 // or move it (will return move -> Map(Map &&))
68 Map map(std::move(_map_data));
69 CreateJunctionBoundingBoxes(map);
70 ComputeJunctionRoadConflicts(map);
71 CheckSignalsOnRoads(map);
72
73 return map;
74 }

670 // return the pointer to a lane object
671 Lane *MapBuilder::GetEdgeLanePointer(RoadId road_id, LaneId lane_id) {
672
673 if (!_map_data.ContainsRoad(road_id)) {
674 return nullptr;
675 }
676 Road &road = _map_data.GetRoad(road_id);
677
678 // Not very pretty as it repeats the IsPositiveDirection logic of the Lane class
679 bool from_start = false;
680 if (road.IsRHT() && lane_id <= 0){
681 from_start = true;
682 } else if (!road.IsRHT() && lane_id >= 0){
683 from_start = true;
684 }
685
686 // get the lane section
687 LaneSection *section;
688 if (from_start) {
689 section = road.GetStartSection(lane_id);
690 } else {
691 section = road.GetEndSection(lane_id);
692 }
693
694 // get the lane
695 DEBUG_ASSERT(section != nullptr);
696 return section->GetLane(lane_id);
697 }

699 // return a list of pointers to all lanes from a lane (using road and junction
700 // info)
701 std::vector<Lane *> MapBuilder::GetLaneNext(
702 RoadId road_id,
703 SectionId section_id,
704 LaneId lane_id) {
705 std::vector<Lane *> result;
706
707 if (!_map_data.ContainsRoad(road_id)) {
708 return result;
709 }
710 Road &road = _map_data.GetRoad(road_id);
711
712 // get the section
713 LaneSection &section = road._lane_sections.GetById(section_id);
714
715 // get the lane
716 Lane *lane = section.GetLane(lane_id);
717 DEBUG_ASSERT(lane != nullptr);
718
719 // successor and predecessor (road and lane)
720 LaneId next;
721 RoadId next_road;
722 if (lane->IsPositiveDirection()) {
723 next_road = road.GetSuccessor();
724 next = lane->GetSuccessor();
725 } else {
726 next_road = road.GetPredecessor();
727 next = lane->GetPredecessor();
728 }
729
730 // check to see if next is a road or a junction
731 bool next_is_junction = !_map_data.ContainsRoad(next_road);
732 double s = section.GetDistance();
733
734 // check if we are in a lane section in the middle
735 if ((!lane->IsPositiveDirection() && s > 0) ||
736 (lane->IsPositiveDirection() && road._lane_sections.upper_bound(s) != road._lane_sections.end())) {
737 // check if lane has a next link (if not, it deads in the middle section)
738 if (next != 0 || (lane_id == 0 && next == 0)) {
739 // change to next / prev section
740 if (lane->IsPositiveDirection()) {
741 result.push_back(road.GetNextLane(s, next));
742 } else {
743 result.push_back(road.GetPrevLane(s, next));
744 }
745 }
746 } else if (!next_is_junction) {
747 // change to another road / junction
748 if (next != 0 || (lane_id == 0 && next == 0)) {
749 // single road
750 result.push_back(GetEdgeLanePointer(next_road, next));
751 }
752 } else {
753 // several roads (junction)
754
755 /// @todo Is it correct to use a road id as section id? (NS: I just added

964 void MapBuilder::CreateJunctionBoundingBoxes(Map &map) {
965 for (auto &junctionpair : map._data.GetJunctions()) {
966 auto* junction = map.GetJunction(junctionpair.first);
967 auto waypoints = map.GetJunctionWaypoints(junction->GetId(), Lane::LaneType::Any);
968 const int number_intervals = 10;
969
970 float minx = std::numeric_limits<float>::max();
971 float miny = std::numeric_limits<float>::max();
972 float minz = std::numeric_limits<float>::max();
973 float maxx = -std::numeric_limits<float>::max();
974 float maxy = -std::numeric_limits<float>::max();
975 float maxz = -std::numeric_limits<float>::max();
976
977 auto get_min_max = [&](geom::Location position) {
978 if (position.x < minx) {
979 minx = position.x;
980 }
981 if (position.y < miny) {
982 miny = position.y;
983 }
984 if (position.z < minz) {
985 minz = position.z;
986 }
987
988 if (position.x > maxx) {
989 maxx = position.x;
990 }
991 if (position.y > maxy) {
992 maxy = position.y;
993 }
994 if (position.z > maxz) {
995 maxz = position.z;
996 }
997 };
998
999 for (auto &waypoint_p : waypoints) {
1000 auto &waypoint_start = waypoint_p.first;
1001 auto &waypoint_end = waypoint_p.second;
1002 double interval = (waypoint_end.s - waypoint_start.s) / static_cast<double>(number_intervals);
1003 auto next_wp = waypoint_end;
1004 auto location = map.ComputeTransform(next_wp).location;
1005
1006 get_min_max(location);
1007
1008 next_wp = waypoint_start;
1009 location = map.ComputeTransform(next_wp).location;
1010
1011 get_min_max(location);
1012
1013 for (int i = 0; i < number_intervals; ++i) {
1014 if (interval < std::numeric_limits<double>::epsilon())
1015 break;
1016 auto next = map.GetNext(next_wp, interval);
1017 if(next.size()){
1018 next_wp = next.back();
1019 }
1020
1021 location = map.ComputeTransform(next_wp).location;
1022 get_min_max(location);
1023 }
1024 }
1025 carla::geom::Location location(0.5f * (maxx + minx), 0.5f * (maxy + miny), 0.5f * (maxz + minz));
1026 carla::geom::Vector3D extent(0.5f * (maxx - minx), 0.5f * (maxy - miny), 0.5f * (maxz - minz));
1027
1028 junction->_bounding_box = carla::geom::BoundingBox(location, extent);
1029 }

1165 void MapBuilder::CheckSignalsOnRoads(Map &map) {
1166 for (auto& signal_pair : map._data._signals) {
1167 auto& signal = signal_pair.second;
1168 auto signal_position = signal->GetTransform().location;
1169 auto signal_rotation = signal->GetTransform().rotation;
1170 auto closest_waypoint_to_signal =
1171 map.GetClosestWaypointOnRoad(signal_position,
1172 static_cast<int32_t>(carla::road::Lane::LaneType::Shoulder) | static_cast<int32_t>(carla::road::Lane::LaneType::Driving));
1173 // workarround to not move stencil stop
1174 if (
1175 signal->GetName().find("Stencil_STOP") != std::string::npos ||
1176 signal->GetName().find("STATIC") != std::string::npos ||
1177 signal->_using_inertial_position) {
1178 continue;
1179 }
1180 if(closest_waypoint_to_signal) {
1181 auto road_transform = map.ComputeTransform(closest_waypoint_to_signal.value());
1182 auto distance_to_road = (road_transform.location -signal_position).Length();
1183 double lane_width = map.GetLaneWidth(closest_waypoint_to_signal.value());
1184 bool is_rht = map.GetLane(closest_waypoint_to_signal.value()).GetRoad()->IsRHT();
1185 int displacement_direction = 1;
1186 int iter = 0;
1187 int MaxIter = 10;
1188 // Displaces signal until it finds a suitable spot
1189 while(distance_to_road < (lane_width * 0.7) && iter < MaxIter && displacement_direction != 0) {
1190 if(iter == 0) {
1191 log_debug("Traffic sign",
1192 signal->GetSignalId(),
1193 "overlaps a driving lane. Moving out of the road...");
1194 }
1195
1196 auto right_waypoint = map.GetRight(closest_waypoint_to_signal.value());
1197 auto right_lane_type = (right_waypoint) ? map.GetLaneType(right_waypoint.value()) : carla::road::Lane::LaneType::None;
1198
1199 auto left_waypoint = map.GetLeft(closest_waypoint_to_signal.value());
1200 auto left_lane_type = (left_waypoint) ? map.GetLaneType(left_waypoint.value()) : carla::road::Lane::LaneType::None;
1201
1202 if (is_rht) {
1203 // Move to the right if possible, then left
1204 if (right_lane_type != carla::road::Lane::LaneType::Driving) {
1205 displacement_direction = 1;
1206 } else if (left_lane_type != carla::road::Lane::LaneType::Driving) {
1207 displacement_direction = -1;
1208 } else {
1209 displacement_direction = 0;
1210 }
1211 } else {
1212 // Move to the left if possible, then right
1213 if (left_lane_type != carla::road::Lane::LaneType::Driving) {
1214 displacement_direction = -1;
1215 } else if (right_lane_type != carla::road::Lane::LaneType::Driving) {
1216 displacement_direction = 1;
1217 } else {
1218 displacement_direction = 0;
1219 }
1220 }
1221
1222 geom::Vector3D displacement = 1.f*(road_transform.GetRightVector()) *
1223 static_cast<float>(abs(lane_width))*0.2f;
1224 signal_position += (displacement * static_cast<float>(displacement_direction));
1225 signal_rotation = road_transform.rotation;
1226 closest_waypoint_to_signal =
1227 map.GetClosestWaypointOnRoad(signal_position,
1228 static_cast<int32_t>(carla::road::Lane::LaneType::Shoulder) | static_cast<int32_t>(carla::road::Lane::LaneType::Driving));
1229 distance_to_road =
1230 (map.ComputeTransform(closest_waypoint_to_signal.value()).location -
1231 signal_position).Length();
1232 lane_width = map.GetLaneWidth(closest_waypoint_to_signal.value());
1233 iter++;
1234 }
1235 if(iter == MaxIter) {
1236 log_debug("Failed to find suitable place for signal.");
1237 } else {
1238 // Only perform the displacement if a good location has been found
1239 signal->_transform.location = signal_position;
1240 signal->_transform.rotation = signal_rotation;
1241 }
1242 }

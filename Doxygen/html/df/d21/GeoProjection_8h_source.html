28 /// All the supported projection types. To add new ones, add them to this enum,
29 /// create the parameters structure and the parser from geo to location (and viceversa).
30 enum class ProjectionType {
31 TransverseMercator,
32 UniversalTransverseMercator,
33 WebMercator,
34 LambertConformalConic,
35 };

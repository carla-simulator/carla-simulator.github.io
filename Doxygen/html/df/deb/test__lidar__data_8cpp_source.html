17// {Lidar,SemanticLidar}Data::ResetMemory rejects callers that supply more
18// points-per-channel entries than the configured channel count; the equal
19// count case is valid. These tests pin the boundary so a future tightening
20// to a strict `>` would be caught. DEBUG_ASSERT compiles away under NDEBUG,
21// so the assertion only fires when the suite is built as Debug.
22
23TEST(LidarData, reset_memory_accepts_one_entry_per_channel) {
24 constexpr uint32_t channel_count{4u};
25 LidarData data{channel_count};
26 std::vector<uint32_t> points_per_channel(channel_count, 1u);
27 data.ResetMemory(points_per_channel);
28 EXPECT_EQ(data.GetChannelCount(), channel_count);
29}
30
31TEST(LidarData, reset_memory_accepts_fewer_entries_than_channels) {
32 constexpr uint32_t channel_count{8u};
33 LidarData data{channel_count};
34 std::vector<uint32_t> points_per_channel(channel_count - 2u, 3u);
35 data.ResetMemory(points_per_channel);

17void ACarlaHUD::DrawHUD()
18{
19 Super::DrawHUD();
20
21 auto Player = GetOwningPlayerController();
22 if (Player == nullptr)
23 {
24 UE_LOG(LogCarla, Error, TEXT("Can't find player controller!"));
25 return;
26 }
27
28 if(DebugVehicle) {
29 float YL = 1600.0f;
30 float Y0 = 0.0f;
31 DrawDebug(YL, Y0);
32 }
33
34 double Now = FPlatformTime::Seconds();
35 int i = 0;
36 while (i < StringList.Num())
37 {
38 // project position from camera
39 FVector2D Screen;
40 if (Player->ProjectWorldLocationToScreen(StringList[i].Location, Screen, true))
41 {
42 // draw text
43 DrawText(StringList[i].Str, StringList[i].Color, Screen.X, Screen.Y);
44 }
45
46 // check to remove the string
47 if (Now >= StringList[i].TimeToDie)
48 {
49 StringList.RemoveAt(i);
50 }
51 else
52 ++i;
53 }
54}

64void ACarlaHUD::DrawDebug(float& YL, float& YPos)
65{
66 using namespace Chaos;
67
68 ensure(IsInGameThread());
69
70 if (!DebugVehicle->HasValidPhysicsState()){
71 UE_LOG(LogCarla, Log, TEXT("Invalid state"));
72 return;
73 }
74
75 UFont* RenderFont = GEngine->GetMediumFont();
76 // draw drive data
77 {
78 Canvas->SetDrawColor(FColor::White);
79
80 YPos += 16;
81 for (int i = 0; i < DebugVehicle->GetNumWheels(); i++)
82 {
83 YPos += Canvas->DrawText(RenderFont, FString::Printf(TEXT("WheelLoad: [%d] %1.f N"), i, CmToM(DebugVehicle->GetWheelState(i).SpringForce)), 4, YPos);
84 }
85
86 YPos += 16;
87 for (int i = 0; i < DebugVehicle->GetNumWheels(); i++)
88 {
89 if (DebugVehicle->GetWheelState(i).PhysMaterial.IsValid())
90 {
91 YPos += Canvas->DrawText(RenderFont, FString::Printf(TEXT("SurfaceFriction: [%d] %.2f"), i, DebugVehicle->GetWheelState(i).PhysMaterial->Friction), 4, YPos);
92 }
93 }
94 }
95}

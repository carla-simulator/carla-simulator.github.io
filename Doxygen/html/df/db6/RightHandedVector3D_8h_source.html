14 /// Internal adapter that holds a `Vector3D` in a right-handed coordinate
15 /// frame. CARLA / Unreal expose vectors in a left-handed frame; `Quaternion`
16 /// and `Rotation` work in right-handed math internally. The Y axis is
17 /// negated on construction and re-negated on conversion back, so the
18 /// handedness flip happens exactly once at the boundary.
19 struct RightHandedVector3D {
20
21 float x = 0.0f;
22 float y = 0.0f;
23 float z = 0.0f;
24
25 RightHandedVector3D() = default;
26
27 constexpr RightHandedVector3D(float ix, float iy, float iz)
28 : x(ix),
29 y(iy),
30 z(iz) {
31 }
32
33 /// Implicit boundary conversion from a left-handed `Vector3D`. Negates
34 /// the Y component so internal math runs right-handed.
35 constexpr RightHandedVector3D(const Vector3D &v)
36 : x(v.x),
37 y(-v.y),
38 z(v.z) {
39 }
40
41 /// Implicit boundary conversion back to a left-handed `Vector3D`.
42 constexpr operator Vector3D() const {
43 return Vector3D(x, -y, z);
44 }
45 };

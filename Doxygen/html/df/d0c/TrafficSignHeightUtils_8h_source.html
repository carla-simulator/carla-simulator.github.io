16/// Helpers to snap OpenDRIVE-spawned traffic signs / lights onto the ground so
17/// they are not left floating in the air or sunk below the terrain.
18namespace TrafficSignHeightUtils
19{
20 /// Line-trace downward to find the ground under \p Location. Every surface
21 /// along the ray is considered and the first one that is ground wins, so a
22 /// fence or other prop beside the pole is skipped instead of being treated
23 /// as the ground. On hit, set Location.Z to the ground hit plus a small
24 /// offset and return true; otherwise leave Location untouched and return
25 /// false.
26 bool AdjustLocationToGround(
27 UWorld* World,
28 FVector& Location,
29 const TArray<AActor*>& IgnoredActors,
30 const TArray<UPrimitiveComponent*>& IgnoredComponents);

32 /// Snap a single traffic sign to the ground and shift its trigger

100// -- FPixelReader::SendPixelsInRenderThread -----------------------------------

103template <typename TSensor, typename TPixel>
104void FPixelReader::SendPixelsInRenderThread(TSensor &Sensor, bool use16BitFormat, std::function<TArray<TPixel>(void *, uint32)> Conversor)
105{
106 TRACE_CPUPROFILER_EVENT_SCOPE(FPixelReader::SendPixelsInRenderThread);
107 check(Sensor.CaptureRenderTarget != nullptr);
108
109 if (!Sensor.HasActorBegunPlay() || !IsValidChecked(&Sensor))
110 {
111 return;
112 }
113
114 /// Blocks until the render thread has finished all it's tasks.
115 Sensor.EnqueueRenderSceneImmediate();
116
117 // Enqueue a command in the render-thread that will write the image buffer to
118 // the data stream. We need to get frame, timestamp and the sensor transform in the capture
119 // (thus executed in the game-thread), so that they reflect the current point in time.
120 // Otherwise the asynchronous execution could send a future header to the client.
121 ENQUEUE_RENDER_COMMAND(FWritePixels_SendPixelsInRenderThread)
122 (
123 [&Sensor, use16BitFormat, Conversor = std::move(Conversor), Frame = FCarlaEngine::GetFrameCounter(),
124 Timestamp = Sensor.GetEpisode().GetElapsedGameTime(), Transform = Sensor.GetActorTransform()](auto &InRHICmdList) mutable
125 {
126 TRACE_CPUPROFILER_EVENT_SCOPE_STR("FWritePixels_SendPixelsInRenderThread");
127
128 /// @todo Can we make sure the sensor is not going to be destroyed?
129 if (IsValidChecked(&Sensor))
130 {
131 FPixelReader::Payload FuncForSending =
132 [&Sensor, Frame, Timestamp, Transform, Conversor = std::move(Conversor)]
133 (void *LockedData, uint32 Size, uint32 Offset, uint32 ExpectedRowBytes)
134 {
135 if (!IsValidChecked(&Sensor))
136 return;
137
138 TArray<TPixel> Converted;
139
140 // optional conversion of data
141 if (Conversor)
142 {
143 TRACE_CPUPROFILER_EVENT_SCOPE_STR("Data conversion");
144 Converted = Conversor(LockedData, Size);
145 LockedData = reinterpret_cast<void *>(Converted.GetData());
146 Size = Converted.Num() * Converted.GetTypeSize();
147 }
148
149 auto Stream = Sensor.GetDataStream(Sensor);
150 Stream.SetFrameNumber(Frame);
151 Stream.SetTimestamp(Timestamp);
152 Stream.SetTransform(Transform);
153 auto Buffer = Stream.PopBufferFromPool();
154
155 uint32 CurrentRowBytes = ExpectedRowBytes;
156
157#ifdef _WIN32
158 // DirectX uses additional bytes to align each row to 256 boundry,
159 // so we need to remove that extra data when sending to the client
160 if (IsD3DPlatform(GMaxRHIShaderPlatform) && Sensor.AreClientsListening())
161 {
162 CurrentRowBytes = Align(ExpectedRowBytes, D3D12_TEXTURE_DATA_PITCH_ALIGNMENT);
163 if (ExpectedRowBytes != CurrentRowBytes)
164 {
165 TRACE_CPUPROFILER_EVENT_SCOPE_STR("Buffer Copy (windows, row by row)");
166 Buffer.reset(Offset + Size);
167 auto DstRow = Buffer.begin() + Offset;
168 const uint8 *SrcRow = reinterpret_cast<uint8 *>(LockedData);
169 uint32 i = 0;
170 while (i < Size)
171 {
172 FMemory::Memcpy(DstRow, SrcRow, ExpectedRowBytes);
173 DstRow += ExpectedRowBytes;
174 SrcRow += CurrentRowBytes;
175 i += ExpectedRowBytes;
176 }
177 }
178 }
179#endif // _WIN32
180
181 if (ExpectedRowBytes == CurrentRowBytes)
182 {
183 check(ExpectedRowBytes == CurrentRowBytes);
184 TRACE_CPUPROFILER_EVENT_SCOPE_STR("Buffer Copy");
185 Buffer.copy_from(Offset, boost::asio::buffer(LockedData, Size));
186 }
187
188 {
189 // send
190 TRACE_CPUPROFILER_EVENT_SCOPE_STR("Sending buffer");
191 if(Buffer.data())
192 {
193 // serialize data
194 carla::Buffer BufferReady(std::move(carla::sensor::SensorRegistry::Serialize(Sensor, std::move(Buffer))));
195 carla::SharedBufferView BufView = carla::BufferView::CreateFrom(std::move(BufferReady));
196
197 // ROS2
198 #if defined(WITH_ROS2)
199 auto ROS2 = carla::ros2::ROS2::GetInstance();
200 if (ROS2->IsEnabled())
201 {
202 TRACE_CPUPROFILER_EVENT_SCOPE_STR("ROS2 Send PixelReader");
203 auto StreamId = carla::streaming::detail::token_type(Sensor.GetToken()).get_stream_id();
204 auto Res = std::async(std::launch::async, [&Sensor, ROS2, &Stream, StreamId, BufView]()
205 {
206 // get resolution of camera
207 int W = -1, H = -1;
208 float Fov = -1.0f;
209 auto WidthOpt = Sensor.GetAttribute("image_size_x");
210 if (WidthOpt.has_value())
211 W = FCString::Atoi(*WidthOpt->Value);
212 auto HeightOpt = Sensor.GetAttribute("image_size_y");
213 if (HeightOpt.has_value())
214 H = FCString::Atoi(*HeightOpt->Value);
215 auto FovOpt = Sensor.GetAttribute("fov");
216 if (FovOpt.has_value())
217 Fov = FCString::Atof(*FovOpt->Value);
218 // send data to ROS2
219 AActor* ParentActor = Sensor.GetAttachParentActor();
220 if (ParentActor)
221 {
222 FTransform LocalTransformRelativeToParent = Sensor.GetActorTransform().GetRelativeTransform(ParentActor->GetActorTransform());
223 ROS2->ProcessDataFromCamera(Stream.GetSensorType(), StreamId, LocalTransformRelativeToParent, W, H, Fov, BufView, &Sensor);
224 }
225 else
226 {
227 ROS2->ProcessDataFromCamera(Stream.GetSensorType(), StreamId, Stream.GetSensorTransform(), W, H, Fov, BufView, &Sensor);
228 }
229 });
230 }
231 #endif
232
233 if (Sensor.AreClientsListening())
234 {
235 // network
236 SCOPE_CYCLE_COUNTER(STAT_CarlaSensorStreamSend);
237 TRACE_CPUPROFILER_EVENT_SCOPE_STR("Stream Send");
238 Stream.Send(Sensor, BufView);
239 }
240 }
241 }
242 };
243
244 WritePixelsToBuffer(
245 *Sensor.CaptureRenderTarget,
246 carla::sensor::SensorRegistry::get<TSensor *>::type::header_offset,
247 InRHICmdList,
248 std::move(FuncForSending),
249 Sensor.GetReadbackPool());
250
251 }
252 }
253 );
254
255 // Blocks until the render thread has finished all it's tasks
256 Sensor.WaitForRenderThreadToFinish();
257}

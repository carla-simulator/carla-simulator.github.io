33void FFrameData::GetFrameData(UCarlaEpisode *ThisEpisode, bool bAdditionalData, bool bIncludeActorsAgain)
34{
35 Episode = ThisEpisode;
36 // PlatformTime.UpdateTime();
37 const FActorRegistry &Registry = Episode->GetActorRegistry();
38
39 if (bIncludeActorsAgain)
40 {
41 AddExistingActors();
42 }
43
44 // through all actors in registry
45 for (auto It = Registry.begin(); It != Registry.end(); ++It)
46 {
47 FCarlaActor* View = It->Value.Get();
48
49 if (bAdditionalData)
50 {
51 AddActorAppearance(View, bIncludeActorsAgain);
52 }
53
54 switch (View->GetActorType())
55 {
56 // save the transform for props
57 case FCarlaActor::ActorType::Other:
58 AddActorPosition(View);
59 // a prop that simulates physics moves: mirrors need its velocities
60 if (bAdditionalData && !View->IsDormant() && (View->GetActor() != nullptr))
61 {
62 const auto *Root = Cast<UPrimitiveComponent>(View->GetActor()->GetRootComponent());
63 if ((Root != nullptr) && Root->IsSimulatingPhysics())
64 {
65 AddActorKinematics(View);
66 }
67 }
68 break;
69
70 case FCarlaActor::ActorType::Sensor:
71 AddActorPosition(View);
72 if (bAdditionalData)
73 {
74 AddSensorTick(View);
75 }
76 break;
77
78 // save the transform of all vehicles
79 case FCarlaActor::ActorType::Vehicle:
80 AddActorPosition(View);
81 AddVehicleAnimation(View);
82 AddVehicleLight(View);
83 AddVehicleParts(View);
84 if (bAdditionalData)
85 {
86 AddActorKinematics(View);
87 }
88 break;
89
90 // save the transform of all walkers
91 case FCarlaActor::ActorType::Walker:
92 AddActorPosition(View);
93 AddWalkerAnimation(View);
94 if (bAdditionalData)
95 {
96 AddActorKinematics(View);
97 }
98 break;
99
100 // save the state of each traffic light
101 case FCarlaActor::ActorType::TrafficLight:
102 AddTrafficLightState(View);
103 if (bAdditionalData)
104 {
105 // its timings too, cheap and self-healing for a secondary
106 if (const auto *TrafficLight = Cast<ATrafficLightBase>(View->GetActor()))
107 {
108 TrafficLightTimes.Add(CarlaRecorderTrafficLightTime{
109 View->GetActorId(),
110 TrafficLight->GetGreenTime(),
111 TrafficLight->GetYellowTime(),
112 TrafficLight->GetRedTime()});
113 }
114 }
115 break;
116 }
117 }
118 if (bAdditionalData)
119 {
120 // Forget the appearance of the actors gone.
121 auto &Appearances = Episode->GetMultiGpuWorldState().Appearances;
122 if (Appearances.size() > static_cast<size_t>(Registry.Num()))
123 {
124 for (auto It = Appearances.begin(); It != Appearances.end();)
125 {
126 It = Registry.Contains(It->first) ? std::next(It) : Appearances.erase(It);
127 }
128 }
129 AddWorldState(bIncludeActorsAgain);
130 }
131 GetFrameCounter();
132}

200void FFrameData::AddActorAppearance(FCarlaActor *CarlaActor, bool bFullResync)
201{
202 const AActor *Actor = CarlaActor->IsDormant() ? nullptr : CarlaActor->GetActor();
203 if (Actor == nullptr)
204 {
205 return;
206 }
207 FMirroredAppearance Appearance;
208 Appearance.bCollisions = Actor->GetActorEnableCollision();
209 Appearance.bHidden = Actor->IsHidden();
210 Appearance.Scale = Actor->GetActorScale3D();
211 auto &Sent = Episode->GetMultiGpuWorldState().Appearances;
212 const auto It = Sent.find(CarlaActor->GetActorId());
213 if (!bFullResync && (It != Sent.end()) && (It->second == Appearance))
214 {
215 return;
216 }
217 Sent[CarlaActor->GetActorId()] = Appearance;
218 carla::multigpu::ActorAppearance Record;
219 Record.actor_id = CarlaActor->GetActorId();
220 Record.collisions = Appearance.bCollisions;
221 Record.hidden = Appearance.bHidden;
222 Record.scale = std::array<double, 3>{Appearance.Scale.X, Appearance.Scale.Y, Appearance.Scale.Z};
223 MirrorAppearances.actors.push_back(std::move(Record));
224}

226void FFrameData::AddWorldState(bool bFullResync)
227{
228 FPrimaryWorldState &Sent = Episode->GetMultiGpuWorldState();
229 UWorld *World = Episode->GetWorld();
230 ACarlaGameModeBase *GameMode = UCarlaStatics::GetGameMode(World);
231 UCarlaLightSubsystem *LightSubsystem = World ? World->GetSubsystem<UCarlaLightSubsystem>() : nullptr;
232
233 // Applied again with the same values is still an apply to mirror.
234 if (AWeather *Weather = Episode->GetWeather())
235 {
236 if (bFullResync || (Weather->GetApplySerial() != Sent.WeatherSerial))
237 {
238 Sent.WeatherSerial = Weather->GetApplySerial();
239 Weathers.Add(FFrameData_ToRecord(Weather->GetCurrentWeather()));
240 }
241 }
242
243 carla::multigpu::WorldGlobals Globals;
244 if (GameMode != nullptr)
245 {
246 Globals.map_layers = static_cast<uint16_t>(GameMode->GetLoadedMapLayers());
247 Globals.imu_gravity = GameMode->IMUSensorGravity;
248 }
249 if (LightSubsystem != nullptr)
250 {
251 Globals.day_night_cycle = LightSubsystem->GetDayNightCycle();
252 Globals.has_day_time_state = LightSubsystem->GetDayTimeState(Globals.last_is_day);
253 }
254#if defined(WITH_ROS2)
255 auto ROS2 = carla::ros2::ROS2::GetInstance();
256 if (ROS2 && ROS2->IsEnabled())
257 {
258 Globals.publish_tf = ROS2->GetPublishTF();
259 }
260#endif
261 MirrorWorldGlobals = Globals;
262
263 if ((GameMode != nullptr) &&
264 (bFullResync || (GameMode->GetEnvironmentObjectsSerial() != Sent.EnvironmentObjectsSerial)))
265 {
266 Sent.EnvironmentObjectsSerial = GameMode->GetEnvironmentObjectsSerial();
267 MirrorWorldObjects.disabled_environment_objects =
268 FFrameData_SortedIds(GameMode->GetDisabledEnvironmentObjects());
269 }
270
271 // Each light's state at the end of this frame, after any weather change.
272 if (LightSubsystem != nullptr)
273 {
274 if (bFullResync)
275 {
276 for (const auto &Light : LightSubsystem->GetLights())
277 {
278 if (IsValid(Light.Value) && IsValid(Light.Value->GetOwner()))
279 {
280 AddEventLightSceneChanged(Light.Value);
281 }
282 }
283 }
284 else
285 {
286 for (const int32 LightId : Sent.ChangedLights)
287 {
288 UCarlaLight *Light = LightSubsystem->GetLight(LightId);
289 if (IsValid(Light) && IsValid(Light->GetOwner()))
290 {
291 AddEventLightSceneChanged(Light);
292 }
293 }
294 }
295 }
296 Sent.ChangedLights.Reset();
297}

310void FFrameData::PlayFrameData(
311 UCarlaEpisode *ThisEpisode,
312 std::unordered_map<uint32_t, uint32_t>& MappedId,
313 FMirrorState &MirrorState,
314 TFunctionRef<void(uint32_t, uint32_t, bool)> OnActorAdded,
315 TFunctionRef<void(uint32_t)> OnActorRemoved)
316{
317 Episode = ThisEpisode;
318 const bool bFullResync = MultiGpuFrameTime && MultiGpuFrameTime->full_resync;
319 std::unordered_set<uint32_t> AddedIds;
320
321 // First: what the rest of the frame sets can live in a map layer.
322 PlayMapLayers();
323
324 for(const CarlaRecorderEventAdd &EventAdd : EventsAdd.GetEvents())
325 {
326 uint32_t OldId = EventAdd.DatabaseId;
327 auto Result = ProcessReplayerEventAdd(
328 EventAdd.Location,
329 EventAdd.Rotation,
330 EventAdd.Description,
331 EventAdd.DatabaseId,
332 false,
333 true,
334 MappedId);
335 switch (Result.first)
336 {
337 // actor not created
338 case 0:
339 UE_LOG(LogCarla, Log, TEXT("actor could not be created"));
340 break;
341
342 // actor created but with different id
343 case 1:
344 // mapping id (recorded Id is a new Id in replayer)
345 MappedId[OldId] = Result.second;
346 MirrorState.CreatedIds.insert(Result.second);
347 PrepareMirroredActor(Episode->FindCarlaActor(Result.second), OldId);
348 UE_LOG(LogCarla, Log, TEXT("actor created"));
349 OnActorAdded(OldId, Result.second, true);
350 break;
351
352 // actor reused from existing
353 case 2:
354 // mapping id (say desired Id is mapped to what)
355 MappedId[OldId] = Result.second;
356 PrepareMirroredActor(Episode->FindCarlaActor(Result.second), OldId);
357 UE_LOG(LogCarla, Log, TEXT("actor reused"));
358 OnActorAdded(OldId, Result.second, false);
359 break;
360 }
361 AddedIds.insert(OldId);
362 }
363
364 // Attachments, retrying those whose actors were not mirrored yet.
365 {
366 std::vector<std::pair<uint32_t, uint32_t>> Parents = std::move(MirrorState.PendingParents);
367 MirrorState.PendingParents.clear();
368 for (const CarlaRecorderEventParent &EventParent : EventsParent.GetEvents())
369 {
370 Parents.emplace_back(EventParent.DatabaseId, EventParent.DatabaseIdParent);
371 }
372 for (const auto &Parent : Parents)
373 {
374 if (!ProcessReplayerEventParent(Parent.first, Parent.second, MappedId))
375 {
376 MirrorState.PendingParents.push_back(Parent);
377 }
378 }
379 }
380
381 std::unordered_set<uint32_t> RemovedIds;
382 for (const CarlaRecorderEventDel &EventDel : EventsDel.GetEvents())
383 {
384 if (const std::optional<uint32_t> Local = FindMirror(MappedId, EventDel.DatabaseId))
385 {
386 ProcessReplayerEventDel(*Local);
387 MirrorState.CreatedIds.erase(*Local);
388 MappedId.erase(EventDel.DatabaseId);
389 }
390 RemovedIds.insert(EventDel.DatabaseId);
391 OnActorRemoved(EventDel.DatabaseId);
392 }
393
394 if (bFullResync)
395 {
396 // A full resync announces every live actor of the primary: a mirror this
397 // secondary spawned for any other one outlived its actor (its removal was
398 // missed) and goes away.
399 for (auto It = MappedId.begin(); It != MappedId.end();)
400 {
401 const uint32_t PrimaryId = It->first;
402 const uint32_t LocalId = It->second;
403 if ((AddedIds.count(PrimaryId) == 0u) && (MirrorState.CreatedIds.count(LocalId) != 0u))
404 {
405 ProcessReplayerEventDel(LocalId);
406 MirrorState.CreatedIds.erase(LocalId);
407 It = MappedId.erase(It);
408 RemovedIds.insert(PrimaryId);
409 OnActorRemoved(PrimaryId);
410 }
411 else
412 {
413 ++It;
414 }
415 }
416 }
417 if (!RemovedIds.empty() || bFullResync)
418 {
419 // An attachment can no longer be replayed once either actor is gone.
420 auto &Pending = MirrorState.PendingParents;
421 Pending.erase(std::remove_if(Pending.begin(), Pending.end(), [&](const auto &Parent)
422 {
423 const auto Gone = [&](uint32_t Id)
424 {
425 return (RemovedIds.count(Id) != 0u) || (bFullResync && (AddedIds.count(Id) == 0u));
426 };
427 return Gone(Parent.first) || Gone(Parent.second);
428 }),
429 Pending.end());
430 }
431
432 PlayAppearances(MappedId);
433 PlayPositions(MappedId, MirrorState);
434 PlayKinematics(MappedId);
435
436 for (const CarlaRecorderTrafficLightTime &Time : TrafficLightTimes.GetTrafficLightTimes())
437 {
438 const std::optional<uint32_t> Local = FindMirror(MappedId, Time.DatabaseId);
439 FCarlaActor *CarlaActor = Local ? Episode->FindCarlaActor(*Local) : nullptr;
440 auto *TrafficLight = CarlaActor ? Cast<ATrafficLightBase>(CarlaActor->GetActor()) : nullptr;
441 if (TrafficLight == nullptr)
442 {
443 continue;
444 }
445 if (TrafficLight->GetGreenTime() != Time.GreenTime)
446 {
447 TrafficLight->SetGreenTime(Time.GreenTime);
448 }
449 if (TrafficLight->GetYellowTime() != Time.YellowTime)
450 {
451 TrafficLight->SetYellowTime(Time.YellowTime);
452 }
453 if (TrafficLight->GetRedTime() != Time.RedTime)
454 {
455 TrafficLight->SetRedTime(Time.RedTime);
456 }
457 }
458
459 for (const CarlaRecorderStateTrafficLight &State : States.GetStates())
460 {
461 if (const std::optional<uint32_t> Local = FindMirror(MappedId, State.DatabaseId))
462 {
463 CarlaRecorderStateTrafficLight StateTrafficLight = State;
464 StateTrafficLight.DatabaseId = *Local;
465 ProcessReplayerStateTrafficLight(StateTrafficLight);
466 }
467 }
468
469 // Before packet 8: the control replayed there takes the manual gear shift
470 // from the drivetrain set here.
471 PlayVehicleParts(MappedId);
472
473 for (const CarlaRecorderAnimVehicle &AnimVehicle : Vehicles.GetVehicles())
474 {
475 if (const std::optional<uint32_t> Local = FindMirror(MappedId, AnimVehicle.DatabaseId))
476 {
477 CarlaRecorderAnimVehicle Vehicle = AnimVehicle;
478 Vehicle.DatabaseId = *Local;
479 ProcessReplayerAnimVehicle(Vehicle);
480 }
481 }
482
483 for (const CarlaRecorderAnimWalker &AnimWalker : Walkers.GetWalkers())
484 {
485 if (const std::optional<uint32_t> Local = FindMirror(MappedId, AnimWalker.DatabaseId))
486 {
487 CarlaRecorderAnimWalker Walker = AnimWalker;
488 Walker.DatabaseId = *Local;
489 ProcessReplayerAnimWalker(Walker);
490 }
491 }
492
493 for (const CarlaRecorderAnimBiker &AnimBiker : Bikers.GetBikers())
494 {
495 if (const std::optional<uint32_t> Local = FindMirror(MappedId, AnimBiker.DatabaseId))
496 {
497 CarlaRecorderAnimBiker Biker = AnimBiker;
498 Biker.DatabaseId = *Local;
499 ProcessReplayerAnimBiker(Biker);
500 }
501 }
502
503 for (const CarlaRecorderLightVehicle &LightVehicle : LightVehicles.GetLightVehicles())
504 {
505 if (const std::optional<uint32_t> Local = FindMirror(MappedId, LightVehicle.DatabaseId))
506 {
507 CarlaRecorderLightVehicle Light = LightVehicle;
508 Light.DatabaseId = *Local;
509 ProcessReplayerLightVehicle(Light);
510 }
511 }
512
513 // Before the scene lights: a day/night change switches lights, and the
514 // lights then get the state the primary's had at the end of the frame.
515 PlayWorldState(MirrorState);
516
517 for (const CarlaRecorderLightScene &Light : LightScenes.GetLights())
518 {
519 ProcessReplayerLightScene(Light);
520 }
521
522 // Sensors last, once their mirrors exist: the state each primary sensor
523 // starts this frame's tick from, then which of them tick, with what delta.
524 const auto FindSensor = [&](uint32_t PrimaryId) -> ASensor *
525 {
526 const std::optional<uint32_t> Local = FindMirror(MappedId, PrimaryId);
527 FCarlaActor *CarlaActor = Local ? Episode->FindCarlaActor(*Local) : nullptr;
528 return CarlaActor ? Cast<ASensor>(CarlaActor->GetActor()) : nullptr;
529 };
530 for (const carla::multigpu::SensorState &State : MirrorSensorStates.states)
531 {
532 if (ASensor *Sensor = FindSensor(State.actor_id))
533 {
534 Sensor->SetMirrorState(State);
535 }
536 }
537 for (const carla::multigpu::SensorTick &Tick : MirrorSensorTicks.ticks)
538 {
539 if (ASensor *Sensor = FindSensor(Tick.actor_id))
540 {
541 Sensor->SetPrimaryTick(Tick.delta_seconds);
542 }
543 }
544
545 SetFrameCounter();
546}

548void FFrameData::PlayMapLayers()
549{
550 ACarlaGameModeBase *GameMode = UCarlaStatics::GetGameMode(Episode->GetWorld());
551 if (!MirrorWorldGlobals || (GameMode == nullptr))
552 {
553 return;
554 }
555 // Both flush level streaming: the layers are there for the rest of the frame.
556 const std::pair<uint16_t, uint16_t> Diff = carla::multigpu::ComputeMapLayerDiff(
557 static_cast<uint16_t>(GameMode->GetLoadedMapLayers()), MirrorWorldGlobals->map_layers);
558 if (Diff.first != 0u)
559 {
560 GameMode->LoadMapLayer(Diff.first);
561 }
562 if (Diff.second != 0u)
563 {
564 GameMode->UnLoadMapLayer(Diff.second);
565 }

568void FFrameData::PlayAppearances(const std::unordered_map<uint32_t, uint32_t>& MappedId)
569{
570 for (const carla::multigpu::ActorAppearance &Appearance : MirrorAppearances.actors)
571 {
572 const std::optional<uint32_t> Local = FindMirror(MappedId, Appearance.actor_id);
573 FCarlaActor *CarlaActor = Local ? Episode->FindCarlaActor(*Local) : nullptr;
574 AActor *Actor = ((CarlaActor != nullptr) && !CarlaActor->IsDormant()) ? CarlaActor->GetActor() : nullptr;
575 if (Actor == nullptr)
576 {
577 continue;
578 }
579 if (Appearance.collisions && (Actor->GetActorEnableCollision() != *Appearance.collisions))
580 {
581 Actor->SetActorEnableCollision(*Appearance.collisions);
582 }
583 if (Appearance.hidden && (Actor->IsHidden() != *Appearance.hidden))
584 {
585 Actor->SetActorHiddenInGame(*Appearance.hidden);
586 }
587 if (Appearance.scale)
588 {
589 const FVector Scale((*Appearance.scale)[0], (*Appearance.scale)[1], (*Appearance.scale)[2]);
590 if (!Actor->GetActorScale3D().Equals(Scale, 0.0))
591 {
592 USceneComponent *Root = Actor->GetRootComponent();
593 if ((Root != nullptr) && (Root->GetMobility() != EComponentMobility::Movable))
594 {
595 Root->SetMobility(EComponentMobility::Movable);
596 }
597 Actor->SetActorScale3D(Scale);
598 }
599 }
600 }

603void FFrameData::PlayWorldState(FMirrorState &MirrorState)
604{
605 UWorld *World = Episode->GetWorld();
606 ACarlaGameModeBase *GameMode = UCarlaStatics::GetGameMode(World);
607 UCarlaLightSubsystem *LightSubsystem = World ? World->GetSubsystem<UCarlaLightSubsystem>() : nullptr;
608
609 if (MirrorWorldGlobals)
610 {
611 const carla::multigpu::WorldGlobals &Globals = *MirrorWorldGlobals;
612 if (GameMode != nullptr)
613 {
614 GameMode->IMUSensorGravity = Globals.imu_gravity;
615 }
616 // Before the weather, which switches lights only with the cycle on.
617 if ((LightSubsystem != nullptr) && (LightSubsystem->GetDayNightCycle() != Globals.day_night_cycle))
618 {
619 LightSubsystem->SetDayNightCycle(Globals.day_night_cycle);
620 }
621#if defined(WITH_ROS2)
622 auto ROS2 = carla::ros2::ROS2::GetInstance();
623 if (ROS2 && ROS2->IsEnabled() && (ROS2->GetPublishTF() != Globals.publish_tf))
624 {
625 ROS2->SetPublishTF(Globals.publish_tf);
626 }
627#endif
628 }
629
630 if (AWeather *Weather = Episode->GetWeather())
631 {
632 for (const CarlaRecorderWeather &Record : Weathers.GetWeathers())
633 {
634 Weather->ApplyWeather(FFrameData_FromRecord(Record));
635 }
636 }
637
638 // The day/night state the primary's lights follow, even where this
639 // server's own threshold decided otherwise.
640 if (MirrorWorldGlobals && (LightSubsystem != nullptr) &&
641 MirrorWorldGlobals->day_night_cycle && MirrorWorldGlobals->has_day_time_state)
642 {
643 bool bIsDay = true;
644 if (!LightSubsystem->GetDayTimeState(bIsDay) || (bIsDay != MirrorWorldGlobals->last_is_day))
645 {
646 LightSubsystem->NotifyDayTimeChange(MirrorWorldGlobals->last_is_day);
647 }
648 }
649
650 // Environment objects: again whenever this server registers its objects
651 // anew, for those of a layer that was still streaming in.
652 bool bApplyObjects = false;
653 if (MirrorWorldObjects.disabled_environment_objects)
654 {
655 MirrorState.DisabledObjects = *MirrorWorldObjects.disabled_environment_objects;
656 bApplyObjects = true;
657 }
658 if ((GameMode != nullptr) && MirrorState.DisabledObjects &&
659 (bApplyObjects || (GameMode->GetEnvironmentObjectsRegistrations() != MirrorState.DisabledObjectsRegistrations)))
660 {
661 MirrorState.DisabledObjectsRegistrations = GameMode->GetEnvironmentObjectsRegistrations();
662 const auto Diff = carla::multigpu::DiffSortedIds(
663 FFrameData_SortedIds(GameMode->GetDisabledEnvironmentObjects()), *MirrorState.DisabledObjects);
664 if (!Diff.second.empty())
665 {
666 GameMode->EnableEnvironmentObjects(FFrameData_IdSet(Diff.second), true);
667 }
668 if (!Diff.first.empty())
669 {
670 GameMode->EnableEnvironmentObjects(FFrameData_IdSet(Diff.first), false);
671 }
672 }

675void FFrameData::PlayPositions(
676 const std::unordered_map<uint32_t, uint32_t>& MappedId,
677 FMirrorState &MirrorState)
678{
679 MirrorState.PlayedPoses.clear();
680 if (MirrorPositions.positions.empty())
681 {
682 // A frame from a primary without packet 34.
683 for (const CarlaRecorderPosition &Position : Positions.GetPositions())
684 {
685 if (const std::optional<uint32_t> Local = FindMirror(MappedId, Position.DatabaseId))
686 {
687 CarlaRecorderPosition Pos = Position;
688 Pos.DatabaseId = *Local;
689 ProcessReplayerPosition(Pos, Pos, 0.0, 0.0);
690 }
691 }
692 return;
693 }
694
695 std::vector<uint32_t> LocalIds;
696 std::unordered_map<uint32_t, FTransform> Poses;
697 std::unordered_map<uint32_t, uint32_t> Parents;
698 LocalIds.reserve(MirrorPositions.positions.size());
699 for (const carla::multigpu::MirroredPosition &Position : MirrorPositions.positions)
700 {
701 const std::optional<uint32_t> Local = FindMirror(MappedId, Position.id);
702 FCarlaActor *CarlaActor = Local ? Episode->FindCarlaActor(*Local) : nullptr;
703 if (CarlaActor == nullptr)
704 {
705 continue;
706 }
707 LocalIds.push_back(*Local);
708 Poses.emplace(*Local, FTransform(
709 FQuat(Position.rotation[0], Position.rotation[1], Position.rotation[2], Position.rotation[3]),
710 FVector(Position.location[0], Position.location[1], Position.location[2])));
711 if (CarlaActor->GetParent() != 0u)
712 {
713 Parents.emplace(*Local, CarlaActor->GetParent());
714 }
715 }
716 // Parents first: moving a parent moves its attached children, which then
717 // get their own exact pose.
718 for (const uint32_t LocalId : carla::multigpu::OrderParentsFirst(LocalIds, Parents))
719 {
720 FCarlaActor *CarlaActor = Episode->FindCarlaActor(LocalId);
721 const FTransform &Pose = Poses[LocalId];
722 PlaceMirror(*CarlaActor, Pose);
723 MirrorState.PlayedPoses.emplace_back(LocalId, Pose);
724 }
725}

753void FFrameData::PlayKinematics(const std::unordered_map<uint32_t, uint32_t>& MappedId)
754{
755 const uint64_t ThisFrame = FrameCounter.FrameCounter;
756 for (const CarlaRecorderKinematics &Kinematic : Kinematics.GetKinematics())
757 {
758 const std::optional<uint32_t> Local = FindMirror(MappedId, Kinematic.DatabaseId);
759 FCarlaActor *CarlaActor = Local ? Episode->FindCarlaActor(*Local) : nullptr;
760 if (CarlaActor != nullptr)
761 {
762 CarlaActor->SetMirroredKinematics(Kinematic.LinearVelocity, Kinematic.AngularVelocity, ThisFrame);
763 FFrameData_SetMirroredMotion(*CarlaActor, Kinematic.LinearVelocity);
764 }
765 }
766 // A mirror this frame carries no velocities for (a prop that came to
767 // rest) stands still, rather than keep the last ones it got.
768 for (const auto &Mapped : MappedId)
769 {
770 FCarlaActor *CarlaActor = Episode->FindCarlaActor(Mapped.second);
771 if ((CarlaActor != nullptr) && CarlaActor->HasMirroredKinematics() &&
772 (CarlaActor->GetMirroredKinematicsFrame() != ThisFrame))
773 {
774 CarlaActor->ClearMirroredKinematics();
775 FFrameData_SetMirroredMotion(*CarlaActor, FVector::ZeroVector);
776 }
777 }

780void FFrameData::PlaceMirror(FCarlaActor &CarlaActor, FTransform Transform)
781{
782 if (CarlaActor.IsDormant())
783 {
784 if (const FActorData *Data = CarlaActor.GetActorData())
785 {
786 Transform.SetScale3D(Data->Scale);
787 }
788 CarlaActor.SetActorGlobalTransform(Transform, ETeleportType::None);
789 return;
790 }
791 AActor *Actor = CarlaActor.GetActor();
792 if (Actor == nullptr)
793 {
794 return;
795 }
796 // The primary's pose, the mirror's own scale (a spawn attribute).
797 Transform.SetScale3D(Actor->GetActorScale3D());
798 USceneComponent *Root = Actor->GetRootComponent();
799 if ((Root != nullptr) && (Root->GetMobility() != EComponentMobility::Movable))
800 {
801 // A static actor that moved on the primary: such a root refuses to move.
802 if (CarlaActor.GetActorGlobalTransform().Equals(Transform, 1e-3))
803 {
804 return;
805 }
806 Root->SetMobility(EComponentMobility::Movable);
807 }
808 // No teleport: the renderer keeps the previous pose for motion vectors.
809 CarlaActor.SetActorGlobalTransform(Transform, ETeleportType::None);
810}

952carla::multigpu::FrameDataReadResult FFrameData::Read(const uint8_t *Data, size_t Size)
953{
954 Clear();

956 [this](uint8_t Id, std::istream &InStream, uint32_t PacketSize)
957 {
958 switch (static_cast<CarlaRecorderPacketId>(Id))
959 {
960 case CarlaRecorderPacketId::EventAdd:
961 EventsAdd.Read(InStream);
962 return true;
963 case CarlaRecorderPacketId::EventDel:
964 EventsDel.Read(InStream);
965 return true;
966 case CarlaRecorderPacketId::EventParent:
967 EventsParent.Read(InStream);
968 return true;
969 case CarlaRecorderPacketId::Position:
970 Positions.Read(InStream);
971 return true;
972 case CarlaRecorderPacketId::State:
973 States.Read(InStream);
974 return true;
975 case CarlaRecorderPacketId::AnimVehicle:
976 Vehicles.Read(InStream);
977 return true;
978 case CarlaRecorderPacketId::AnimWalker:
979 Walkers.Read(InStream);
980 return true;
981 case CarlaRecorderPacketId::MultiGpuVehicleParts:
982 MirrorVehicleParts.Read(InStream, PacketSize);
983 return true;
984 case CarlaRecorderPacketId::AnimBiker:
985 Bikers.Read(InStream);
986 return true;
987 case CarlaRecorderPacketId::VehicleLight:
988 LightVehicles.Read(InStream);
989 return true;
990 case CarlaRecorderPacketId::SceneLight:
991 LightScenes.Read(InStream);
992 return true;
993 case CarlaRecorderPacketId::Kinematics:
994 Kinematics.Read(InStream);
995 return true;
996 case CarlaRecorderPacketId::TrafficLightTime:
997 TrafficLightTimes.Read(InStream);
998 return true;
999 case CarlaRecorderPacketId::Weather:
1000 Weathers.Read(InStream);
1001 return true;
1002 case CarlaRecorderPacketId::MultiGpuWorldGlobals:
1003 {
1004 carla::multigpu::WorldGlobals Globals;
1005 // A short packet fails InStream, so the codec stops there.
1006 if (Globals.Read(InStream, PacketSize))
1007 {
1008 MirrorWorldGlobals = Globals;
1009 }
1010 return true;
1011 }
1012 case CarlaRecorderPacketId::MultiGpuWorldObjects:
1013 MirrorWorldObjects.Read(InStream, PacketSize);
1014 return true;
1015 case CarlaRecorderPacketId::MultiGpuActorAppearance:
1016 MirrorAppearances.Read(InStream, PacketSize);
1017 return true;
1018 case CarlaRecorderPacketId::MultiGpuSensorTicks:
1019 bHasSensorTicks = MirrorSensorTicks.Read(InStream, PacketSize);
1020 return true;
1021 case CarlaRecorderPacketId::MultiGpuSensorState:
1022 MirrorSensorStates.Read(InStream, PacketSize);
1023 return true;
1024 case CarlaRecorderPacketId::MultiGpuPositions:
1025 // A short packet fails InStream, so the codec stops there.
1026 MirrorPositions.Read(InStream, PacketSize);
1027 return true;
1028 case CarlaRecorderPacketId::MultiGpuFrameTime:
1029 {
1030 carla::multigpu::FrameTime Time;
1031 // A short packet fails InStream, so the codec stops there.
1032 if (Time.Read(InStream, PacketSize))
1033 {
1034 MultiGpuFrameTime = Time;
1035 }
1036 return true;
1037 }
1038 default:
1039 // Unknown here (written by another build): the codec skips it.
1040 return false;
1041 }
1042 });
1043 FrameCounter.FrameCounter = Result.frame.value_or(0u);
1044 carla::multigpu::FrameDataReadResult Checked = Result;
1045 if (Checked.IsComplete() && MultiGpuFrameTime && (MultiGpuFrameTime->frame != FrameCounter.FrameCounter))
1046 {
1047 Checked.status = carla::multigpu::FrameDataStatus::Inconsistent;
1048 Checked.packet_id = carla::multigpu::FrameTime::kPacketId;
1049 }
1050 return Checked;
1051}

1627// create or reuse an actor for replaying
1628std::pair<int, FCarlaActor*> FFrameData::CreateOrReuseActor(
1629 FVector &Location,
1630 FVector &Rotation,
1631 FActorDescription &ActorDesc,
1632 uint32_t DesiredId,
1633 bool SpawnSensors,
1634 std::unordered_map<uint32_t, uint32_t>& MappedId)
1635{
1636 check(Episode != nullptr);
1637
1638 // check type of actor we need
1639 if (ActorDesc.Id.StartsWith("traffic."))
1640 {
1641 FCarlaActor* CarlaActor = FindTrafficSignAt(Location);
1642 if (CarlaActor != nullptr)
1643 {
1644 // reuse that actor
1645 UE_LOG(LogCarla, Log, TEXT("TrafficLight found"));
1646 return std::pair<int, FCarlaActor*>(2, CarlaActor);
1647 }
1648 else
1649 {
1650 // actor not found
1651 UE_LOG(LogCarla, Log, TEXT("TrafficLight not found"));
1652 return std::pair<int, FCarlaActor*>(0, nullptr);
1653 }
1654 }
1655 else if (SpawnSensors || !ActorDesc.Id.StartsWith("sensor."))
1656 {
1657 const carla::multigpu::ReplayedActorPlan Plan = carla::multigpu::PlanReplayedActor(
1658 MappedId, DesiredId, FReplayedActorLookup(*Episode, ActorDesc, Location));
1659 if (Plan.reuse)
1660 {
1661 FCarlaActor *CarlaActor = Episode->FindCarlaActor(*Plan.reuse);
1662 FRotator Rot = FRotator::MakeFromEuler(Rotation);
1663 PlaceMirror(*CarlaActor, FTransform(Rot, Location));
1664 return std::pair<int, FCarlaActor*>(2, CarlaActor);
1665 }
1666 // create new actor
1667 // create the transform
1668 FRotator Rot = FRotator::MakeFromEuler(Rotation);
1669 FTransform Trans(Rot, FVector(0, 0, 100000), FVector(1, 1, 1));
1670 // create as new actor
1671 TPair<EActorSpawnResultStatus, FCarlaActor*> Result = Episode->SpawnActorWithInfo(Trans, ActorDesc, Plan.spawn_id);
1672 if (Result.Key == EActorSpawnResultStatus::Success)
1673 {
1674 // relocate, keeping the scale the actor was spawned with (and moving
1675 // a static root, e.g. a mass-less prop, which refuses to move otherwise)
1676 PlaceMirror(*Result.Value, FTransform(Rot, Location));
1677 ALargeMapManager * LargeMapManager = UCarlaStatics::GetLargeMapManager(Episode->GetWorld());
1678 if (LargeMapManager)
1679 {
1680 LargeMapManager->OnActorSpawned(*Result.Value);
1681 }
1682 return std::pair<int, FCarlaActor*>(1, Result.Value);
1683 }
1684 else
1685 {
1686 UE_LOG(LogCarla, Log, TEXT("Actor could't be created"));
1687 return std::pair<int, FCarlaActor*>(0, Result.Value);
1688 }
1689 }
1690 else
1691 {
1692 // actor ignored
1693 return std::pair<int, FCarlaActor*>(0, nullptr);
1694 }
1695}

1697// replay event for creating actor
1698std::pair<int, uint32_t> FFrameData::ProcessReplayerEventAdd(
1699 FVector Location,
1700 FVector Rotation,
1701 CarlaRecorderActorDescription Description,
1702 uint32_t DesiredId,
1703 bool bIgnoreHero,
1704 bool ReplaySensors,
1705 std::unordered_map<uint32_t, uint32_t>& MappedId)
1706{
1707 check(Episode != nullptr);
1708 FActorDescription ActorDesc;
1709 bool IsHero = false;
1710
1711 // prepare actor description
1712 ActorDesc.UId = Description.UId;
1713 ActorDesc.Id = Description.Id;
1714 for (const auto &Item : Description.Attributes)
1715 {
1716 FActorAttribute Attr;
1717 Attr.Type = static_cast<EActorAttributeType>(Item.Type);
1718 Attr.Id = Item.Id;
1719 Attr.Value = Item.Value;
1720 ActorDesc.Variations.Add(Attr.Id, std::move(Attr));
1721 // check for hero
1722 if (Item.Id == "role_name" && Item.Value == "hero")
1723 IsHero = true;
1724 }
1725
1726 auto result = CreateOrReuseActor(
1727 Location,
1728 Rotation,
1729 ActorDesc,
1730 DesiredId,
1731 ReplaySensors,
1732 MappedId);
1733
1734 if (result.first != 0)
1735 {
1736 // disable physics and autopilot on vehicles
1737 if (result.second->GetActorType() == FCarlaActor::ActorType::Vehicle)
1738 {
1739 // ignore hero ?
1740 if (!(bIgnoreHero && IsHero))
1741 {
1742 // disable physics
1743 SetActorSimulatePhysics(result.second, false);
1744 // disable autopilot
1745 // SetActorAutopilot(result.second, false, false);
1746 }
1747 else
1748 {
1749 // reenable physics just in case
1750 SetActorSimulatePhysics(result.second, true);
1751 }
1752 }
1753 return std::make_pair(result.first, result.second->GetActorId());
1754 }
1755 return std::make_pair(result.first, 0);

1772// replay event for parenting actors
1773bool FFrameData::ProcessReplayerEventParent(
1774 uint32_t ChildId,
1775 uint32_t ParentId,
1776 const std::unordered_map<uint32_t, uint32_t>& MappedId)
1777{
1778 check(Episode != nullptr);
1779 const std::optional<uint32_t> LocalChild = FindMirror(MappedId, ChildId);
1780 const std::optional<uint32_t> LocalParent = FindMirror(MappedId, ParentId);
1781 FCarlaActor *Child = LocalChild ? Episode->FindCarlaActor(*LocalChild) : nullptr;
1782 FCarlaActor *Parent = LocalParent ? Episode->FindCarlaActor(*LocalParent) : nullptr;
1783 if ((Child == nullptr) || (Parent == nullptr))
1784 {
1785 return false;
1786 }
1787 // Every full resync replays every attachment: only a new one registers the
1788 // ROS 2 parent names, which pile up otherwise ("hero/hero").
1789 const bool bNewParent = (Child->GetParent() != Parent->GetActorId());
1790 Child->SetParent(Parent->GetActorId());
1791 Child->SetAttachmentType(carla::rpc::AttachmentType::Rigid);
1792 Parent->AddChildren(Child->GetActorId());
1793 if (Parent->IsDormant())
1794 {
1795 // Putting a parent to sleep puts its children to sleep too.
1796 if (!Child->IsDormant())
1797 {
1798 Episode->PutActorToSleep(Child->GetActorId());
1799 }
1800 }
1801 else if (!Child->IsDormant() && (Child->GetActor() != nullptr) && (Parent->GetActor() != nullptr))
1802 {
1803 // Always rigid, keeping the child where it is: every frame sets its exact
1804 // pose (after its parent's), whatever attachment the primary used.
1805 AActor *ChildActor = Child->GetActor();
1806 AActor *ParentActor = Parent->GetActor();
1807 if (ChildActor->GetAttachParentActor() != ParentActor)
1808 {
1809 ChildActor->AttachToActor(ParentActor, FAttachmentTransformRules::KeepWorldTransform);
1810 }
1811 // Sensors read their parent's motion through their owner (the IMU). A
1812 // collision sensor binds its owner's hit event in SetOwner.
1813 if (ChildActor->GetOwner() != ParentActor)
1814 {
1815 ChildActor->SetOwner(ParentActor);
1816 }
1817 }
1818 if (bNewParent)
1819 {
1820 Episode->AddRosParentNames(*Child);
1821 }
1822 return true;
1823}

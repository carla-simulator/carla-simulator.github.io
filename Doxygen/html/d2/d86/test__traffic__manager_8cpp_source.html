138 // before the first waypoint, so the helper brackets with the first segment

206constexpr float kSampleSpacing{5.0f};
207constexpr float kLateralAcceleration{3.0f};
208constexpr float kBrakingDeceleration{2.0f};
209constexpr float kFreeSpeed{30.0f};
210
211/// A straight run along +x followed by a left arc, sampled every `spacing` m.
212std::vector<cg::Location> StraightThenArc(
213 float straight_length,
214 float radius,
215 float arc_degrees,
216 float spacing) {
217
218 std::vector<cg::Location> path;
219 for (float s = 0.0f; s < straight_length; s += spacing) {
220 path.push_back(cg::Location{s, 0.0f, 0.0f});
221 }
222 const cg::Location centre{straight_length, radius, 0.0f};
223 const float arc_radians = arc_degrees * 3.14159265f / 180.0f;
224 for (float travelled = 0.0f; travelled <= radius * arc_radians; travelled += spacing) {
225 const float angle = travelled / radius;
226 path.push_back(cg::Location{centre.x + radius * std::sin(angle),
227 centre.y - radius * std::cos(angle),
228 0.0f});
229 }
230 return path;
231}

373 // t = 1 - max_offset_point.

378TEST(TrafficManagerWideTurn, OffsetProfile_InboardExcursionIsCapped) {
379 // The inboard (positive) excursion is the exit cut-in that drives the rear
380 // into the inside shoulder. Across the whole junction it must never exceed
381 // inboard_scale * max_offset, and never fall below -max_offset.
382 const float inboard_cap{kProfileInboardScale * kProfileMaxOffset};
383 for (int i = 0; i <= 100; ++i) {
384 const float t{static_cast<float>(i) / 100.0f};
385 const float offset{SampleProfile(t, kProfileInboardScale)};
386 EXPECT_LE(offset, inboard_cap + kProfileTolerance);
387 EXPECT_GE(offset, -kProfileMaxOffset - kProfileTolerance);
388 }
389}

531struct LightRefreshTrace {
532 int refreshes{0};
533 int steps_from_registration_to_a_refresh{-1};
534};

536/// Mirrors VehicleLightStage's refresh cadence for a vehicle registered on
537/// @a registration_step that the server never reports; pass @a steps for none.
538LightRefreshTrace TraceLightRefreshes(
539 const int steps,
540 const double step_period,
541 const int registration_step) {
542
543 LightRefreshTrace trace{};
544 double last_update{kNeverRead};
545 bool missing_from_the_last_refresh{false};
546
547 for (int step{0}; step < steps; ++step) {
548 const double now{step_period * static_cast<double>(step + 1)};
549 const bool refreshed{
550 refresh::IsRefreshDue(now, last_update, refresh::VEHICLE_LIGHT_STATES_REFRESH_PERIOD)};
551 if (refreshed) {
552 last_update = now;
553 missing_from_the_last_refresh = false;
554 ++trace.refreshes;
555 }
556
557 if (step < registration_step) {
558 continue;
559 }
560 if (refreshed && trace.steps_from_registration_to_a_refresh < 0) {
561 trace.steps_from_registration_to_a_refresh = step - registration_step;
562 }
563 if (refresh::IsEarlyRefreshDue(refreshed, missing_from_the_last_refresh)) {
564 last_update = kNeverRead;
565 }
566 if (refreshed) {
567 missing_from_the_last_refresh = true;
568 }
569 }
570 return trace;
571}

574constexpr double kThirtyFpsPeriod{1.0 / 30.0};

614struct BatchSyncTrace {
615 int syncs{0};
616 uint64_t deepest_queue{0u};
617};
618
619BatchSyncTrace TraceBatchSyncs(const int steps, const uint64_t limit) {
620 BatchSyncTrace trace{};
621 uint64_t unwaited{limit};
622 for (int step{0}; step < steps; ++step) {
623 if (refresh::IsBatchSyncDue(unwaited, limit)) {
624 unwaited = 0u;
625 ++trace.syncs;
626 } else {
627 ++unwaited;
628 }
629 trace.deepest_queue = std::max(trace.deepest_queue, unwaited);
630 }
631 return trace;
632}
633
634/// The same queue bounded by simulation time instead.
635BatchSyncTrace TraceTimeGatedBatchSyncs(
636 const int steps,
637 const double fixed_delta_seconds,
638 const double period) {
639
640 BatchSyncTrace trace{};
641 uint64_t unwaited{0u};
642 double last_sync{kNeverRead};
643 for (int step{0}; step < steps; ++step) {
644 const double now{fixed_delta_seconds * static_cast<double>(step + 1)};
645 if (refresh::IsRefreshDue(now, last_sync, period)) {
646 unwaited = 0u;
647 last_sync = now;
648 ++trace.syncs;
649 } else {
650 ++unwaited;
651 }
652 trace.deepest_queue = std::max(trace.deepest_queue, unwaited);
653 }
654 return trace;
655}

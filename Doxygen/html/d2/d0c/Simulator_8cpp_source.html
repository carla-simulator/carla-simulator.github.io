38 static void ValidateVersions(Client &client) {
39 const auto vc = client.GetClientVersion();
40 const auto vs = client.GetServerVersion();
41 if (vc != vs) {
42 log_warning(
43 "Version mismatch detected: You are trying to connect to a simulator",
44 "that might be incompatible with this API");
45 log_warning("Client API version =", vc);
46 log_warning("Simulator API version =", vs);
47 }
48 }

50 static bool SynchronizeFrame(uint64_t frame, const Episode &episode, time_duration timeout) {
51 bool result = true;
52 auto start = std::chrono::system_clock::now();
53 while (frame > episode.GetState()->GetTimestamp().frame) {
54 std::this_thread::sleep_for(std::chrono::microseconds(100));
55 auto end = std::chrono::system_clock::now();
56 auto diff = std::chrono::duration_cast<std::chrono::milliseconds>(end-start);
57 if(timeout.to_chrono() < diff) {
58 result = false;
59 break;
60 }
61 }
62 if(result) {
63 carla::traffic_manager::TrafficManager::Tick();
64 }
65
66 return result;
67 }

93 EpisodeProxy Simulator::LoadEpisode(std::string map_name, bool reset_settings, rpc::MapLayer map_layers) {
94 const auto id = GetCurrentEpisode().GetId();
95 _client.LoadEpisode(std::move(map_name), reset_settings, map_layers);
96
97 // We are waiting 50ms for the server to reload the episode.
98 // If in this time we have not detected a change of episode, we try again
99 // 'number_of_attempts' times.
100 // TODO This time is completly arbitrary so we need to improve
101 // this pipeline to not depend in this time because this timeout
102 // could result that the client resume the simulation in different
103 // initial ticks when loading a map in syncronous mode.
104 size_t number_of_attempts = _client.GetTimeout().milliseconds() / 50u;
105
106 for (auto i = 0u; i < number_of_attempts; ++i) {
107 using namespace std::literals::chrono_literals;
108 if (_client.GetEpisodeSettings().synchronous_mode)
109 _client.SendTickCue();
110
111 _episode->WaitForState(50ms);
112 auto episode = GetCurrentEpisode();
113
114 if (episode.GetId() != id) {
115 // Wait for the stream thread to reset the Traffic Manager for it.
116 const auto deadline = std::chrono::steady_clock::now() + _client.GetTimeout().to_chrono();
117 while (!_episode->IsEpisodeChangeHandled()) {
118 if (std::chrono::steady_clock::now() > deadline) {
119 log_warning("the Traffic Manager has not finished resetting for the new episode");
120 break;
121 }
122 std::this_thread::sleep_for(10ms);
123 }
124 return episode;
125 }
126 }
127 throw_exception(std::runtime_error("failed to connect to newly created map"));
128 }

172 SharedPtr<Map> Simulator::GetCurrentMap() {
173 DEBUG_ASSERT(_episode != nullptr);
174 // Check-and-refetch must be atomic: the Traffic Manager refreshes the map
175 // from its own thread on an episode change, and a caller that ran while
176 // that refresh was still downloading the OpenDRIVE found the change flag
177 // already consumed and the cache still holding the previous map. The
178 // cache is also keyed on the episode id, so a missed change flag cannot
179 // leave a stale map behind either.
180 std::lock_guard<std::mutex> lock(_cached_map_mutex);
181 const auto episode_id = _episode->GetId();
182 if (!_cached_map || _cached_map_episode_id != episode_id || _episode->HasMapChangedSinceLastCall()) {
183 rpc::MapInfo map_info = _client.GetMapInfo();
184 std::string map_name;
185 std::string map_base_path;
186 bool fill_base_string = false;
187 for (size_t ri = map_info.name.size(); ri > 0; --ri) {
188 size_t i = ri - 1;
189 if (fill_base_string == false && map_info.name[i] != '/') {
190 map_name += map_info.name[i];
191 } else {
192 map_base_path += map_info.name[i];
193 fill_base_string = true;
194 }
195 }
196 std::reverse(map_name.begin(), map_name.end());
197 std::reverse(map_base_path.begin(), map_base_path.end());
198 std::string XODRFolder = map_base_path + "/OpenDrive/" + map_name + ".xodr";
199 if (FileTransfer::FileExists(XODRFolder) == false) _client.GetRequiredFiles();
200 _open_drive_file = _client.GetMapData();
201 _cached_map = MakeShared<Map>(map_info, _open_drive_file);
202 _cached_map_episode_id = episode_id;
203 }
204
205 return _cached_map;
206 }

233 WorldSnapshot Simulator::WaitForTick(time_duration timeout) {
234 DEBUG_ASSERT(_episode != nullptr);
235
236 // tick pedestrian navigation
237 NavigationTick();
238
239 auto result = _episode->WaitForState(timeout);
240 if (!result.has_value()) {
241 throw_exception(TimeoutException(_client.GetEndpoint(), timeout));
242 }

246 uint64_t Simulator::Tick(time_duration timeout) {
247 DEBUG_ASSERT(_episode != nullptr);
248
249 // tick pedestrian navigation
250 NavigationTick();
251
252 // send tick command
253 const auto frame = _client.SendTickCue();
254
255 // waits until new episode is received
256 bool result = SynchronizeFrame(frame, *_episode, timeout);
257 if (!result) {
258 throw_exception(TimeoutException(_client.GetEndpoint(), timeout));
259 }
260 return frame;
261 }

280 uint64_t Simulator::SetEpisodeSettings(const rpc::EpisodeSettings &settings) {
281 if (settings.synchronous_mode && !settings.fixed_delta_seconds) {
282 log_warning(
283 "synchronous mode enabled with variable delta seconds. It is highly "
284 "recommended to set 'fixed_delta_seconds' when running on synchronous mode.");
285 }
286 else if (settings.synchronous_mode && settings.substepping) {
287 if(settings.max_substeps < 1 || settings.max_substeps > 16) {
288 log_warning(
289 "synchronous mode and substepping are enabled but the number of substeps is not valid. "
290 "Please be aware that this value needs to be in the range [1-16].");
291 }
292 double n_substeps = settings.fixed_delta_seconds.value() / settings.max_substep_delta_time;
293
294 if (n_substeps > static_cast<double>(settings.max_substeps)) {
295 log_warning(
296 "synchronous mode and substepping are enabled but the values for the simulation are not valid. "
297 "The values should fulfil fixed_delta_seconds <= max_substep_delta_time * max_substeps. "
298 "Be very careful about that, the time deltas are not guaranteed.");
299 }
300 }
301 const auto frame = _client.SetEpisodeSettings(settings);
302
303 using namespace std::literals::chrono_literals;
304 SynchronizeFrame(frame, *_episode, 1s);
305
306 return frame;
307 }

353 std::optional<geom::Location> Simulator::GetRandomLocationFromNavigation() {
354 DEBUG_ASSERT(_episode != nullptr);
355 auto nav = _episode->CreateNavigationIfMissing();
356 if (nav->IsServerSide()) {
357 // The server samples the loaded navmesh; the sentinel z = -1e6 means it
358 // found no point (e.g. tiles still streaming in), so try a few times
359 // before giving up like the legacy path does (empty optional).
360 constexpr int attempts = 3;
361 for (int i = 0; i < attempts; ++i) {
362 auto location = _client.GetRandomLocationFromNavigation();
363 if (location.z > -9.0e5f) {
364 return location;
365 }
366 }
367 return {};
368 }
369 return nav->GetRandomLocation();
370 }

396 SharedPtr<Actor> Simulator::SpawnActor(
397 const ActorBlueprint &blueprint,
398 const geom::Transform &transform,
399 Actor *parent,
400 rpc::AttachmentType attachment_type,
401 GarbageCollectionPolicy gc) {
402 rpc::Actor actor;
403 if (parent != nullptr) {
404 actor = _client.SpawnActorWithParent(
405 blueprint.MakeActorDescription(),
406 transform,
407 parent->GetId(),
408 attachment_type);
409 } else {
410 actor = _client.SpawnActor(
411 blueprint.MakeActorDescription(),
412 transform);
413 }
414 DEBUG_ASSERT(_episode != nullptr);
415 _episode->RegisterActor(actor);
416 const auto gca = (gc == GarbageCollectionPolicy::Inherit ? _gc_policy : gc);
417 auto result = ActorFactory::MakeActor(GetCurrentEpisode(), actor, gca);
418 // Snapshots up to (and including) this frame predate the spawn; see
419 // Simulator::GetActorState.
420 static_cast<ActorState &>(*result)._spawn_frame = _episode->GetState()->GetFrame();
421 log_debug(
422 result->GetDisplayId(),
423 "created",
424 gca == GarbageCollectionPolicy::Enabled ? "with" : "without",
425 "garbage collection");
426 return result;
427 }

430 bool success = true;

432 if (success) {
433 // Remove it's persistent state so it cannot access the client anymore.
434 actor.GetEpisode().Clear();
435 log_debug(actor.GetDisplayId(), "destroyed.");
436 } else {
437 log_debug("failed to destroy", actor.GetDisplayId());
438 }
439 return success;

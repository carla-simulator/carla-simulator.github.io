57/// What a Multi-GPU secondary keeps between the primary frames it plays.
58struct FMirrorState
59{
60 /// Local ids of the mirrors this secondary spawned, as opposed to map actors
61 /// it reused (traffic lights, the spectator): only these may be swept.
62 std::unordered_set<uint32_t> CreatedIds;
63
64 /// Attachments not replayed yet, as (child, parent) primary ids: one of the
65 /// two is not mirrored yet.
66 std::vector<std::pair<uint32_t, uint32_t>> PendingParents;
67
68 /// The poses the last frame played set, by local id, to check before the
69 /// capture that nothing moved a mirror in between.
70 std::vector<std::pair<uint32_t, FTransform>> PlayedPoses;
71
72 /// The environment objects the primary has disabled (sorted), once a frame
73 /// said so, and the registration of this server's objects they were last
74 /// applied to: one in a map layer still streaming in is disabled once it
75 /// is registered.
76 std::optional<std::vector<uint64_t>> DisabledObjects;
77 uint64 DisabledObjectsRegistrations = 0u;
78
79 void Reset()
80 {
81 CreatedIds.clear();
82 PendingParents.clear();
83 PlayedPoses.clear();
84 DisabledObjects.reset();
85 DisabledObjectsRegistrations = 0u;
86 }
87};

89/// An actor's appearance, as a Multi-GPU primary last sent it.
90struct FMirroredAppearance
91{
92 bool bCollisions = true;
93 bool bHidden = false;
94 FVector Scale = FVector::OneVector;
95
96 bool operator==(const FMirroredAppearance &Other) const
97 {
98 return (bCollisions == Other.bCollisions) && (bHidden == Other.bHidden) &&
99 Scale.Equals(Other.Scale, 0.0);
100 }
101};

28static bool ValidateStaticMesh(UStaticMesh *Mesh)
29{
30 const FString AssetName = Mesh->GetName();
31
32 if (AssetName.Contains(TEXT("light"), ESearchCase::IgnoreCase) ||
33 AssetName.Contains(TEXT("sign"), ESearchCase::IgnoreCase))
34 {
35 return false;
36 }
37
38 for (int i = 0; i < Mesh->GetStaticMaterials().Num(); i++)
39 {
40 UMaterialInterface *Material = Mesh->GetMaterial(i);
41 const FString MaterialName = Material->GetName();
42
43 if (MaterialName.Contains(TEXT("light"), ESearchCase::IgnoreCase) ||
44 MaterialName.Contains(TEXT("sign"), ESearchCase::IgnoreCase))
45 {
46 return false;
47 }
48 }
49
50 return true;
51}

165TArray<AStaticMeshActor *> UPrepareAssetsForCookingCommandlet::SpawnMeshesToWorld(
166 const TArray<FString> &AssetsPaths,
167 bool bUseCarlaMaterials,
168 int i,
169 int j)
170{
171 TArray<AStaticMeshActor *> SpawnedMeshes;
172
173 // Create default Transform for all assets to spawn
174 const FTransform ZeroTransform = FTransform();
175
176 // Load assets specified in AssetsPaths by using an object library
177 // for building map world
178 AssetsObjectLibrary = UObjectLibrary::CreateLibrary(UStaticMesh::StaticClass(), false, GIsEditor);
179 AssetsObjectLibrary->AddToRoot();
180 AssetsObjectLibrary->LoadAssetDataFromPaths(AssetsPaths);
181 AssetsObjectLibrary->LoadAssetsFromAssetData();
182 MapContents.Empty();
183 AssetsObjectLibrary->GetAssetDataList(MapContents);
184
185 UStaticMesh *MeshAsset;
186 AStaticMeshActor *MeshActor;
187
188 // name of current tile to cook
189 FString TileName;
190 if (i != -1)
191 {
192 TileName = FString::Printf(TEXT("_Tile_%d_%d"), i, j);
193 }
194
195 // try to get the name of the map that precedes all assets name
196 FString AssetName;
197 for (auto MapAsset : MapContents)
198 {
199 // Spawn Static Mesh
200 MeshAsset = Cast<UStaticMesh>(MapAsset.GetAsset());
201 if (MeshAsset && ValidateStaticMesh(MeshAsset))
202 {
203 // get asset name
204 MapAsset.AssetName.ToString(AssetName);
205
206 // check to ignore meshes from other tiles
207 if (i == -1 || (i != -1 && (AssetName.EndsWith(TileName) || AssetName.Contains(TileName + "_"))))
208 {
209 MeshActor = World->SpawnActor<AStaticMeshActor>(AStaticMeshActor::StaticClass(), ZeroTransform);
210 UStaticMeshComponent *MeshComponent = MeshActor->GetStaticMeshComponent();
211 MeshComponent->SetStaticMesh(CastChecked<UStaticMesh>(MeshAsset));
212 MeshActor->SetActorLabel(AssetName, true);
213
214 // set complex collision as simple in asset
215 UBodySetup *BodySetup = MeshAsset->GetBodySetup();
216 if (BodySetup)
217 {
218 BodySetup->CollisionTraceFlag = CTF_UseComplexAsSimple;
219 MeshAsset->MarkPackageDirty();
220 }
221
222 SpawnedMeshes.Add(MeshActor);
223
224 if (bUseCarlaMaterials)
225 {
226 // Set Carla Materials depending on RoadRunner's Semantic Segmentation
227 // tag
228 if (AssetName.Contains(SSTags::R_MARKING1) || AssetName.Contains(SSTags::R_MARKING2))
229 {
230 for (int32 materialIndex = 0; materialIndex < MeshActor->GetStaticMeshComponent()->GetStaticMesh()->GetStaticMaterials().Num(); ++materialIndex)
231 {
232 if (MeshActor->GetStaticMeshComponent()->GetStaticMesh()->GetStaticMaterials()[materialIndex].ImportedMaterialSlotName.ToString().Contains("Yellow"))
233 {
234 MeshActor->GetStaticMeshComponent()->SetMaterial(materialIndex, MarkingNodeYellow);
235 }
236 else
237 {
238 MeshActor->GetStaticMeshComponent()->SetMaterial(materialIndex, MarkingNodeWhite);
239 }
240 }
241 }
242 else if (AssetName.Contains(SSTags::R_ROAD1) || AssetName.Contains(SSTags::R_ROAD2))
243 {
244 MeshActor->GetStaticMeshComponent()->SetMaterial(0, RoadNodeMaterial);
245 }
246 else if (AssetName.Contains(SSTags::R_TERRAIN))
247 {
248 MeshActor->GetStaticMeshComponent()->SetMaterial(0, TerrainNodeMaterialInstance);
249 MeshActor->GetStaticMeshComponent()->bReceivesDecals = false;
250 }
251 else if (AssetName.Contains(SSTags::R_SIDEWALK1) || AssetName.Contains(SSTags::R_SIDEWALK2))
252 {
253 MeshActor->GetStaticMeshComponent()->SetMaterial(0, SidewalkNodeMaterialInstance);
254 MeshActor->GetStaticMeshComponent()->bReceivesDecals = false;
255 }
256 else if (AssetName.Contains(SSTags::R_CURB1) || AssetName.Contains(SSTags::R_CURB2)) {
257
258 MeshActor->GetStaticMeshComponent()->SetMaterial(0, CurbNodeMaterialInstance);
259 MeshActor->GetStaticMeshComponent()->bReceivesDecals = false;
260 }
261 else if (AssetName.Contains(SSTags::R_GUTTER1) || AssetName.Contains(SSTags::R_GUTTER2)) {
262
263 MeshActor->GetStaticMeshComponent()->SetMaterial(0, GutterNodeMaterialInstance);
264 MeshActor->GetStaticMeshComponent()->bReceivesDecals = false;
265 }
266 }
267 }
268 }
269 }
270
271 // Clear loaded assets in library
272 AssetsObjectLibrary->ClearLoaded();
273
274 // Mark package dirty
275 World->MarkPackageDirty();
276
277 return SpawnedMeshes;
278}

280bool UPrepareAssetsForCookingCommandlet::IsMapInTiles(const TArray<FString> &AssetsPaths)
281{
282 // Load assets specified in AssetsPaths by using an object library
283 // for building map world
284 AssetsObjectLibrary = UObjectLibrary::CreateLibrary(UStaticMesh::StaticClass(), false, GIsEditor);
285 AssetsObjectLibrary->AddToRoot();
286 AssetsObjectLibrary->LoadAssetDataFromPaths(AssetsPaths);
287 AssetsObjectLibrary->LoadAssetsFromAssetData();
288 MapContents.Empty();
289 AssetsObjectLibrary->GetAssetDataList(MapContents);
290
291 UStaticMesh *MeshAsset;
292
293 FString AssetName;
294 bool Found = false;
295 for (auto MapAsset : MapContents)
296 {
297 // Spawn Static Mesh
298 MeshAsset = Cast<UStaticMesh>(MapAsset.GetAsset());
299 if (MeshAsset && ValidateStaticMesh(MeshAsset))
300 {
301 // get asset name
302 MapAsset.AssetName.ToString(AssetName);
303
304 // check if the asset is a tile
305 if (AssetName.Contains("_Tile_"))
306 {
307 Found = true;
308 break;
309 }
310 }
311 }
312
313 // Clear loaded assets in library
314 AssetsObjectLibrary->ClearLoaded();
315
316 return Found;
317}

332bool UPrepareAssetsForCookingCommandlet::SaveWorld(
333 FAssetData &AssetData,
334 const FString &PackageName,
335 const FString &DestPath,
336 const FString &WorldName,
337 bool bGenerateSpawnPoints)
338{
339 // Create Package to save
340 UPackage *Package = AssetData.GetPackage();
341 Package->FullyLoad();
342 Package->MarkPackageDirty();
343 FAssetRegistryModule::AssetCreated(World);
344
345 // Renaming map
346 World->Rename(*WorldName, World->GetOuter());
347 const FString PackagePath = DestPath + "/" + WorldName;
348 FAssetRegistryModule::AssetRenamed(World, *PackagePath);
349 World->MarkPackageDirty();
350 World->GetOuter()->MarkPackageDirty();
351
352 // Check if OpenDrive file exists
353 const FString PathXODR = FPaths::ProjectContentDir() + PackageName + TEXT("/Maps/") +
354 WorldName + TEXT("/OpenDrive/") + WorldName + TEXT(".xodr");
355
356 bool bPackageSaved = false;
357 if (FPaths::FileExists(PathXODR) && bGenerateSpawnPoints)
358 {
359 // We need to spawn OpenDrive assets before saving the map
360 AOpenDriveActor *OpenWorldActor = CastChecked<AOpenDriveActor>(
361 World->SpawnActor(AOpenDriveActor::StaticClass(),
362 new FVector(),
363 NULL));
364
365 OpenWorldActor->BuildRoutes(WorldName);
366 OpenWorldActor->AddSpawners();
367
368 bPackageSaved = SavePackage(PackagePath, Package);
369
370 // We need to destroy OpenDrive assets once saved the map
371 OpenWorldActor->RemoveRoutes();
372 OpenWorldActor->RemoveSpawners();
373 OpenWorldActor->Destroy();
374 }
375 else
376 {
377 bPackageSaved = SavePackage(PackagePath, Package);
378 }
379
380 return bPackageSaved;
381}

399FAssetsPaths UPrepareAssetsForCookingCommandlet::GetAssetsPathFromPackage(const FString &PackageName) const
400{
401 const FString PackageJsonFilePath = GetFirstPackagePath(PackageName);
402
403 FAssetsPaths AssetsPaths;
404
405 // Get All Maps Path
406 FString MapsFileJsonContent;
407 if (FFileHelper::LoadFileToString(MapsFileJsonContent, *PackageJsonFilePath))
408 {
409 TSharedPtr<FJsonObject> JsonParsed;
410 TSharedRef<TJsonReader<TCHAR>> JsonReader = TJsonReaderFactory<TCHAR>::Create(MapsFileJsonContent);
411 if (FJsonSerializer::Deserialize(JsonReader, JsonParsed))
412 {
413 // Add Maps Path
414 auto MapsJsonArray = JsonParsed->GetArrayField(TEXT("maps"));
415
416 for (auto &MapJsonValue : MapsJsonArray)
417 {
418 TSharedPtr<FJsonObject> MapJsonObject = MapJsonValue->AsObject();
419
420 FMapData MapData;
421 MapData.Name = MapJsonObject->GetStringField(TEXT("name"));
422 MapData.Path = MapJsonObject->GetStringField(TEXT("path"));
423 MapData.bUseCarlaMapMaterials = MapJsonObject->GetBoolField(TEXT("use_carla_materials"));
424 MapJsonObject->TryGetBoolField(TEXT("merge_tiles"), MapData.bMergeTiles);
425
426 AssetsPaths.MapsPaths.Add(std::move(MapData));
427 }
428
429 // Add Props Path
430 auto PropJsonArray = JsonParsed->GetArrayField(TEXT("props"));
431
432 for (auto &PropJsonValue : PropJsonArray)
433 {
434 TSharedPtr<FJsonObject> PropJsonObject = PropJsonValue->AsObject();
435
436 const FString PropAssetPath = PropJsonObject->GetStringField(TEXT("path"));
437
438 AssetsPaths.PropsPaths.Add(std::move(PropAssetPath));
439 }
440 }
441 }
442 return AssetsPaths;
443}

552void UPrepareAssetsForCookingCommandlet::PrepareMapsForCooking(
553 const FString &PackageName,
554 const TArray<FMapData> &MapsPaths)
555{
556 FString BasePath = TEXT("/Game/") + PackageName + TEXT("/Static/");
557
558 for (const auto &Map : MapsPaths)
559 {
560 const FString MapPath = TEXT("/") + Map.Name;
561
562 const FString DefaultPath = TEXT("/Game/") + PackageName + TEXT("/Maps/") + Map.Name;
563 const FString RoadsPath = BasePath + SSTags::ROAD + MapPath;
564 const FString RoadLinesPath = BasePath + SSTags::ROADLINE + MapPath;
565 const FString TerrainPath = BasePath + SSTags::TERRAIN + MapPath;
566 const FString SidewalkPath = BasePath + SSTags::SIDEWALK + MapPath;
567
568 // Spawn assets located in semantic segmentation folders
569 TArray<FString> DataPath = {DefaultPath, RoadsPath, RoadLinesPath, TerrainPath, SidewalkPath};
570
571 // check whether we have a single map or a map in tiles
572 // A merged tiled map imports its tiles into one world: the .fbx vertices are
573 // already in world space, so only the sublevel split has to be skipped.
574 if (Map.bMergeTiles || !IsMapInTiles(DataPath))
575 {
576 UE_LOG(LogTemp, Log, TEXT("Cooking map"));
577 // Load World
578 FAssetData AssetData;
579 LoadWorld(AssetData);
580 UObjectRedirector *BaseMapRedirector = Cast<UObjectRedirector>(AssetData.GetAsset());
581 if (BaseMapRedirector != nullptr) {
582 World = CastChecked<UWorld>(BaseMapRedirector->DestinationObject);
583 }
584 else {
585 World = CastChecked<UWorld>(AssetData.GetAsset());
586 }
587 // try to cook the whole map (no tiles)
588 TArray<AStaticMeshActor *> SpawnedActors = SpawnMeshesToWorld(DataPath, Map.bUseCarlaMapMaterials, -1, -1);
589 // Save the World in specified path
590 SaveWorld(AssetData, PackageName, Map.Path, Map.Name);
591 // Remove spawned actors from world to keep equal as BaseMap
592 DestroySpawnedActorsInWorld(SpawnedActors);
593 }
594 else
595 {
596 TArray<TPair<FString, FIntVector>> MapPathsIds;
597
598 FVector PositionTile0 = FVector();
599 float TileSize = 200000.f;
600 FString TxtFile;
601 FString TilesInfoPath = FPaths::ProjectContentDir() + PackageName + TEXT("/Maps/") + Map.Name + "/TilesInfo.txt";
602 UE_LOG(LogTemp, Warning, TEXT("Loading %s ..."), *TilesInfoPath);
603 if (FFileHelper::LoadFileToString(TxtFile, *(TilesInfoPath)) == true) {
604
605 TArray<FString> Out;
606 TxtFile.ParseIntoArray(Out, TEXT(","), true);
607 if (Out.Num() >= 3)
608 {
609 const float METERSTOCM = 100.f;
610 PositionTile0.X = METERSTOCM * FCString::Atof(*Out[0]);
611 PositionTile0.Y = METERSTOCM * FCString::Atof(*Out[1]);
612 TileSize = METERSTOCM * FCString::Atof(*Out[2]);
613 }
614 else
615 {
616 UE_LOG(LogTemp, Warning, TEXT("TilesInfo.txt format is invalid file"));
617 }
618 }
619 else {
620 UE_LOG(LogTemp, Warning, TEXT("Could not find TilesInfo.txt file"));
621 }
622
623 UE_LOG(LogTemp, Log, TEXT("Cooking tiles:"));
624 // Load World
625 FAssetData AssetData;
626 LoadWorldTile(AssetData);
627 UObjectRedirector *BaseMapRedirector = Cast<UObjectRedirector>(AssetData.GetAsset());
628 if (BaseMapRedirector != nullptr) {
629 World = CastChecked<UWorld>(BaseMapRedirector->DestinationObject);
630 }
631 else {
632 World = CastChecked<UWorld>(AssetData.GetAsset());
633 }
634 // try to create each possible tile of the map
635 int i, j;
636 bool Res;
637 j = 0;
638 do
639 {
640 i = 0;
641 do
642 {
643 // Spawn
644 TArray<AStaticMeshActor *> SpawnedActors = SpawnMeshesToWorld(DataPath, Map.bUseCarlaMapMaterials, i, j);
645 Res = SpawnedActors.Num() > 0;
646 if (Res)
647 {
648 UE_LOG(LogTemp, Log, TEXT(" Tile %d,%d found"), i, j);
649 FString TileName;
650 TileName = FString::Printf(TEXT("%s_Tile_%d_%d"), *Map.Name, i, j);
651 // Save the World in specified path
652 // UE_LOG(LogTemp, Log, TEXT("Saving as %s to %s"), *TileName, *Map.Path);
653 SaveWorld(AssetData, PackageName, Map.Path, TileName);
654 MapPathsIds.Add(
655 TPair<FString, FIntVector>(
656 Map.Path + "/" + TileName, FIntVector(i, j, 0)));
657 // Remove spawned actors from world to keep equal as BaseMap
658 DestroySpawnedActorsInWorld(SpawnedActors);
659 ++i;
660 }
661 }
662 while (Res);
663 ++j;
664 }

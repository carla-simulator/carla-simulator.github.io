32TEST(multigpu_secondary, retrying_a_refused_connection_does_not_add_threads) {

35 constexpr auto first_attempt = 500ms;
36 constexpr auto retry_window = 3000ms;

38 std::uint16_t unused_port = 0u;

40 boost::asio::io_context io_context;
41 boost::asio::ip::tcp::acceptor acceptor(
42 io_context,
43 boost::asio::ip::tcp::endpoint(boost::asio::ip::address_v4::loopback(), 0));
44 unused_port = acceptor.local_endpoint().port();

46
47 auto secondary = std::make_shared<Secondary>("127.0.0.1", unused_port, [](carla::multigpu::MultiGPUCommand, carla::Buffer) {});
48 secondary->Connect();

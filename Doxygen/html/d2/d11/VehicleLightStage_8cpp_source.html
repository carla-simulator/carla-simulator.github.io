27void VehicleLightStage::UpdateWorldInfo(const double current_time, const bool synchronous_mode) {
28 const bool any_vehicle_updates_lights = std::any_of(
29 vehicle_id_list.begin(),
30 vehicle_id_list.end(),
31 [this](const ActorId actor_id) { return parameters.GetUpdateVehicleLights(actor_id); });
32 if (!any_vehicle_updates_lights) {
33 all_light_states.clear();
34 missing_from_last_refresh.clear();
35 last_light_states_update = -std::numeric_limits<double>::infinity();
36 light_states_refreshed = false;
37 return;
38 }
39
40 // Set only once the read has succeeded; it guards the unknown-state path.
41 light_states_refreshed = false;
42 if (synchronous_mode ||
43 IsRefreshDue(current_time, last_light_states_update, VEHICLE_LIGHT_STATES_REFRESH_PERIOD)) {
44 all_light_states = world.GetVehiclesLightStates();
45 missing_from_last_refresh.clear();
46 last_light_states_update = current_time;
47 light_states_refreshed = true;
48 }
49
50 if (synchronous_mode ||
51 IsRefreshDue(current_time, last_weather_update, WEATHER_REFRESH_PERIOD)) {
52 is_weather_enabled = world.IsWeatherEnabled();
53 if (is_weather_enabled) {
54 weather = world.GetWeather();
55 }
56 last_weather_update = current_time;
57 }
58}

60void VehicleLightStage::SetCachedLightState(
61 const ActorId actor_id,
62 const rpc::VehicleLightState::flag_type light_state) {
63 for (auto &vls : all_light_states) {
64 if (vls.first == actor_id) {
65 vls.second = light_state;
66 return;
67 }
68 }
69 all_light_states.emplace_back(actor_id, light_state);

72void VehicleLightStage::Update(const unsigned long index) {
73 ActorId actor_id = vehicle_id_list.at(index);
74
75 if (!parameters.GetUpdateVehicleLights(actor_id))
76 return; // this vehicle is not set to have automatic lights update
77
78 rpc::VehicleLightState::flag_type light_states = uint32_t(-1);
79 bool brake_lights = false;
80 bool left_turn_indicator = false;
81 bool right_turn_indicator = false;
82 bool position = false;
83 bool low_beam = false;
84 bool high_beam = false;
85 bool fog_lights = false;
86
87 // search the current light state of the vehicle
88 bool found_light_state = false;
89 for (auto&& vls : all_light_states) {
90 if (vls.first == actor_id) {
91 light_states = vls.second;
92 found_light_state = true;
93 break;
94 }
95 }
96
97 if (!found_light_state) {
98 // Unknown state: deriving a command from the sentinel would switch on the
99 // unmanaged bits (reverse, interior, special) for good.
100 if (IsEarlyRefreshDue(light_states_refreshed, missing_from_last_refresh.contains(actor_id))) {
101 last_light_states_update = -std::numeric_limits<double>::infinity();
102 }
103 if (light_states_refreshed) {
104 missing_from_last_refresh.insert(actor_id);
105 }
106 return;
107 }
108
109 // Determine if the vehicle is truning left or right by checking the close waypoints
110
111 const Buffer& waypoint_buffer = buffer_map.at(actor_id);
112 cg::Location front_location = waypoint_buffer.front()->GetLocation();
113
114 for (const SimpleWaypointPtr& waypoint : waypoint_buffer) {
115 if (waypoint->CheckJunction()) {
116 RoadOption target_ro = waypoint->GetRoadOption();
117 if (target_ro == RoadOption::Left) left_turn_indicator = true;
118 else if (target_ro == RoadOption::Right) right_turn_indicator = true;
119 break;
120 }
121 if (cg::Math::DistanceSquared(front_location, waypoint->GetLocation()) > MAX_DISTANCE_LIGHT_CHECK) {
122 break;
123 }
124 }
125
126 // Determine brake light state
127 for (size_t cc = 0; cc < control_frame.size(); cc++) {
128 if (auto* maybe_ctrl = std::get_if<carla::rpc::Command::ApplyVehicleControl>(&control_frame[cc].command)) {
129 carla::rpc::Command::ApplyVehicleControl& ctrl = *maybe_ctrl;
130 if (ctrl.actor == actor_id) {
131 brake_lights = (ctrl.control.brake > 0.5); // hard braking, avoid blinking for throttle control
132 break;
133 }
134 }
135 }
136
137 // Determine position, fog and beams. Do nothing if the weather is disabled.
138 if (is_weather_enabled) {
139
140 // Turn on beams & positions from sunset to dawn
141 if (weather.sun_altitude_angle < SUN_ALTITUDE_DEGREES_BEFORE_DAWN ||
142 weather.sun_altitude_angle > SUN_ALTITUDE_DEGREES_AFTER_SUNSET)
143 {
144 position = true;
145 low_beam = true;
146 }
147 else if (weather.sun_altitude_angle < SUN_ALTITUDE_DEGREES_JUST_AFTER_DAWN ||
148 weather.sun_altitude_angle > SUN_ALTITUDE_DEGREES_JUST_BEFORE_SUNSET)
149 {
150 position = true;
151 }
152 // Turn on lights under heavy rain
153 if (weather.precipitation > HEAVY_PRECIPITATION_THRESHOLD) {
154 position = true;
155 low_beam = true;
156 }
157 // Turn on fog lights
158 if (weather.fog_density > FOG_DENSITY_THRESHOLD) {
159 position = true;
160 low_beam = true;
161 fog_lights = true;
162 }
163 }
164
165 // Determine the new vehicle light state
166 rpc::VehicleLightState::flag_type new_light_states = light_states;
167 if (brake_lights)
168 new_light_states |= rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::Brake);
169 else
170 new_light_states &= ~rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::Brake);
171
172 if (left_turn_indicator)
173 new_light_states |= rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::LeftBlinker);
174 else
175 new_light_states &= ~rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::LeftBlinker);
176
177 if (right_turn_indicator)
178 new_light_states |= rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::RightBlinker);
179 else
180 new_light_states &= ~rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::RightBlinker);
181
182 if (position)
183 new_light_states |= rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::Position);
184 else
185 new_light_states &= ~rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::Position);
186
187 if (low_beam)
188 new_light_states |= rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::LowBeam);
189 else
190 new_light_states &= ~rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::LowBeam);
191
192 if (high_beam)
193 new_light_states |= rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::HighBeam);
194 else
195 new_light_states &= ~rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::HighBeam);
196
197 if (fog_lights)
198 new_light_states |= rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::Fog);
199 else
200 new_light_states &= ~rpc::VehicleLightState::flag_type(rpc::VehicleLightState::LightState::Fog);
201
202 // Update the vehicle light state if it has changed
203 if (new_light_states != light_states) {
204 control_frame.push_back(carla::rpc::Command::SetVehicleLightState(actor_id, new_light_states));
205 // Otherwise the same command is re-queued every step until the next refresh.
206 SetCachedLightState(actor_id, new_light_states);
207 }
208}

210void VehicleLightStage::RemoveActor(const ActorId actor_id) {
211 all_light_states.erase(
212 std::remove_if(
213 all_light_states.begin(),
214 all_light_states.end(),
215 [actor_id](const rpc::VehicleLightStateList::value_type &light_state) {
216 return light_state.first == actor_id;
217 }),
218 all_light_states.end());
219 missing_from_last_refresh.erase(actor_id);
220}

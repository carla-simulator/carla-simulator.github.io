21TArray<FActorDefinition> ABlueprintActorFactory::GetDefinitions()
22{
23 // Rebuilt from scratch every episode, otherwise entries accumulate.
24 Definitions.Reset();
25 BlueprintsParams.Reset();
26 LoadBlueprintParametersArrayFromFile("BlueprintParameters.json", BlueprintsParams);
27 // Content packs: <Pack>/Content/Config/BlueprintParameters.json, later
28 // packs override earlier entries with the same Name.
29 for (const FString &PackFile : UCarlaContentPackManager::FindPackCatalogFiles("BlueprintParameters.json"))
30 {
31 TArray<FBlueprintParameters> PackBlueprintsParams;
32 LoadBlueprintParametersArrayFromFile(PackFile, PackBlueprintsParams);
33 UCarlaContentPackManager::MergeCatalog(BlueprintsParams, PackBlueprintsParams,
34 [](const FBlueprintParameters &B) { return B.Name; });
35 }
36
37 UActorBlueprintFunctionLibrary::MakeBlueprintDefinitions(BlueprintsParams, Definitions);
38 return Definitions;
39}

41FActorSpawnResult ABlueprintActorFactory::SpawnActor(
42 const FTransform &SpawnAtTransform,
43 const FActorDescription &ActorDescription)
44{
45 FActorSpawnResult SpawnResult;
46 if(!IsValid(ActorDescription.Class))
47 {
48 UE_LOG(LogCarla, Error, TEXT("Actor Description Class is null."));
49 SpawnResult.Status = EActorSpawnResultStatus::InvalidDescription;
50 return SpawnResult;
51 }
52
53 AActor* SpawnedActor = GetWorld()->SpawnActor<AActor>(ActorDescription.Class, SpawnAtTransform);
54 SpawnResult.Actor = SpawnedActor;
55
56 if(SpawnedActor == nullptr)
57 {
58 SpawnResult.Status = EActorSpawnResultStatus::Collision;
59 return SpawnResult;
60 }
61
62 if(PostProcessBlueprint(SpawnedActor, ActorDescription))
63 {
64 SpawnResult.Status = EActorSpawnResultStatus::Success;
65 return SpawnResult;
66 }
67
68 SpawnResult.Status = EActorSpawnResultStatus::UnknownError;
69 return SpawnResult;
70}

87FString ABlueprintActorFactory::FBlueprintParametersArrayToJson(const TArray<FBlueprintParameters>& BlueprintParamsArray)
88{
89 TArray<TSharedPtr<FJsonValue>> JsonArray;
90
91 for (const FBlueprintParameters& BlueprintParams : BlueprintParamsArray)
92 {
93 // Convert each FBlueprintParameters to a JSON object
94 TSharedPtr<FJsonObject> JsonObject = FBlueprintParametersToJsonObject(BlueprintParams);
95 JsonArray.Add(MakeShareable(new FJsonValueObject(JsonObject)));
96 }
97
98 // Convert the array of JSON objects into a single JSON object
99 TSharedRef<FJsonObject> RootObject = MakeShareable(new FJsonObject);
100 RootObject->SetArrayField(TEXT("Blueprints"), JsonArray);
101
102 // Serialize the JSON object into an FString
103 FString OutputString;
104 TSharedRef<TJsonWriter<>> Writer = TJsonWriterFactory<>::Create(&OutputString);
105 FJsonSerializer::Serialize(RootObject, Writer);
106
107 return OutputString;
108}

137bool ABlueprintActorFactory::JsonToFBlueprintParametersArray(const FString& JsonString, TArray<FBlueprintParameters>& OutBlueprintParamsArray)
138{
139 TSharedPtr<FJsonObject> RootObject;
140 TSharedRef<TJsonReader<>> Reader = TJsonReaderFactory<>::Create(JsonString);
141
142 if (!FJsonSerializer::Deserialize(Reader, RootObject) || !RootObject.IsValid())
143 {
144 UE_LOG(LogCarla, Error, TEXT("Failed to parse JSON."));
145 return false;
146 }
147
148 // Get the "Blueprints" array from the JSON root object
149 const TArray<TSharedPtr<FJsonValue>>* BlueprintsArray;
150 if (RootObject->TryGetArrayField(TEXT("Blueprints"), BlueprintsArray))
151 {
152 OutBlueprintParamsArray.Empty();
153 for (const TSharedPtr<FJsonValue>& BlueprintValue : *BlueprintsArray)
154 {
155 TSharedPtr<FJsonObject> BlueprintObject = BlueprintValue->AsObject();
156 if (BlueprintObject.IsValid())
157 {
158 FBlueprintParameters BlueprintParams;
159 if (JsonToFBlueprintParameters(BlueprintObject, BlueprintParams))
160 {
161 OutBlueprintParamsArray.Add(BlueprintParams);
162 }
163 }
164 }
165 }
166
167 return true;
168}

170void ABlueprintActorFactory::LoadBlueprintParametersArrayFromFile(const FString& FileName, TArray<FBlueprintParameters>& OutBlueprintParamsArray)
171{
172 FString JsonString;
173 // A relative name is resolved against the project catalog dir; an absolute
174 // path (content pack catalogs) is used as is.
175 FString FilePath = FPaths::IsRelative(FileName)
176 ? FPaths::ProjectContentDir() + TEXT("Carla/Config/") + FileName
177 : FileName;
178 FString JsonContent;
179
180 // Load the JSON file content into an FString
181 if (FFileHelper::LoadFileToString(JsonContent, *FilePath))
182 {
183 // Parse the JSON and populate the TArray<FBlueprintParameters>
184 if (JsonToFBlueprintParametersArray(JsonContent, OutBlueprintParamsArray))
185 {
186 UE_LOG(LogCarla, Log, TEXT("Blueprint parameters loaded successfully from %s"), *FilePath);
187 }
188 else
189 {
190 UE_LOG(LogCarla, Error, TEXT("Failed to parse Blueprint parameters from %s"), *FilePath);
191 }
192 }
193 else
194 {
195 UE_LOG(LogCarla, Error, TEXT("Failed to load file: %s"), *FilePath);
196 }
197}

56void UCarlaLidarSubsystem::StartLidarSimulations(UWorld *World, [[maybe_unused]] float DeltaSeconds)
57{
58 TRACE_CPUPROFILER_EVENT_SCOPE(UCarlaLidarSubsystem::StartLidarSimulations);
59
60 if (World == nullptr || World->GetPhysicsScene() == nullptr || !HasActiveLidars())
61 {
62 return;
63 }
64
65 // Snapshot on the game thread; the task must not read ActiveLidars while the game thread can modify it.
66 LidarsToSimulate.Reset();
67 for (const TWeakObjectPtr<ARayCastSemanticLidar> &WeakLidar : ActiveLidars)
68 {
69 ARayCastSemanticLidar *Lidar = WeakLidar.Get();
70 if (Lidar != nullptr && Lidar->IsReadyToTick())
71 {
72 LidarsToSimulate.Add(Lidar);
73 }
74 }
75
76 if (LidarsToSimulate.Num() == 0)
77 {
78 return;
79 }
80
81 const bool bBatch = (CVarLidarBatch.GetValueOnGameThread() != 0) && (LidarsToSimulate.Num() > 1);
82
83 // Dispatch raycast simulation as a high-priority Task Graph job on worker threads.
84 // Each LiDAR uses its own tick-interval delta, not the physics substep delta.
85 LidarSimulationTask = FFunctionGraphTask::CreateAndDispatchWhenReady(
86 [this, World, bBatch]()
87 {
88 TRACE_CPUPROFILER_EVENT_SCOPE(TaskGraphLidarSimulations);
89 auto LockedPhysObject = FPhysicsObjectExternalInterface::LockRead(World->GetPhysicsScene());
90
91 if (bBatch)
92 {
93 ParallelFor(LidarsToSimulate.Num(), [this](int32 Index)
94 {
95 ARayCastSemanticLidar *Lidar = LidarsToSimulate[Index];
96 Lidar->SimulateLidar(Lidar->GetReadyToTickDeltaSeconds(), /*bLockPhysics=*/false);
97 });
98 }
99 else
100 {
101 for (ARayCastSemanticLidar *Lidar : LidarsToSimulate)
102 {
103 Lidar->SimulateLidar(Lidar->GetReadyToTickDeltaSeconds(), /*bLockPhysics=*/false);
104 }
105 }
106
107 LockedPhysObject.Release();
108 },
109 TStatId(),
110 nullptr,
111 ENamedThreads::AnyBackgroundHiPriTask
112 );
113}

115void UCarlaLidarSubsystem::FinishLidarSimulations(UWorld *World, [[maybe_unused]] float DeltaSeconds)
116{
117 TRACE_CPUPROFILER_EVENT_SCOPE(UCarlaLidarSubsystem::FinishLidarSimulations);
118
119 if (LidarSimulationTask.IsValid())
120 {
121 FTaskGraphInterface::Get().WaitUntilTaskCompletes(LidarSimulationTask);
122 LidarSimulationTask = nullptr;
123 }
124
125 // Streaming and ROS2 publishing stay on the game thread.
126 {
127 TRACE_CPUPROFILER_EVENT_SCOPE(LidarSendData);
128 for (ARayCastSemanticLidar *Lidar : LidarsToSimulate)
129 {
130 if (IsValid(Lidar))
131 {
132 Lidar->SendData(Lidar->GetReadyToTickDeltaSeconds());
133 Lidar->ClearReadyToTick();
134 }
135 }
136 }
137 LidarsToSimulate.Reset();
138}

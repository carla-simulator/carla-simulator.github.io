53void AV2XSensor::EndPlay(const EEndPlayReason::Type EndPlayReason)
54{
55 // Drop our entry from the shared static map so a destroyed sensor cannot
56 // leave a dangling AActor* key for the surviving V2X sensors to dereference.
57 AV2XSensor::mActorV2XDataMap.erase(this);
58
59 // CaServiceObj / PathLossModelObj are std::unique_ptr: they are released
60 // automatically in the destructor, which runs after UE's teardown clears
61 // ownership (SetOwner(nullptr) -> InitModel). Do not reset them here, or
62 // that teardown call would dereference a freed object.
63 Super::EndPlay(EndPlayReason);
64}

108/*
109 * Function stores the actor details in to the static list.
110 * Calls the CaService object to generate CAM message
111 * Stores the message in static map
112 */
113void AV2XSensor::PrePhysTick(float DeltaSeconds)
114{
115 if ( !CaServiceObj->CarlaActorInitialized() ) {
116 // ensure a CarlaActor to be the owner if no parent actor
117 InitModel(this);
118 }
119
120 Super::PrePhysTick(DeltaSeconds);
121 // Clear the message created during the last sim cycle
122 AV2XSensor::mActorV2XDataMap.erase(this);
123
124 // Step 0: Create message to send, if triggering conditions fulfilled
125 // this needs to be done in pre phys tick to enable synchronous reception in all other v2x sensors
126 // Check whether the message is generated
127 if (CaServiceObj->Trigger(DeltaSeconds))
128 {
129 // If message is generated store it
130 // make a pair of message and sending power
131 // if different v2x sensors send with different power, we need to store that
132 carla::sensor::data::CAMData cam_pw;
133 cam_pw.Message = CaServiceObj->GetCamMessage();
134 cam_pw.Power = PathLossModelObj->GetTransmitPower();
135 AV2XSensor::mActorV2XDataMap.insert({this, cam_pw});
136 }
137}

173/*
174 * Function takes care of sending messages to the current actor.
175 * First simulates the communication by calling LOSComm object.
176 * If there is a list present then messages from those list are sent to the current actor
177 */
178void AV2XSensor::PostPhysTick(UWorld *World, ELevelTick TickType, float DeltaTime)
179{
180 TRACE_CPUPROFILER_EVENT_SCOPE(AV2XSensor::PostPhysTick);
181 // Step 1: Create an actor list which has messages to send targeting this v2x sensor instance
182 std::vector<ActorPowerPair> ActorPowerList;
183 for (const auto &pair : AV2XSensor::mActorV2XDataMap)
184 {
185 if (pair.first != this)
186 {
187 ActorPowerPair actor_power_pair;
188 actor_power_pair.first = pair.first;
189 // actor sending with transmit power
190 actor_power_pair.second = pair.second.Power;
191 ActorPowerList.push_back(actor_power_pair);
192 }
193 }
194
195 carla::sensor::data::CAMDataS msg_received_power_list;
196 // Step 2: Simulate the communication for the actors in actor list to current actor.
197 if (!ActorPowerList.empty())
198 {
199 UCarlaEpisode *carla_episode = UCarlaStatics::GetCurrentEpisode(GetWorld());
200 PathLossModelObj->Simulate(ActorPowerList, carla_episode, GetWorld());
201 // Step 3: Get the list of actors who can send message to current actor, and the receive power of their messages.
202 ActorPowerMap actor_receivepower_map = PathLossModelObj->GetReceiveActorPowerList();
203 // Step 4: Retrieve the messages of the actors that are received
204
205 // get registry to retrieve carla actor IDs
206 const FActorRegistry &Registry = carla_episode->GetActorRegistry();
207
208 for (const auto &pair : actor_receivepower_map)
209 {
210 // Note: AActor* sender_actor = pair.first;
211 // find + skip-on-miss: at() would throw std::out_of_range if the
212 // sender dropped out of the map between simulate and retrieval.
213 const auto sender_it = AV2XSensor::mActorV2XDataMap.find(pair.first);
214 if (sender_it == AV2XSensor::mActorV2XDataMap.end())
215 {
216 continue;
217 }
218 carla::sensor::data::CAMData const &send_msg_and_pw = sender_it->second;
219 carla::sensor::data::CAMData received_msg_and_pw;
220 // sent CAM
221 received_msg_and_pw.Message = send_msg_and_pw.Message;
222 // receive power
223 received_msg_and_pw.Power = pair.second;
224
225 msg_received_power_list.WriteMessage(received_msg_and_pw);
226 }
227 }
228 // Step 5: Send message
229
230 if (msg_received_power_list.GetMessageCount() > 0)
231 {
232 auto DataStream = GetDataStream(*this);
233 DataStream.SerializeAndSend(*this, msg_received_power_list, DataStream.PopBufferFromPool());
234 }
235}

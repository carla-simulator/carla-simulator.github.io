365// The night sky is pure black on every map (verified content-side, not a
366// generated-map defect): BP_Carla_Sky's SetSkySphere respawns a child actor
367// of the stock engine /Engine/EngineSky/BP_Sky_Sphere on every weather push
368// and stores it in its own "SkySphere" object variable (see PushWeatherToSky
369// below). That sphere's material, M_Sky_Panning_Clouds2, already ships star
370// rendering -- confirmed by dumping strings from the .uasset directly: a
371// MaterialExpressionScalarParameter named "Stars brightness" and a
372// TextureSample referencing /Engine/EngineSky/T_Sky_Stars. The sphere
373// blueprint exposes it through two of its own variables (also confirmed via
374// strings, including their tooltips baked into the asset): "Colors
375// Determined By Sun Position" (bool -- "If enabled, sky colors will change
376// according to the sun's position"; when off, the sphere's "Sun Height"
377// value -- whose own tooltip is "If no directional light is assigned, this
378// value determines the height of the sun" -- stays frozen at its authored
379// fallback instead of tracking the Directional Light Actor BP_Carla_Sky
380// already assigns every push, which would permanently suppress the
381// sun-height-gated star blend regardless of the actual time of day) and
382// "Stars Brightness" (float -- "Multiplier for the brightness of the stars
383// when the sun is below the horizon"). CARLA's own BP_Carla_Sky/BP_
384// CarlaWeather carry a dead "Stars Intensity Over Sun Altitude" comment/graph
385// fragment (same severed-exec-chain pattern documented throughout this file)
386// but, unlike SunIntensity_Curve/SkyIntensity_Curve, there is no backing
387// UCurveFloat asset anywhere in the content for it -- confirmed by searching
388// the plugin Content tree -- so there is nothing to re-wire; drive the two
389// stock engine variables directly instead, the same way this file already
390// routes around every other severed exec chain in the rig.
391// Both are held gated to SunAltitudeAngle < 0, deliberately not applied
392// during the day even though a fresh sphere actor makes it safe either way
393// (SetSkySphere respawns it every push, so nothing here can leak across
394// pushes): "Colors Determined By Sun Position" may also drive the sphere's
395// OWN internal Horizon/Zenith/Cloud color curves, which would compete with
396// CARLA's curve-driven day colors already pushed by UpdateSkySphereColor
397// earlier in the same push -- unverifiable without an editor, so day is left
398// completely untouched to avoid any color-grading regression risk.
399// StarsBrightness is a material-scalar knob, not a physical unit. The
400// white-texel estimate (~100, from the EV100=10 exposure multiplier above)
401// rendered stars at max pixel ~7: the star texture's lit texels sample far
402// below white. 5000 was calibrated visually on a generated Town03 world at
403// sun -30 -- a dense readable star field (sky-crop max ~145) with no bloom
404// halos or day-side effect (the whole block is gated to SunAltitudeAngle<0).
405// City light pollution scattered by the clouds: emission of the volumetric
406// cloud material (WeatherMaterialParameters.CityGlow, M_BasicClouds Emissive)
407// times NightFactor. Kept very low and near-grey: on camera, real overcast city
408// nights show the clouds as barely visible dark grey (sky L* ~2), not lit
409// orange. Per unit of cloud density, weighted to the cloud base in the material
410// ((1 - NormAltitudeInLayer)^4): the city lights it from below.

876// Applied on top of what the rig wrote. Tracks its own last write, so a value
877// the rig did not rewrite since is not scaled twice.
878static void UpdateMoonHalo(AActor* SkyActor, float HaloBlend)
879{
880 FObjectProperty* AtmosphereProperty = CastField<FObjectProperty>(
881 SkyActor->GetClass()->FindPropertyByName(TEXT("SkyAtmosphereComponent")));
882 USkyAtmosphereComponent* Atmosphere = AtmosphereProperty != nullptr
883 ? Cast<USkyAtmosphereComponent>(AtmosphereProperty->GetObjectPropertyValue_InContainer(SkyActor))
884 : nullptr;
885 if (Atmosphere == nullptr)
886 return;
887
888 static TWeakObjectPtr<USkyAtmosphereComponent> Tracked;
889 static float BaseScale = 0.0f, WrittenScale = -1.0f;
890 static float BaseAnisotropy = 0.0f, WrittenAnisotropy = -2.0f;
891 if (Tracked.Get() != Atmosphere)
892 {
893 Tracked = Atmosphere;
894 WrittenScale = -1.0f;
895 WrittenAnisotropy = -2.0f;
896 }
897 if (Atmosphere->MieScatteringScale != WrittenScale)
898 BaseScale = Atmosphere->MieScatteringScale;
899 if (Atmosphere->MieAnisotropy != WrittenAnisotropy)
900 BaseAnisotropy = Atmosphere->MieAnisotropy;
901
902 const float ScaleFactor = FMath::Lerp(1.0f,
903 FMath::Max(CVarCarlaWeatherMoonHaloMieScale.GetValueOnGameThread(), 0.0f), HaloBlend);
904 const float NightAnisotropy = CVarCarlaWeatherMoonHaloMieAnisotropy.GetValueOnGameThread();

906 ? FMath::Lerp(BaseAnisotropy, FMath::Min(NightAnisotropy, 0.999f), HaloBlend)
907 : BaseAnisotropy;
908
909 WrittenScale = BaseScale * ScaleFactor;
910 WrittenAnisotropy = Anisotropy;
911 if (Atmosphere->MieScatteringScale != WrittenScale)
912 Atmosphere->SetMieScatteringScale(WrittenScale);
913 if (Atmosphere->MieAnisotropy != WrittenAnisotropy)
914 Atmosphere->SetMieAnisotropy(WrittenAnisotropy);
915}

1043// Applied before the overcast blend, so its peak stays the curve's own.
1044static float ApplySunAltitudeFalloff(float CurveValue, float CurvePeak, float SunAltitudeAngle)
1045{
1046 const float Strength = FMath::Clamp(
1047 CVarCarlaWeatherSunAltitudeFalloff.GetValueOnGameThread(), 0.0f, 1.0f);
1048 if (Strength <= 0.0f || CurvePeak <= UE_KINDA_SMALL_NUMBER)
1049 return CurveValue;
1050
1051 // Both fractions are of the curve's peak, not of its value at this altitude.
1052 const float SinElevation =
1053 FMath::Sin(FMath::DegreesToRadians(FMath::Max(SunAltitudeAngle, 0.0f)));
1054 const float Exponent = FMath::Max(CVarCarlaWeatherSunFalloffExponent.GetValueOnGameThread(), 0.0f);
1055 const float Physical = CurvePeak * FMath::Max(
1056 FMath::Pow(SinElevation, Exponent), CVarCarlaWeatherSunHorizonFloor.GetValueOnGameThread());
1057
1058 // Min so it only darkens; below the horizon the curve governs.
1059 return FMath::Min(CurveValue, FMath::Lerp(CurveValue, Physical, Strength));
1060}

1064{

1067 for (float Altitude = 0.0f; Altitude <= 90.0f; Altitude += 5.0f)

1072// CurvePeak > 0 treats OvercastValue as a factor on the curve, 0 as an absolute.
1073static float ApplyOvercastBlend(float CurveValue, float OvercastValue,
1074 float Cloudiness, bool bOnlyDarken, float CurvePeak)
1075{
1076 if (OvercastValue < 0.0f)
1077 return CurveValue;
1078 const float Overcast = FMath::Clamp(Cloudiness / 100.0f, 0.0f, 1.0f);
1079 const float Blended = CurvePeak > UE_KINDA_SMALL_NUMBER
1080 ? CurveValue * FMath::Lerp(1.0f, OvercastValue / CurvePeak, Overcast)
1081 : FMath::Lerp(CurveValue, OvercastValue, Overcast);
1082 return bOnlyDarken ? FMath::Min(CurveValue, Blended) : FMath::Max(CurveValue, Blended);
1083}

1085// White when there is nothing to cancel, so writing it every push also undoes
1086// an earlier tint. The sky light stores an 8-bit colour, so the tint comes
1087// back scaled to a peak of 1 and OutIntensityScale carries the difference.
1088static FLinearColor ComputeOvercastNeutralTint(const USkyAtmosphereComponent* Atmosphere,
1089 float SunAltitudeAngle, float Cloudiness,
1090 float NeutralizeAmount, float& OutIntensityScale)
1091{
1092 OutIntensityScale = 1.0f;
1093 const float Release = CVarCarlaWeatherTransmittanceReleaseDeg.GetValueOnGameThread();
1094 const float DayFade = Release < 0.0f
1095 ? 1.0f - FMath::SmoothStep(0.0f, 1.0f, FMath::Clamp(SunAltitudeAngle / Release, 0.0f, 1.0f))
1096 : (SunAltitudeAngle >= 0.0f ? 1.0f : 0.0f);
1097 const float Strength = FMath::Clamp(NeutralizeAmount, 0.0f, 1.0f)
1098 * FMath::Clamp(Cloudiness / 100.0f, 0.0f, 1.0f) * DayFade;
1099 if (Atmosphere == nullptr || Strength <= 0.0f)
1100 return FLinearColor::White;
1101
1102 // Held at the horizon or above: a ray through the ground comes back black.
1103 FAtmosphereSetup Setup(*Atmosphere);
1104 Setup.TransmittanceMinLightElevationAngle = 0.0f;
1105 const float Elevation = FMath::DegreesToRadians(FMath::Max(SunAltitudeAngle, 0.0f));
1106 const FLinearColor TSun = Setup.GetTransmittanceAtGroundLevel(
1107 FVector(FMath::Cos(Elevation), 0.0f, FMath::Sin(Elevation)));
1108 // Relative to the zenith, or every overcast midday would turn blue.
1109 const FLinearColor TZenith = Setup.GetTransmittanceAtGroundLevel(FVector::UpVector);
1110 const FLinearColor T(
1111 TSun.R / FMath::Max(TZenith.R, UE_KINDA_SMALL_NUMBER),
1112 TSun.G / FMath::Max(TZenith.G, UE_KINDA_SMALL_NUMBER),
1113 TSun.B / FMath::Max(TZenith.B, UE_KINDA_SMALL_NUMBER));
1114 const float Luminance = T.GetLuminance();
1115 if (Luminance <= UE_KINDA_SMALL_NUMBER)
1116 return FLinearColor::White;
1117
1118 const float MaxGain = FMath::Max(CVarCarlaWeatherOvercastAmbientMaxGain.GetValueOnGameThread(), 1.0f);
1119 FLinearColor Gain(
1120 FMath::Min(Luminance / FMath::Max(T.R, UE_KINDA_SMALL_NUMBER), MaxGain),
1121 FMath::Min(Luminance / FMath::Max(T.G, UE_KINDA_SMALL_NUMBER), MaxGain),
1122 FMath::Min(Luminance / FMath::Max(T.B, UE_KINDA_SMALL_NUMBER), MaxGain),
1123 1.0f);
1124 const float GainLuminance = Gain.GetLuminance();
1125 if (GainLuminance > UE_KINDA_SMALL_NUMBER)
1126 Gain = Gain * (1.0f / GainLuminance);
1127 Gain.A = 1.0f;
1128 FLinearColor Tint = FMath::Lerp(FLinearColor::White, Gain, Strength);
1129 const float Peak = FMath::Max3(Tint.R, Tint.G, Tint.B);
1130 if (Peak > 1.0f)
1131 {
1132 Tint = Tint * (1.0f / Peak);
1133 OutIntensityScale = Peak;
1134 }
1135 Tint.A = 1.0f;
1136 return Tint;
1137}

1227void AWeather::UpdateRain()
1228{
1229 if (!GetWorld() || !GetWorld()->IsGameWorld())
1230 return;
1231
1232 const float Rain = FMath::Clamp(Weather.Precipitation / 100.0f, 0.0f, 1.0f);
1233 // Explicitly drive the lens collection without depending on a town's sky BP.
1234 if (RainParameters)
1235 UKismetMaterialLibrary::SetScalarParameterValue(GetWorld(), RainParameters, TEXT("Precipitation"), Rain);
1236
1237 auto UpdateEmitter = [this, Rain](UParticleSystemComponent* Emitter, const FVector& Location, float Yaw)
1238 {
1239 if (!Emitter) return;
1240 // PS_Rain's scaled spawn box is X[-1000,4000], Y[-2000,2000] cm. Turned with
1241 // the view so it reaches 40 m ahead of it: left axis-aligned it rained on
1242 // one side of the camera only, and the drops looked parked in one place.
1243 // World-space, vertically falling drops.
1244 const FRotator Heading(0.0f, Yaw, 0.0f);
1245 const FVector Origin = Location + FVector(0.0f, 0.0f, 400.0f);
1246 Emitter->SetWorldRotation(Heading);
1247 if (FVector::DistSquared(Emitter->GetComponentLocation(), Origin) > FMath::Square(2000.0f))
1248 Emitter->DeactivateImmediate(); // Do not leave a trail after teleporting.
1249 Emitter->SetWorldLocation(Origin);
1250 Emitter->SetFloatParameter(TEXT("RainDensity"), Rain * (4000.0f / 0.85f));
1251 // Legacy distribution maps Z [0,1] to [0,-1000] cm/s. Negative input clamps to zero.
1252 Emitter->SetVectorParameter(TEXT("RainVelocity"),
1253 FVector(FMath::Clamp(Weather.WindIntensity, 0.0f, 100.0f) * 3.0f, 0.0f, 1.0f));
1254 if (Rain <= 0.0f)
1255 Emitter->DeactivateImmediate();
1256 else if (!Emitter->IsActive())
1257 Emitter->ActivateSystem(true);
1258 };
1259
1260 if (auto* Camera = UGameplayStatics::GetPlayerCameraManager(GetWorld(), 0))
1261 {
1262 if (!IsValid(RainCameraModifier) || RainCameraModifier->GetOuter() != Camera)
1263 {
1264 if (IsValid(RainCameraModifier))
1265 if (auto* PreviousCamera = Cast<APlayerCameraManager>(RainCameraModifier->GetOuter()))
1266 PreviousCamera->RemoveCameraModifier(RainCameraModifier);
1267 RainCameraModifier = Cast<UCarlaRainCameraModifier>(
1268 Camera->AddNewCameraModifier(UCarlaRainCameraModifier::StaticClass()));
1269 }
1270 if (RainCameraModifier)
1271 {
1272 RainCameraModifier->RainMaterial = PrecipitationPostProcessMaterial;
1273 RainCameraModifier->RainWeight = Rain;
1274 }
1275 if (!ViewportRain && Rain > 0.0f)
1276 ViewportRain = CreateRainEmitter(this, false);
1277 UpdateEmitter(ViewportRain, Camera->GetCameraLocation(), Camera->GetCameraRotation().Yaw);
1278 }
1279 else if (ViewportRain)
1280 ViewportRain->DeactivateImmediate();
1281
1282 for (auto It = SensorRain.CreateIterator(); It; ++It)
1283 if (!It.Key().IsValid() || !It.Value().IsValid())
1284 {
1285 if (It.Value().IsValid()) It.Value()->DestroyComponent();
1286 It.RemoveCurrent();
1287 }
1288 for (TActorIterator<ASceneCaptureCamera> It(GetWorld()); It; ++It)
1289 {
1290 auto* Sensor = *It;
1291 auto* Emitter = SensorRain.FindRef(Sensor).Get();
1292 if (!Emitter && Rain > 0.0f)
1293 {
1294 Emitter = CreateRainEmitter(Sensor, true);
1295 SensorRain.Add(Sensor, Emitter);
1296 }
1297 UpdateEmitter(Emitter, Sensor->GetCaptureComponent2D()->GetComponentLocation(),
1298 Sensor->GetCaptureComponent2D()->GetComponentRotation().Yaw);
1299 }

1363// Environment-map override for the sky light (set_sky_light_map RPC). The
1364// cubemap lives in the transient package and is rooted while set, but the
1365// override is scoped to the world it was set on: it is re-applied to that
1366// world's sky rig on every weather push, and dropped when that world is
1367// cleaned up (load_world / reload_world), so a freshly loaded level always
1368// starts on its own real-time atmosphere capture -- a client that loaded a
1369// new map never asked for the previous map's lighting.
1370namespace
1371{
1372 struct FSkyLightMapOverride
1373 {
1374 UTextureCube* Cubemap = nullptr;
1375 float Intensity = 1.0f;
1376 TWeakObjectPtr<UWorld> World;
1377 };
1378 FSkyLightMapOverride GSkyLightMapOverride;
1379 FDelegateHandle GSkyLightMapWorldCleanupHandle;
1380
1381 bool SkyLightMapAppliesTo(const UWorld* World)
1382 {
1383 return GSkyLightMapOverride.Cubemap != nullptr && World != nullptr &&
1384 GSkyLightMapOverride.World.Get() == World;
1385 }
1386
1387 // What the override rewrites on a sky light, saved the first time it is
1388 // applied to that component and put back by ClearSkyLightMap. The
1389 // real-time capture honours bLowerHemisphereIsBlack too, so leaving it
1390 // cleared would change the rig's ambient after the map is removed.
1391 struct FSkyLightSavedState
1392 {
1393 TEnumAsByte<ESkyLightSourceType> SourceType;
1394 bool bLowerHemisphereIsBlack;
1395 };
1396 TMap<TWeakObjectPtr<USkyLightComponent>, FSkyLightSavedState> GSkyLightSavedStates;
1397
1398 void ApplySkyLightMapToComponent(USkyLightComponent* SkyLightComponent)
1399 {
1400 if (!GSkyLightSavedStates.Contains(SkyLightComponent))
1401 {
1402 GSkyLightSavedStates.Add(SkyLightComponent,
1403 {SkyLightComponent->SourceType, SkyLightComponent->bLowerHemisphereIsBlack});
1404 }
1405 if (SkyLightComponent->bRealTimeCapture)
1406 SkyLightComponent->SetRealTimeCaptureEnabled(false);
1407 if (SkyLightComponent->SourceType != SLS_SpecifiedCubemap)
1408 {
1409 SkyLightComponent->SourceType = SLS_SpecifiedCubemap;
1410 SkyLightComponent->MarkRenderStateDirty();
1411 SkyLightComponent->SetCaptureIsDirty();
1412 }
1413 if (SkyLightComponent->bLowerHemisphereIsBlack)
1414 {
1415 SkyLightComponent->bLowerHemisphereIsBlack = false;
1416 SkyLightComponent->MarkRenderStateDirty();
1417 }
1418 SkyLightComponent->SetSourceCubemapAngle(0.0f);
1419 SkyLightComponent->SetCubemap(GSkyLightMapOverride.Cubemap);
1420 if (SkyLightComponent->Cubemap != GSkyLightMapOverride.Cubemap)
1421 {
1422 // SetCubemap is a no-op on a Static-mobility light; the rig's
1423 // skylight is meant to be Stationary, but do not depend on it.
1424 SkyLightComponent->Cubemap = GSkyLightMapOverride.Cubemap;
1425 }
1426 // Always re-process: the cubemap contents may be new even when the
1427 // pointer is not (see SkyLightMap.cpp on object naming).
1428 SkyLightComponent->MarkRenderStateDirty();
1429 SkyLightComponent->SetCaptureIsDirty();
1430 SkyLightComponent->SetLightColor(FLinearColor::White);
1431 SkyLightComponent->SetIntensity(GSkyLightMapOverride.Intensity);
1432 if (!SkyLightComponent->IsActive())
1433 SkyLightComponent->SetActive(true);
1434 }

1436 // Forget the override: unroot the cubemap (the transient object is then
1437 // collected with the next GC pass) and drop the saved component states,
1438 // whose weak keys are dead once the owning world has gone anyway.
1439 void DropSkyLightMapOverride()
1440 {
1441 if (GSkyLightMapOverride.Cubemap != nullptr)
1442 GSkyLightMapOverride.Cubemap->RemoveFromRoot();

1444 GSkyLightSavedStates.Empty();
1445 if (GSkyLightMapWorldCleanupHandle.IsValid())
1446 {
1447 FWorldDelegates::OnWorldCleanup.Remove(GSkyLightMapWorldCleanupHandle);
1448 GSkyLightMapWorldCleanupHandle.Reset();
1449 }
1450 }
1451
1452 // load_world / reload_world tear the current world down through
1453 // UWorld::CleanupWorld before the next level is brought up; that is the
1454 // end of the override's scope. Other worlds (the editor world, a level
1455 // instance) are not ours and are left alone.
1456 void OnSkyLightMapWorldCleanup(UWorld* World, bool /*bSessionEnded*/, bool /*bCleanupResources*/)
1457 {
1458 if (World == nullptr || GSkyLightMapOverride.World.Get() != World)
1459 return;
1460 DropSkyLightMapOverride();
1461 UE_LOG(LogCarla, Log, TEXT("AWeather: sky light map dropped with world %s"), *World->GetName());
1462 }

1553 // lookup in this function.

1555 [](const UClass* Class, const TCHAR* AuthoredName) -> FProperty*
1556 {
1557 for (TFieldIterator<FProperty> It(Class); It; ++It)
1558 {

1566 // and refreshes every component from its blueprint Update function.
1567 // Matched by type via ASkyBase::FindWeatherParameters, not by name --
1568 // a name-based lookup here silently broke (no warning triggered,
1569 // because "SkyParameters"/"Sky Parameters" both still resolved to
1570 // *something* is wrong; it just returned null and this whole write
1571 // no-opped) the moment that Blueprint variable got renamed to
1572 // "WeatherParameters". Precipitation/Wetness visuals -- driven by the
1573 // legacy UpdateAtmosphereAndPrecipitation Blueprint call below, which

1590 // exactly once at a day->night transition. Running it on every
1591 // weather push collapses the sphere to pure black (no stars, no
1592 // horizon glow) after a single call. The sphere is respawned fresh
1593 // by SetSkySphere each push, and UpdateSkySphereColor plus the

1617 // BP_Sky_Sphere and tries to attach it to this rig's root

1620 // child to a non-Static parent ("AttachTo: ... is not static ...

1624 // instead of following the rig. Finish the attach here natively,
1625 // forcing the spawned instance's root to Movable first (a fresh

23void FSensorManager::PostPhysTick(UWorld *World, ELevelTick TickType, float DeltaSeconds)
24{
25 TRACE_CPUPROFILER_EVENT_SCOPE(FSensorManager::PostPhysTick);
26
27 UCarlaLidarSubsystem* LidarSubsystem = World ? World->GetSubsystem<UCarlaLidarSubsystem>() : nullptr;
28 if (LidarSubsystem != nullptr)
29 {
30 LidarSubsystem->StartLidarSimulations(World, DeltaSeconds);
31 }
32
33 for (ASensor * Sensor : SensorList)
34 {
35 Sensor->PostPhysTickInternal(World, TickType, DeltaSeconds);
36 }
37 // Every camera above only recorded its GPU copy; synchronize the GPU once
38 // and deliver the whole batch (one pipeline drain per tick instead of one
39 // per camera).
40 ImageUtil::FlushBatchedReadbacks();
41
42 if (LidarSubsystem != nullptr)
43 {
44 LidarSubsystem->FinishLidarSimulations(World, DeltaSeconds);
45 }
46}

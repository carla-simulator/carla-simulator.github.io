13constexpr std::uint32_t SizeOf(carla::ros2::PointFieldDataType datatype) {
14 switch (datatype) {
15 case carla::ros2::PointFieldDataType::Int8: return 1u;
16 case carla::ros2::PointFieldDataType::UInt8: return 1u;
17 case carla::ros2::PointFieldDataType::Int16: return 2u;
18 case carla::ros2::PointFieldDataType::UInt16: return 2u;
19 case carla::ros2::PointFieldDataType::Int32: return 4u;
20 case carla::ros2::PointFieldDataType::UInt32: return 4u;
21 case carla::ros2::PointFieldDataType::Float32: return 4u;
22 case carla::ros2::PointFieldDataType::Float64: return 8u;
23 }
24 return 0u;
25}

27// Compile-time check that within a layout the fields are tightly packed
28// (each field's offset equals the running sum of previous field sizes).
29template <std::size_t N>
30constexpr std::uint32_t LastFieldEnd(
31 const std::array<carla::ros2::PointFieldDescriptor, N> &layout) {
32 return layout.back().offset + SizeOf(layout.back().datatype) * layout.back().count;
33}

109TEST(PointCloudFields, layouts_offsets_are_monotonically_increasing) {

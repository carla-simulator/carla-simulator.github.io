38ETrafficSignState AGeoTrafficSign::StateForSignal(const FString& Type, const FString& Subtype)
39{
40 if (Type == TEXT("206"))
41 {
42 return ETrafficSignState::StopSign;
43 }
44 if (Type == TEXT("205"))
45 {
46 return ETrafficSignState::YieldSign;
47 }
48 if (Type == TEXT("274"))
49 {
50 return SpeedLimitStateForKmh(FCString::Atoi(*Subtype));
51 }
52 return ETrafficSignState::UNKNOWN;
53}

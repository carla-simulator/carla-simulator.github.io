32TEST(BoostRtree, InsertAndNearestNeighbour2D) {
33 bgi::rtree<Value2D, bgi::quadratic<16>> tree;
34 tree.insert({Point2D{0.0, 0.0}, 0u});
35 tree.insert({Point2D{1.0, 1.0}, 1u});
36 tree.insert({Point2D{10.0, 10.0}, 2u});
37 tree.insert({Point2D{-5.0, -5.0}, 3u});
38
39 std::vector<Value2D> result;
40 tree.query(bgi::nearest(Point2D{0.9, 1.1}, 1), std::back_inserter(result));
41 ASSERT_EQ(result.size(), 1u);
42 EXPECT_EQ(result.front().second, 1u);
43}

63 tree.insert({Point2D{5.0, 5.0}, 2u});

66 const Box2D box{Point2D{0.0, 0.0}, Point2D{3.0, 3.0}};

80TEST(BoostRtree, KNearestReturnsExpectedSet) {
81 bgi::rtree<Value2D, bgi::quadratic<16>> tree;
82 tree.insert({Point2D{0.0, 0.0}, 0u});
83 tree.insert({Point2D{1.0, 0.0}, 1u});
84 tree.insert({Point2D{2.0, 0.0}, 2u});
85 tree.insert({Point2D{3.0, 0.0}, 3u});
86
87 std::vector<Value2D> result;
88 tree.query(bgi::nearest(Point2D{0.0, 0.0}, 3), std::back_inserter(result));
89 ASSERT_EQ(result.size(), 3u);
90
91 // The rtree's nearest() query does not guarantee ordering of its k results,
92 // only that they are the k nearest. Verify the expected set membership.
93 std::vector<std::size_t> ids;
94 for (const auto &entry : result) {
95 ids.push_back(entry.second);
96 }
97 std::sort(ids.begin(), ids.end());
98 EXPECT_EQ(ids[0], 0u);
99 EXPECT_EQ(ids[1], 1u);
100 EXPECT_EQ(ids[2], 2u);

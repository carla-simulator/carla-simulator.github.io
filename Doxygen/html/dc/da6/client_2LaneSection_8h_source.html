23 /// A stretch of road over which the lane layout does not change.
24 ///
25 /// An OpenDRIVE road is divided along s into lane sections; lanes are added
26 /// or dropped only at a section boundary. This is why a lane's identity
27 /// carries a section id, and why Lane::GetNextLanes() can step into a
28 /// different section.
29 class LaneSection
30 : public EnableSharedFromThis<LaneSection>,
31 private NonCopyable {
32 public:
33

35 return _section_id;
36 }
37
38 road::RoadId GetRoadId() const {
39 return _road_id;
40 }

43 uint64_t GetUniqueId() const;
44
45 SharedPtr<Road> GetRoad() const;
46
47 /// The s coordinate, along the road, at which this section begins.
48 double GetDistance() const;
49
50 /// Length of the section, in metres.
51 double GetLength() const;
52
53 /// Every lane in this section of the given type. The centre lane (id 0)
54 /// carries no width and is never returned.
55 std::vector<SharedPtr<Lane>> GetLanes(
56 road::Lane::LaneType type = road::Lane::LaneType::Driving) const;
57
58 /// The lane with this id, or nullptr when the section has no such lane.
59 SharedPtr<Lane> GetLane(road::LaneId lane_id) const;
60
61 bool ContainsLane(road::LaneId lane_id) const;
62
63 /// The next section along the road, or nullptr at the end.
64 SharedPtr<LaneSection> GetNextSection() const;
65
66 /// The previous section along the road, or nullptr at the start.
67 SharedPtr<LaneSection> GetPreviousSection() const;
68
69 bool operator==(const LaneSection &rhs) const {
70 return (_road_id == rhs._road_id) && (_section_id == rhs._section_id);
71 }
72
73 bool operator!=(const LaneSection &rhs) const {

75 }

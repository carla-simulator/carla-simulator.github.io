230bool AProceduralVegetationTool::Bake()
231{
232 if (IsGenerating() || !bPreviewReady || PreviewInstanceCount() != Points.Num())
233 { Status = TEXT("Generate a complete preview before baking."); return false; }
234 Modify();
235 // Upgrade existing tool actors too: static instances cannot attach to a
236 // movable root, and must follow the region actor's transform after baking.
237 RootComponent->Modify();
238 RootComponent->SetMobility(EComponentMobility::Static);
239 TArray<UInstancedStaticMeshComponent*> PreviewComponents;
240 GetComponents(PreviewComponents);
241 TArray<TObjectPtr<UInstancedStaticMeshComponent>> NewBaked;
242 int32 Count = 0;
243 for (auto* Source : PreviewComponents)
244 {
245 if (BakedComponents.Contains(Source) || Source->GetInstanceCount() == 0) continue;
246 auto* Dest = NewObject<UInstancedStaticMeshComponent>(this, NAME_None, RF_Transactional);
247 Dest->SetStaticMesh(Source->GetStaticMesh());
248 Dest->SetMobility(EComponentMobility::Static);
249 Dest->SetupAttachment(RootComponent);
250 Dest->SetRelativeTransform(FTransform::Identity);
251 Dest->SetCollisionProfileName(TEXT("BlockAll"));
252 Dest->SetGenerateOverlapEvents(false);
253 Dest->SetCanEverAffectNavigation(true);
254 for (int32 M=0; M<Source->GetNumMaterials(); ++M) Dest->SetMaterial(M, Source->GetMaterial(M));
255 AddInstanceComponent(Dest);
256 Dest->RegisterComponent();
257 for (int32 I=0; I<Source->GetInstanceCount(); ++I)
258 {
259 FTransform Transform;
260 Source->GetInstanceTransform(I, Transform, true);
261 Dest->AddInstance(Transform, true);
262 ++Count;
263 }
264 NewBaked.Add(Dest);
265 }
266 for (UInstancedStaticMeshComponent* Old : BakedComponents) if (Old) { RemoveInstanceComponent(Old); Old->DestroyComponent(); }
267 BakedComponents = MoveTemp(NewBaked);
268 BakedInstanceCount = Count;
269 PCG->CleanupLocalImmediate(true);
270 bPreviewReady = false;
271 ATagger::TagActor(*this, true);
272 MarkPackageDirty();
273 Status = FString::Printf(TEXT("Baked %d plant and surround instances. Save the level."), Count);
274 return Count == Points.Num();
275}

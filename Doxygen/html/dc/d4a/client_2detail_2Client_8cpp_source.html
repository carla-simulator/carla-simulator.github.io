48 class Client::Pimpl {
49 public:
50
51 Pimpl(const std::string &host, uint16_t port, size_t worker_threads)
52 : endpoint(host + ":" + std::to_string(port)),
53 rpc_client(host, port),
54 streaming_client(host) {
55 rpc_client.set_timeout(5000u);
56 streaming_client.AsyncRun(
57 worker_threads > 0u ? worker_threads : std::thread::hardware_concurrency());
58 }
59
60 template <typename ... Args>
61 auto RawCall(const std::string &function, Args && ... args) {
62 try {
63 return rpc_client.call(function, std::forward<Args>(args) ...);
64 } catch (const ::rpc::timeout &) {
65 throw_exception(TimeoutException(endpoint, GetTimeout()));
66 }
67 }
68
69 template <typename T, typename ... Args>
70 auto CallAndWait(const std::string &function, Args && ... args) {
71 auto object = RawCall(function, std::forward<Args>(args) ...);
72 using R = typename carla::rpc::Response<T>;
73 auto response = object.template as<R>();
74 if (response.HasError()) {
75 throw_exception(std::runtime_error(response.GetError().What()));
76 }
77 return Get(response);
78 }
79
80 template <typename ... Args>
81 void AsyncCall(const std::string &function, Args && ... args) {
82 // Discard returned future.
83 rpc_client.async_call(function, std::forward<Args>(args) ...);
84 }
85
86 time_duration GetTimeout() const {
87 auto timeout = rpc_client.get_timeout();
88 DEBUG_ASSERT(timeout.has_value());
89 return time_duration::milliseconds(static_cast<size_t>(*timeout));
90 }
91
92 const std::string endpoint;
93
94 rpc::Client rpc_client;
95
96 streaming::Client streaming_client;
97 };

272 std::vector<std::string> Client::GetRequiredFiles(const std::string &folder, const bool download) const {
273 // Get the list of required files
274 auto requiredFiles = _pimpl->CallAndWait<std::vector<std::string>>("get_required_files", folder);
275
276 if (download) {
277
278 // For each required file, check if it exists and request it otherwise
279 for (auto requiredFile : requiredFiles) {
280 if (!FileTransfer::FileExists(requiredFile)) {
281 RequestFile(requiredFile);
282 log_info("Could not find the required file in cache, downloading... ", requiredFile);
283 } else {
284 log_info("Found the required file in cache! ", requiredFile);
285 }
286 }
287 }
288 return requiredFiles;
289 }

297 std::vector<uint8_t> Client::GetCacheFile(const std::string &name, const bool request_otherwise) const {
298 // Get the file from the cache in the file transfer
299 std::vector<uint8_t> file = FileTransfer::ReadFile(name);
300
301 // If it isn't in the cache, download it if request otherwise is true
302 if (file.empty() && request_otherwise) {
303 RequestFile(name);
304 file = FileTransfer::ReadFile(name);
305 }
306 return file;
307 }

467 rpc::Actor Client::SpawnActorWithParent(
468 const rpc::ActorDescription &description,
469 const geom::Transform &transform,
470 rpc::ActorId parent,
471 rpc::AttachmentType attachment_type) {
472
473 if (attachment_type == rpc::AttachmentType::SpringArm ||
474 attachment_type == rpc::AttachmentType::SpringArmGhost)
475 {
476 const auto a = transform.location.MakeUnitVector(std::numeric_limits<float>::epsilon());
477 const auto z = geom::Vector3D(0.0f, 0.f, 1.0f);
478 constexpr float OneEps = 1.0f - std::numeric_limits<float>::epsilon();
479 if (geom::Math::Dot(a, z) > OneEps) {
480 std::cout << "WARNING: Transformations with translation only in the 'z' axis are ill-formed when \
481 using SpringArm or SpringArmGhost attachment. Please, be careful with that." << std::endl;
482 }
483 }
484
485 return _pimpl->CallAndWait<rpc::Actor>("spawn_actor_with_parent",
486 description,
487 transform,
488 parent,
489 attachment_type);
490 }

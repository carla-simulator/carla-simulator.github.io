41static EQualityLevel QualityLevelFromString(
42 const FString &SQualitySettingsLevel,
43 const EQualityLevel Default = EQualityLevel::INVALID)
44{
45 if (SQualitySettingsLevel.Equals("Low"))
46 {
47 return EQualityLevel::Low;
48 }
49 if (SQualitySettingsLevel.Equals("Medium"))
50 {
51 return EQualityLevel::Medium;
52 }
53 if (SQualitySettingsLevel.Equals("High"))
54 {
55 return EQualityLevel::High;
56 }
57 if (SQualitySettingsLevel.Equals("Epic"))
58 {
59 return EQualityLevel::Epic;
60 }
61 return Default;
62}

70static void LoadSettingsFromConfig(
71 const FIniFile &ConfigFile,
72 UCarlaSettings &Settings,
73 const bool bLoadCarlaServerSection)
74{
75 // CarlaServer.
76 if (bLoadCarlaServerSection)
77 {
78 ConfigFile.GetInt(S_CARLA_SERVER, TEXT("WorldPort"), Settings.RPCPort);
79 ConfigFile.GetInt(S_CARLA_SERVER, TEXT("RPCPort"), Settings.RPCPort);
80 Settings.StreamingPort = Settings.RPCPort + 1u;
81 Settings.SecondaryPort = Settings.RPCPort + 2u;
82 ConfigFile.GetInt(S_CARLA_SERVER, TEXT("StreamingPort"), Settings.StreamingPort);
83 ConfigFile.GetInt(S_CARLA_SERVER, TEXT("SecondaryPort"), Settings.SecondaryPort);
84 FString Tmp;
85 ConfigFile.GetString(S_CARLA_SERVER, TEXT("PrimaryIP"), Tmp);
86 Settings.PrimaryIP = TCHAR_TO_UTF8(*Tmp);
87 ConfigFile.GetInt(S_CARLA_SERVER, TEXT("PrimaryPort"), Settings.PrimaryPort);
88 }
89 ConfigFile.GetBool(S_CARLA_SERVER, TEXT("SynchronousMode"), Settings.bSynchronousMode);
90 ConfigFile.GetBool(S_CARLA_SERVER, TEXT("DisableRendering"), Settings.bDisableRendering);
91 // QualitySettings.
92 FString sQualityLevel;
93 ConfigFile.GetString(S_CARLA_QUALITYSETTINGS, TEXT("QualityLevel"), sQualityLevel);
94 Settings.SetQualityLevel(QualityLevelFromString(sQualityLevel));
95}

97static bool GetSettingsFilePathFromCommandLine(FString &Value)
98{
99 if (FParse::Value(FCommandLine::Get(), TEXT("-carla-settings="), Value))
100 {
101 if (FPaths::IsRelative(Value))
102 {
103 Value = FPaths::ConvertRelativePathToFull(FPaths::LaunchDir(), Value);
104 }
105 return true;
106 }
107 return false;
108}

114void UCarlaSettings::LoadSettings()
115{
116 CurrentFileName = TEXT("");
117 // Load settings from project Config folder if present.
118 LoadSettingsFromFile(FPaths::Combine(FPaths::ProjectConfigDir(), TEXT("CarlaSettings.ini")), false);
119 // Load settings given by command-line arg if provided.
120 {
121 FString FilePath;
122 if (GetSettingsFilePathFromCommandLine(FilePath))
123 {
124 LoadSettingsFromFile(FilePath, true);
125 }
126 }
127 // Override settings from command-line.
128 {
129 uint32 Value;
130 if (FParse::Value(FCommandLine::Get(), TEXT("-world-port="), Value) ||
131 FParse::Value(FCommandLine::Get(), TEXT("-carla-port="), Value) ||
132 FParse::Value(FCommandLine::Get(), TEXT("-carla-rpc-port="), Value) ||
133 FParse::Value(FCommandLine::Get(), TEXT("-carla-world-port="), Value))
134 {
135 RPCPort = Value;
136 StreamingPort = Value + 1u;
137 SecondaryPort = Value + 2u;
138 }
139 if (FParse::Value(FCommandLine::Get(), TEXT("-carla-streaming-port="), Value))
140 {
141 StreamingPort = Value;
142 }
143 if (FParse::Value(FCommandLine::Get(), TEXT("-carla-secondary-port="), Value))
144 {
145 SecondaryPort = Value;
146 }
147 FString Tmp;
148 if (FParse::Value(FCommandLine::Get(), TEXT("-carla-primary-host="), Tmp))
149 {
150 PrimaryIP = TCHAR_TO_UTF8(*Tmp);
151 }
152 if (FParse::Value(FCommandLine::Get(), TEXT("-carla-primary-port="), Value))
153 {
154 PrimaryPort = Value;
155 }
156 FString StringQualityLevel;
157 if (FParse::Value(FCommandLine::Get(), TEXT("-quality-level="), StringQualityLevel))
158 {
159 QualityLevel = QualityLevelFromString(StringQualityLevel, EQualityLevel::Epic);
160 }
161 if (FParse::Param(FCommandLine::Get(), TEXT("-no-rendering")))
162 {
163 bDisableRendering = true;
164 }
165 // FParse::Param only matches single-dash switches; also accept the
166 // historically documented "--ros2" spelling so it does not silently no-op.
167 if (FParse::Param(FCommandLine::Get(), TEXT("-ros2")) ||
168 FCString::Strifind(FCommandLine::Get(), TEXT("--ros2")) != nullptr)
169 {
170 ROS2 = true;
171 }
172 FString RmwValue;
173 if (FParse::Value(FCommandLine::Get(), TEXT("-rmw="), RmwValue))
174 {
175 RmwName = RmwValue.ToLower();
176 }
177 int32 DomainIdValue;
178 if (FParse::Value(FCommandLine::Get(), TEXT("-ros-domain-id="), DomainIdValue))
179 {
180 ROS2DomainId = DomainIdValue;
181 }
182 }
183}

185void UCarlaSettings::LoadSettingsFromString(const FString &INIFileContents)
186{
187 UE_LOG(LogCarla, Log, TEXT("Loading CARLA settings from string"));
188 FIniFile ConfigFile;
189 CurrentFileName = TEXT("<string-provided-by-client>");
190 ConfigFile.ProcessInputFileContents(INIFileContents, CurrentFileName);
191 constexpr bool bLoadCarlaServerSection = false;
192 LoadSettingsFromConfig(ConfigFile, *this, bLoadCarlaServerSection);

216void UCarlaSettings::LoadSettingsFromFile(const FString &FilePath, const bool bLogOnFailure)
217{
218 if (FPaths::FileExists(FilePath))
219 {
220 UE_LOG(LogCarla, Log, TEXT("Loading CARLA settings from \"%s\""), *FilePath);
221 const FIniFile ConfigFile(FilePath);
222 constexpr bool bLoadCarlaServerSection = true;
223 LoadSettingsFromConfig(ConfigFile, *this, bLoadCarlaServerSection);
224 CurrentFileName = FilePath;
225 }
226 else if (bLogOnFailure)
227 {
228 UE_LOG(LogCarla, Error, TEXT("Unable to find settings file \"%s\""), *FilePath);
229 }
230}

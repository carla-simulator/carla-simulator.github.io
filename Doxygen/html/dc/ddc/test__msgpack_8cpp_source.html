17TEST(msgpack, response) {
18 using mp = carla::MsgPack;
19 const std::string error = "Uh ah an error!";
20 Response<int> r = ResponseError(error);
21 auto s = mp::UnPack<decltype(r)>(mp::Pack(r));
22 ASSERT_TRUE(r.HasError());
23 ASSERT_EQ(r.GetError().What(), error);
24 ASSERT_TRUE(s.HasError());
25 ASSERT_EQ(s.GetError().What(), error);
26 r.Reset(42);
27 s = mp::UnPack<decltype(r)>(mp::Pack(r));
28 ASSERT_FALSE(r.HasError());
29 ASSERT_EQ(r.Get(), 42);
30 ASSERT_FALSE(s.HasError());
31 ASSERT_EQ(s.Get(), 42);
32 r.SetError(error);
33 s = mp::UnPack<decltype(r)>(mp::Pack(r));
34 ASSERT_FALSE(r);
35 ASSERT_EQ(r.GetError().What(), error);
36 ASSERT_FALSE(s);
37 ASSERT_EQ(s.GetError().What(), error);
38 Response<std::vector<float>> rv;
39 auto sv = mp::UnPack<decltype(rv)>(mp::Pack(rv));
40 ASSERT_TRUE(rv.HasError());
41 ASSERT_TRUE(sv.HasError());
42 Response<void> er;
43 ASSERT_TRUE(er.HasError());
44 er = Response<void>::Success();
45 auto es = mp::UnPack<decltype(er)>(mp::Pack(er));
46 ASSERT_FALSE(er.HasError());
47 ASSERT_FALSE(es.HasError());
48 er.SetError(error);
49 es = mp::UnPack<decltype(er)>(mp::Pack(er));
50 ASSERT_FALSE(er);
51 ASSERT_EQ(er.GetError().What(), error);
52 ASSERT_FALSE(es);
53 ASSERT_EQ(es.GetError().What(), error);
54}

87TEST(msgpack, actor_spawn_frame_is_compatible_both_ways) {
88 namespace c = carla;
89 Actor actor;
90 actor.id = 42u;
91 actor.parent_id = 7u;
92 actor.description.id = "vehicle.tesla.model3";
93 actor.spawn_frame = 123456u;
94
95 auto round_trip = c::MsgPack::UnPack<Actor>(c::MsgPack::Pack(actor));
96 ASSERT_EQ(round_trip.spawn_frame, actor.spawn_frame);
97
98 // A new client reading an old server's reply: no spawn_frame, defaults to 0.
99 ActorWithoutSpawnFrame old_actor;
100 old_actor.id = 42u;
101 old_actor.parent_id = 7u;
102 old_actor.description.id = "vehicle.tesla.model3";
103 auto from_old = c::MsgPack::UnPack<Actor>(c::MsgPack::Pack(old_actor));
104 ASSERT_EQ(from_old.id, old_actor.id);
105 ASSERT_EQ(from_old.parent_id, old_actor.parent_id);
106 ASSERT_EQ(from_old.description.id, old_actor.description.id);
107 ASSERT_EQ(from_old.spawn_frame, 0u);
108
109 // An old client reading a new server's reply ignores the extra field.
110 auto from_new = c::MsgPack::UnPack<ActorWithoutSpawnFrame>(c::MsgPack::Pack(actor));
111 ASSERT_EQ(from_new.id, actor.id);
112 ASSERT_EQ(from_new.parent_id, actor.parent_id);
113 ASSERT_EQ(from_new.description.id, actor.description.id);

17TEST(CameraIntrinsics, hd_90deg_fov_focal_length) {
18 // 1920x1080 with 90deg horizontal FOV: fx = 1920 / (2 * tan(45deg)) = 960.
19 const auto k = carla::ros2::ComputeIntrinsics(1920u, 1080u, 90.0f);
20 EXPECT_NEAR(k.fx, 960.0, kEps);
21 EXPECT_NEAR(k.fy, 960.0, kEps);
22 EXPECT_NEAR(k.cx, 960.0, kEps);
23 EXPECT_NEAR(k.cy, 540.0, kEps);
24}

26TEST(CameraIntrinsics, square_60deg_fov) {
27 // 800x800 with 60deg FOV: fx = 800 / (2 * tan(30deg)) ~ 692.82.
28 const auto k = carla::ros2::ComputeIntrinsics(800u, 800u, 60.0f);
29 const double expected_fx = 800.0 / (2.0 * std::tan(M_PI / 6.0));
30 EXPECT_NEAR(k.fx, expected_fx, kEps);
31 EXPECT_NEAR(k.fy, k.fx, kEps);
32 EXPECT_NEAR(k.cx, 400.0, kEps);
33 EXPECT_NEAR(k.cy, 400.0, kEps);
34}

36TEST(CameraIntrinsics, regression_guard_against_buggy_precedence) {
37 // The pre-86ebadcee spelling computed fx = width / (2 * tan(fov) * M_PI / 360),
38 // applying the degree-to-radian conversion to tan(fov) instead of to fov.
39 // For fov=90 that returned ~0.06; the corrected form returns 960. Guard against
40 // a regression to the buggy spelling.
41 const auto k = carla::ros2::ComputeIntrinsics(1920u, 1080u, 90.0f);
42 ASSERT_GT(k.fx, 100.0);
43}

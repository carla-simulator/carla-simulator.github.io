20bool FCarlaShowOnlyFilter::ParseLabels(
21 const FString &CommaSeparated,
22 TSet<Label> &OutLabels,
23 FString &OutUnknown)
24{
25 OutLabels.Reset();
26 OutUnknown.Reset();
27 TArray<FString> Tokens;
28 CommaSeparated.ParseIntoArray(Tokens, TEXT(","), true);
29 bool bAllKnown = true;
30 for (FString &Token : Tokens)
31 {
32 Token.TrimStartAndEndInline();
33 if (Token.IsEmpty())
34 {
35 continue;
36 }
37 // ATagger::GetTagFromString is the mapping the tagger itself uses for
38 // component tags; it is a case-insensitive substring match, so "car",
39 // "Car" and "Cars" all name CityObjectLabel::Car.
40 const Label Parsed = ATagger::GetTagFromString(Token);
41 if (Parsed == Label::None)
42 {
43 bAllKnown = false;
44 OutUnknown += (OutUnknown.IsEmpty() ? TEXT("") : TEXT(",")) + Token;
45 continue;
46 }
47 OutLabels.Add(Parsed);
48 }
49 return bAllKnown;
50}

77void FCarlaShowOnlyFilter::SetShowOnlyActorIds(const FString &CommaSeparated)
78{
79 ActorIds.Reset();
80 TArray<FString> Tokens;
81 CommaSeparated.ParseIntoArray(Tokens, TEXT(","), true);
82 for (FString &Token : Tokens)
83 {
84 Token.TrimStartAndEndInline();
85 if (Token.IsEmpty())
86 {
87 continue;
88 }
89 if (Token.IsNumeric())
90 {
91 ActorIds.Add(static_cast<uint32>(FCString::Atoi64(*Token)));
92 }
93 else
94 {
95 UE_LOG(LogCarla, Warning,
96 TEXT("show_only_actor_ids: \"%s\" is not an actor id, ignored"), *Token);
97 }
98 }
99 bDirty = true;
100}

117void FCarlaShowOnlyFilter::Rescan(UWorld *World, const UCarlaEpisode *Episode)
118{
119 const double Start = FPlatformTime::Seconds();
120 CachedActors.Reset();
121
122 TSet<Label> Wanted = ShowOnlyTags;
123 Wanted.Append(ShadowCatcherTags);
124
125 if (Wanted.Num() > 0)
126 {
127 TInlineComponentArray<UPrimitiveComponent *> Primitives;
128 for (TActorIterator<AActor> It(World); It; ++It)
129 {
130 AActor *Actor = *It;
131 if (!IsValid(Actor))
132 {
133 continue;
134 }
135 Primitives.Reset();
136 Actor->GetComponents<UPrimitiveComponent>(Primitives);
137 for (UPrimitiveComponent *Primitive : Primitives)
138 {
139 if (Primitive != nullptr &&
140 Wanted.Contains(ATagger::GetTagOfTaggedComponent(*Primitive)))
141 {
142 CachedActors.Add(Actor);
143 break;
144 }
145 }
146 }
147 }
148
149 if (ActorIds.Num() > 0 && Episode != nullptr)
150 {
151 const FActorRegistry &Registry = Episode->GetActorRegistry();
152 for (const uint32 Id : ActorIds)
153 {
154 const FCarlaActor *View = Registry.FindCarlaActor(Id);
155 if (View == nullptr)
156 {
157 continue;
158 }
159 AActor *Actor = const_cast<AActor *>(View->GetActor());
160 if (IsValid(Actor))
161 {
162 CachedActors.AddUnique(Actor);
163 }
164 }
165 }
166
167 UE_LOG(LogCarla, Verbose,
168 TEXT("ShowOnlyFilter: %d actor(s) selected out of %d in %.2f ms"),
169 CachedActors.Num(), World->GetActorCount(),
170 (FPlatformTime::Seconds() - Start) * 1000.0);
171}

173void FCarlaShowOnlyFilter::ApplyTo(USceneCaptureComponent2D &Capture) const
174{
175 if (!IsEnabled())
176 {
177 if (Capture.PrimitiveRenderMode == ESceneCapturePrimitiveRenderMode::PRM_UseShowOnlyList)
178 {
179 Capture.PrimitiveRenderMode = ESceneCapturePrimitiveRenderMode::PRM_RenderScenePrimitives;
180 Capture.ShowOnlyActors.Reset();
181 Capture.ClearShowOnlyComponents();
182 }
183 return;
184 }
185 Capture.PrimitiveRenderMode = ESceneCapturePrimitiveRenderMode::PRM_UseShowOnlyList;
186 Capture.ShowOnlyActors.Reset(CachedActors.Num());
187 for (const TWeakObjectPtr<AActor> &Weak : CachedActors)
188 {
189 if (AActor *Actor = Weak.Get())
190 {
191 Capture.ShowOnlyActors.Add(Actor);
192 }
193 }

196 /// Call on the game thread right after CaptureScene().
197 static FSensorCaptureContext MakeCaptureContext(const ASensor &Sensor)
198 {
199 FSensorCaptureContext Context;
200 Context.FrameIndex = FCarlaEngine::GetFrameCounter();
201 Context.Timestamp = Sensor.GetEpisode().GetElapsedGameTime();
202 Context.Transform = Sensor.GetActorTransform();
203 const AActor *ParentActor = Sensor.GetAttachParentActor();
204 Context.RelativeTransform = ParentActor ?
205 Context.Transform.GetRelativeTransform(ParentActor->GetActorTransform()) :
206 Context.Transform;
207 return Context;
208 }

229 // Send sensor data to the client.
230 template <
231 typename SensorType,
232 typename ElementType>
233 static void SendDataToClient(
234 SensorType&& Sensor, // The data's owning sensor.
235 TArrayView<ElementType> SensorData, // Data to send to the client.
236 const FSensorCaptureContext &CaptureContext // Metadata snapshotted at capture time.
237 )
238 {
239 using carla::sensor::SensorRegistry;
240 using SensorT = std::remove_const_t<std::remove_reference_t<SensorType>>;
241 constexpr size_t HeaderOffset = SensorRegistry::get<SensorT*>::type::header_offset;
242
243 if (!Sensor.AreClientsListening())
244 return;
245
246 auto Stream = Sensor.GetDataStream(Sensor);
247 Stream.SetFrameNumber(CaptureContext.FrameIndex);
248 Stream.SetTimestamp(CaptureContext.Timestamp);
249 Stream.SetTransform(CaptureContext.Transform);
250
251 auto Buffer = Stream.PopBufferFromPool();
252 Buffer.copy_from(
253 HeaderOffset,
254 boost::asio::buffer(
255 SensorData.GetData(),
256 SensorData.Num() * sizeof(ElementType)));
257
258 if (!Buffer.data())
259 return;
260
261 auto Serialized = SensorRegistry::Serialize(Sensor, std::move(Buffer));
262 auto SerializedBuffer = carla::Buffer(std::move(Serialized));
263 auto BufferView = carla::BufferView::CreateFrom(std::move(SerializedBuffer));
264
265#if defined(WITH_ROS2)
266 auto ROS2 = carla::ros2::ROS2::GetInstance();
267 if (ROS2->IsEnabled())
268 {
269 TRACE_CPUPROFILER_EVENT_SCOPE_STR("ROS2 SendDataToClient");
270 auto StreamId = carla::streaming::detail::token_type(Sensor.GetToken()).get_stream_id();
271 auto SensorTypeId = Stream.GetSensorType();
272 const FTransform &Transform = CaptureContext.RelativeTransform;
273
274 int W = -1, H = -1;
275 float Fov = -1.0f;
276 auto WidthOpt = Sensor.GetAttribute("image_size_x");
277 if (WidthOpt.has_value())
278 W = FCString::Atoi(*WidthOpt->Value);
279 auto HeightOpt = Sensor.GetAttribute("image_size_y");
280 if (HeightOpt.has_value())
281 H = FCString::Atoi(*HeightOpt->Value);
282 auto FovOpt = Sensor.GetAttribute("fov");
283 if (FovOpt.has_value())
284 Fov = FCString::Atof(*FovOpt->Value);
285
286 ROS2->ProcessDataFromCamera(
287 SensorTypeId,
288 StreamId,
289 Transform,
290 W, H,
291 Fov,
292 BufferView,
293 &Sensor);
294 }
295#endif
296
297 if (Sensor.AreClientsListening())
298 Stream.Send(Sensor, BufferView);
299 }

23 /// path-traced distance camera. The header is layout-identical to
24 /// ImageSerializer::ImageHeader (every image serializer shares it, see
25 /// data::ImageTmpl); only the pixel type differs.

49 template <typename Sensor>
50 inline Buffer DistanceImageSerializer::Serialize(const Sensor &sensor, Buffer &&bitmap) {
51 DEBUG_ASSERT(bitmap.size() > sizeof(ImageHeader));
52 ImageHeader header = {
53 sensor.GetImageWidth(),
54 sensor.GetImageHeight(),
55 sensor.GetFOVAngle()
56 };
57 std::memcpy(bitmap.data(), reinterpret_cast<const void *>(&header), sizeof(header));
58 return std::move(bitmap);

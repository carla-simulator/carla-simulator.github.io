37 static std::unordered_map<std::string, std::string> ParseProjectionParameters(const std::string &s){
38
39 // +key[=value]; value can be "quoted" or 'quoted' or an unquoted token (no spaces)
40 // This also captures bare flags like +south
41 std::unordered_map<std::string, std::string> geo_parameters_map;
42
43 static const std::regex re(R"(\+([A-Za-z0-9_]+)(?:=("[^"]*"|'[^']*'|[^ \t\r\n+]+))?)");
44 for (std::sregex_iterator it(s.begin(), s.end(), re), end; it != end; ++it) {
45 std::string key = (*it)[1].str();
46 std::string val = (*it)[2].str();
47 geo_parameters_map[key] = val;
48 }
49
50 return geo_parameters_map;
51 }

53 static std::unordered_map<std::string, double> ParseOffsetParameters(const pugi::xml_node &geo_offset){
54
55 std::unordered_map<std::string, double> geo_offset_result;
56
57 for (auto attr = geo_offset.first_attribute(); attr; attr = attr.next_attribute()){

124 static geom::Ellipsoid CreateEllipsoid(std::unordered_map<std::string, std::string> parameters){
125
126 geom::Ellipsoid ellps;
127 std::string value;
128 bool initialized=false;
129 double x;
130
131 if (TryGetParameter(value, parameters, "ellps") || TryGetParameter(value, parameters, "datum")) {
132 value = ToLowerCase(value);
133 auto it = geom::custom_ellipsoids.find(value);
134 if (it != geom::custom_ellipsoids.end()) {
135 ellps.a = it->second.first;
136 ellps.f_inv = it->second.second;
137 initialized=true;
138 }
139 }
140
141 initialized |= TryGetParameter(ellps.a, parameters, "a");
142 // Specific semi-major axis
143 if (TryGetParameter(x, parameters, "b")) {
144 initialized = true;
145 ellps.fromb(x);
146 }
147 if (TryGetParameter(x, parameters, "f")) {
148 initialized = true;
149 ellps.fromf(x);
150 }
151 initialized |= TryGetParameter(ellps.f_inv, parameters, "rf");
152
153 if (!initialized) {
154 auto val = geom::custom_ellipsoids.find("wgs84");
155 ellps.a = val->second.first;
156 ellps.f_inv = val->second.second;
157 }
158
159 return ellps;
160 }

162 static boost::optional<geom::OffsetTransform> CreateOffsetTransform(std::unordered_map<std::string, double>& offset_parameters){
163
164 if(offset_parameters.empty()){
165 return boost::none;
166 }
167
168 double x = 0.0;
169 double y = 0.0;
170 double z = 0.0;
171 double hdg = 0.0;
172
173 TryGetParameter(x, offset_parameters, "x");
174 TryGetParameter(y, offset_parameters, "y");
175 TryGetParameter(z, offset_parameters, "z");
176 TryGetParameter(hdg, offset_parameters, "hdg");
177
178 return geom::OffsetTransform{x, y, z, hdg};
179 }

288 static auto ParseGeoProjectionAndReference(const std::string& proj_string, const pugi::xml_node &geo_offset){
289 auto parameters = ParseProjectionParameters(proj_string);
290 auto offset_parameters = ParseOffsetParameters(geo_offset);
291 auto ellipsoid = CreateEllipsoid(parameters);
292 auto offset_transform = CreateOffsetTransform(offset_parameters);
293
294 // Get the projection type
295 std::string proj;
296 if (!TryGetParameter(proj, parameters, "proj")) {
297 log_warning("cannot find the type of projection, using default transverse mercator");
298 return std::make_pair(
299 CreateDefaultProjection(ellipsoid),
300 CreateDefaultGeoReference());
301 }
302
303 // Parse the parameters
304 if (proj == "tmerc") {
305 return std::make_pair(
306 CreateTransverseMercatorProjection(parameters, proj_string, ellipsoid),
307 CreateTransverseMercatorGeoReference(parameters));
308 } else if (proj == "utm") {
309 return std::make_pair(
310 CreateUniversalTransverseMercatorProjection(parameters, proj_string, ellipsoid, offset_transform),
311 CreateUniversalTransverseMercatorGeoReference(parameters));
312 } else if (proj == "merc") {
313 return std::make_pair(
314 CreateWebMercatorProjection(proj_string, ellipsoid),
315 CreateWebMercatorGeoReference());
316 } else if (proj == "lcc") {
317 return std::make_pair(
318 CreateLambertConformalConicProjection(parameters, proj_string, ellipsoid),
319 CreateLambertConic2SPGeoReference(parameters));
320 } else {
321 log_debug("projection '" + proj + "' is not supported, using default transverse mercator.");
322 return std::make_pair(
323 CreateDefaultProjection(ellipsoid),
324 CreateDefaultGeoReference());
325 }

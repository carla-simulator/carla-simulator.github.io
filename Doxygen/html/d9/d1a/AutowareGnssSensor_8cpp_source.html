44void AAutowareGnssSensor::PostPhysTick(UWorld *World, ELevelTick TickType, float DeltaSeconds)
45{
46 TRACE_CPUPROFILER_EVENT_SCOPE(AAutowareGnssSensor::PostPhysTick);
47
48 FVector ActorLocation = GetActorLocation();
49 ALargeMapManager * LargeMap = UCarlaStatics::GetLargeMapManager(GetWorld());
50 if (LargeMap)
51 {
52 ActorLocation = LargeMap->LocalToGlobalLocation(ActorLocation);
53 }
54 carla::geom::Location Location = ActorLocation;
55 carla::geom::GeoLocation CurrentLocation = CurrentGeoProjection.TransformToGeoLocation(Location);
56
57 LatitudeValue = CurrentLocation.latitude;
58 LongitudeValue = CurrentLocation.longitude;
59 AltitudeValue = CurrentLocation.altitude;
60
61 if (bApplyNoiseError)
62 {
63 // Compute the noise for the sensor
64 const float LatError = RandomEngine->GetNormalDistribution(0.0f, LatitudeDeviation);
65 const float LonError = RandomEngine->GetNormalDistribution(0.0f, LongitudeDeviation);
66 const float AltError = RandomEngine->GetNormalDistribution(0.0f, AltitudeDeviation);
67
68 // Apply the noise to the sensor
69 LatitudeValue += LatitudeBias + LatError;
70 LongitudeValue += LongitudeBias + LonError;
71 AltitudeValue += AltitudeBias + AltError;
72 }
73
74 auto DataStream = GetDataStream(*this);
75
76 // ROS2
77 #if defined(WITH_ROS2)
78 auto ROS2 = carla::ros2::ROS2::GetInstance();
79 if (ROS2->IsEnabled())
80 {
81 TRACE_CPUPROFILER_EVENT_SCOPE_STR("ROS2 Send");
82 auto StreamId = carla::streaming::detail::token_type(GetToken()).get_stream_id();
83 AActor* ParentActor = GetAttachParentActor();
84 const FTransform SensorWorldTransform = GetActorTransform();
85 double MgrsOffsetPosition[3] = {0.0, 0.0, 0.0};
86 if (MgrsDataAsset)
87 {
88 MgrsOffsetPosition[0] = static_cast<double>(MgrsDataAsset->MgrsOffsetPosition.X);
89 MgrsOffsetPosition[1] = static_cast<double>(MgrsDataAsset->MgrsOffsetPosition.Y);
90 MgrsOffsetPosition[2] = static_cast<double>(MgrsDataAsset->MgrsOffsetPosition.Z);
91 }
92
93 if (ParentActor)
94 {
95 FTransform LocalTransformRelativeToParent = GetActorTransform().GetRelativeTransform(ParentActor->GetActorTransform());
96 ROS2->ProcessDataFromAutowareGNSS(DataStream.GetSensorType(), StreamId, LocalTransformRelativeToParent, carla::geom::GeoLocation{LatitudeValue, LongitudeValue, AltitudeValue}, SensorWorldTransform, MgrsOffsetPosition, this);
97 }
98 else
99 {
100 ROS2->ProcessDataFromAutowareGNSS(DataStream.GetSensorType(), StreamId, DataStream.GetSensorTransform(), carla::geom::GeoLocation{LatitudeValue, LongitudeValue, AltitudeValue}, SensorWorldTransform, MgrsOffsetPosition, this);
101 }
102 }
103 #endif
104
105 {
106 TRACE_CPUPROFILER_EVENT_SCOPE_STR("AAutowareGnssSensor Stream Send");
107 DataStream.SerializeAndSend(*this, carla::geom::GeoLocation{LatitudeValue, LongitudeValue, AltitudeValue});
108 }
109}

111void AAutowareGnssSensor::LoadMgrsData()
112{
113 if (MgrsDataAsset)
114 {
115 return; // Already loaded
116 }
117
118 if (const auto* WS = Cast<AAutowareWorldSettings>(GetWorld()->GetWorldSettings()))
119 {
120 if (WS->MgrsDataAssetSoftPtr.IsNull())
121 {
122 UE_LOG(LogCarla, Warning, TEXT("MGRS Data Asset SoftPtr not set in WorldSettings."));
123 return;
124 }
125
126 UMgrsDataAsset* Data = WS->MgrsDataAssetSoftPtr.Get();
127 if (!IsValid(Data))
128 {
129 Data = WS->MgrsDataAssetSoftPtr.LoadSynchronous();
130 }
131
132 MgrsDataAsset = Data;
133 }
134}

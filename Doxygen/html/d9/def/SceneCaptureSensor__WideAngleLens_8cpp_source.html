433uint8 ASceneCaptureSensor_WideAngleLens::FindFaceIndex(FVector2D UV) const
434{
435 const float R = hypotf(UV.X, UV.Y);
436 float Theta;
437 switch (CameraModel)
438 {
439 case ECameraModel::LUT1D:
440 Theta = CameraModelUtil::LUT1D::SampleInverse(R, LensLUT, LensThetaMax);
441 break;
442 case ECameraModel::BrownConrady:
443 Theta = CameraModelUtil::ComputeAngle(ECameraModel::Perspective, R, KannalaBrandtCameraCoefficients);
444 break;
445 default:
446 Theta = CameraModelUtil::ComputeAngle(CameraModel, R, KannalaBrandtCameraCoefficients);
447 break;
448 }
449 const float HalfPi = PI / 2.0f;
450 const float Phi = HalfPi - Theta;
451 const float Rho = atan2f(UV.Y, UV.X);
452
453 float PhiSin = 0.0F;
454 float PhiCos = 0.0F;
455 float RhoSin = 0.0F;
456 float RhoCos = 0.0F;
457
458 FMath::SinCos(&PhiSin, &PhiCos, Phi);
459 FMath::SinCos(&RhoSin, &RhoCos, Rho);
460
461 auto Direction = FVector(PhiSin, PhiCos * RhoCos, -PhiCos * RhoSin);
462 Direction.Normalize();
463
464 auto DirectionAbs = Direction.GetAbs();
465
466 if (DirectionAbs.X >= DirectionAbs.Y && DirectionAbs.X >= DirectionAbs.Z)
467 return Direction.X < 0 ? 1U : 0U;
468 else if (DirectionAbs.Y >= DirectionAbs.X && DirectionAbs.Y >= DirectionAbs.Z)
469 return Direction.Y < 0 ? 3U : 2U;
470 else
471 return Direction.Z < 0 ? 5U : 4U;
472}

474uint8 ASceneCaptureSensor_WideAngleLens::ComputeCubemapRenderMask() const
475{
476 // Equirectangular projection samples the full sphere, so every cube face
477 // must be rendered regardless of the configured FOV.
478 if (bRenderEquirectangular)
479 return (1U << CubeFace_PosX) | (1U << CubeFace_NegX) |
480 (1U << CubeFace_PosY) | (1U << CubeFace_NegY) |
481 (1U << CubeFace_PosZ) | (1U << CubeFace_NegZ);
482
483 static const float Pi = PI;
484 static const float HalfPi = Pi / 2.0f;
485 static const float Sqrt2 = sqrtf(2.0f);
486
487 auto Mask = 1U << CubeFace_PosX; // Render front face by default.
488
489 // LUT1D: the table's acceptance angle bounds the field of view on both
490 // axes (the principal point may sit anywhere, so assume the full circle).
491 // Otherwise the fov attribute does, widened to the frame corners unless
492 // the FOV mask crops them.
493 const auto FOV = CameraModel == ECameraModel::LUT1D
494 ? FVector2D(2.0F * LensThetaMax, 2.0F * LensThetaMax)
495 : FVector2D(GetFOVAngleX(), GetFOVAngleY()) * (GetFOVMaskEnable() ? 1 : Sqrt2);
496
497 if (FOV.Y > HalfPi)
498 {
499 Mask |= 1U << CubeFace_PosZ;
500 Mask |= 1U << CubeFace_NegZ;
501 }
502
503 if (FOV.X > HalfPi)
504 {
505 Mask |= 1U << CubeFace_PosY;
506 Mask |= 1U << CubeFace_NegY;
507 }
508
509 if (FOV.X > Pi || FOV.Y > Pi)
510 {
511 Mask |= 1U << CubeFace_NegX;
512 }
513
514 return (uint8)Mask;
515}

517void ASceneCaptureSensor_WideAngleLens::CaptureSceneExtended()
518{
519 TRACE_CPUPROFILER_EVENT_SCOPE(ASceneCaptureSensor_WideAngleLens::CaptureSceneExtended);
520
521 bool SkipVFTR = CVarWideAngleSensorSkipVFTR.GetValueOnAnyThread() != 0;
522
523 // Capture the previous value so it can be restored afterwards, instead of
524 // unconditionally forcing the cvar back on and clobbering the project /
525 // scalability setting for every other view in the scene.
526 IConsoleVariable* VolumetricFogTRVar = nullptr;
527 int32 PreviousVolumetricFogTR = 1;
528
529 if (!SkipVFTR)
530 {
531 VolumetricFogTRVar = IConsoleManager::Get().FindConsoleVariable(
532 TEXT("r.VolumetricFog.TemporalReprojection"));
533 if (VolumetricFogTRVar != nullptr)
534 {
535 PreviousVolumetricFogTR = VolumetricFogTRVar->GetInt();
536 FlushRenderingCommands();
537 VolumetricFogTRVar->Set(TEXT("0"), ECVF_SetByCode);
538 }
539 }
540
541 // Show-only mode: refresh the actor list (rescans only when the world's
542 // actor count changed) and push it into every face capture.
543 if (ShowOnlyFilter.IsEnabled())
544 {
545 ShowOnlyFilter.Refresh(GetWorld(), &GetEpisode());
546 for (auto FaceCapture : FaceCaptures)
547 if (FaceCapture != nullptr)
548 ShowOnlyFilter.ApplyTo(*FaceCapture);
549 }
550
551 // Same camera-cut warm-up as ASceneCaptureSensor::EnqueueRenderSceneImmediate, on every face,
552 // so a new fisheye does not start from a default exposure and adapt for seconds.
553 static IConsoleVariable* SpawnCameraCutsVar =
554 IConsoleManager::Get().FindConsoleVariable(TEXT("carla.Camera.SpawnCameraCuts"));
555 const bool bCameraCut = SpawnCameraCutsVar != nullptr && NumCaptures < SpawnCameraCutsVar->GetInt();
556 if (bCameraCut)
557 ++NumCaptures;
558
559 for (uint8 i = 0; i < 6; ++i)
560 if (CubemapRenderMask & (1U << i))
561 {
562 // Reset by the renderer after the capture.
563 if (bCameraCut)
564 FaceCaptures[i]->bCameraCutThisFrame = true;
565 FaceCaptures[i]->CaptureScene();
566 }
567
568 // Snapshot the data the render-thread pass needs so the lambda does not
569 // dereference `this` after the actor may have been destroyed. The render
570 // targets and sampler are still resolved through a TWeakObjectPtr so we
571 // skip the dispatch cleanly if the owning actor is gone.
572 TWeakObjectPtr<ASceneCaptureSensor_WideAngleLens> WeakSelf(this);
573 UTextureRenderTarget2D* RenderTargetsSnapshot[] =
574 {
575 FaceRenderTargets[0],
576 FaceRenderTargets[1],
577 FaceRenderTargets[2],
578 FaceRenderTargets[3],
579 FaceRenderTargets[4],
580 FaceRenderTargets[5]
581 };
582 UTextureRenderTarget2D* CaptureRenderTargetSnapshot = CaptureRenderTarget;
583 FRHISamplerState* CubemapSamplerSnapshot = CubemapSampler;
584
585 CameraModelUtil::FDistortCubemapToImageOptions DistortedOptions = { };
586 DistortedOptions.KannalaBrandtCoefficients = KannalaBrandtCameraCoefficients;
587 DistortedOptions.LUT = LensLUT;
588 DistortedOptions.ThetaMax = LensThetaMax;
589 DistortedOptions.bExplicitIntrinsics = bExplicitIntrinsics;
590 // Normalized viewport units -> pixels (x by width, y by height).
591 DistortedOptions.Intrinsics = FVector4f(
592 LensIntrinsics.X * static_cast<float>(ImageWidth),
593 LensIntrinsics.Y * static_cast<float>(ImageHeight),
594 LensIntrinsics.Z * static_cast<float>(ImageWidth),
595 LensIntrinsics.W * static_cast<float>(ImageHeight));
596 {
597 const FLinearColor Invalid = GetInvalidPixelColor();
598 DistortedOptions.InvalidColor = FVector4f(Invalid.R, Invalid.G, Invalid.B, Invalid.A);
599 }
600 DistortedOptions.YFOVAngle = YFOVAngle;
601 DistortedOptions.YFocalLength = YFocalLength;
602 DistortedOptions.LongitudeOffset = LongitudeOffset;
603 DistortedOptions.FOVFadeSize = FOVFadeSize;
604 DistortedOptions.CameraModel = CameraModel;
605 DistortedOptions.bRenderEquirectangular = bRenderEquirectangular;
606 DistortedOptions.bFOVMaskEnable = bFOVMaskEnable;
607 DistortedOptions.bRenderPerspective = bRenderPerspective;
608
609 ENQUEUE_RENDER_COMMAND(WideAngleLensCommand)(
610 [WeakSelf,
611 RenderTargetsSnapshot,
612 CaptureRenderTargetSnapshot,
613 CubemapSamplerSnapshot,
614 DistortedOptions](FRHICommandListImmediate& RHICmdList)
615 {
616 TRACE_CPUPROFILER_EVENT_SCOPE(WideAngleLensCommand);
617
618 if (!WeakSelf.IsValid())
619 return;
620
621 FRDGBuilder GraphBuilder(RHICmdList);
622
623 UTextureRenderTarget2D* RenderTargets[] =
624 {
625 RenderTargetsSnapshot[0],
626 RenderTargetsSnapshot[1],
627 RenderTargetsSnapshot[2],
628 RenderTargetsSnapshot[3],
629 RenderTargetsSnapshot[4],
630 RenderTargetsSnapshot[5]
631 };
632
633 CameraModelUtil::DistortCubemapToImage(
634 GraphBuilder,
635 CaptureRenderTargetSnapshot,
636 RenderTargets,
637 CubemapSamplerSnapshot,
638 DistortedOptions);
639
640 GraphBuilder.Execute();
641 });
642
643 if (CVarWideAngleSensorDumpAllFrames.GetValueOnAnyThread() == 1)
644 {
645 static thread_local auto FrameCounter = 0U;
646
647 const TCHAR* Names[(size_t)ECameraModel::MaxEnum] =
648 {
649 TEXT("Perspective"),
650 TEXT("Stereographic"),
651 TEXT("Equidistant"),
652 TEXT("Equisolid"),
653 TEXT("Orthographic"),
654 TEXT("KannalaBrandt"),
655 TEXT("BrownConrady"),
656 TEXT("LUT1D"),
657 TEXT("FTheta"),
658 };
659
660 auto CameraTypeName = Names[(uint8)CameraModel];
661 auto Path = CVarWideAngleSensorDumpAllFramesPath.GetValueOnAnyThread();
662
663 if (CVarWideAngleSensorDumpAllFramesCubemap.GetValueOnAnyThread())
664 {
665 for (uint8 FaceIndex = 0; FaceIndex != 6; ++FaceIndex)
666 {
667 FPixelReader::SavePixelsToDisk(
668 *FaceRenderTargets[FaceIndex],
669 FString::Printf(

696void ASceneCaptureSensor_WideAngleLens::BeginPlay()
697{
698 const bool bInForceLinearGamma = !bEnablePostProcessingEffects;
699
700 const auto Format = bEnable16BitFormat ? PF_FloatRGBA : PF_B8G8R8A8;
701 const auto Side = FaceSize > 0 ?
702 static_cast<uint32>(FaceSize) :
703 std::max(GetImageWidth(), GetImageHeight());
704
705 CaptureRenderTarget->InitCustomFormat(
706 GetImageWidth(),
707 GetImageHeight(),
708 Format,
709 bInForceLinearGamma);
710
711 for (auto FaceRenderTarget : FaceRenderTargets)
712 {
713 FaceRenderTarget->InitCustomFormat(
714 Side, Side,
715 Format,
716 bInForceLinearGamma);
717 }
718
719 if (bEnablePostProcessingEffects)
720 {
721 for (auto Face : FaceRenderTargets)
722 Face->TargetGamma = TargetGamma;
723 CaptureRenderTarget->TargetGamma = TargetGamma;
724 }
725
726 for (uint8 i = 0; i != 6; ++i)
727 {
728 FaceCaptures[i]->Deactivate();
729 FaceCaptures[i]->TextureTarget = FaceRenderTargets[i];
730 }
731
732 SetUpSceneCaptureComponents(FaceCaptures);
733
734 for (auto FaceCapture : FaceCaptures)
735 {
736 // Use PostProcessConfig for each face capture to honour the
737 // ue5-dev convention (mirrors ASceneCaptureSensor::BeginPlay).
738 auto PostProcessConfig = FPostProcessConfig(
739 FaceCapture->PostProcessSettings,
740 FaceCapture->ShowFlags);
741 PostProcessConfig.UpdateFromSceneCaptureComponent2D(*FaceCapture);
742 PostProcessConfig.EnablePostProcessingEffects(bEnablePostProcessingEffects);
743 FaceCapture->ShowFlags = PostProcessConfig.EngineShowFlags;
744 FaceCapture->PostProcessSettings = PostProcessConfig.PostProcessSettings;
745
746 // The assignment above replaces the settings the constructor put on
747 // this face, auto exposure included: each face would then adapt to its
748 // own view and the cube's faces meet at visible brightness seams.
749 SceneCaptureSensorWideAngleLens_local_ns::SetCameraDefaultOverrides(*FaceCapture);
750
751 FaceCapture->CaptureSource = ESceneCaptureSource::SCS_FinalColorLDR;
752 FaceCapture->UpdateContent();
753 FaceCapture->Activate();
754 }
755
756 // Raise the render-fence block timeout: a single wide-angle capture
757 // enqueues up to 6 face scene-captures plus the RDG composite pass, which
758 // can exceed the default timeout on heavy scenes and trip a false-positive
759 // render-fence fatal assert. This is a process-wide setting; CARLA already
760 // runs the render pipeline outside interactive-latency budgets, so a wider
761 // watchdog is harmless for other views.
762 UKismetSystemLibrary::ExecuteConsoleCommand(
763 GetWorld(),
764 FString("g.TimeoutForBlockOnRenderFence 300000"));
765
766 // This ensures the camera is always spawning the raindrops in case the
767 // weather was previously set to have rain.
768 auto Weather = GetEpisode().GetWeather();
769 if (Weather != nullptr)
770 Weather->NotifyWeather(this);
771
772 ReadbackPool = MakeShared<FRHIGPUReadbackPool, ESPMode::ThreadSafe>(
773 TEXT("WideAngleLensReadback"));
774
775 Super::BeginPlay();
776}

778void ASceneCaptureSensor_WideAngleLens::PrePhysTick(float DeltaSeconds)
779{
780 Super::PrePhysTick(DeltaSeconds);
781
782 auto CaptureComponents = GetCaptureComponents2D();
783
784 // Add the view information every tick. It's only used for one tick and then
785 // removed by the streamer.
786 // FOVAngle is stored in degrees; convert to radians and use the
787 // half-angle tangent, guarding against a non-positive denominator.
788 const float HalfFOVTan = FMath::Tan(
789 0.5f * FMath::DegreesToRadians(CaptureComponents[0]->FOVAngle));
790 IStreamingManager::Get().AddViewInformation(
791 CaptureComponents[0]->GetComponentLocation(),
792 ImageWidth,
793 HalfFOVTan > 0.0f ? ImageWidth / HalfFOVTan : ImageWidth);
794}

801void ASceneCaptureSensor_WideAngleLens::EndPlay(const EEndPlayReason::Type EndPlayReason)
802{
803 Super::EndPlay(EndPlayReason);
804 // The sensor counter is process-wide and only used to derive unique
805 // subobject names at construction. Resetting it here would let a
806 // surviving sibling sensor collide with the indices of the next spawned
807 // sensor.
808 FlushRenderingCommands();
809
810 // Release the GPU memory now rather than at the next garbage collection:
811 // six persistent face view states (Lumen scene, TSR history, shadow
812 // caches; several GB at 2048^2) plus the seven render targets. The
813 // UObjects stay alive until GC, but without their resources.
814 for (auto FaceCapture : FaceCaptures)
815 {
816 if (FaceCapture == nullptr)
817 continue;
818 FaceCapture->Deactivate();
819 FaceCapture->TextureTarget = nullptr;
820 FaceCapture->ReleaseViewStates();
821 }
822 for (auto FaceRenderTarget : FaceRenderTargets)
823 {
824 if (FaceRenderTarget != nullptr)
825 FaceRenderTarget->ReleaseResource();
826 }
827 if (CaptureRenderTarget != nullptr)
828 CaptureRenderTarget->ReleaseResource();
829 FlushRenderingCommands();
830 // In-flight readback tasks keep their own reference; the pool dies with the
831 // last of them (see ASceneCaptureSensor::EndPlay).
832 ReadbackPool.Reset();
833}

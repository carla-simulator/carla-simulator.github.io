49/// Mounts CARLA content packs (content-only Unreal plugins cooked as DLC
50/// against a base release, wrapped with a carla-pack.json manifest) into the
51/// running process, editor or cooked game alike.
52///
53/// Implemented as a UEngineSubsystem: it outlives game instances and worlds,
54/// so packs survive load_world, and it exists before the first map loads.
55/// Auto-discovery runs from Initialize() in game processes (packaged binary
56/// or editor -game); FCarlaEngine::NotifyInitGame calls Discover() again as
57/// a no-op safety net so the RPC server never starts before the packs are in.
58/// The full editor and commandlets (the cooker) do not auto-mount packs.

105 /// Merge catalog entries: an entry whose key already exists replaces it in
106 /// place (later files override by id), new keys are appended.
107 template <typename T, typename KeyFn>
108 static void MergeCatalog(TArray<T> &Into, const TArray<T> &From, KeyFn Key)
109 {
110 for (const T &Entry : From)
111 {
112 const FString EntryKey = Key(Entry);
113 const int32 Existing = Into.IndexOfByPredicate([&](const T &Other)
114 {
115 return Key(Other).Equals(EntryKey, ESearchCase::IgnoreCase);
116 });
117 if (Existing != INDEX_NONE)
118 {
119 Into[Existing] = Entry;
120 }
121 else
122 {
123 Into.Add(Entry);
124 }
125 }
126 }

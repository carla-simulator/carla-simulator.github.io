22 // Plain XYZI path (4 float32 per point). Shadows the base method so a call
23 // through a CarlaLidarPublisher pointer always resets the extended mode
24 // before the base implementation consults the layout virtuals.
25 bool WritePointCloud(
26 std::int32_t seconds,
27 std::uint32_t nanoseconds,
28 std::uint32_t height,
29 std::uint32_t width,
30 const std::uint8_t *data) {
31 _extended = false;
32 return CarlaPointCloudPublisher::WritePointCloud(
33 seconds, nanoseconds, height, width, data);
34 }
35
36 // Extended XYZIRCAEDT path (tier4 SetDataEx port): expands each 16 B XYZI
37 // detection into the 32 B Autoware PointXYZIRCAEDT layout declared by
38 // kLidarFieldsEx. `data` is the raw XYZI buffer (`width` points of 4
39 // float32s), `channel_point_counts` holds `channel_count` per-channel point
40 // counts (the LidarData header tail), and `vertical_angles` provides one
41 // elevation (radians) per channel.
42 bool WriteExtendedPointCloud(
43 std::int32_t seconds,
44 std::uint32_t nanoseconds,
45 std::uint32_t height,
46 std::uint32_t width,
47 const std::uint8_t *data,
48 const std::uint32_t *channel_point_counts,
49 std::size_t channel_count,
50 const std::vector<float> &vertical_angles);
51
52private:
53 [[nodiscard]] std::size_t GetPointSize() const override;
54 [[nodiscard]] const PointFieldDescriptor *GetFieldDescriptors() const override;
55 [[nodiscard]] std::size_t GetFieldDescriptorCount() const override;
56 [[nodiscard]] std::vector<std::uint8_t> ComputePointCloud(
57 std::uint32_t height, std::uint32_t width, const std::uint8_t *data) const override;
58
59 // Extended-mode state, set by WriteExtendedPointCloud before it delegates

61 bool _extended = false;
62 std::vector<std::uint32_t> _channel_point_counts;
63 std::vector<float> _vertical_angles;
64};

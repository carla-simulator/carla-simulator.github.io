25void Router::Stop() {
26 ClearSessions();
27 // Joins worker threads before the acceptor closes, so no in-flight accept
28 // can race with teardown.
29 _pool.Stop();
30 // ~Router() never runs in production (PrimaryCommands holds a shared_ptr
31 // back to this Router, see set_router()), so Stop() must release the
32 // listening socket itself. Listener::Stop() is idempotent, so this is
33 // safe even if ~Listener() also runs later.
34 if (_listener) {
35 _listener->Stop();
36 }
37 _listener.reset();
38}

89void Router::ConnectSession(std::shared_ptr<Primary> session) {
90 DEBUG_ASSERT(session != nullptr);
91 std::scoped_lock<std::mutex> lock(_mutex);
92 _new_sessions.emplace_back(session);
93 _sessions.emplace_back(std::move(session));
94 _session_count = _sessions.size();
95 log_info("Connected secondary servers:", _sessions.size());
96 if (_last_map) {
97 // The resync callback runs from HandleResponse once the map is loaded.
98 const load_map_id_type load_id = ++_last_load_id;
99 SendLoadMap(_sessions.back(), *_last_map, load_id, _last_map_extras);
100 _loading[_sessions.back().get()] = load_id;
101 return;
102 }
103 if (_callback)
104 _callback();
105}

107void Router::DisconnectSession(std::shared_ptr<Primary> session) {
108 DEBUG_ASSERT(session != nullptr);
109 std::scoped_lock<std::mutex> lock(_mutex);
110 if (_sessions.size() > 0) {
111 _sessions.erase(
112 std::remove(_sessions.begin(), _sessions.end(), session),
113 _sessions.end());
114 _session_count = _sessions.size();
115 log_info("Connected secondary servers:", _sessions.size());
116 }
117 _loading.erase(session.get());
118
119 // A request may still be waiting on this session's response; without this,
120 // the caller blocked in std::future::get() would hang forever.
121 auto pending = _promises.find(session.get());
122 if (pending != _promises.end()) {
123 RejectPromise(pending->second, "secondary server disconnected before responding");
124 _promises.erase(pending);
125 }
126}

142void Router::HandleResponse(std::shared_ptr<Primary> session, Buffer buffer) {
143 std::scoped_lock<std::mutex> lock(_mutex);
144
145 // Checked before the promise lookup: no command reply can equal this
146 // exact marker (see commands.h), so it can never be misdelivered as data
147 // for whatever happens to be pending on this session.
148 const std::string_view payload(
149 reinterpret_cast<const char *>(buffer.data()), buffer.size());
150 if (const auto ready_load_id = ParseEpisodeReadyMessage(payload)) {
151 log_info("Secondary episode ready, re-arming full resync");
152 auto loading = _loading.find(session.get());
153 if ((loading != _loading.end()) && (loading->second == *ready_load_id)) {
154 _loading.erase(loading);
155 } else if (loading != _loading.end()) {
156 log_info("multigpu router: ignoring episode ready of load ", *ready_load_id,
157 ", waiting for load ", loading->second);
158 }
159 if (_callback) {
160 _callback();
161 }
162 return;
163 }
164
165 auto prom = _promises.find(session.get());
166 if (prom != _promises.end()) {
167 log_info("Got data from secondary (with promise): ", buffer.size());
168 prom->second->set_value({std::move(session), std::move(buffer)});
169 _promises.erase(prom);
170 } else {
171 log_warning("multigpu router: ignoring unsolicited data with no pending request: ", buffer.size(), " bytes");
172 }
173}

175void Router::ClearSessions() {
176 std::scoped_lock<std::mutex> lock(_mutex);
177 _sessions.clear();
178 _session_count = 0u;
179 _new_sessions.clear();
180 _loading.clear();
181 // Symmetric with DisconnectSession: any request still waiting on a
182 // response from a session that is about to disappear must not be left to
183 // hang forever.
184 RejectAllPending("router is shutting down");
185 log_info("Disconnecting all secondary servers");
186}

188void Router::Write(MultiGPUCommand id, Buffer &&buffer) {
189 // define the command header
190 CommandHeader header;
191 header.id = id;
192 header.size = buffer.size();
193 Buffer buf_header(reinterpret_cast<uint8_t *>(&header), sizeof(header));
194
195 auto view_header = carla::BufferView::CreateFrom(std::move(buf_header));
196 auto view_data = carla::BufferView::CreateFrom(std::move(buffer));
197 auto message = Primary::MakeMessage(view_header, view_data);
198
199 // write to multiple servers
200 std::scoped_lock<std::mutex> lock(_mutex);
201 for (auto &s : _sessions) {
202 if (s != nullptr) {
203 s->Write(message);
204 }
205 }
206}

208void Router::SendLoadMap(
209 const std::shared_ptr<Primary> &session,
210 std::string_view map,
211 load_map_id_type load_id,
212 const LoadMapExtras &extras) {
213 const std::string payload = MakeLoadMapPayload(map, load_id, extras);
214 Buffer buffer(reinterpret_cast<const unsigned char *>(payload.data()), payload.size());
215
216 CommandHeader header;
217 header.id = MultiGPUCommand::LOAD_MAP;
218 header.size = static_cast<uint32_t>(buffer.size());
219 Buffer buf_header(reinterpret_cast<uint8_t *>(&header), sizeof(header));
220
221 auto view_header = carla::BufferView::CreateFrom(std::move(buf_header));
222 auto view_data = carla::BufferView::CreateFrom(std::move(buffer));
223 if (_request_observer) {
224 _request_observer(session.get(), MultiGPUCommand::LOAD_MAP, *view_data);
225 }
226 session->Write(Primary::MakeMessage(view_header, view_data));
227}

253std::future<SessionInfo> Router::WriteToNext(MultiGPUCommand id, Buffer &&buffer) {
254 // define the command header
255 CommandHeader header;
256 header.id = id;
257 header.size = buffer.size();
258 Buffer buf_header(reinterpret_cast<uint8_t *>(&header), sizeof(header));
259
260 auto view_header = carla::BufferView::CreateFrom(std::move(buf_header));
261 auto view_data = carla::BufferView::CreateFrom(std::move(buffer));
262 auto message = Primary::MakeMessage(view_header, view_data);
263
264 // create the promise for the posible answer
265 auto response = std::make_shared<std::promise<SessionInfo>>();
266
267 // write to the next server only
268 std::scoped_lock<std::mutex> lock(_mutex);
269 if (_next >= _sessions.size()) {
270 _next = 0;
271 }
272 if (_next < _sessions.size()) {
273 auto s = _sessions[_next];
274 if (s == nullptr) {
275 RejectPromise(response, "no secondary session available to route the command");
276 } else if (_promises.contains(s.get())) {
277 // One slot per session (no per-request correlation id in the wire
278 // protocol): overwriting it would orphan the earlier promise forever,
279 // so refuse the newer request instead.
280 RejectPromise(response, "a request is already pending on this secondary session");
281 } else {
282 _promises[s.get()] = response;
283 if (_request_observer) {
284 _request_observer(s.get(), id, *view_data);
285 }
286 s->Write(message);
287 }
288 } else {
289 RejectPromise(response, "no secondary server connected");
290 }
291 ++_next;
292 return response->get_future();
293}

306std::future<SessionInfo> Router::WriteToOne(std::weak_ptr<Primary> server, MultiGPUCommand id, Buffer &&buffer) {
307 // define the command header
308 CommandHeader header;
309 header.id = id;
310 header.size = buffer.size();
311 Buffer buf_header(reinterpret_cast<uint8_t *>(&header), sizeof(header));
312
313 auto view_header = carla::BufferView::CreateFrom(std::move(buf_header));
314 auto view_data = carla::BufferView::CreateFrom(std::move(buffer));
315 auto message = Primary::MakeMessage(view_header, view_data);
316
317 // create the promise for the posible answer
318 auto response = std::make_shared<std::promise<SessionInfo>>();
319
320 // write to the specific server only
321 std::scoped_lock<std::mutex> lock(_mutex);
322 auto s = server.lock();
323 // A session's socket can already be closed while the Primary object is
324 // kept alive a little longer elsewhere, so also check _sessions
325 // membership; otherwise Write() would silently no-op and this promise
326 // would never resolve.
327 const bool still_connected =
328 s && (std::find(_sessions.begin(), _sessions.end(), s) != _sessions.end());
329 if (!still_connected) {
330 RejectPromise(response, "secondary session is no longer connected");
331 } else if (_promises.contains(s.get())) {
332 // Same reasoning as WriteToNext: one slot per session.
333 RejectPromise(response, "a request is already pending on this secondary session");
334 } else {
335 _promises[s.get()] = response;
336 if (_request_observer) {
337 _request_observer(s.get(), id, *view_data);
338 }
339 s->Write(message);
340 }
341 return response->get_future();
342}

58void ASensorSpawnerActor::SpawnSensors()
59{
60 // Check if we are doing a delayed spawn. If so, don't do nothing.
61 if(!SensorsToSpawnCopy.IsEmpty())
62 {
63 UE_LOG(LogSensorSpawnerActor, Warning, TEXT("Warning: ASensorSpawnerActor::SpawnSensors - Delayed spawn already in progress, wait until it ends"));
64 return;
65 }
66
67 if(DelayBetweenSpawns > 0.f)
68 {
69 SensorsToSpawnCopy = SensorsToSpawn;
70 GetWorldTimerManager().SetTimer(SpawnSensorsDelayedTimerHandle, this, &ASensorSpawnerActor::SpawnSensorsDelayed, DelayBetweenSpawns, true);
71 return;
72 }
73
74 for(const auto& SensorStruct : SensorsToSpawn)
75 {
76 if(const FActorDefinition* SensorDefinition = GetActorDefinitionByClass(SensorStruct.SensorClass))
77 {
78 FActorDescription SensorDescription;
79 GenerateSensorActorDescription(SensorDefinition, SensorDescription);
80
81 for(int i = 0; i < SensorStruct.Amount; i++)
82 {
83 SpawnSensorActor(SensorDescription, SensorStruct.bAttachToActor);
84 }
85 }
86 }
87}

120const FActorDefinition* ASensorSpawnerActor::GetActorDefinitionByClass(const TSubclassOf<AActor> ActorClass) const
121{
122 if(!ActorClass || !IsValid(CarlaEpisode))
123 {
124 return nullptr;
125 }
126
127 const TArray<FActorDefinition>& ActorDefinitions = CarlaEpisode->GetActorDefinitions();
128 // Checks that the class is exactly the same. If we want to allow also child classes use: ActorDef.Class->IsChildOf(ActorClass)
129 const FActorDefinition* ActorDefinition = ActorDefinitions.FindByPredicate([&](const FActorDefinition& ActorDef){ return ActorDef.Class == ActorClass; });
130
131 return ActorDefinition;
132}

135{

156void ASensorSpawnerActor::GenerateSensorActorDescription(const FActorDefinition* Definition, FActorDescription& SensorDescription) const
157{
158 SensorDescription.UId = Definition->UId;
159 SensorDescription.Id = Definition->Id;
160 SensorDescription.Class = Definition->Class;
161 SensorDescription.Variations.Reserve(Definition->Variations.Num());
162
163 FActorAttribute CreatedAttribute;
164 for(const FActorVariation& Variation : Definition->Variations)
165 {
166 if(Variation.RecommendedValues.IsValidIndex(0))
167 {
168 CreatedAttribute.Id = Variation.Id;
169 CreatedAttribute.Type = Variation.Type;
170 CreatedAttribute.Value = Variation.RecommendedValues[0];
171 SensorDescription.Variations.Emplace(CreatedAttribute.Id, CreatedAttribute);
172 }
173 }
174}

185void ASensorSpawnerActor::SpawnSensorsDelayed()
186{
187 if(SensorsToSpawnCopy.IsEmpty())
188 {
189 GetWorldTimerManager().ClearTimer(SpawnSensorsDelayedTimerHandle);
190 return;
191 }
192
193 if(const FActorDefinition* SensorDefinition = GetActorDefinitionByClass(SensorsToSpawnCopy[0].SensorClass))
194 {
195 FActorDescription SensorDescription;
196 GenerateSensorActorDescription(SensorDefinition, SensorDescription);
197 SpawnSensorActor(SensorDescription, SensorsToSpawnCopy[0].bAttachToActor);
198 }
199
200 SensorsToSpawnCopy[0].Amount--;
201
202 if(SensorsToSpawnCopy[0].Amount <= 0)
203 {
204 SensorsToSpawnCopy.RemoveAt(0);
205 }
206}

235void ASensorSpawnerActor::SaveSensorData(float DeltaSeconds)
236{
237 const FString FrameNumber = FString::Printf(TEXT("%lld"), FCarlaEngine::GetFrameCounter());
238 for(ASensor* CurrentSensor : SpawnedSensorsArray)
239 {
240 if(ASceneCaptureSensor* CaptureSensor = Cast<ASceneCaptureSensor>(CurrentSensor))
241 {
242 const FString FinalPath = FPaths::Combine(SaveImagePath, CaptureSensor->GetName(), FString::Printf(TEXT("%lld"), FDateTime::Now().ToUnixTimestamp()) + "-Frame_" + FrameNumber + ".png");
243 CaptureSensor->SaveCaptureToDisk(FinalPath);
244 continue;
245 }
246
247 if(const AInertialMeasurementUnit* IMUSensor = Cast<AInertialMeasurementUnit>(CurrentSensor))
248 {
249 const FVector Accelerometer = IMUSensor->GetAccelerometerValue().ToFVector();
250 const FVector Gyroscope = IMUSensor->GetGyroscopeValue().ToFVector();
251 float Compass = IMUSensor->GetCompassValue();
252 Compass = FMath::RadiansToDegrees(Compass);
253
254 const FString FilePath = FPaths::Combine(SaveImagePath, IMUSensor->GetName(), IMUSensor->GetName() + ".json");
255 UJsonFileManagerLibrary::SaveIMUDataToJson(FilePath, Accelerometer, Gyroscope, Compass, FrameNumber);
256 continue;
257 }
258
259 if(const AGnssSensor* GnssSensor = Cast<AGnssSensor>(CurrentSensor))
260 {
261 const FString FilePath = FPaths::Combine(SaveImagePath, GnssSensor->GetName(), GnssSensor->GetName() + ".json");
262 UJsonFileManagerLibrary::SaveGnssDataToJson(FilePath, GnssSensor->GetAltitudeValue(), GnssSensor->GetLatitudeValue(), GnssSensor->GetLongitudeValue(), FrameNumber);
263 }
264
265 if(const ARayCastLidar* LidarSensor = Cast<ARayCastLidar>(CurrentSensor))
266 {
267 const FString FilePath = FPaths::Combine(SaveImagePath, LidarSensor->GetName(), FString::Printf(TEXT("%lld"), FDateTime::Now().ToUnixTimestamp()) + "-Frame_" + FrameNumber + ".ply");
268 UJsonFileManagerLibrary::SaveLidarDataToPly(FilePath, LidarSensor->GetTestPointCloud(), 4);
269 }
270
271 if(const ARadar* RadarSensor = Cast<ARadar>(CurrentSensor))
272 {
273 const FString FilePath = FPaths::Combine(SaveImagePath, RadarSensor->GetName(), FString::Printf(TEXT("%lld"), FDateTime::Now().ToUnixTimestamp()) + "-Frame_" + FrameNumber + ".json");
274 UJsonFileManagerLibrary::SaveRadarDataToJson(FilePath, RadarSensor->GetRadarData(), FrameNumber);
275 }
276 }

294void ASensorSpawnerActor::DrawRadarSensorPoints() const
295{
296 float velocity_range = VelocityRange; // m/s
297 for(const ASensor* CurrentSensor : SpawnedSensorsArray)
298 {
299 if(const ARadar* RadarSensor = Cast<ARadar>(CurrentSensor); CurrentSensor)
300 {
301 FRotator CurrentRot = RadarSensor->GetActorRotation();
302 for( auto& detect : RadarSensor->GetRadarData().GetDetections() )
303 {
304 float Azi = FMath::RadiansToDegrees(detect.azimuth);
305 float Alt = FMath::RadiansToDegrees(detect.altitude);
306
307 FVector FWVector( (detect.depth - 0.25f) * 100.0f, 0.f, 0.f);
308 FTransform SensorTransform(FRotator(CurrentRot.Pitch + Alt,
309 CurrentRot.Yaw + Azi,
310 CurrentRot.Roll));
311
312 FWVector = SensorTransform.GetRotation().RotateVector(FWVector);
313
314 float Norm_velocity = detect.velocity / velocity_range;
315 FColor ColorPoint;
316 ColorPoint.R = FMath::Clamp(Norm_velocity, 0.0f, 1.0f) * 255;
317 ColorPoint.G = FMath::Clamp( 1.0f - abs(Norm_velocity), 0.0f, 1.0f) * 255;
318 ColorPoint.B = FMath::Clamp(abs(- 1.0f - Norm_velocity), 0.0f, 1.0f) * 255;
319
320 DrawDebugPoint(GetWorld(), RadarSensor->GetActorLocation() + FWVector , PointSize, ColorPoint, false, 0.06);
321 }
322 }

248 /// Capture the six faces, project them into the lens, and deliver the
249 /// image through ImageUtil's per-tick batched readback -- the same path as
250 /// every raster camera (ASceneCaptureCamera::PostPhysTick). The previous
251 /// FPixelReader::SendPixelsInRenderThread path locked the GPU readback on a
252 /// worker thread, which asserts IsInRenderingThread() on UE 5.8.
253 template <typename TSensor>
254 static void CaptureAndSendToClient(TSensor& Sensor)
255 {
256 // This path does not go through ASceneCaptureSensor::ShouldCaptureThisFrame(),
257 // so it needs its own check for a process without an RHI (-nullrhi).
258 if (!Sensor.AreClientsListening() || !FApp::CanEverRender())
259 {
260 return;
261 }
262 Sensor.EnqueueRenderSceneImmediate();
263 const auto CaptureContext = ASensor::MakeCaptureContext(Sensor);
264 ImageUtil::ReadImageDataAsyncFColor(
265 *Sensor.GetCaptureRenderTarget(),
266 [&Sensor, CaptureContext](TArrayView<const FColor> Pixels, FIntPoint) -> bool
267 {
268 if (!IsValid(&Sensor))
269 return false;
270 ASensor::SendDataToClient(Sensor, Pixels, CaptureContext);
271 return true;
272 },
273 /*bNonBlocking=*/false,
274 Sensor.GetReadbackPool()); // recycle staging buffers, as the pinhole cameras do
275 }

277 /// Blocks until the render thread has finished all its tasks.

281 FlushRenderingCommands();
282 }

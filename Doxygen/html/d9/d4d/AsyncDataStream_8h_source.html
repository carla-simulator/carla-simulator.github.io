26/// A streaming channel for sending sensor data to clients, supports sending
27/// data asynchronously. Data sent by the "Send" functions is passed to the
28/// serializer registered with the sensor at carla::sensor:SensorRegistry before
29/// being sent down the stream.
30///
31/// @warning This is a single-use object, a new one needs to be created for each
32/// new message.
33///
34/// FAsyncDataStream also has a pool of carla::Buffer that allows reusing the
35/// allocated memory, use it whenever possible.
36template <typename T>
37class FAsyncDataStreamTmpl
38{
39public:
40
41 using StreamType = T;
42
43 FAsyncDataStreamTmpl(FAsyncDataStreamTmpl &&) = default;
44
45 /// Return the token that allows subscribing to this stream.
46 auto GetToken() const
47 {
48 return Stream.GetToken();
49 }
50
51 /// Pop a Buffer from the pool. Buffers in the pool can reuse the memory
52 /// allocated by previous messages, significantly improving performance for
53 /// big messages.
54 carla::Buffer PopBufferFromPool()
55 {
56 return Stream.MakeBuffer();
57 }

66 /// allow to change the frame number of the header

175template <typename T>
176template <typename SensorT, typename... ArgsT>
177inline void FAsyncDataStreamTmpl<T>::SerializeAndSend(SensorT &Sensor, ArgsT &&... Args)
178{
179 // serialize data
180 carla::Buffer Data(carla::sensor::SensorRegistry::Serialize(Sensor, std::forward<ArgsT>(Args)...));
181
182 // create views of buffers
183 auto ViewHeader = carla::BufferView::CreateFrom(std::move(Header));
184 auto ViewData = carla::BufferView::CreateFrom(std::move(Data));
185
186 // send views
187 Stream.Write(ViewHeader, ViewData);
188}
189
190template <typename T>
191template <typename SensorT, typename... ArgsT>
192inline void FAsyncDataStreamTmpl<T>::Send(SensorT &Sensor, ArgsT &&... Args)
193{
194 // create views of buffers
195 auto ViewHeader = carla::BufferView::CreateFrom(std::move(Header));
196
197 // send views
198 Stream.Write(ViewHeader, std::forward<ArgsT>(Args)...);
199}

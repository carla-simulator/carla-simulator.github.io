32 * @note Steering compensation lookup table.

60// Linear interpolation function
61inline float Lerp(const float a, const float b, const float t) {
62 return a + t * (b - a);
63}

65// Get steering compensation ratio from lookup table with LERP
66template<std::size_t key_idx, std::size_t value_idx>
67float InterpolateSteeringData(const float angle) {
68 const auto AdjustSign = [angle](const float value) -> float {
69 return angle < 0.0f ? -value : value;
70 };
71
72 // Use absolute value for symmetric steering
73 const float key_angle = std::abs(angle);
74
75 // Handle edge cases
76 if (const auto front_key = std::get<key_idx>(DATA.front()); key_angle <= front_key) {
77 // Interpolate between 0 and first data point
78 const float t = key_angle / front_key;
79 return AdjustSign(Lerp(0.0f, std::get<value_idx>(DATA.front()), t));
80 }
81 if (key_angle >= std::get<key_idx>(DATA.back())) {
82 return AdjustSign(std::get<value_idx>(DATA.back()));
83 }
84
85 // Find the two points to interpolate between
86 for (size_t i = 0; i < DATA.size() - 1; ++i) {
87 const auto current_key_angle = std::get<key_idx >(DATA[i]);
88 const auto current_value_angle = std::get<value_idx>(DATA[i]);
89 const auto next_key_angle = std::get<key_idx >(DATA[i + 1]);
90 const auto next_value_angle = std::get<value_idx>(DATA[i + 1]);
91
92 if (current_key_angle <= key_angle && key_angle <= next_key_angle) {
93 // Calculate interpolation factor
94 const float t = (key_angle - current_key_angle) / (next_key_angle - current_key_angle);
95 // Interpolate between the two ratio values
96 return AdjustSign(Lerp(current_value_angle, next_value_angle, t));
97 }
98 }
99
100 // Default fallback (should not reach here)
101 return angle;
102}
103} // namespace detail

105/// Desired vehicle-input angle -> observed (actual) tire angle. Used when
106/// publishing the SteeringReport so Autoware sees the angle the wheels really
107/// reached.
108inline float GetSteeringOutput(const float input_angle) {
109 return detail::InterpolateSteeringData<0, 1>(input_angle);
110}
111
112/// Target (actual) tire angle -> vehicle input that produces it. Used when
113/// applying Autoware's steering command to the vehicle.
114inline float GetSteeringInput(const float output_angle) {
115 return detail::InterpolateSteeringData<1, 0>(output_angle);
116}

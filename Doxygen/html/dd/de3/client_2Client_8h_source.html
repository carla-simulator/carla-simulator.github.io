102 /// Return the current world, loading @a map_name first unless it is already

107 rpc::MapLayer map_layers = rpc::MapLayer::All) const {
108 const std::string current_map_name = GetWorld().GetMap()->GetName();
109 if (current_map_name == map_name || current_map_name == "Carla/Maps/" + map_name) {

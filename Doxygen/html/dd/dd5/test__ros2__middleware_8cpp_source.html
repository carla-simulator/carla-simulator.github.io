86// Byte-exact wire-format guard. The symmetric round-trip tests above cannot
87// detect an encoding drift (e.g. Fast-CDR defaulting to XCDRv2, which inserts
88// DHEADERs) because deserialize_from_cdr would consume whatever encoding
89// serialize_to_cdr emitted. This test pins the literal bytes so a regression
90// to XCDRv2 or big-endian is caught deterministically. These exact bytes are
91// what every backend puts on the wire: FastDDS via write_serialized_payload,
92// CycloneDDS via dds_writecdr (raw passthrough of serialize_to_cdr), and Zenoh.
93// Layout (classic CDR, encoding version 1, little-endian):
94// 00 01 00 00 encapsulation header (PLAIN_CDR little-endian + 2 option bytes)
95// 2A 00 00 00 sec = int32 42 (0x0000002A, LE)
96// 15 CD 5B 07 nanosec= uint32 123456789 (0x075BCD15, LE)
97TEST(cdr_serialization, time_golden_xcdrv1_bytes) {
98 carla::ros2::msg::Time original{};
99 original.sec = 42;
100 original.nanosec = 123456789u;
101
102 const auto buf = carla::ros2::serialize_to_cdr(original);
103
104 const std::vector<uint8_t> expected{
105 0x00u, 0x01u, 0x00u, 0x00u,
106 0x2Au, 0x00u, 0x00u, 0x00u,
107 0x15u, 0xCDu, 0x5Bu, 0x07u};
108 // Assert the size first: a failed or short serialization then yields a
109 // targeted size mismatch instead of a noisy full-vector diff.
110 ASSERT_EQ(buf.size(), expected.size());
111 EXPECT_EQ(buf, expected);
112 // Discriminator byte: classic CDR_LE is 0x01; any XCDRv2 representation
113 // (PLAIN_CDR2 / DELIMIT_CDR2 / PL_CDR2) would change byte 1.
114 EXPECT_EQ(buf[0], 0x00u);
115 EXPECT_EQ(buf[1], 0x01u);
116}

470TEST(cdr_serialization, deserialize_pointcloud2_hostile_length_returns_false) {
471 // Hand-craft a PointCloud2 buffer that claims its sequence has a hostile
472 // length (max uint32). Without the kMaxCdrSequenceElements cap, the
473 // call would attempt a multi-GB resize and abort the process.
474 std::vector<uint8_t> buf;
475 // Encapsulation header: CDR_LE + options.
476 buf.push_back(0x00u);
477 buf.push_back(0x01u);
478 buf.push_back(0x00u);
479 buf.push_back(0x00u);
480 // Header.stamp.sec (int32) + nanosec (uint32) = 8 bytes of zeros.
481 for (int i = 0; i < 8; ++i) buf.push_back(0x00u);
482 // Header.frame_id (string): length 1 (NUL only) + "\0" + 3 padding bytes
483 // to keep alignment for the next uint32.
484 buf.push_back(0x01u);
485 buf.push_back(0x00u);
486 buf.push_back(0x00u);
487 buf.push_back(0x00u);
488 buf.push_back(0x00u);
489 buf.push_back(0x00u);
490 buf.push_back(0x00u);
491 buf.push_back(0x00u);
492 // PointCloud2.height (uint32) + width (uint32).
493 for (int i = 0; i < 8; ++i) buf.push_back(0x00u);
494 // fields_size = 0xFFFFFFFF (hostile).
495 buf.push_back(0xFFu);
496 buf.push_back(0xFFu);
497 buf.push_back(0xFFu);
498 buf.push_back(0xFFu);
499

1553struct FakeFrag {
1554 const FakeFrag* nextfrag{nullptr};
1555 uint32_t min{0u};
1556 uint32_t maxp1{0u};
1557 const uint8_t* payload{nullptr}; ///< bytes for [min, maxp1)
1558};

1560const uint8_t* FakeFragPayload(const FakeFrag* f) { return f->payload; }

1562std::vector<uint8_t> MakePatternBytes(size_t size) {
1563 std::vector<uint8_t> bytes(size);
1564 for (size_t i = 0u; i < size; ++i) {
1565 bytes[i] = static_cast<uint8_t>((i * 31u + 7u) & 0xFFu);
1566 }
1567 return bytes;
1568}

1570/// Split [0, bytes.size()) into a nextfrag-linked chain of fragments of at
1571/// most frag_size bytes each, all viewing the same backing buffer.
1572std::vector<FakeFrag> MakeFragChain(
1573 const std::vector<uint8_t>& bytes,
1574 size_t frag_size)
1575{
1576 std::vector<FakeFrag> frags;
1577 for (size_t off = 0u; off < bytes.size(); off += frag_size) {
1578 const size_t end = std::min(bytes.size(), off + frag_size);
1579 FakeFrag f;
1580 f.min = static_cast<uint32_t>(off);
1581 f.maxp1 = static_cast<uint32_t>(end);
1582 f.payload = bytes.data() + off;
1583 frags.push_back(f);
1584 }
1585 for (size_t i = 1u; i < frags.size(); ++i) {
1586 frags[i - 1u].nextfrag = &frags[i];
1587 }
1588 return frags;
1589}

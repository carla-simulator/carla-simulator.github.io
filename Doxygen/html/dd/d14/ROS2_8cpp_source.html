101bool ROS2::Enable(bool enable, Middleware middleware, int domain_id) {
102 // Select the ROS 2 middleware before any publisher or subscriber is created.
103 // SetActiveMiddleware is the DDS-free bridge that resolves availability inside
104 // the shared library (the CARLA_ROS2_MIDDLEWARE_* macros are not visible here
105 // in carla-server) and keeps vendor headers out of carla-server.
106 if (enable && !SetActiveMiddleware(middleware)) {
107 log_error("ROS2: requested middleware '", MiddlewareToString(middleware),
108 "' is not available. Compiled in: ", GetAvailableMiddleware(), ".");
109 _enabled = false;
110 return false;
111 }
112 if (enable) {
113 // Configure the domain id before any transport context is created (the
114 // shared participants/sessions are created lazily on the first publisher,
115 // i.e. the clock publisher below). SetActiveDomainId is the DDS-free bridge
116 // so the value lands in the shared library's MiddlewareConfig, which the
117 // middlewares read.
118 const ResolvedDomainId resolved = SetActiveDomainId(domain_id);
119 const char* domain_source =
120 (resolved.source == DomainIdSource::CommandLine) ? "--ros-domain-id"
121 : (resolved.source == DomainIdSource::Environment) ? "ROS_DOMAIN_ID"
122 : "default";
123 log_info("ROS2: using middleware '", MiddlewareToString(middleware),
124 "', domain id: ", resolved.id, " (", domain_source, ")");
125 }
126 _enabled = enable;
127 log_info("ROS2 enabled: ", _enabled.load());
128 {
129 std::lock_guard<std::mutex> lock(_mutex);
130 _clock_publisher = std::make_shared<CarlaClockPublisher>();
131 }
132#if defined(WITH_ROS2_DEMO)
133 _basic_publisher = std::make_shared<BasicPublisher>();
134 _basic_publisher->Init();
135#endif
136 return true;
137}

139void ROS2::SetFrame(uint64_t frame) {
140 _frame = frame;
141 // The control callbacks run outside the lock so applying vehicle control
142 // never stalls the camera deliveries.
143 std::vector<std::pair<std::shared_ptr<BaseSubscriber>, ActorCallback>> pending;
144 {
145 std::lock_guard<std::mutex> lock(_mutex);
146 for (auto &element : _subscribers) {
147 auto callback_it = _actor_callbacks.find(element.first);
148 if (callback_it != _actor_callbacks.end()) {
149 pending.emplace_back(element.second, callback_it->second);
150 }
151 }
152 }
153 for (auto &element : pending) {
154 element.first->ProcessMessages(element.second);
155 }
156#if defined(WITH_ROS2_DEMO)
157 if (_basic_subscriber) {
158 void *actor = _basic_subscriber->GetActor();
159 if (!_basic_subscriber->IsAlive()) {
160 RemoveBasicSubscriberCallback(actor);
161 }
162 if (actor && _basic_subscriber->HasNewMessage()) {
163 auto it = _actor_message_callbacks.find(actor);
164 if (it != _actor_message_callbacks.end()) {
165 MessageControl control;
166 control.message = _basic_subscriber->GetMessage();
167 it->second(actor, control);
168 }
169 }
170 }
171#endif
172}

174void ROS2::SetTimestamp(double timestamp) {
175 double integral;
176 const double fractional = std::modf(timestamp, &integral);
177 const double multiplier = 1000000000.0;
178 const Stamp stamp{
179 static_cast<int32_t>(integral),
180 static_cast<uint32_t>(fractional * multiplier)};
181 std::shared_ptr<CarlaClockPublisher> clock_publisher;
182 {
183 std::lock_guard<std::mutex> lock(_mutex);
184 _stamp = stamp;
185 clock_publisher = _clock_publisher;
186 }
187 if (clock_publisher) {
188 clock_publisher->Write(stamp.seconds, stamp.nanoseconds);
189 clock_publisher->Publish();
190 }
191#if defined(WITH_ROS2_DEMO)
192 _basic_publisher->SetData("Hello from Carla!");
193 _basic_publisher->Publish();
194#endif
195}

236void ROS2::RegisterVehicle(
237 void *actor, std::string ros_name, std::string frame_id, ActorCallback callback,
238 bool enable_ackermann_control,
239 bool enable_autoware_control) {
240 std::lock_guard<std::mutex> lock(_mutex);
241 DropPublishersLocked(actor);
242 _registrations.insert_or_assign(
243 actor, ActorRegistration{ros_name, frame_id, true});
244
245 // Idempotency: drop any prior subscribers / callbacks bound to this actor
246 // so a re-registration does not accumulate duplicate DataReaders nor leave
247 // the previous callback wired.
248 _subscribers.erase(actor);
249 _autoware_vehicles.erase(actor);
250 _actor_callbacks.insert_or_assign(actor, std::move(callback));
251
252 // The legacy CarlaEgoVehicleControlSubscriber::Init built its topic as
253 // "rt/carla/" + [parent + "/"] + name + "/vehicle_control_cmd". With the
254 // new template constructors the suffix is appended inside each subscriber,
255 // so we hand them the base path only. BuildBaseTopicName also honors an
256 // exact-topic override registered via AddActorRosTopicName.
257 const std::string base_topic_name = BuildBaseTopicName(actor);
258
259 // The control modes are mutually exclusive: a vehicle listens on exactly one
260 // command surface, so two topics can never contend frame to frame. The
261 // Autoware command set wins over Ackermann (tier4 gave it priority over the
262 // ego control subscriber); Ackermann stays opt-in per youtalk's review.
263 if (enable_autoware_control) {
264 _autoware_vehicles.insert(actor);
265 _subscribers.insert({actor, std::make_shared<AutowareControlSubscriber>(actor, std::move(frame_id))});
266 } else if (enable_ackermann_control) {
267 _subscribers.insert({actor, std::make_shared<AckermannControlSubscriber>(actor, base_topic_name, std::move(frame_id))});
268 } else {
269 _subscribers.insert({actor, std::make_shared<CarlaEgoVehicleControlSubscriber>(actor, base_topic_name, std::move(frame_id))});
270 }
271}

346std::string ROS2::BuildParentChain(void *actor) const {
347 auto it = _actor_parents.find(actor);
348 if (it == _actor_parents.end()) {
349 return std::string{};
350 }
351 const std::string current_actor_name = LookupRosName(actor);
352 std::string parent_name;
353 for (auto *parent : it->second) {
354 const std::string name = LookupRosName(parent);
355 if (name.empty() || name == current_actor_name) {
356 continue;
357 }
358 parent_name = name + '/' + parent_name;
359 }
360 if (!parent_name.empty() && parent_name.back() == '/') {
361 parent_name.pop_back();
362 }
363 return parent_name;
364}

366std::string ROS2::BuildBaseTopicName(void *actor) const {
367 // Exact-topic override (ros_topic_name attribute flow): a non-empty custom
368 // name replaces the whole "rt/carla/[parent/]ros_name" convention. Suffixes
369 // for multi-topic sensors are appended by the publishers on top of this.
370 auto override_it = _actor_ros_topic_names.find(actor);
371 if (override_it != _actor_ros_topic_names.end() && !override_it->second.empty()) {
372 const std::string &custom = override_it->second;
373 std::string base_topic_name = "rt";
374 if (custom.front() != '/') {
375 base_topic_name += '/';
376 }
377 base_topic_name += custom;
378 return base_topic_name;
379 }
380
381 const std::string ros_name = LookupRosName(actor);
382 if (ros_name.empty()) {
383 return std::string{};
384 }
385 const std::string parent_chain = BuildParentChain(actor);
386 std::string base_topic_name = "rt/carla/";
387 if (!parent_chain.empty()) {
388 base_topic_name += parent_chain + "/";
389 }
390 base_topic_name += ros_name;
391 return base_topic_name;
392}

394void ROS2::ResolveAutoStreamSuffix(
395 void *actor,
396 const std::string &prefix,
397 carla::streaming::detail::stream_id_type id) {
398 auto it = _registrations.find(actor);
399 if (it == _registrations.end()) {
400 return;
401 }
402 const std::string placeholder = prefix + "__";
403 if (it->second.ros_name != placeholder) {
404 return;
405 }
406 std::string resolved = prefix + std::to_string(id);
407 it->second.ros_name = resolved;
408 if (it->second.frame_id == placeholder) {
409 it->second.frame_id = std::move(resolved);
410 }

413template <typename CameraT>
414std::shared_ptr<CarlaCameraPublisher> ROS2::GetOrCreateCameraSensor(
415 carla::streaming::detail::stream_id_type id,
416 void *actor,
417 const std::string &default_prefix) {
418 std::lock_guard<std::mutex> lock(_mutex);
419 auto it_camera = _camera_publishers.find(actor);
420 if (it_camera != _camera_publishers.end()) {
421 // Enforce the one-actor-one-camera-type invariant on the cache-hit path.
422 // RGB/Depth/SS/IS/Normals all alias CarlaRGBCameraPublisher (shared BGRA
423 // passthrough), so a hit across those types casts cleanly and is expected.
424 // Only an RGB <-> OpticalFlow mismatch fails the cast: that would route
425 // optical-flow float bytes through an RGB publisher (or vice versa). Surface
426 // it and skip the sample instead of letting the first-created type silently
427 // win and corrupt the published image.
428 auto typed = std::dynamic_pointer_cast<CameraT>(it_camera->second);
429 if (typed == nullptr) {
430 log_error(
431 "ROS2 camera publisher type mismatch for actor", actor,
432 "- the actor was dispatched as two different camera types; ignoring this sample.");
433 return nullptr;
434 }
435 return typed;
436 }
437 if (!IsKnownActorLocked(actor)) {
438 return nullptr;
439 }
440
441 ResolveAutoStreamSuffix(actor, default_prefix, id);
442 const std::string base_topic_name = BuildBaseTopicName(actor);
443 const std::string frame_id = LookupFrameId(actor);
444
445 auto new_publisher = std::make_shared<CameraT>(base_topic_name, frame_id);
446 _camera_publishers.insert({actor, new_publisher});
447 return new_publisher;
448}

450std::shared_ptr<BasePublisher> ROS2::GetOrCreateSensor(
451 int type, carla::streaming::detail::stream_id_type id, void *actor) {
452 std::lock_guard<std::mutex> lock(_mutex);
453 auto it_publishers = _publishers.find(actor);
454 if (it_publishers != _publishers.end()) {
455 return it_publishers->second;
456 }
457 if (!IsKnownActorLocked(actor)) {
458 return nullptr;
459 }
460
461 // Resolve auto-naming "prefix__" -> "prefix<stream_id>" before computing the
462 // topic name. Each enum case names its own prefix so the resolved ros_name
463 // stays stable across ticks.
464 auto resolve = [this, actor, id](const std::string &prefix) {
465 ResolveAutoStreamSuffix(actor, prefix, id);
466 };
467
468 std::shared_ptr<BasePublisher> publisher;
469 switch (type) {
470 case ESensors::CollisionSensor: {
471 resolve("collision");
472 publisher = std::make_shared<CarlaCollisionPublisher>(
473 BuildBaseTopicName(actor), LookupFrameId(actor));
474 break;
475 }
476 case ESensors::DVSCamera: {
477 resolve("dvs");
478 publisher = std::make_shared<CarlaDVSCameraPublisher>(
479 BuildBaseTopicName(actor), LookupFrameId(actor));
480 break;
481 }
482 case ESensors::GnssSensor: {
483 resolve("gnss");
484 publisher = std::make_shared<CarlaGNSSPublisher>(
485 BuildBaseTopicName(actor), LookupFrameId(actor));
486 break;
487 }
488 case ESensors::InertialMeasurementUnit: {
489 resolve("imu");
490 publisher = std::make_shared<CarlaIMUPublisher>(
491 BuildBaseTopicName(actor), LookupFrameId(actor));
492 break;
493 }
494 case ESensors::Radar: {
495 resolve("radar");
496 publisher = std::make_shared<CarlaRadarPublisher>(
497 BuildBaseTopicName(actor), LookupFrameId(actor), HasTopicOverride(actor));
498 break;
499 }
500 case ESensors::RayCastSemanticLidar: {
501 resolve("ray_cast_semantic");
502 publisher = std::make_shared<CarlaSemanticLidarPublisher>(
503 BuildBaseTopicName(actor), LookupFrameId(actor), HasTopicOverride(actor));
504 break;
505 }
506 case ESensors::RayCastLidar: {
507 // Both ray-cast and HSS lidars dispatch here; resolve either placeholder.
508 resolve("ray_cast");
509 resolve("hss_lidar");
510 publisher = std::make_shared<CarlaLidarPublisher>(
511 BuildBaseTopicName(actor), LookupFrameId(actor), HasTopicOverride(actor));
512 break;
513 }
514 case ESensors::AutowareGnssSensor: {
515 resolve("autoware_gnss");
516 // frame_id "map" (not the sensor's own frame): the published poses are
517 // world/map-frame quantities including the MGRS offset. Architecture
518 // decision — tier4's AutowarePublisherBase used the sensor ros_name as
519 // frame_id, which was accidental; triage confirmed "map" is the intent.
520 publisher = std::make_shared<AutowareGNSSPublisher>(
521 BuildBaseTopicName(actor), "map");
522 break;
523 }
524 case ESensors::LaneInvasionSensor:
525 case ESensors::ObstacleDetectionSensor:
526 case ESensors::RssSensor:
527 case ESensors::WorldObserver:
528 case ESensors::CameraGBufferUint8:
529 case ESensors::CameraGBufferFloat:
530 // Sensors without a publisher on ue5-dev today; the dispatch in
531 // ProcessDataFrom* logs and exits cleanly.
532 return nullptr;
533 default:
534 log_error("ROS2::GetOrCreateSensor: unknown sensor type", type);
535 return nullptr;
536 }
537
538 if (publisher) {
539 _publishers.insert({actor, publisher});
540 }
541 return publisher;
542}

544std::shared_ptr<CarlaTransformPublisher> ROS2::GetOrCreateTransformPublisherLocked(void *actor) {
545 // Global TF gate (tier4's _publish_tf): checked before the cache so
546 // toggling at runtime silences every sensor immediately.
547 if (!_publish_tf) {
548 return nullptr;
549 }
550 auto it = _transforms.find(actor);
551 if (it != _transforms.end()) {
552 return it->second;
553 }
554 auto registration_it = _registrations.find(actor);
555 if (registration_it == _registrations.end() || !registration_it->second.publish_tf) {
556 return nullptr;
557 }
558 auto transform = std::make_shared<CarlaTransformPublisher>();
559 _transforms.insert({actor, transform});
560 return transform;
561}

573void ROS2::PublishTransform(
574 void *actor, const Stamp &stamp, const carla::geom::Transform &sensor_transform) {
575 std::shared_ptr<CarlaTransformPublisher> transform_publisher;
576 std::string parent_frame;
577 std::string frame_id;
578 {
579 std::lock_guard<std::mutex> lock(_mutex);
580 transform_publisher = GetOrCreateTransformPublisherLocked(actor);
581 if (!transform_publisher) {
582 return;
583 }
584 parent_frame = ParentFrameOrMap(BuildParentChain(actor));
585 frame_id = LookupFrameId(actor);
586 }
587 transform_publisher->Write(
588 stamp.seconds, stamp.nanoseconds, parent_frame, frame_id,
589 sensor_transform.location.x, sensor_transform.location.y, sensor_transform.location.z,
590 sensor_transform.rotation.pitch, sensor_transform.rotation.yaw, sensor_transform.rotation.roll);
591 transform_publisher->Publish();
592}

594void ROS2::ProcessDataFromCamera(
595 uint64_t sensor_type,
596 carla::streaming::detail::stream_id_type stream_id,
597 const carla::geom::Transform sensor_transform,
598 int W, int H, float Fov,
599 const carla::SharedBufferView buffer,
600 void *actor) {
601 const Stamp stamp = GetStamp();
602 // Image dimensions + FOV are now read straight from ImageSerializer's
603 // per-frame header inside the camera publisher's WriteCameraInfo call;
604 // the W/H/Fov arguments survive for ABI compatibility with the
605 // Unreal-side dispatcher.
606 (void)W;
607 (void)H;
608 (void)Fov;
609
610 std::shared_ptr<CarlaCameraPublisher> publisher;
611 switch (sensor_type) {
612 case ESensors::SceneCaptureCamera:
613 publisher = GetOrCreateCameraSensor<CarlaRGBCameraPublisher>(stream_id, actor, "rgb");
614 break;
615 case ESensors::DepthCamera:
616 publisher = GetOrCreateCameraSensor<CarlaDepthCameraPublisher>(stream_id, actor, "depth");
617 break;
618 case ESensors::NormalsCamera:
619 publisher = GetOrCreateCameraSensor<CarlaNormalsCameraPublisher>(stream_id, actor, "normals");
620 break;
621 case ESensors::SemanticSegmentationCamera:
622 publisher = GetOrCreateCameraSensor<CarlaSSCameraPublisher>(stream_id, actor, "semantic_segmentation");
623 break;
624 case ESensors::InstanceSegmentationCamera:
625 publisher = GetOrCreateCameraSensor<CarlaISCameraPublisher>(stream_id, actor, "instance_segmentation");
626 break;
627 case ESensors::OpticalFlowCamera:
628 publisher = GetOrCreateCameraSensor<CarlaOpticalFlowCameraPublisher>(
629 stream_id, actor, "optical_flow");
630 break;
631 case ESensors::CollisionSensor:
632 case ESensors::RssSensor:

753void ROS2::ProcessDataFromLidar(
754 uint64_t /*sensor_type*/,
755 carla::streaming::detail::stream_id_type stream_id,
756 const carla::geom::Transform sensor_transform,
757 uint32_t channel_count,
758 float upper_fov_limit,
759 float lower_fov_limit,
760 carla::sensor::data::LidarData &data,
761 void *actor) {
762 const Stamp stamp = GetStamp();
763 if (auto base = GetOrCreateSensor(ESensors::RayCastLidar, stream_id, actor)) {
764 auto publisher = std::dynamic_pointer_cast<CarlaLidarPublisher>(base);
765 // The lidar returns a flat list of floats rather than structured detection
766 // points. Each detection is 4 floats: x, y, z, intensity. Divide the total
767 // float count by 4 to recover the number of detections.
768 const auto width = static_cast<std::uint32_t>(data._points.size() / 4u);
769 if (channel_count >= 2u) {
770 // tier4 routing: every raycast lidar whose description is known goes
771 // through the extended Autoware XYZIRCAEDT layout. Per-channel
772 // elevations span [upper_fov, lower_fov] (degrees in, radians out),
773 // uniformly spaced, top channel first — same as tier4's ROS2.cpp.
774 const float upper_rad = upper_fov_limit * static_cast<float>(M_PI) / 180.0f;
775 const float lower_rad = lower_fov_limit * static_cast<float>(M_PI) / 180.0f;
776 std::vector<float> vertical_angles;
777 vertical_angles.reserve(channel_count);
778 const float vertical_step =
779 (upper_rad - lower_rad) / static_cast<float>(channel_count - 1u);
780 for (uint32_t i = 0u; i < channel_count; ++i) {
781 vertical_angles.push_back(upper_rad - static_cast<float>(i) * vertical_step);
782 }
783 publisher->WriteExtendedPointCloud(
784 stamp.seconds, stamp.nanoseconds, 1u, width,
785 reinterpret_cast<const std::uint8_t *>(data._points.data()),
786 data._header.data() + carla::sensor::data::LidarData::Index::SIZE,
787 data._header.size() - carla::sensor::data::LidarData::Index::SIZE,
788 vertical_angles);
789 } else {
790 publisher->WritePointCloud(
791 stamp.seconds, stamp.nanoseconds, 1u, width,
792 reinterpret_cast<const std::uint8_t *>(data._points.data()));
793 }
794 publisher->Publish();
795 }
796 PublishTransform(actor, stamp, sensor_transform);
797}

50void AHSSLidar::Set(const FLidarDescription &LidarDescription)
51{
52 Description = LidarDescription;
53 LidarData = FLidarData(Description.Channels);
54 CreateLasers();
55 PointsPerChannel.resize(Description.Channels);
56
57 // Apply the user-supplied noise_seed; the constructor's SetSeed ran before
58 // Description was populated from the blueprint attributes, so the engine
59 // would otherwise be stuck on the FLidarDescription default seed.
60 SetSeed(Description.RandomSeed);
61
62 DropOffBeta = 1.0f - Description.DropOffAtZeroIntensity;
63 // Guard against dropoff_intensity_limit == 0: alpha = x/0 would propagate
64 // inf/NaN through PostprocessDetection and silently disable the
65 // intensity-based dropoff branch. With alpha = 0 the postprocess simply
66 // falls back to the constant beta probability for low-intensity points.
67 DropOffAlpha = (Description.DropOffIntensityLimit > std::numeric_limits<float>::epsilon())
68 ? Description.DropOffAtZeroIntensity / Description.DropOffIntensityLimit
69 : 0.0f;
70 DropOffGenActive = Description.DropOffGenRate > std::numeric_limits<float>::epsilon();
71}

104AHSSLidar::FDetection AHSSLidar::ComputeDetection(
105 const FHitResult& HitInfo, const FTransform& InverseSensorTransform) const
106{
107 FDetection Detection;
108 const FVector HitPoint = HitInfo.ImpactPoint;
109 Detection.point = InverseSensorTransform.TransformPosition(HitPoint);
110
111 const float Distance = Detection.point.Length();
112
113 const float AttenAtm = Description.AtmospAttenRate;
114 const float AbsAtm = exp(-AttenAtm * Distance);
115
116 const float IntRec = AbsAtm;
117
118 Detection.intensity = IntRec;
119
120 return Detection;
121}

123void AHSSLidar::PreprocessRays(uint32_t Channels, uint32_t MaxPointsPerChannel)
124{
125 Super::PreprocessRays(Channels, MaxPointsPerChannel);
126
127 if (!DropOffGenActive)
128 {
129 return;
130 }
131
132 for (auto ch = 0u; ch < Channels; ch++) {
133 for (auto p = 0u; p < MaxPointsPerChannel; p++) {
134 RayPreprocessCondition[ch][p] =
135 !(DropOffGenActive && RandomEngine->GetUniformFloat() < Description.DropOffGenRate);
136 }
137 }
138}

140bool AHSSLidar::PostprocessDetection(FDetection& Detection) const
141{
142 if (Description.NoiseStdDev > std::numeric_limits<float>::epsilon()) {
143 const auto ForwardVector = Detection.point.MakeUnitVector();
144 const auto Noise = ForwardVector * RandomEngine->GetNormalDistribution(0.0f, Description.NoiseStdDev);
145 Detection.point += Noise;
146 }
147
148 const float Intensity = Detection.intensity;
149 if(Intensity > Description.DropOffIntensityLimit)
150 return true;
151 else
152 return RandomEngine->GetUniformFloat() < DropOffAlpha * Intensity + DropOffBeta;
153}

181// SimulateLidar(float DeltaTime)
182//
183// Fixed-sweep scan centred on the sensor's forward axis. Casts
184// HorizontalFov / HorizontalResolution rays per vertical channel each tick;
185// no horizontal-angle accumulation across ticks (the sensor does not rotate).
186static float SnapToStep(float value, float step)
187{
188 return std::round(value / step) * step;
189}

191void AHSSLidar::SimulateLidar(const float DeltaTime, bool bLockPhysics)
192{
193 TRACE_CPUPROFILER_EVENT_SCOPE(AHSSLidar::SimulateLidar);
194 const uint32 ChannelCount = Description.Channels;
195
196 // horizontal_resolution and horizontal_fov are user-supplied attributes;
197 // clamp them to safe positive values before dividing. A zero or negative
198 // resolution would divide to inf/NaN and then cast to a huge uint32,
199 // triggering an OOM-sized allocation in ResetDetections/PreprocessRays.
200 constexpr float MinHorizontalResolution = 0.01f;
201 const float HorizontalResolution = FMath::Max(
202 SnapToStep(Description.HorizontalResolution, MinHorizontalResolution),
203 MinHorizontalResolution);
204 const float HorizontalFov = FMath::Max(Description.HorizontalFov, 0.0f);
205 const uint32 PointsToScanWithOneLaser = static_cast<uint32>(
206 FMath::RoundHalfFromZero(HorizontalFov / HorizontalResolution));
207
208 if (PointsToScanWithOneLaser == 0)
209 {
210 UE_LOG(
211 LogCarla,
212 Warning,
213 TEXT("%s: no points requested this frame, increase horizontal_fov or decrease horizontal_resolution."),
214 *GetName());
215 return;
216 }
217
218 check(ChannelCount == LaserAngles.Num());
219
220 ResetDetections(ChannelCount, PointsToScanWithOneLaser);
221 PreprocessRays(ChannelCount, PointsToScanWithOneLaser);
222
223 const FTransform ActorTransform = GetTransform();
224 const FTransform InverseSensorTransform = ActorTransform.Inverse();
225 const FVector LidarBodyLocation = ActorTransform.GetLocation();
226 const FRotator LidarBodyRotation = ActorTransform.Rotator();
227
228 auto RunChannelRaycasts = [&]()
229 {
230 TRACE_CPUPROFILER_EVENT_SCOPE(ParallelFor);
231 ParallelFor(ChannelCount, [&](int32 idxChannel) {
232 TRACE_CPUPROFILER_EVENT_SCOPE(ParallelForTask);
233
234 FCollisionQueryParams TraceParams = FCollisionQueryParams(FName(TEXT("Laser_Trace")), true, this);
235 TraceParams.bTraceComplex = true;
236 TraceParams.bReturnPhysicalMaterial = false;
237
238 const float VertAngle = LaserAngles[idxChannel];
239 const std::vector<bool>& ChannelPreprocessCondition = RayPreprocessCondition[idxChannel];
240
241 for (auto idxPtsOneLaser = 0u; idxPtsOneLaser < PointsToScanWithOneLaser; idxPtsOneLaser++) {
242 if (!ChannelPreprocessCondition[idxPtsOneLaser])
243 {
244 continue;
245 }
246
247 FHitResult HitResult;
248 const float HorizAngle =
249 -HorizontalFov / 2.0f + static_cast<float>(idxPtsOneLaser) * HorizontalResolution;
250
251 if (ShootLaser(VertAngle, HorizAngle, HitResult, TraceParams, LidarBodyLocation, LidarBodyRotation)) {
252 WriteDetectionAsync(idxChannel, HitResult, InverseSensorTransform, LidarBodyLocation);
253 }
254 };
255 });
256 };
257
258 if (bLockPhysics)
259 {
260 auto LockedPhysObject = FPhysicsObjectExternalInterface::LockRead(GetWorld()->GetPhysicsScene());
261 RunChannelRaycasts();
262 LockedPhysObject.Release();
263 }
264 else
265 {
266 RunChannelRaycasts();
267 }
268
269 ComputeAndSaveDetections(ActorTransform);
270}

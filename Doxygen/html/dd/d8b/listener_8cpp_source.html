34 void Listener::Stop() {
35 // Non-throwing: Stop() can run twice (explicitly, then again from
36 // ~Listener() if another reference outlives it); the throwing overloads
37 // raise on the second call, since the acceptor is already closed by then.
38 boost::system::error_code ec;
39 _acceptor.cancel(ec);
40 _acceptor.close(ec);
41 // io_context is owned by whoever constructed it (Router's ThreadPool);
42 // stopping/restarting it is not this class's responsibility.
43 }

45 void Listener::OpenSession(
46 time_duration timeout,
47 callback_function_type on_opened,
48 callback_function_type on_closed,
49 callback_function_type_response on_response) {
50
51 using boost::system::error_code;
52
53 auto session = std::make_shared<Primary>(_io_context, timeout, *this);
54 auto self = shared_from_this();
55
56 auto handle_query = [on_opened, on_closed, on_response, session, self](const error_code &ec)
57 {
58 if (!ec) {
59 session->Open(std::move(on_opened), std::move(on_closed), std::move(on_response));
60 } else {
61 log_error("Secondary server:", ec.message());
62 }
63 };
64
65 _acceptor.async_accept(
66 session->_socket,
67 [this, handle_query, timeout, on_opened, on_closed, on_response](error_code ec)
68 {
69 boost::asio::post(
70 _io_context,
71 [ec, handle_query]()
72 {
73 handle_query(ec);
74 });
75
76 OpenSession(timeout, on_opened, on_closed, on_response);
77 });
78 }

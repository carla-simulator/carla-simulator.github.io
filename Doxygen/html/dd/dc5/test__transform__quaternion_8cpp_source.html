98TEST(TransformQuaternion, regression_guard_against_halved_deg_to_rad) {
99 // The pre-d9cfc79fb spelling used M_PI_2 / 180 in place of M_PI / 180,
100 // halving every rotation. A 180-degree pitch should land at quaternion
101 // w=0 (pi/2 half-angle, cos(pi/2)=0); the buggy spelling would land at
102 // w=cos(pi/4)=sqrt(2)/2. Guard against a regression to that spelling.
103 const auto tf = carla::ros2::TransformFromCarlaRotation(
104 0.0f, 0.0f, 0.0f,
105 180.0f, 0.0f, 0.0f);
106 EXPECT_NEAR(tf.rotation[0], 0.0f, 1e-3f);
107 EXPECT_GT(std::abs(tf.rotation[2]), 0.9f);
108}

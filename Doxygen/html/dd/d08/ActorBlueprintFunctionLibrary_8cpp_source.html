58 /// If @a Predicate is false, print an error message. If possible the message
59 /// is printed to the editor window.
60 template <typename T, typename... ARGS>
61 bool OnScreenAssert(bool Predicate, const T &Format, ARGS &&...Args) const
62 {
63 if (!Predicate)
64 {
65 FString Message;
66 for (auto &String : Stack)
67 {
68 Message += String;
69 }
70 Message += TEXT(" ");
71 // FString::Printf validates its format string at compile time in UE 5.8,
72 // which rejects a format forwarded through a function parameter.
73 // FCString::Snprintf accepts the forwarded literal reference.
74 TCHAR Formatted[4096];
75 FCString::Snprintf(Formatted, UE_ARRAY_COUNT(Formatted), Format, std::forward<ARGS>(Args)...);
76 Message += Formatted;
77
78 UE_LOG(LogCarla, Error, TEXT("%s"), *Message);
79#if WITH_EDITOR
80 if (GEngine)
81 {
82 GEngine->AddOnScreenDebugMessage(42, 15.0f, FColor::Red, Message);
83 }
84#endif // WITH_EDITOR
85 }
86 return Predicate;
87 }

100 /// Applies @a Validator to each item in @a Array. Pushes a new context to the
101 /// stack for each item.
102 template <typename T, typename F>
103 bool ForEach(const FString &Type, const TArray<T> &Array, F Validator)
104 {
105 bool Result = true;
106 auto Counter = 0u;
107 for (const auto &Item : Array)
108 {
109 auto Scope = Stack.PushScope(GetDisplayId(Type, Counter, Item));
110 Result &= Validator(Item);
111 ++Counter;
112 }
113 return Result;
114 }

116 /// Applies @a IsValid to each item in @a Array. Pushes a new context to the
117 /// stack for each item.
118 template <typename T>
119 bool AreValid(const FString &Type, const TArray<T> &Array)
120 {
121 return ForEach(Type, Array, [this](const auto &Item)
122 { return IsValid(Item); });
123 }

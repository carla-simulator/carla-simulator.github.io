20/// Secondary-side map from a primary sensor actor to its local stream, exposed
21/// to clients only through an alias of the primary's stream id. Entries are
22/// per primary episode. Resolve() and BeginEpisodeChange() run on the multi-GPU
23/// command thread, the rest on the game thread.
24class SensorStreamRegistry {
25public:
26 using actor_id_type = uint32_t;
27 using epoch_type = uint64_t;
28 using stream_id_type = carla::streaming::detail::stream_id_type;
29 using token_type = carla::streaming::detail::token_type;
30
31 /// Token that serves primary actor @a actor_id under @a primary_stream_id,
32 /// reserving a local stream if the sensor has not spawned here yet.
33 [[nodiscard]]
34 std::optional<token_type> Resolve(
35 carla::streaming::Server &server,
36 actor_id_type actor_id,
37 stream_id_type primary_stream_id);
38
39 /// Returns the stream reserved for @a actor_id if @a can_adopt (the sensor
40 /// has never ticked) and the sensor must adopt it; otherwise the alias is
41 /// retargeted to @a sensor_stream_id. A stream bound to another actor is refused.
42 [[nodiscard]]
43 std::optional<carla::streaming::Stream> Bind(
44 carla::streaming::Server &server,
45 actor_id_type actor_id,
46 stream_id_type sensor_stream_id,
47 bool can_adopt);
48
49 /// Forgets @a actor_id, closing its reserved stream if no sensor adopted it.
50 void Unbind(carla::streaming::Server &server, actor_id_type actor_id);
51
52 /// The primary switched episode: forgets every actor and returns the epoch
53 /// the secondary must pass to OpenEpisode() once it runs the new episode.
54 [[nodiscard]]
55 epoch_type BeginEpisodeChange(carla::streaming::Server &server);
56
57 /// Accepts binding if @a epoch is the latest one BeginEpisodeChange() issued.
58 void OpenEpisode(epoch_type epoch);
59
60 void CloseEpisode();
61
62 /// Whether Bind()/Unbind() are accepted, i.e. the episode of the latest
63 /// BeginEpisodeChange() is open.
64 [[nodiscard]]
65 bool IsEpisodeOpen();
66
67private:
68 struct Entry {
69 stream_id_type stream_id;
70 std::optional<carla::streaming::Stream> reserved;
71 std::optional<stream_id_type> primary_stream_id;
72 };
73
74 static void Forget(carla::streaming::Server &server, const Entry &entry);
75
76 std::mutex _mutex;
77 epoch_type _epoch = 0u;
78 bool _bindable = false;
79 std::unordered_map<actor_id_type, Entry> _entries;
80};

102void CarlaRecorderPhysicsControls::Write(std::ostream& OutFile)
103{
104 if (PhysicsControls.size() == 0)
105 {
106 return;
107 }
108 // write the packet id
109 WriteValue<char>(OutFile, static_cast<char>(CarlaRecorderPacketId::PhysicsControl));
110
111 std::streampos PosStart = OutFile.tellp();
112 // write dummy packet size
113 uint32_t Total = 0;
114 WriteValue<uint32_t>(OutFile, Total);
115
116 // write total records
117 Total = PhysicsControls.size();
118 WriteValue<uint16_t>(OutFile, Total);
119
120 // write records
121 for (auto& PhysicsControl : PhysicsControls)
122 {
123 PhysicsControl.Write(OutFile);
124 }
125
126 // write the real packet size
127 std::streampos PosEnd = OutFile.tellp();
128 Total = PosEnd - PosStart - sizeof(uint32_t);
129 OutFile.seekp(PosStart, std::ios::beg);
130 WriteValue<uint32_t>(OutFile, Total);
131 OutFile.seekp(PosEnd, std::ios::beg);
132}

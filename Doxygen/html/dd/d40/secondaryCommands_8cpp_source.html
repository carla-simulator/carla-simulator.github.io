23void SecondaryCommands::process_command(Buffer buffer) {
24 // Each message is framed by the transport's size prefix, so dropping a
25 // malformed one never desynchronizes the stream.
26 CommandHeader header;
27 if (buffer.size() < sizeof(header)) {
28 log_error("multigpu: dropping a", buffer.size(), "byte command shorter than its header");
29 return;
30 }
31 std::memcpy(&header, buffer.data(), sizeof(header));
32 const std::size_t available = buffer.size() - sizeof(header);
33 if (header.size > available) {
34 log_error("multigpu: dropping command", static_cast<uint32_t>(header.id), "declaring",
35 header.size, "payload bytes in a message holding", available);
36 return;
37 }
38 if (header.size < available) {
39 log_warning("multigpu: command", static_cast<uint32_t>(header.id), "has",
40 available - header.size, "trailing bytes; ignored");
41 }
42
43 // send only data to the callback
44 Buffer data(buffer.data() + sizeof(header), header.size);
45 _callback(header.id, std::move(data));
46}

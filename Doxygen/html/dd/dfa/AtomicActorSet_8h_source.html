23 class AtomicActorSet {
24
25 private:
26
27 std::mutex modification_mutex;
28 std::map<ActorId, ActorPtr> actor_set;
29 int state_counter;
30 /// Removed actors, kept alive until TakeReleased() so they die outside TM locks.
31 std::vector<ActorPtr> released;

33 public:
34
35 AtomicActorSet(): state_counter(0) {}
36
37 std::vector<ActorPtr> GetList() {
38
39 std::scoped_lock<std::mutex> lock(modification_mutex);
40 std::vector<ActorPtr> actor_list;
41 for (auto it = actor_set.begin(); it != actor_set.end(); ++it) {
42 actor_list.push_back(it->second);
43 }
44 return actor_list;
45 }

52 actor_list.push_back(it->first);
53 }
54 return actor_list;
55 }

57 void Insert(std::vector<ActorPtr> actor_list) {
58
59 std::scoped_lock<std::mutex> lock(modification_mutex);
60 for (auto &actor: actor_list) {
61 actor_set.insert({actor->GetId(), actor});
62 }
63 ++state_counter;
64 }
65
66 void Remove(std::vector<ActorId> actor_id_list) {
67
68 std::scoped_lock<std::mutex> lock(modification_mutex);
69 for (auto& actor_id: actor_id_list) {
70 auto it = actor_set.find(actor_id);
71 if (it != actor_set.end()){
72 released.push_back(std::move(it->second));
73 actor_set.erase(it);
74 }
75 }
76 ++state_counter;
77 }
78
79 void Destroy(ActorId actor_id) {
80
81 std::scoped_lock<std::mutex> lock(modification_mutex);
82 if (actor_set.find(actor_id) != actor_set.end()) {
83 ActorPtr actor = actor_set.at(actor_id);
84 actor->Destroy();
85 actor_set.erase(actor_id);
86 released.push_back(std::move(actor));
87 ++state_counter;
88 }
89 }

91 int GetState() {
92
93 std::scoped_lock<std::mutex> lock(modification_mutex);
94 return state_counter;
95 }
96
97 bool Contains(ActorId id) {
98
99 std::scoped_lock<std::mutex> lock(modification_mutex);
100 return actor_set.find(id) != actor_set.end();
101 }
102
103 size_t Size() {
104
105 std::scoped_lock<std::mutex> lock(modification_mutex);
106 return actor_set.size();
107 }
108
109 void Clear() {
110
111 std::scoped_lock<std::mutex> lock(modification_mutex);
112 for (auto &entry : actor_set) {
113 released.push_back(std::move(entry.second));
114 }
115 actor_set.clear();
116 }
117
118 /// Actors removed since the last call; drop them with no lock held.
119 std::vector<ActorPtr> TakeReleased() {
120
121 std::scoped_lock<std::mutex> lock(modification_mutex);
122 std::vector<ActorPtr> result;
123 result.swap(released);
124 return result;
125 }

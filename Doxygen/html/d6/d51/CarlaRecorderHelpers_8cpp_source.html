56// write binary data from FString (length + text)
57void WriteFString(std::ostream &OutFile, const FString &InObj)
58{
59 // encode the string to UTF8 to know the final length
60 FTCHARToUTF8 EncodedString(*InObj);
61 int16_t Length = EncodedString.Length();
62 // write
63 WriteValue<uint16_t>(OutFile, Length);
64 OutFile.write(reinterpret_cast<char *>(TCHAR_TO_UTF8(*InObj)), Length);

89// read binary data to FString (length + text)
90void ReadFString(std::istream &InFile, FString &OutObj)
91{
92 uint16_t Length = 0;
93 ReadValue<uint16_t>(InFile, Length);
94 // make room in vector buffer
95 if (CarlaRecorderHelperBuffer.capacity() < Length + 1)
96 {
97 CarlaRecorderHelperBuffer.reserve(Length + 1);
98 }
99 CarlaRecorderHelperBuffer.clear();
100 // initialize the vector space with 0
101 CarlaRecorderHelperBuffer.resize(Length + 1);
102 // read
103 InFile.read(reinterpret_cast<char *>(CarlaRecorderHelperBuffer.data()), Length);
104 // convert from UTF8 to FString
105 OutObj = FString(UTF8_TO_TCHAR(CarlaRecorderHelperBuffer.data()));
106}

454 /// State of @a actor according to the latest world snapshot.
455 ///
456 /// An actor spawned by this client is not part of any snapshot produced
457 /// before the spawn was processed (asynchronous mode: until the next frame
458 /// arrives; synchronous mode: until the next tick). Reporting such an actor
459 /// as Invalid made is_alive/is_active False right after spawn_actor(), so
460 /// while the latest snapshot is not newer than the frame recorded at spawn
461 /// time the actor is reported Active. Any snapshot newer than that frame
462 /// is authoritative: an actor missing from it is gone (Invalid). A handle
463 /// that did not come from a spawn call gets no such grace: it is only ever
464 /// as good as the snapshots, including before the first one arrives.
465 rpc::ActorState GetActorState(const Actor &actor) const {
466 DEBUG_ASSERT(_episode != nullptr);
467 const auto state = _episode->GetState();
468 const auto snapshot = state->GetActorSnapshotIfPresent(actor.GetId());
469 if (snapshot.has_value()) {
470 return snapshot->actor_state;
471 }
472 if (actor.WasSpawnedByThisClient() && state->GetFrame() <= actor.GetSpawnFrame()) {
473 return rpc::ActorState::Active;
474 }
475 return rpc::ActorState::Invalid;
476 }

457 /// State of @a actor according to the latest world snapshot.
458 ///
459 /// An actor spawned by this client is not part of any snapshot produced
460 /// before the spawn was processed (asynchronous mode: until the next frame
461 /// arrives; synchronous mode: until the next tick). Reporting such an actor
462 /// as Invalid made is_alive/is_active False right after spawn_actor(), so
463 /// while the latest snapshot is not newer than the frame recorded at spawn
464 /// time the actor is reported Active. Any snapshot newer than that frame
465 /// is authoritative: an actor missing from it is gone (Invalid). A handle
466 /// that did not come from a spawn call gets no such grace: it is only ever
467 /// as good as the snapshots, including before the first one arrives.
468 rpc::ActorState GetActorState(const Actor &actor) const {
469 DEBUG_ASSERT(_episode != nullptr);
470 const auto state = _episode->GetState();
471 const auto snapshot = state->GetActorSnapshotIfPresent(actor.GetId());
472 if (snapshot.has_value()) {
473 return snapshot->actor_state;
474 }
475 if (actor.WasSpawnedByThisClient() && state->GetFrame() <= actor.GetSpawnFrame()) {
476 return rpc::ActorState::Active;
477 }
478 return rpc::ActorState::Invalid;
479 }

30 /// LocalPlanner reproduces `agents.navigation.local_planner.LocalPlanner`.
31 /// It maintains a queue of (waypoint, RoadOption) pairs, refills it as
32 /// needed, and drives the vehicle along it via a VehiclePIDController.
33 class LocalPlanner {
34 public:
35 using PlanItem = std::pair<SharedPtr<client::Waypoint>, RoadOption>;
36
37 /// Aggregated configuration. Mirrors the `opt_dict` in local_planner.py.
38 struct Options {
39 float dt = 1.0f / 20.0f;
40 float target_speed = 20.0f; // km/h
41 float sampling_radius = 2.0f;
42 // LateralArgs/LongitudinalArgs supply their own defaults (1.95/0.05/0.20
43 // and 1.50/0.05/0.20 with dt=0.05) — leaving these value-initialized.
44 // Earlier brace-init member defaults here triggered a GCC aggregate-init
45 // quirk when Options was used as `= {}` default arg downstream.
46 VehiclePIDController::LateralArgs lateral{};
47 VehiclePIDController::LongitudinalArgs longitudinal{};
48 float max_throttle = 0.75f;
49 float max_brake = 0.30f;
50 float max_steering = 0.80f;
51 float offset = 0.0f;
52 float base_min_distance = 3.0f;
53 float distance_ratio = 0.5f;
54 bool follow_speed_limits = false;
55 };
56
57 /// Construct a LocalPlanner. If @a map is null we pull the map from the
58 /// vehicle's world (matches the Python `map_inst` fallback).
59 /// Returns a value-initialized Options. Defined out-of-line because GCC's
60 /// default-arg parser refuses `= {}` / `= Options()` when the struct
61 /// inherits from another aggregate with default member initializers.
62 static Options DefaultOptions();

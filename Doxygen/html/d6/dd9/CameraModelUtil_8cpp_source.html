223template <typename FShaderType>
224static auto CreateDistortionParameters(
225 FRDGBuilder& GraphBuilder,
226 FRDGTextureRef Destination,
227 FRDGTextureRef CubeTextures[6],
228 FRHISamplerState* Sampler,
229 const CameraModelUtil::FDistortCubemapToImageOptions& Options,
230 FVector4f CameraParams)
231{
232 auto Parameters = GraphBuilder.AllocParameters<typename FShaderType::FParameters>();
233 Parameters->OutImage = GraphBuilder.CreateUAV(Destination);
234 Parameters->CubeFront = GraphBuilder.CreateSRV(FRDGTextureSRVDesc::Create(CubeTextures[0]));
235 Parameters->CubeBack = GraphBuilder.CreateSRV(FRDGTextureSRVDesc::Create(CubeTextures[1]));
236 Parameters->CubeRight = GraphBuilder.CreateSRV(FRDGTextureSRVDesc::Create(CubeTextures[2]));
237 Parameters->CubeLeft = GraphBuilder.CreateSRV(FRDGTextureSRVDesc::Create(CubeTextures[3]));
238 Parameters->CubeTop = GraphBuilder.CreateSRV(FRDGTextureSRVDesc::Create(CubeTextures[4]));
239 Parameters->CubeBottom = GraphBuilder.CreateSRV(FRDGTextureSRVDesc::Create(CubeTextures[5]));
240 Parameters->FaceSampler = Sampler;
241 Parameters->CameraParams = CameraParams;
242 Parameters->YFOVAngle = Options.YFOVAngle;
243 Parameters->LongitudeOffset = Options.LongitudeOffset;
244 Parameters->FOVFadeSize = Options.FOVFadeSize;
245 Parameters->InvalidColor = Options.InvalidColor;
246 Parameters->Flags = 0;
247 if (Options.bRenderEquirectangular)
248 Parameters->Flags |= WAL_SHADER_FLAGS_EQUIRECTANGULAR;
249 if (Options.bFOVMaskEnable)
250 Parameters->Flags |= WAL_SHADER_FLAGS_FOV_MASK;
251 return Parameters;
252}

254template <typename FShaderType>
255static void DispatchDistortion(
256 FRDGBuilder& GraphBuilder,
257 typename FShaderType::FParameters* Parameters,
258 FIntPoint Size)
259{
260 GraphBuilder.AddPass(
261 RDG_EVENT_NAME("WideAngleLens-Dispatch"),
262 Parameters,
263 ERDGPassFlags::Compute,
264 [Parameters, Size](FRHICommandListImmediate& RHICmdList)
265 {
266 TShaderMapRef<FShaderType> ComputeShader(GetGlobalShaderMap(GMaxRHIFeatureLevel));
267 check(ComputeShader.IsValid());
268
269 FComputeShaderUtils::Dispatch(
270 RHICmdList,
271 ComputeShader,
272 *Parameters,
273 FComputeShaderUtils::GetGroupCount(
274 FIntVector(Size.X, Size.Y, 1),
275 FIntVector(SubgroupSize, 1, 1)));
276 });
277}

279template <typename FShaderType>
280static void ApplyDistortion(
281 FRDGBuilder& GraphBuilder,
282 FRDGTextureRef Destination,
283 FRDGTextureRef CubeTextures[6],
284 FRHISamplerState* Sampler,
285 FIntPoint Size,
286 const CameraModelUtil::FDistortCubemapToImageOptions& Options,
287 FVector4f CameraParams)
288{
289 auto Parameters = CreateDistortionParameters<FShaderType>(
290 GraphBuilder, Destination, CubeTextures, Sampler, Options, CameraParams);
291 DispatchDistortion<FShaderType>(GraphBuilder, Parameters, Size);
292}

294static void ApplyDistortionKannalaBrandt(
295 FRDGBuilder& GraphBuilder,
296 FRDGTextureRef Destination,
297 FRDGTextureRef CubeTextures[6],
298 FRHISamplerState* Sampler,
299 FIntPoint Size,
300 const CameraModelUtil::FDistortCubemapToImageOptions& Options,
301 FVector4f CameraParams)
302{
303 using FShaderType = FWideAngleLensShader_Custom;
304
305 auto Parameters = CreateDistortionParameters<FShaderType>(
306 GraphBuilder, Destination, CubeTextures, Sampler, Options, CameraParams);
307
308 const auto& Coefficients = Options.KannalaBrandtCoefficients;
309 auto CoefficientBuffer = CreateStructuredBuffer(
310 GraphBuilder,
311 TEXT("CoefficientBuffer"),
312 sizeof(float),
313 Coefficients.Num(),
314 Coefficients.GetData(),
315 Coefficients.Num() * sizeof(float),
316 ERDGInitialDataFlags::None);
317
318 Parameters->Coefficients = GraphBuilder.CreateSRV(FRDGBufferSRVDesc(CoefficientBuffer));
319 Parameters->CoefficientCount = static_cast<uint32>(Coefficients.Num());
320 DispatchDistortion<FShaderType>(GraphBuilder, Parameters, Size);
321}

323static void ApplyDistortionLUT1D(
324 FRDGBuilder& GraphBuilder,
325 FRDGTextureRef Destination,
326 FRDGTextureRef CubeTextures[6],
327 FRHISamplerState* Sampler,
328 FIntPoint Size,
329 const CameraModelUtil::FDistortCubemapToImageOptions& Options,
330 FVector4f CameraParams)
331{
332 using FShaderType = FWideAngleLensShader_LUT1D;
333
334 auto Parameters = CreateDistortionParameters<FShaderType>(
335 GraphBuilder, Destination, CubeTextures, Sampler, Options, CameraParams);
336
337 // An empty table would map every pixel to Theta = 0; substitute a single
338 // zero sample so the shader's out-of-range test marks the frame invalid
339 // instead of sampling a zero-sized buffer.
340 static const float EmptyLUT[1] = { 0.0F };
341 const TArrayView<const float> LUT = Options.LUT.Num() > 0
342 ? Options.LUT
343 : TArrayView<const float>(EmptyLUT, 1);
344
345 auto LUTBuffer = CreateStructuredBuffer(
346 GraphBuilder,
347 TEXT("LensLUTBuffer"),
348 sizeof(float),
349 LUT.Num(),
350 LUT.GetData(),
351 LUT.Num() * sizeof(float),
352 ERDGInitialDataFlags::None);
353
354 Parameters->LensLUT = GraphBuilder.CreateSRV(FRDGBufferSRVDesc(LUTBuffer));
355 Parameters->LensLUTSize = static_cast<uint32>(LUT.Num());
356 Parameters->LensThetaMax = Options.ThetaMax;
357 DispatchDistortion<FShaderType>(GraphBuilder, Parameters, Size);
358}

451 float ComputeCameraPolynomial(
452 float Theta,
453 TArrayView<const float> Coefficients)
454 {
455 auto Result = 1.0F;
456 auto Theta2 = Theta * Theta;
457 auto ThetaN = 1.0F;
458
459 for (auto K : Coefficients)
460 {
461 ThetaN *= Theta2;
462 Result += K * ThetaN;
463 }
464
465 return Result * Theta;
466 }
467
468 float ComputeCameraPolynomialDerivative(
469 float Theta,
470 TArrayView<const float> Coefficients)
471 {
472 auto Result = 1.0F;
473 auto Theta2 = Theta * Theta;
474 auto ThetaN = Theta2;
475 auto A = 3.0F;
476
477 for (auto K : Coefficients)
478 {
479 Result += A * K * ThetaN;
480 A += 2.0F;
481 ThetaN *= Theta2;
482 }
483
484 return Result;
485 }

489 float ComputeAngle(
490 ECameraModel CameraModel,
491 float Distance,
492 TArrayView<const float> Coefficients)
493 {
494 switch (CameraModel)
495 {
496 case ECameraModel::Perspective:
497 return std::atan(Distance);
498 case ECameraModel::Stereographic:
499 return std::atan(Distance * 0.5F) * 2;
500 case ECameraModel::Equidistant:
501 return Distance;
502 case ECameraModel::Equisolid:
503 return std::asin(FMath::Clamp(Distance * 0.5F, -1.0F, 1.0F)) * 2;
504 case ECameraModel::Orthographic:
505 return std::asin(FMath::Clamp(Distance, -1.0F, 1.0F));
506 case ECameraModel::KannalaBrandt:
507 {
508 float Theta = Distance;
509 for (uint8 i = 0; i != KannalaBrandtSolverIterations; ++i)
510 {
511 float N = Distance - KannalaBrandt::ComputeCameraPolynomial(Theta, Coefficients);
512 float D = -KannalaBrandt::ComputeCameraPolynomialDerivative(Theta, Coefficients);
513 Theta -= N / D;
514 }
515 return Theta;
516 }
517 default:
518 check(false);
519 return 0.0F;
520 }
521 }
522
523 float ComputeDistance(
524 ECameraModel CameraModel,
525 float Angle,
526 int32 ImageHeight,
527 TArrayView<const float> Coefficients)
528 {
529 const auto H = static_cast<float>(ImageHeight);
530 const auto R = H * 0.5F;
531 float F = 0.0F;
532
533 Angle *= 0.5F;
534 switch (CameraModel)
535 {
536 case ECameraModel::Perspective:
537 F = R / std::tan(Angle);
538 break;
539 case ECameraModel::Stereographic:
540 F = R / (std::tan(Angle * 0.5F) * 2);
541 break;
542 case ECameraModel::Equidistant:
543 F = R / Angle;
544 break;
545 case ECameraModel::Equisolid:
546 F = R / (std::sin(Angle * 0.5F) * 2);
547 break;
548 case ECameraModel::Orthographic:
549 F = R / std::sin(Angle);
550 break;
551 case ECameraModel::KannalaBrandt:
552 F = R / KannalaBrandt::ComputeCameraPolynomial(Angle, Coefficients);
553 break;
554 default:
555 check(false);
556 break;
557 }
558 return F;
559 }

561 void DistortCubemapToImage(
562 FRDGBuilder& GraphBuilder,
563 FRDGTexture* Destination,
564 FRDGTexture** CubeTextures,
565 FRHISamplerState* Sampler,
566 const FDistortCubemapToImageOptions& Options)
567 {
568 auto& DestinationDesc = Destination->Desc;
569 auto SizeVector = DestinationDesc.GetSize();
570 auto Size = FIntPoint(SizeVector.X, SizeVector.Y);
571
572 auto DistortedTexture =
573 GraphBuilder.CreateTexture(
574 FRDGTextureDesc::Create2D(
575 Size,
576 DestinationDesc.Format,
577 DestinationDesc.ClearValue,
578 TexCreate_ShaderResource | TexCreate_UAV),
579 TEXT("DistortedTexture"));
580
581 // (fx, fy, cx, cy) in pixels. Legacy path: square focal length from
582 // the FOV, principal point at the image centre.
583 const FVector4f CameraParams = Options.bExplicitIntrinsics
584 ? Options.Intrinsics
585 : FVector4f(Options.YFocalLength, Options.YFocalLength, (Size / 2).X, (Size / 2).Y);
586
587 switch (Options.CameraModel)
588 {
589 case ECameraModel::Perspective:
590 case ECameraModel::BrownConrady: // no cubemap implementation; undistorted pinhole
591 ApplyDistortion<FWideAngleLensShader_Perspective>(
592 GraphBuilder, DistortedTexture, CubeTextures, Sampler, Size, Options, CameraParams);
593 break;
594 case ECameraModel::Stereographic:
595 ApplyDistortion<FWideAngleLensShader_Stereographic>(
596 GraphBuilder, DistortedTexture, CubeTextures, Sampler, Size, Options, CameraParams);
597 break;
598 case ECameraModel::Equidistant:
599 ApplyDistortion<FWideAngleLensShader_Equidistance>(
600 GraphBuilder, DistortedTexture, CubeTextures, Sampler, Size, Options, CameraParams);
601 break;
602 case ECameraModel::Equisolid:
603 ApplyDistortion<FWideAngleLensShader_Equisolid>(
604 GraphBuilder, DistortedTexture, CubeTextures, Sampler, Size, Options, CameraParams);
605 break;
606 case ECameraModel::Orthographic:
607 ApplyDistortion<FWideAngleLensShader_Orthogonal>(

809 // Evaluates Distort() and its Jacobian in one pass, so Undistort()'s
810 // Newton-Raphson loop only pays for the distortion math once per iteration.
811 static void EvalWithJacobian(
812 float Xn,
813 float Yn,
814 TArrayView<const float> C,
815 float& OutXd,
816 float& OutYd,
817 float OutJ[4]) // row-major [dXd/dXn, dXd/dYn, dYd/dXn, dYd/dYn]
818 {
819 const float K1 = C.Num() > 0 ? C[0] : 0.0F;
820 const float K2 = C.Num() > 1 ? C[1] : 0.0F;
821 const float K3 = C.Num() > 2 ? C[2] : 0.0F;
822 const float P1 = C.Num() > 3 ? C[3] : 0.0F;
823 const float P2 = C.Num() > 4 ? C[4] : 0.0F;
824
825 const float R2 = Xn * Xn + Yn * Yn;
826 const float R4 = R2 * R2;
827 const float Radial = 1.0F + K1 * R2 + K2 * R4 + K3 * R4 * R2;
828 const float DRadialDR2 = K1 + 2.0F * K2 * R2 + 3.0F * K3 * R4;
829
830 OutXd = Xn * Radial + 2.0F * P1 * Xn * Yn + P2 * (R2 + 2.0F * Xn * Xn);
831 OutYd = Yn * Radial + P1 * (R2 + 2.0F * Yn * Yn) + 2.0F * P2 * Xn * Yn;
832
833 const float DRadialDXn = 2.0F * Xn * DRadialDR2;
834 const float DRadialDYn = 2.0F * Yn * DRadialDR2;
835
836 OutJ[0] = Radial + Xn * DRadialDXn + 2.0F * P1 * Yn + 6.0F * P2 * Xn; // dXd/dXn
837 OutJ[1] = Xn * DRadialDYn + 2.0F * P1 * Xn + 2.0F * P2 * Yn; // dXd/dYn
838 OutJ[2] = Yn * DRadialDXn + 2.0F * P1 * Xn + 2.0F * P2 * Yn; // dYd/dXn
839 OutJ[3] = Radial + Yn * DRadialDYn + 6.0F * P1 * Yn + 2.0F * P2 * Xn; // dYd/dYn
840 }
841
842 FVector2f Distort(
843 float Xn,
844 float Yn,
845 TArrayView<const float> Coefficients)
846 {
847 float Xd, Yd, J[4];
848 EvalWithJacobian(Xn, Yn, Coefficients, Xd, Yd, J);
849 return FVector2f(Xd, Yd);
850 }
851
852 FVector2f Undistort(
853 float Xd,
854 float Yd,
855 TArrayView<const float> Coefficients,
856 int32 Iterations)
857 {
858 float Xn = Xd, Yn = Yd; // initial guess: undistorted ~ distorted for small distortion
859 for (int32 i = 0; i != Iterations; ++i)
860 {
861 float Fx, Fy, J[4];
862 EvalWithJacobian(Xn, Yn, Coefficients, Fx, Fy, J);
863 Fx -= Xd;
864 Fy -= Yd;
865
866 const float Det = J[0] * J[3] - J[1] * J[2];
867 if (FMath::Abs(Det) < 1e-12F)
868 break;
869
870 const float InvDet = 1.0F / Det;
871 const float DXn = InvDet * (J[3] * Fx - J[1] * Fy);
872 const float DYn = InvDet * (-J[2] * Fx + J[0] * Fy);
873
874 Xn -= DXn;
875 Yn -= DYn;
876 }
877 return FVector2f(Xn, Yn);
878 }

883 float Polynomial(float Distance, TArrayView<const float> Coefficients)
884 {
885 float Result = 0.0F, RN = 1.0F;
886 const int32 N = FMath::Min(Coefficients.Num(), 8);
887 for (int32 i = 0; i < N; ++i) { Result += Coefficients[i] * RN; RN *= Distance; }
888 return Result;
889 }
890 float SolveRadius(float Theta, TArrayView<const float> Coefficients, int32 Iterations)
891 {
892 const int32 N = FMath::Min(Coefficients.Num(), 8);
893 const float C0 = N > 0 ? Coefficients[0] : 0.0F, C1 = N > 1 ? Coefficients[1] : 0.0F;
894 float R = FMath::Abs(C1) > 1e-12F ? (Theta - C0) / C1 : Theta;
895 for (int32 It = 0; It < Iterations; ++It)
896 {
897 float D = 0.0F, RN = 1.0F;
898 for (int32 i = 1; i < N; ++i) { D += float(i) * Coefficients[i] * RN; RN *= R; }
899 if (FMath::Abs(D) < 1e-12F) break;
900 R -= (Polynomial(R, Coefficients) - Theta) / D;
901 }
902 return FMath::Max(R, 0.0F);
903 }

908 float SampleForward(
909 float Theta,
910 TArrayView<const float> Samples,
911 float ThetaMax)
912 {
913 const int32 Count = Samples.Num();
914 if (Count == 0)
915 return 0.0F;
916 if (Count == 1)
917 return Samples[0];
918
919 const float Clamped = FMath::Clamp(Theta, 0.0F, ThetaMax);
920 const float T = ThetaMax > 0.0F ? (Clamped / ThetaMax) * static_cast<float>(Count - 1) : 0.0F;
921 const int32 I0 = FMath::Clamp(static_cast<int32>(FMath::FloorToFloat(T)), 0, Count - 1);
922 const int32 I1 = FMath::Min(I0 + 1, Count - 1);
923 const float Frac = T - static_cast<float>(I0);
924 return FMath::Lerp(Samples[I0], Samples[I1], Frac);
925 }
926
927 float SampleInverse(
928 float Distance,
929 TArrayView<const float> Samples,
930 float ThetaMax)
931 {
932 const int32 Count = Samples.Num();
933 if (Count < 2)
934 return 0.0F;
935 if (Distance <= Samples[0])
936 return 0.0F;
937 if (Distance >= Samples[Count - 1])
938 return ThetaMax;
939
940 // Samples is assumed monotonically non-decreasing.
941 int32 Lo = 0;
942 int32 Hi = Count - 1;
943 while (Hi - Lo > 1)
944 {
945 const int32 Mid = (Lo + Hi) / 2;
946 if (Samples[Mid] <= Distance)
947 Lo = Mid;
948 else
949 Hi = Mid;
950 }
951
952 const float R0 = Samples[Lo];
953 const float R1 = Samples[Hi];
954 const float Frac = R1 > R0 ? (Distance - R0) / (R1 - R0) : 0.0F;
955 const float IndexF = static_cast<float>(Lo) + Frac;
956 return (IndexF / static_cast<float>(Count - 1)) * ThetaMax;
957 }

993 static float RadialInverse(
994 ECameraModel Model,
995 float Distance,
996 TArrayView<const float> Coeffs,
997 TArrayView<const float> LUT,
998 float ThetaMax)
999 {
1000 switch (Model)
1001 {
1002 case ECameraModel::Perspective:
1003 return std::atan(Distance);
1004 case ECameraModel::Stereographic:
1005 return std::atan(Distance * 0.5F) * 2.0F;
1006 case ECameraModel::Equidistant:
1007 return Distance;

1046 FVector2f ComputeLensImagePoint(
1047 const FLensModelDescriptor& Descriptor,
1048 FVector Direction,
1049 ELensColorChannel Channel)
1050 {
1051 Direction = Direction.GetSafeNormal();
1052 if (Direction.IsNearlyZero())
1053 Direction = FVector(0, 0, 1);
1054
1055 const float CAScale = SelectCAScale(Descriptor, Channel);
1056
1057 float Xn, Yn;
1058
1059 if (Descriptor.Model == ECameraModel::BrownConrady)
1060 {
1061 const float Z = FMath::Max(static_cast<float>(Direction.Z), KINDA_SMALL_NUMBER);
1062 const float PinholeX = static_cast<float>(Direction.X) / Z;
1063 const float PinholeY = static_cast<float>(Direction.Y) / Z;
1064 const FVector2f Distorted = BrownConrady::Distort(PinholeX, PinholeY, Descriptor.Coeffs);
1065 Xn = Distorted.X;
1066 Yn = Distorted.Y;
1067 }
1068 else
1069 {
1070 const float Theta = std::acos(FMath::Clamp(static_cast<float>(Direction.Z), -1.0F, 1.0F));
1071 const float Phi = std::atan2(static_cast<float>(Direction.Y), static_cast<float>(Direction.X));
1072 const float R = RadialForward(Descriptor.Model, Theta, Descriptor.Coeffs, Descriptor.LUT, Descriptor.ThetaMax);
1073 Xn = R * std::cos(Phi);
1074 Yn = R * std::sin(Phi);
1075 }
1076
1077 Xn *= CAScale;
1078 Yn *= CAScale;
1079
1080 return FVector2f(
1081 Descriptor.CenterX + Descriptor.FocalX * Xn,
1082 Descriptor.CenterY + Descriptor.FocalY * Yn);
1083 }

1085 FVector ComputeLensRayDirection(
1086 const FLensModelDescriptor& Descriptor,
1087 FVector2f ImagePoint,
1088 ELensColorChannel Channel)
1089 {
1090 const float CAScale = SelectCAScale(Descriptor, Channel);
1091 const float SafeCAScale = FMath::IsNearlyZero(CAScale) ? 1.0F : CAScale;
1092
1093 const float Xn = (ImagePoint.X - Descriptor.CenterX) / (Descriptor.FocalX * SafeCAScale);
1094 const float Yn = (ImagePoint.Y - Descriptor.CenterY) / (Descriptor.FocalY * SafeCAScale);
1095
1096 if (Descriptor.Model == ECameraModel::BrownConrady)
1097 {
1098 const FVector2f Pinhole = BrownConrady::Undistort(Xn, Yn, Descriptor.Coeffs);
1099 return FVector(Pinhole.X, Pinhole.Y, 1.0F).GetSafeNormal();
1100 }
1101
1102 const float R = std::sqrt(Xn * Xn + Yn * Yn);
1103 const float Phi = std::atan2(Yn, Xn);
1104 const float Theta = RadialInverse(Descriptor.Model, R, Descriptor.Coeffs, Descriptor.LUT, Descriptor.ThetaMax);
1105 const float SinTheta = std::sin(Theta);
1106 return FVector(
1107 SinTheta * std::cos(Phi),
1108 SinTheta * std::sin(Phi),
1109 std::cos(Theta));
1110 }

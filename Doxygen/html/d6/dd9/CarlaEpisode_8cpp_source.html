65// State of the background RecastBuilder run of the last generated OpenDRIVE
66// world (there is at most one). Readers of the navmesh wait on it.

68static bool GNavBuildPending = false;
69static bool GNavBuildSucceeded = false;

71bool UCarlaEpisode::WaitForPendingNavigationBuild(double TimeoutSeconds)
72{
73 const double Start = FPlatformTime::Seconds();
74 bool bLogged = false;
75 for (;;)
76 {
77 {
78 FScopeLock Lock(&GNavBuildLock);
79 if (!GNavBuildPending)
80 {
81 return GNavBuildSucceeded;
82 }
83 }
84 if (!bLogged)
85 {
86 UE_LOG(LogCarla, Log, TEXT("Waiting for RecastBuilder to finish the pedestrian navigation (up to %.0f s)"),
87 TimeoutSeconds);
88 bLogged = true;
89 }
90 if (FPlatformTime::Seconds() - Start > TimeoutSeconds)
91 {
92 UE_LOG(LogCarla, Error, TEXT("RecastBuilder did not finish within %.0f s; "
93 "the pedestrian navigation of this OpenDRIVE world is not available yet"), TimeoutSeconds);
94 return false;
95 }
96 FPlatformProcess::Sleep(0.05f);
97 }
98}

118static FString UCarlaEpisode_GetTrafficSignId(const ATrafficSignBase &Sign)
119{
120 using TSS = ETrafficSignState;
121 const int32 Kmh = Sign.GetSpeedLimitKmh();
122 if (Kmh > 0)
123 {
124 // km/h always (the OpenDRIVE subtype vocabulary); mph plates are a look, not a unit.
125 return FString::Printf(TEXT("traffic.speed_limit.%d"), Kmh);
126 }
127 switch (Sign.GetTrafficSignState())
128 {
129 case TSS::TrafficLightRed:
130 case TSS::TrafficLightYellow:
131 case TSS::TrafficLightGreen: return TEXT("traffic.traffic_light");
132 case TSS::StopSign: return TEXT("traffic.stop");
133 case TSS::YieldSign: return TEXT("traffic.yield");
134 default: return TEXT("traffic.unknown");
135 }

155bool UCarlaEpisode::LoadNewEpisode(const FString &MapString, bool ResetSettings)
156{
157 FString FinalPath = UCarlaStatics::FindMapPath(MapString);
158 const bool bIsFileFound = !FinalPath.IsEmpty();
159
160 if (bIsFileFound)
161 {
162 UE_LOG(LogCarla, Warning, TEXT("Loading a new episode: %s"), *FinalPath);
163 UGameplayStatics::OpenLevel(GetWorld(), *FinalPath, true);
164 if (ResetSettings)
165 ApplySettings(FEpisodeSettings{});
166
167 // send 'LOAD_MAP' command to all secondary servers (if any)
168 if (bIsPrimaryServer)
169 {
170 UCarlaGameInstance *GameInstance = UCarlaStatics::GetGameInstance(GetWorld());
171 if(GameInstance)
172 {
173 FCarlaEngine *CarlaEngine = GameInstance->GetCarlaEngine();
174 auto SecondaryServer = CarlaEngine->GetSecondaryServer();
175 // A world generated from OpenDRIVE: the secondaries generate the same.
176 carla::multigpu::LoadMapExtras Extras;
177 TArray<uint8> OpenDrive;
178 if (FPaths::GetBaseFilename(FinalPath).Equals(TEXT("OpenDriveMap"), ESearchCase::IgnoreCase) &&
179 FFileHelper::LoadFileToArray(OpenDrive, *UOpenDrive::GetSavedXODRPath(TEXT("OpenDriveMap")), FILEREAD_Silent))
180 {
181 Extras.opendrive = std::string(reinterpret_cast<const char *>(OpenDrive.GetData()), OpenDrive.Num());
182 const carla::Buffer Parameters = carla::MsgPack::Pack(GameInstance->GetOpendriveGenerationParameters());
183 Extras.generation_parameters =
184 std::string(reinterpret_cast<const char *>(Parameters.data()), Parameters.size());
185 }
186 SecondaryServer->GetCommander().SendLoadMap(std::string(TCHAR_TO_UTF8(*FinalPath)), Extras);
187 }
188 }
189 }
190 return bIsFileFound;
191}

195bool UCarlaEpisode::LoadNewOpendriveEpisode(
196 const FString &OpenDriveString,
197 const carla::rpc::OpendriveGenerationParameters &Params)
198{
199 if (OpenDriveString.IsEmpty())
200 {
201 UE_LOG(LogCarla, Error, TEXT("The OpenDrive string is empty."));
202 return false;
203 }
204
205 // Build the Map from the OpenDRIVE data
206 const auto CarlaMap = carla::opendrive::OpenDriveParser::Load(
207 carla::rpc::FromLongFString(OpenDriveString));
208
209 // Check the Map is correclty generated
210 if (!CarlaMap.has_value())
211 {
212 UE_LOG(LogCarla, Error, TEXT("The OpenDrive string is invalid or not supported"));
213 return false;
214 }
215
216 // Generate the OBJ (as string)
217 const auto RoadMesh = CarlaMap->GenerateMesh(Params.vertex_distance);
218 const auto CrosswalksMesh = CarlaMap->GetAllCrosswalkMesh();
219 const auto RecastOBJ = (RoadMesh + CrosswalksMesh).GenerateOBJForRecast();
220
221 // Both files go to Saved/: in a packaged build the content dir is backed
222 // by the pak and not writable. The readers (UOpenDrive::FindPathToXODRFile,
223 // FNavigationMesh::Load) look in Saved/ first. RecastBuilder writes the
224 // .bin next to the .obj. Start from a clean slate so a .bin can never
225 // belong to a different .xodr.
226 WaitForPendingNavigationBuild(60.0);
227 ClearGeneratedWorldFiles();
228 const FString AbsoluteOBJPath = FPaths::ConvertRelativePathToFull(
229 FPaths::ProjectSavedDir() / TEXT("Nav/OpenDriveMap.obj"));
230
231 // Store the OBJ string to a file in order to that RecastBuilder can load it
232 FFileHelper::SaveStringToFile(
233 carla::rpc::ToLongFString(RecastOBJ),
234 *AbsoluteOBJPath,
235 FFileHelper::EEncodingOptions::ForceUTF8,
236 &IFileManager::Get());
237
238 const FString AbsoluteXODRPath = UOpenDrive::GetSavedXODRPath(TEXT("OpenDriveMap"));
239
240 // Copy the OpenDrive as a file in the serverside
241 FFileHelper::SaveStringToFile(
242 OpenDriveString,
243 *AbsoluteXODRPath,
244 FFileHelper::EEncodingOptions::ForceUTF8,
245 &IFileManager::Get());
246
247 if (!FPaths::FileExists(AbsoluteXODRPath))
248 {
249 UE_LOG(LogCarla, Error, TEXT("ERROR: XODR not copied!"));
250 return false;
251 }
252
253 UCarlaGameInstance * GameInstance = UCarlaStatics::GetGameInstance(GetWorld());
254 if(GameInstance)
255 {
256 GameInstance->SetOpendriveGenerationParameters(Params);
257 }
258 else
259 {
260 carla::log_warning("Missing game instance");
261 }
262
263 const FString AbsoluteRecastBuilderPath = BuildRecastBuilderFile();
264 if (FPaths::FileExists(AbsoluteRecastBuilderPath) &&
265 Params.enable_pedestrian_navigation)
266 {
267 // RecastBuilder writes <obj stem>.bin next to the .obj and always exits
268 // 0, so the .bin is the only success signal: drop the previous one first
269 // so a failed build can never hand out last session's navmesh, then watch
270 // the process off the game thread and log the outcome loudly.
271 const FString AbsoluteBinPath = FPaths::ChangeExtension(AbsoluteOBJPath, TEXT("bin"));
272 IFileManager::Get().Delete(*AbsoluteBinPath, false, true, true);
273
274 void *PipeRead = nullptr;
275 void *PipeWrite = nullptr;
276 FPlatformProcess::CreatePipe(PipeRead, PipeWrite);
277
278 /// @todo this can take too long to finish, clients need a method
279 /// to know if the navigation is available or not.
280 FProcHandle RecastHandle = FPlatformProcess::CreateProc(
281 *AbsoluteRecastBuilderPath, *AbsoluteOBJPath,
282 true, true, true, nullptr, 0, nullptr, PipeWrite, nullptr, PipeWrite);
283 if (!RecastHandle.IsValid())
284 {
285 FPlatformProcess::ClosePipe(PipeRead, PipeWrite);
286 UE_LOG(LogCarla, Error, TEXT("RecastBuilder could not be launched ('%s' '%s'); "
287 "no pedestrian navigation for this OpenDRIVE world."),
288 *AbsoluteRecastBuilderPath, *AbsoluteOBJPath);
289 }
290 else
291 {
292 UE_LOG(LogCarla, Log, TEXT("RecastBuilder started: '%s' '%s' -> '%s'"),
293 *AbsoluteRecastBuilderPath, *AbsoluteOBJPath, *AbsoluteBinPath);
294 {
295 FScopeLock Lock(&GNavBuildLock);
296 GNavBuildPending = true;
297 GNavBuildSucceeded = false;
298 }
299 Async(EAsyncExecution::Thread,
300 [RecastHandle, PipeRead, PipeWrite, AbsoluteBinPath, AbsoluteRecastBuilderPath]() mutable
301 {
302 FString Output;
303 while (FPlatformProcess::IsProcRunning(RecastHandle))
304 {
305 Output += FPlatformProcess::ReadPipe(PipeRead);
306 FPlatformProcess::Sleep(0.1f);
307 }
308 Output += FPlatformProcess::ReadPipe(PipeRead);
309 int32 ReturnCode = -1;
310 FPlatformProcess::GetProcReturnCode(RecastHandle, &ReturnCode);
311 FPlatformProcess::CloseProc(RecastHandle);
312 FPlatformProcess::ClosePipe(PipeRead, PipeWrite);
313 Output.TrimEndInline();
314 const bool bSucceeded = ReturnCode == 0 && FPaths::FileExists(AbsoluteBinPath);
315 if (!bSucceeded)
316 {
317 UE_LOG(LogCarla, Error, TEXT("RecastBuilder failed (exit code %d, '%s' %s): no pedestrian "
318 "navigation for this OpenDRIVE world. Output:\n%s"),
319 ReturnCode, *AbsoluteBinPath,
320 FPaths::FileExists(AbsoluteBinPath) ? TEXT("written") : TEXT("missing"), *Output);
321 }
322 else
323 {
324 UE_LOG(LogCarla, Log, TEXT("RecastBuilder finished: navmesh '%s' ready. Output:\n%s"),
325 *AbsoluteBinPath, *Output);
326 }
327 FScopeLock Lock(&GNavBuildLock);
328 GNavBuildPending = false;
329 GNavBuildSucceeded = bSucceeded;
330 });
331 }
332 }
333 else
334 {
335 UE_LOG(LogCarla, Warning, TEXT("'RecastBuilder' not present under '%s', "
336 "the binaries for pedestrian navigation will not be created."),
337 *AbsoluteRecastBuilderPath);
338 }
339
340 return true;
341}

401void UCarlaEpisode::AddRosParentNames(FCarlaActor &Child)
402{
403#if defined(WITH_ROS2)
404 auto ROS2 = carla::ros2::ROS2::GetInstance();
405 if (!ROS2->IsEnabled())
406 {
407 return;
408 }
409 FCarlaActor *CurrentActor = FindCarlaActor(Child.GetParent());
410 while (CurrentActor)
411 {
412 for (const auto &Attr : CurrentActor->GetActorInfo()->Description.Variations)
413 {
414 if (Attr.Key == "ros_name")
415 {
416 ROS2->AddActorParentRosName(static_cast<void*>(Child.GetActor()), static_cast<void*>(CurrentActor->GetActor()));
417 }
418 // NOTE: tier4 also propagated each ancestor's ros_topic_name onto the
419 // child here, but their AddActorRosTopicName used map::insert, so the
420 // dispatcher's earlier per-actor registration (own override or "")
421 // always won and the call was a no-op. Our AddActorRosTopicName is
422 // insert_or_assign (re-registration idempotency), which made the
423 // propagation live and clobbered every attached sensor's own
424 // ros_topic_name with its ancestor's — so it is dropped entirely.
425 }
426 CurrentActor = FindCarlaActor(CurrentActor->GetParent());
427 }
428#endif

486void UCarlaEpisode::InitializeAtBeginPlay()
487{
488 auto World = GetWorld();
489 check(World != nullptr);
490 auto PlayerController = UGameplayStatics::GetPlayerController(World, 0);
491 if (PlayerController == nullptr)
492 {
493 UE_LOG(LogCarla, Error, TEXT("Can't find player controller!"));
494 return;
495 }
496 Spectator = PlayerController->GetPawn();
497 if (Spectator != nullptr)
498 {
499 FActorDescription Description;
500 Description.Id = TEXT("spectator");
501 Description.Class = Spectator->GetClass();
502 ActorDispatcher->RegisterActor(*Spectator, Description);
503 }
504 else
505 {
506 UE_LOG(LogCarla, Error, TEXT("Can't find spectator!"));
507 }
508
509 // material parameters collection
510 UMaterialParameterCollection *Collection = LoadObject<UMaterialParameterCollection>(nullptr, TEXT("/Game/Carla/Blueprints/Game/CarlaParameters.CarlaParameters"), nullptr, LOAD_None, nullptr);
511 if (Collection != nullptr)
512 {
513 MaterialParameters = World->GetParameterCollectionInstance(Collection);
514 if (MaterialParameters == nullptr)
515 {
516 UE_LOG(LogCarla, Error, TEXT("Can't find CarlaParameters instance!"));
517 }
518 }
519 else
520 {
521 UE_LOG(LogCarla, Error, TEXT("Can't find CarlaParameters asset!"));
522 }
523
524 for (TActorIterator<ATrafficSignBase> It(World); It; ++It)
525 {
526 ATrafficSignBase *Actor = *It;
527 check(Actor != nullptr);
528 FActorDescription Description;
529 Description.Id = UCarlaEpisode_GetTrafficSignId(*Actor);
530 Description.Class = Actor->GetClass();
531 ActorDispatcher->RegisterActor(*Actor, Description);
532 }
533
534 // get the definition id for static.prop.mesh
535 auto Definitions = GetActorDefinitions();
536 uint32 StaticMeshUId = 0;
537 for (auto& Definition : Definitions)
538 {
539 if (Definition.Id == "static.prop.mesh")
540 {
541 StaticMeshUId = Definition.UId;
542 break;
543 }
544 }
545
546 for (TActorIterator<AStaticMeshActor> It(World); It; ++It)
547 {
548 auto Actor = *It;
549 check(Actor != nullptr);
550 auto MeshComponent = Actor->GetStaticMeshComponent();
551 check(MeshComponent != nullptr);
552 if (MeshComponent->Mobility == EComponentMobility::Movable)
553 {
554 FActorDescription Description;
555 Description.Id = TEXT("static.prop.mesh");
556 Description.UId = StaticMeshUId;
557 Description.Class = Actor->GetClass();
558 Description.Variations.Add("mesh_path",
559 FActorAttribute{"mesh_path", EActorAttributeType::String,
560 MeshComponent->GetStaticMesh()->GetPathName()});
561 Description.Variations.Add("mass",
562 FActorAttribute{"mass", EActorAttributeType::Float,
563 FString::SanitizeFloat(MeshComponent->GetMass())});
564 ActorDispatcher->RegisterActor(*Actor, Description);
565 }
566 }

598TPair<EActorSpawnResultStatus, FCarlaActor*> UCarlaEpisode::SpawnActorWithInfo(
599 const FTransform &Transform,
600 FActorDescription thisActorDescription,
601 FCarlaActor::IdType DesiredId)
602{
603 ALargeMapManager* LargeMap = UCarlaStatics::GetLargeMapManager(GetWorld());
604 FTransform LocalTransform = Transform;
605 if(LargeMap)
606 {
607 LocalTransform = LargeMap->GlobalToLocalTransform(LocalTransform);
608 }
609
610 // NewTransform.AddToTranslation(-1.0f * FVector(CurrentMapOrigin));
611 auto result = ActorDispatcher->SpawnActor(LocalTransform, thisActorDescription, DesiredId);
612 if (result.Key == EActorSpawnResultStatus::Success &&
613 GetWorld()->GetWorldPartition() != nullptr)
614 {
615 // World Partition map: every physics actor needs the ground under it
616 // resident, mirroring the tile streaming the legacy LargeMapManager
617 // provided on tiled maps. The engine only streams cells around registered
618 // streaming sources; without one, a vehicle spawned away from the
619 // spectator free-falls through the unloaded road.
620 AActor* Actor = result.Value->GetActor();
621 const bool bNeedsGround =
622 Cast<ACarlaWheeledVehicle>(Actor) != nullptr ||
623 Cast<AWalkerBase>(Actor) != nullptr;
624 if (bNeedsGround &&
625 !Actor->FindComponentByClass<UWorldPartitionStreamingSourceComponent>())
626 {
627 // The hero keeps a wide loading ring like the legacy tiled maps gave
628 // it; background traffic only needs the ground in its vicinity.
629 const FActorAttribute* Attribute =
630 thisActorDescription.Variations.Find("role_name");
631 const bool bIsHero = Attribute && (Attribute->Value.Contains("hero") ||
632 Attribute->Value.Contains("ego_vehicle"));
633 auto* Source = NewObject<UWorldPartitionStreamingSourceComponent>(Actor);
634 FStreamingSourceShape Shape;
635 Shape.bUseGridLoadingRange = false;
636 Shape.Radius = bIsHero ? EpisodeSettings.TileStreamingDistance : 20000.0f;
637 Source->Shapes.Add(Shape);
638 Source->RegisterComponent();
639
640 // Physics drops the actor immediately, but cells only stream during the
641 // world partition subsystem update; run one update now and block until
642 // the requested cells are resident so the actor never outruns its
643 // ground.
644 UWorldPartitionSubsystem* WPSubsystem =
645 GetWorld()->GetSubsystem<UWorldPartitionSubsystem>();
646 if (WPSubsystem != nullptr && !WPSubsystem->IsStreamingCompleted(Source))
647 {
648 WPSubsystem->OnUpdateStreamingState();
649 GetWorld()->BlockTillLevelStreamingCompleted();
650 }
651 UE_LOG(LogCarla, Log, TEXT(
652 "WP streaming source on %s (role %s, radius %.0f cm): completed=%d"),
653 *Actor->GetName(), Attribute ? *Attribute->Value : TEXT("none"), Shape.Radius,
654 WPSubsystem ? WPSubsystem->IsStreamingCompleted(Source) : -1);
655 }
656 }
657 if (result.Key == EActorSpawnResultStatus::Success && bIsPrimaryServer)
658 {
659 if (Recorder->IsEnabled())
660 {
661 Recorder->CreateRecorderEventAdd(
662 result.Value->GetActorId(),
663 static_cast<uint8_t>(result.Value->GetActorType()),
664 Transform,
665 thisActorDescription
666 );
667 }
668 if (bIsPrimaryServer)
669 {
670 GetFrameData().CreateRecorderEventAdd(
671 result.Value->GetActorId(),
672 static_cast<uint8_t>(result.Value->GetActorType()),
673 Transform,
674 std::move(thisActorDescription));
675 }
676 }
677
678 return result;
679}

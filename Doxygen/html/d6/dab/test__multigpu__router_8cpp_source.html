70 bool WaitUntilPending(
71 carla::multigpu::Router &router,
72 const std::shared_ptr<carla::multigpu::Primary> &session) {
73 const auto deadline = std::chrono::steady_clock::now() + std::chrono::seconds(2);
74 while (std::chrono::steady_clock::now() < deadline) {
75 if (router.TestHasPendingRequest(session)) {
76 return true;
77 }
78 std::this_thread::sleep_for(std::chrono::milliseconds(1));
79 }
80 return false;

147 /// Records every request the router sends to one secondary.
148 class RequestLog {
149 public:
150 explicit RequestLog(carla::multigpu::Router &router) {
151 router.TestSetRequestObserver([state = _state](
152 const carla::multigpu::Primary *session,
153 carla::multigpu::MultiGPUCommand command,
154 const carla::BufferView &payload) {
155 std::scoped_lock<std::mutex> lock(state->mutex);
156 state->sent.push_back(SentRequest{
157 session, command, std::vector<unsigned char>(payload.data(), payload.data() + payload.size())});
158 });
159 }
160
161 std::vector<SentRequest> Take() {
162 std::scoped_lock<std::mutex> lock(_state->mutex);
163 return std::exchange(_state->sent, {});
164 }
165
166 private:

168 std::mutex mutex;
169 std::vector<SentRequest> sent;
170 };
171 std::shared_ptr<State> _state = std::make_shared<State>();
172 };

176 const std::shared_ptr<carla::multigpu::Primary> &session,
177 carla::multigpu::stream_id sensor_id,

572 router.AsyncRun(1u);
573 auto session = MakeFakeSession();

575
576 // Only one session is connected, so both requests round-robin to it.
577 // There is no per-request correlation id in the wire protocol (a session
578 // gets exactly one _promises slot, not one per request), so the second
579 // request must be refused explicitly rather than silently overwriting --

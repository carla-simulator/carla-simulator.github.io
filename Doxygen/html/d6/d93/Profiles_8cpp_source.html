84 std::optional<double> ParseMaxSpeed(const std::string &value) {
85 if (value.empty() || value == "none" || value == "signals") {
86 return std::nullopt;
87 }
88 double number = 0.0;
89 if (!ParseDouble(value, number) || number <= 0.0) {
90 return std::nullopt;
91 }
92 const bool mph = value.find("mph") != std::string::npos;
93 return mph ? number * 0.44704 : number / 3.6;
94 }

96 LaneProfile ResolveProfile(const std::string &highway, const Tags &tags) {
97 LaneProfile profile;
98 const ClassDefaults *defaults = FindClassDefaults(highway);
99 const ClassDefaults fallback;
100 const ClassDefaults &base = (defaults != nullptr) ? *defaults : fallback;
101
102 profile.lane_width = base.lane_width;
103 profile.sidewalk_width = base.sidewalk_width;
104 profile.shoulder = base.shoulder;
105 profile.shoulder_inner = base.shoulder_inner;
106
107 const std::string *oneway = FindTag(tags, "oneway");
108 profile.oneway = (oneway != nullptr) && (*oneway == "yes" || *oneway == "1" || *oneway == "-1");
109 if (highway == "motorway" || highway == "motorway_link") {
110 profile.oneway = true;
111 }
112
113 int total = base.lanes;
114 const std::string *lanes_tag = FindTag(tags, "lanes");
115 if (lanes_tag != nullptr) {
116 int parsed = 0;
117 if (ParseInt(*lanes_tag, parsed) && parsed > 0) {
118 total = parsed;
119 }
120 }
121
122 if (profile.oneway) {
123 profile.forward_lanes = std::max(1, total);
124 profile.backward_lanes = 0;
125 } else {
126 profile.forward_lanes = std::max(1, total / 2);
127 profile.backward_lanes = std::max(1, total - profile.forward_lanes);
128 }
129
130 const std::string *forward_tag = FindTag(tags, "lanes:forward");
131 const std::string *backward_tag = FindTag(tags, "lanes:backward");
132 int parsed = 0;
133 if (forward_tag != nullptr && ParseInt(*forward_tag, parsed) && parsed >= 0) {
134 profile.forward_lanes = parsed;
135 }
136 if (backward_tag != nullptr && ParseInt(*backward_tag, parsed) && parsed >= 0) {
137 profile.backward_lanes = parsed;
138 }
139
140 const std::string *width_tag = FindTag(tags, "width");
141 if (width_tag != nullptr) {
142 double width = 0.0;
143 const int count = std::max(1, profile.forward_lanes + profile.backward_lanes);
144 if (ParseDouble(*width_tag, width) && width > 0.0) {
145 profile.lane_width = std::max(2.5, width / count);
146 }
147 }
148
149 const std::string *maxspeed = FindTag(tags, "maxspeed");
150 if (maxspeed != nullptr) {
151 profile.speed_limit = ParseMaxSpeed(*maxspeed);
152 }
153
154 const std::string *sidewalk = FindTag(tags, "sidewalk");
155 if (sidewalk != nullptr && (*sidewalk == "no" || *sidewalk == "none")) {
156 profile.sidewalk_width = std::nullopt;
157 }
158 return profile;
159 }

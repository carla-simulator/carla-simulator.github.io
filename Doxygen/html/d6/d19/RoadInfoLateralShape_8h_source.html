16 /// One <lateralProfile><shape> record: at station s, the road surface
17 /// height relative to the elevation profile is a cubic polynomial in
18 /// (t - t_record), valid from t_record until the next record at the same
19 /// station. Together the records describe the road cross-section (crown,
20 /// banking); reconstructed real-world maps rely on them for crossing
21 /// roads to agree on surface height inside junctions.
22 class RoadInfoLateralShape final : public RoadInfo {
23 public:
24
25 RoadInfoLateralShape(
26 double s,
27 double t,
28 double a,
29 double b,
30 double c,
31 double d)
32 : RoadInfo(s),
33 _t(t),
34 _shape(a, b, c, d, t) {}
35
36 void AcceptVisitor(RoadInfoVisitor &v) final {
37 v.Visit(*this);
38 }
39
40 double GetT() const {
41 return _t;
42 }
43
44 /// Evaluate the surface height offset at lateral position t (the
45 /// polynomial is defined over t - t_record).
46 double Evaluate(const double t) const {
47 return _shape.Evaluate(t);
48 }
49
50 const geom::CubicPolynomial &GetPolynomial() const {
51 return _shape;
52 }
53
54 private:
55
56 const double _t;
57
58 const geom::CubicPolynomial _shape;
59 };

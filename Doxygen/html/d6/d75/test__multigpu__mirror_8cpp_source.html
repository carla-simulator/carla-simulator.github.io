33TEST(MultiGpuMirror, positions_round_trip_bit_exact) {
34 MirroredPositions sent;
35 MirroredPosition camera;
36 camera.id = 7u;
37 camera.location[0] = 12345.678901234;
38 camera.location[1] = -0.000001;
39 camera.location[2] = 250.0;
40 // Looking straight down: the pose an Euler round trip cannot keep.
41 camera.rotation[0] = 0.0;
42 camera.rotation[1] = 0.70710678118654757;
43 camera.rotation[2] = 0.0;
44 camera.rotation[3] = 0.70710678118654746;
45 sent.positions.push_back(camera);
46 MirroredPosition other;
47 other.id = 9u;
48 sent.positions.push_back(other);
49
50 std::ostringstream out;
51 sent.Write(out);
52 const std::string bytes = out.str();
53 ASSERT_EQ(static_cast<uint8_t>(bytes[0]), MirroredPositions::kPacketId);
54 ASSERT_EQ(bytes.size(), carla::multigpu::kFrameDataHeaderSize + 6u + 2u * MirroredPositions::kRecordSizeV1);
55
56 MirroredPositions got;
57 ASSERT_TRUE(ReadPositions(bytes, got));
58 ASSERT_EQ(got.positions.size(), 2u);
59 EXPECT_EQ(got.positions[0].id, 7u);
60 for (int i = 0; i < 3; ++i) {
61 EXPECT_EQ(got.positions[0].location[i], camera.location[i]);
62 }
63 for (int i = 0; i < 4; ++i) {
64 EXPECT_EQ(got.positions[0].rotation[i], camera.rotation[i]);
65 }
66 EXPECT_EQ(got.positions[1].id, 9u);
67}

112TEST(MultiGpuMirror, missing_parent_and_cycle_are_guarded) {
113 // 5's parent is not in the frame; 6 and 7 are each other's parent.
114 const std::vector<uint32_t> ids = {5u, 6u, 7u, 8u};
115 const std::unordered_map<uint32_t, uint32_t> parents = {{5u, 99u}, {6u, 7u}, {7u, 6u}, {8u, 7u}};
116 const std::vector<uint32_t> order = OrderParentsFirst(ids, parents);
117 ASSERT_EQ(order.size(), ids.size());
118 // Every id once; the actor attached to the cycle comes after it.
119 const auto pos = [&](uint32_t id) {
120 return std::find(order.begin(), order.end(), id) - order.begin();
121 };
122 for (uint32_t id : ids) {
123 EXPECT_EQ(std::count(order.begin(), order.end(), id), 1);
124 }
125 EXPECT_LT(pos(7u), pos(8u));
126}

142 car.actor_id = 21u;
143 car.wheels = {{12.5f, 361.25f, 3.5f}, {-12.5f, 0.125f, -1.0f}, {0.0f, 90.0f, 0.0f}, {0.0f, 180.0f, 2.25f}};

145 door.door = 1u;
146 door.location = {10.0, -85.5, 40.25};

148 car.open_doors = {door};
149 car.current_gear = -1;
150 car.manual_gear_shift = true;
151 car.simulating_physics = true;
152 carla::multigpu::VehicleParts bike;
153 bike.actor_id = 22u;
154 bike.wheels = {{5.0f, 10.0f, 0.5f}, {0.0f, 10.0f, 0.25f}};
155 sent.vehicles = {car, bike};
156

168 ASSERT_EQ(got.vehicles[0].open_doors.size(), 1u);
169 EXPECT_EQ(got.vehicles[0].open_doors[0].door, 1u);
170 EXPECT_EQ(got.vehicles[0].open_doors[0].location, door.location);
171 EXPECT_EQ(got.vehicles[0].open_doors[0].rotation, door.rotation);
172 EXPECT_EQ(got.vehicles[0].current_gear, std::optional<int32_t>(-1));
173 EXPECT_TRUE(got.vehicles[0].manual_gear_shift);
174 EXPECT_TRUE(got.vehicles[0].simulating_physics);
175 EXPECT_EQ(got.vehicles[1].actor_id, 22u);
176 EXPECT_EQ(got.vehicles[1].wheels.size(), 2u);
177 EXPECT_TRUE(got.vehicles[1].open_doors.empty());
178 EXPECT_FALSE(got.vehicles[1].current_gear.has_value());

181TEST(MultiGpuMirror, vehicle_parts_from_a_newer_build_keep_their_known_fields) {
182 // One vehicle: wheel records of 16 bytes (v1's 12 and a new float) and an
183 // unknown field.
184 std::ostringstream wheels;
185 const uint32_t count = 1u;
186 const uint16_t record_size = 16u;
187 wheels.write(reinterpret_cast<const char *>(&count), sizeof(count));
188 wheels.write(reinterpret_cast<const char *>(&record_size), sizeof(record_size));
189 const float values[4] = {1.0f, 2.0f, 3.0f, 99.0f};
190 wheels.write(reinterpret_cast<const char *>(values), sizeof(values));
191 std::ostringstream record;
192 const uint32_t id = 5u;
193 record.write(reinterpret_cast<const char *>(&id), sizeof(id));
194 carla::multigpu::WriteField(record, 9u, std::string("future"));
195 carla::multigpu::WriteField(record, carla::multigpu::MirroredVehicleParts::kWheels, wheels.str());
196 std::ostringstream payload;
197 payload.write(reinterpret_cast<const char *>(&count), sizeof(count));
198 carla::multigpu::WriteSizedBytes(payload, record.str());
199
200 const std::string body = payload.str();
201 std::istringstream in(body);
202 carla::multigpu::MirroredVehicleParts got;
203 ASSERT_TRUE(got.Read(in, static_cast<uint32_t>(body.size())));
204 ASSERT_EQ(got.vehicles.size(), 1u);
205 ASSERT_EQ(got.vehicles[0].wheels.size(), 1u);
206 EXPECT_EQ(got.vehicles[0].wheels[0].suspension_offset, 3.0f);
207}

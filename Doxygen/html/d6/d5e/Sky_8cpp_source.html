24 AWeather* FindWeatherActor(const UWorld* World)
25 {
26 return World != nullptr
27 ? Cast<AWeather>(UGameplayStatics::GetActorOfClass(World, AWeather::StaticClass()))
28 : nullptr;
29 }

31 // "Current weather" for this sky rig: if a BP_CarlaWeather happens to be
32 // placed in the level, its Weather member is authoritative (same value
33 // the API/recorder see). Otherwise fall back to this actor's own "Sky
34 // Parameters" -- kept in sync by ApplyWeatherTo below -- so none of this
35 // requires an AWeather actor to be placed at all.
36 FWeatherParameters GetEffectiveCurrentWeather(AActor* SkyActor)
37 {
38 if (AWeather* Weather = FindWeatherActor(SkyActor->GetWorld()))
39 return Weather->GetCurrentWeather();
40 if (FWeatherParameters* Params = ASkyBase::FindWeatherParameters(SkyActor))
41 return *Params;
42 return FWeatherParameters();
43 }
44
45 // Applies NewWeather to this sky rig. Prefers routing through a placed
46 // AWeather (full pipeline: post-process camera blendables, day/night
47 // light broadcast, recorder) when one exists; otherwise pushes directly
48 // to the rig via the same per-actor logic AWeather itself uses.
49 void ApplyWeatherTo(AActor* SkyActor, const FWeatherParameters& NewWeather)
50 {
51 if (AWeather* Weather = FindWeatherActor(SkyActor->GetWorld()))
52 Weather->ApplyWeather(NewWeather);
53 else
54 AWeather::ApplyWeatherToSkyActor(SkyActor, NewWeather);
55 }
56}

156void ASkyBase::ApplyTimeAndConditionPresets()
157{
158 FWeatherTimePreset TimePreset;
159 FWeatherConditionPreset ConditionPreset;
160 const bool bTimeOk = UWeatherJsonUtils::LoadTimePreset(TimePreset, TimePresetName);
161 const bool bConditionOk = UWeatherJsonUtils::LoadConditionPreset(ConditionPreset, ConditionPresetName);
162 if (!bTimeOk && !bConditionOk)
163 return;
164
165 // Only overwrite the half that actually loaded, so a missing condition
166 // preset (say) doesn't reset the sun position to defaults.
167 FWeatherParameters Base = GetEffectiveCurrentWeather(this);
168 if (!bTimeOk)
169 {
170 TimePreset.SunAzimuthAngle = Base.SunAzimuthAngle;
171 TimePreset.SunAltitudeAngle = Base.SunAltitudeAngle;
172 }
173 if (!bConditionOk)
174 {
175 ConditionPreset.Cloudiness = Base.Cloudiness;
176 ConditionPreset.Precipitation = Base.Precipitation;
177 ConditionPreset.PrecipitationDeposits = Base.PrecipitationDeposits;
178 ConditionPreset.WindIntensity = Base.WindIntensity;
179 ConditionPreset.FogDensity = Base.FogDensity;
180 ConditionPreset.FogDistance = Base.FogDistance;
181 ConditionPreset.FogFalloff = Base.FogFalloff;
182 ConditionPreset.Wetness = Base.Wetness;
183 ConditionPreset.ScatteringIntensity = Base.ScatteringIntensity;
184 ConditionPreset.MieScatteringScale = Base.MieScatteringScale;
185 ConditionPreset.RayleighScatteringScale = Base.RayleighScatteringScale;
186 }
187
188 ApplyWeatherTo(this, UWeatherJsonUtils::MergeWeatherPresets(Base, TimePreset, ConditionPreset));
189
190 // A hand-edit of a WeatherParameters field always gets a FULL editor
191 // reconstruction cycle for free (property edit -> OnConstruction -> the
192 // Blueprint's own construction graph runs, THEN this class's C++ override
193 // re-pushes). This function updates the struct and pushes natively, same
194 // as OnConstruction's own push, but skips the Blueprint graph's own pass
195 // entirely -- if that pass does any one-time setup the raw push doesn't
196 // replicate (respawning/re-registering the sky sphere, stars material,
197 // etc.), the result can look wrong until some later, unrelated edit
198 // finally triggers a real reconstruction. Force that same full cycle here
199 // instead of leaving it to chance. Editor-only API -- irrelevant at
200 // runtime anyway, OnConstruction only ever runs once there, at spawn.
201#if WITH_EDITOR
202 RerunConstructionScripts();
203#endif
204}

281 // visually inert. Flip ExposureMode below to Manual to make them matter
282 // (that's what GoPro.json does). A separate attempt at making them work

375void ASkyBase::OnConstruction(const FTransform& Transform)
376{
377 Super::OnConstruction(Transform);
378
379 // OnConstruction also fires once at actor spawn in a game world (PIE or
380 // packaged), not just on editor property edits. Pushing here unconditionally
381 // meant BOTH this AND ACarlaGameModeBase::BeginPlay's own weather push ran
382 // at Play start, and SetSkySphere's respawn (destroy-old, spawn-new) isn't
383 // safe against two independent, near-simultaneous callers -- it produced
384 // duplicate BP_Sky_Sphere/DirectionalLight actors and the "multiple
385 // directional lights competing" warning. Editor-only: let BeginPlay's push
386 // be the sole driver once actually playing.
387 if (GetWorld() != nullptr && GetWorld()->IsGameWorld())
388 return;
389
390 // Direct push, not ApplyWeatherTo: this must never depend on finding an
391 // AWeather actor (irrelevant here) or go through its full pipeline (camera
392 // blendables, recorder) -- just put this rig's own components back in sync
393 // with whatever "Sky Parameters" already holds.
394 if (FWeatherParameters* SkyParameters = ASkyBase::FindWeatherParameters(this))
395 AWeather::ApplyWeatherToSkyActor(this, *SkyParameters);
396
397 // Same self-healing story as Sky Parameters above: any edit reruns
398 // Construction and would otherwise leave CameraParameters' last pushed
399 // values sitting unapplied against a Settings struct that just got reset.
400 PushCameraParameters();
401}

405void ASkyBase::BeginPlay()
406{
407 Super::BeginPlay();
408
409 UWorld* World = GetWorld();
410 // The viewport / spectator renders through this unbound post process with
411 // the viewport profile (carla.PostProcess.ViewportProfile); RGB sensors load
412 // their own (carla.PostProcess.Profile or post_process_profile). What the map
413 // saved in the rig is only the editor view.
414 if (World != nullptr && World->IsGameWorld() && PostProcessComponent != nullptr)
415 {
416 const FString Profile = UPostProcessJsonUtils::GetViewportProfileName();
417 if (UPostProcessJsonUtils::LoadAllPostProcessFromJsonToPostProcessComponent(PostProcessComponent, Profile))
418 {
419 bBloomFromWeatherFallback = false;
420 UE_LOG(LogCarla, Log, TEXT("ASkyBase: %s loaded camera profile '%s'"), *GetName(), *Profile);
421 }
422 else
423 {
424 UE_LOG(LogCarla, Warning, TEXT("ASkyBase: %s could not load camera profile '%s'"), *GetName(), *Profile);
425 }
426 AWeather::FillSkyPostProcessFallback(PostProcessComponent);
427 }
428
429 if (World == nullptr || ActorHasTag(FallbackSkyTag))
430 return;
431
432 TArray<AActor*> Fallbacks;
433 UGameplayStatics::GetAllActorsWithTag(World, FallbackSkyTag, Fallbacks);
434 if (Fallbacks.IsEmpty())
435 return;
436 for (AActor* Fallback : Fallbacks)
437 {
438 // SetSunActorReference attaches a DirectionalLight actor to the rig,
439 // which Destroy would leave behind.
440 TArray<AActor*> Attached;
441 Fallback->GetAttachedActors(Attached, /*bResetArray=*/true, /*bRecursivelyIncludeAttachedActors=*/true);
442 for (AActor* Child : Attached)
443 Child->Destroy();
444 Fallback->Destroy();
445 }
446 UE_LOG(LogCarla, Log, TEXT("ASkyBase: %s streamed in, removed %d fallback sky rig(s)"),
447 *GetName(), Fallbacks.Num());
448
449 // Streamed in with the state saved in the map; catch up with the weather.
450 if (AWeather* Weather = Cast<AWeather>(UGameplayStatics::GetActorOfClass(World, AWeather::StaticClass())))
451 AWeather::ApplyWeatherToSkyActor(this, Weather->GetCurrentWeather());
452}

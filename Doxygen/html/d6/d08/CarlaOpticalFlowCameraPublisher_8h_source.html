19// Non-alias subclass: the UE optical flow sensor produces a height*width buffer
20// of (vx, vy) float pairs (8 bytes per pixel) which CARLA encodes on the wire
21// as a 4-channel BGRA uint8 image via the HSV colour wheel. The conversion is
22// the same one that ue5-dev shipped before this refactor; it now lives in
23// OpticalFlowEncoding.h so the math is unit-testable without FastDDS, and the
24// publisher only has to call ComputeImage(...) to plug it into the unified
25// base.
26class CarlaOpticalFlowCameraPublisher : public CarlaCameraPublisher {
27public:
28 CarlaOpticalFlowCameraPublisher(std::string base_topic_name, std::string frame_id)
29 : CarlaCameraPublisher(std::move(base_topic_name), std::move(frame_id)) {}
30
31protected:
32 [[nodiscard]] uint8_t GetChannels() const override { return 4; }
33 [[nodiscard]] std::string GetEncoding() const override { return "bgra8"; }
34 [[nodiscard]] std::vector<uint8_t> ComputeImage(
35 uint32_t height, uint32_t width, const uint8_t *data) const override;
36};

16// PointField datatype constants mirror the numeric constants in
17// carla/ros2/types/msg/PointField.h (carla::ros2::msg::PointField::*). They are
18// kept here as an enum class so this header does not pull the message types into
19// the Build-Tests/ compile unit. Producers in publishers/Carla*Publisher.cpp map
20// these values onto carla::ros2::msg::PointField; this enum mirrors the same
21// numeric values so layout tests can verify field offsets / sizes.
22enum class PointFieldDataType : std::uint8_t {
23 Int8 = 1,
24 UInt8 = 2,
25 Int16 = 3,
26 UInt16 = 4,
27 Int32 = 5,
28 UInt32 = 6,
29 Float32 = 7,
30 Float64 = 8,
31};

90// DVS events: x/y UINT16, t FLOAT64, pol INT8. The producer converts the

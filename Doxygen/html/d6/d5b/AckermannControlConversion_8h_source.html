14// Converts the five scalar fields of a ROS ackermann_msgs/AckermannDriveStamped
15// message payload into the CARLA AckermannControl POD. The wire-side
16// AckermannDriveStamped FastDDS type is decomposed in AckermannControlSubscriber.cpp;
17// this seam exists so the numeric mapping can be exercised by a unit test that does
18// not link FastDDS.
19inline AckermannControl FromAckermannDrive(
20 float steering_angle,
21 float steering_angle_velocity,
22 float speed,
23 float acceleration,
24 float jerk) {
25 AckermannControl control{};
26 control.steer = steering_angle;
27 control.steer_speed = steering_angle_velocity;
28 control.speed = speed;
29 control.acceleration = acceleration;
30 control.jerk = jerk;

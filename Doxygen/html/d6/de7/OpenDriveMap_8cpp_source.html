31 template <typename T, typename RangeT, typename FuncT>
32 static TArray<T> TransformToTArray(RangeT &&Range, FuncT &&TransformFunction)
33 {
34 TArray<T> Result;
35 Result.Reserve(GetSize(Range));
36 for (auto &&Item : Range)
37 {
38 Result.Emplace(TransformFunction(Item));
39 }
40 return Result;
41 }

43 template <typename T, typename RangeT>
44 static TArray<T> TransformToTArray(RangeT &&Range)
45 {
46 return TransformToTArray<T>(
47 std::forward<RangeT>(Range),
48 [](auto &&Item) { return T{Item}; });
49 }

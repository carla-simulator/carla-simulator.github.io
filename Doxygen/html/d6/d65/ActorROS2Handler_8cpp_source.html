73bool ActorROS2Handler::FlattenSteeringCurve(AActor *Actor)
74{
75 if (!IsValid(Actor)) return false;
76
77 ACarlaWheeledVehicle * const Vehicle = Cast<ACarlaWheeledVehicle>(Actor);
78 if (!IsValid(Vehicle)) return false;
79
80 auto VehiclePhysicsControl = Vehicle->GetVehiclePhysicsControl();
81
82 /// @note Flatten steering curve to be always 1.0 to properly map wheel angle to steering at all speeds
83 VehiclePhysicsControl.SteeringCurve.Reset();
84 VehiclePhysicsControl.SteeringCurve.AddKey(0.f, 1.f);
85 VehiclePhysicsControl.SteeringCurve.AddKey(120.f, 1.f);
86
87 Vehicle->ApplyVehiclePhysicsControl(VehiclePhysicsControl);
88 UE_LOG(LogCarla, Log, TEXT("Resetting SteeringCurve!"));
89
90 return true;
91}

28template <typename V>
29requires (
30 std::remove_reference_t<V>::Dim <= 3)
31static std::string FormatVectorLike(V&& v)
32{
33#ifndef HAS_FORMAT
34 char buffer[256];
35#endif
36 constexpr auto Dim = std::remove_reference_t<V>::Dim;
37 if constexpr (Dim == 1)
38 {
39 const auto& [x] = v;
40#ifdef HAS_FORMAT
41 return std::format("({})", x);
42#else
43 auto n = snprintf(buffer, sizeof(buffer), "(%f)", (double)x);
44 if (n < 0)
45 return "<FORMAT-ERRROR>";
46 return std::string(buffer, n);
47#endif
48 }
49 else if constexpr (Dim == 2)
50 {
51 const auto& [x, y] = v;
52#ifdef HAS_FORMAT
53 return std::format("({}, {})", x, y);
54#else
55 auto n = snprintf(buffer, sizeof(buffer), "(%f, %f)", (double)x, (double)y);
56 if (n < 0)
57 return "<FORMAT-ERRROR>";
58 return std::string(buffer, n);
59#endif
60 }
61 else if constexpr (Dim == 3)
62 {
63 const auto& [x, y, z] = v;
64#ifdef HAS_FORMAT
65 return std::format("({}, {}, {})", x, y, z);
66#else
67 auto n = snprintf(buffer, sizeof(buffer), "(%f, %f, %f)", (double)x, (double)y, (double)z);
68 if (n < 0)
69 return "<FORMAT-ERRROR>";
70 return std::string(buffer, n);
71#endif
72 }
73}

75template <typename V>
76static std::string FormatCurveLike(V&& v)
77{
78 std::string r;
79 r += "[";
80 for (auto& point : v)
81 {
82 r += FormatVectorLike(point);
83 r += ',';
84 }
85 r += "]";
86 return r;
87}

89inline bool CarlaRecorderQuery::ReadHeader(void)
90{
91 if (File.eof())
92 {
93 return false;
94 }
95
96 ReadValue<char>(File, Header.Id);
97 ReadValue<uint32_t>(File, Header.Size);
98
99 return true;
100}

103{
104 File.seekg(Header.Size, std::ios::cur);

107inline bool CarlaRecorderQuery::CheckFileInfo(std::stringstream& Info)
108{
109 // read Info
110 RecInfo.Read(File);
111
112 // check magic string
113 if (RecInfo.Magic != "CARLA_RECORDER")
114 {
115 Info << "File is not a CARLA recorder" << std::endl;
116 return false;
117 }
118
119 // show general Info
120 Info << "Version: " << RecInfo.Version << std::endl;
121 Info << "Map: " << TCHAR_TO_UTF8(*RecInfo.Mapfile) << std::endl;
122#ifdef _WIN32
123 struct tm Tmp;
124 localtime_s(&Tmp, &RecInfo.Date);
125 auto TimeInfo = &Tmp;
126#else
127 auto TimeInfo = localtime(&RecInfo.Date);
128#endif
129 char DateStr[100];
130 strftime(DateStr, sizeof(DateStr), "%x %X", TimeInfo);
131 Info << "Date: " << DateStr << std::endl << std::endl;
132
133 return true;
134}

136std::string CarlaRecorderQuery::QueryInfo(std::string Filename, bool bShowAll)
137{
138 std::stringstream Info;
139
140 // get the final path + filename
141 std::string Filename2 = GetRecorderFilename(Filename);
142
143 // try to open
144 File.open(Filename2, std::ios::binary);
145 if (!File.is_open())
146 {
147 Info << "File " << Filename2 << " not found on server\n";
148 return Info.str();
149 }
150
151 uint16_t i, Total;
152 bool bFramePrinted = false;
153
154 // lambda for repeating task
155 auto PrintFrame = [this](std::stringstream& Info)
156 {
157 Info << "Frame " << Frame.Id << " at " << Frame.Elapsed << " seconds\n";
158 };
159
160 if (!CheckFileInfo(Info))
161 return Info.str();
162
163 // parse only frames
164 while (File)
165 {
166 // get header
167 if (!ReadHeader())
168 {
169 break;
170 }
171
172 // check for a frame packet
173 switch (Header.Id)
174 {
175 // frame
176 case static_cast<char>(CarlaRecorderPacketId::FrameStart):
177 Frame.Read(File);
178 if (bShowAll)
179 {
180 PrintFrame(Info);
181 bFramePrinted = true;
182 }
183 else
184 bFramePrinted = false;
185 break;
186
187 // events add
188 case static_cast<char>(CarlaRecorderPacketId::EventAdd):
189 ReadValue<uint16_t>(File, Total);
190 if (Total > 0 && !bFramePrinted)
191 {
192 PrintFrame(Info);
193 bFramePrinted = true;
194 }
195 for (i = 0; i < Total; ++i)
196 {
197 // add
198 EventAdd.Read(File);
199 Info << " Create " << EventAdd.DatabaseId << ": " << TCHAR_TO_UTF8(*EventAdd.Description.Id) <<
200 " (" <<
201 static_cast<int>(EventAdd.Type) << ") at (" << EventAdd.Location.X << ", " <<
202 EventAdd.Location.Y << ", " << EventAdd.Location.Z << ")" << std::endl;
203 for (auto& Att : EventAdd.Description.Attributes)

206 }

877 std::size_t operator()(const std::pair<uint32_t, uint32_t>& P) const
878 {
879 std::size_t hash = P.first;
880 hash <<= 32;
881 hash += P.second;
882 return hash;

962 Type2 = 'o'; // other non-actor object

968 ++Valid;
969 else if (Category1 == 'h' && Collision.IsActor1Hero)
970 ++Valid;
971
972 // filter actor 2
973 if (Category2 == 'a')
974 ++Valid;
975 else if (Category2 == Type2)
976 ++Valid;
977 else if (Category2 == 'h' && Collision.IsActor2Hero)
978 ++Valid;

981 if (Valid == 2)

988 Info << " " << " " << Type1 << " " << Type2 << " ";

993 Info << std::endl;

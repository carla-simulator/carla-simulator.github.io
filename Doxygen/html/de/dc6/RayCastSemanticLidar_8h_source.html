79 /// Clears every channel buffer and reserves room for a full scan.
80 template <typename DetectionT>
81 static void ResetChannelDetections(std::vector<std::vector<DetectionT>> &Detections, uint32_t Channels, uint32_t MaxPointsPerChannel)
82 {
83 Detections.resize(Channels);
84 for (auto &ChannelDetections : Detections)
85 {
86 ChannelDetections.clear();
87 ChannelDetections.reserve(MaxPointsPerChannel);
88 }
89 }

91 /// Post-processes the detections channel by channel, drops the rejected ones and writes the rest to LidarData.
92 /// Runs sequentially in channel and point order, so the random sequence for noise and drop-off is unchanged.
93 template <typename DetectionT, typename LidarDataT, typename PostprocessT>
94 static void CompactAndWriteDetections(
95 std::vector<std::vector<DetectionT>> &Detections,
96 std::vector<uint32_t> &OutPointsPerChannel,
97 uint32_t Channels,
98 LidarDataT &LidarData,
99 PostprocessT &&Postprocess)
100 {
101 for (uint32_t Channel = 0u; Channel < Channels; ++Channel)
102 {
103 auto &ChannelDetections = Detections[Channel];
104 size_t Kept = 0;
105 for (DetectionT &Detection : ChannelDetections)
106 {
107 if (Postprocess(Detection))
108 {
109 ChannelDetections[Kept++] = Detection;
110 }
111 }
112 ChannelDetections.resize(Kept);
113 OutPointsPerChannel[Channel] = static_cast<uint32_t>(Kept);
114 }
115
116 LidarData.ResetMemory(OutPointsPerChannel);
117 for (uint32_t Channel = 0u; Channel < Channels; ++Channel)
118 {
119 LidarData.WritePoints(Detections[Channel]);
120 }
121 LidarData.WriteChannelCount(OutPointsPerChannel);
122 }
123
124 /// Copies the kept detections into a flat x, y, z, intensity array (editor debug point cloud).
125 template <typename DetectionT>
126 static void CopyDetectionsToPointCloud(
127 const std::vector<std::vector<DetectionT>> &Detections,
128 const std::vector<uint32_t> &InPointsPerChannel,
129 TArray<float> &PointCloud)
130 {
131 static_assert(sizeof(DetectionT) == sizeof(float) * 4, "Detection must be 4 packed floats");
132 static_assert(std::is_trivially_copyable_v<DetectionT>, "Detection must be trivially copyable");
133
134 const uint32_t TotalPoints = std::accumulate(InPointsPerChannel.begin(), InPointsPerChannel.end(), 0u);
135 PointCloud.SetNumUninitialized(static_cast<int32>(TotalPoints * 4));
136
137 float *Dest = PointCloud.GetData();
138 for (const auto &ChannelDetections : Detections)
139 {
140 if (ChannelDetections.empty())
141 {
142 continue;
143 }
144 FMemory::Memcpy(Dest, ChannelDetections.data(), ChannelDetections.size() * sizeof(DetectionT));
145 Dest += ChannelDetections.size() * 4;
146 }
147 }

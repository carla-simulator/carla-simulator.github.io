67 /// Return a unit-length copy of this vector. When the vector's length
68 /// is at or below `epsilon`, the input is returned unchanged (no
69 /// runtime exception, no NaN). Callers that need the strict
70 /// `1.0 / length` normalization should pass `epsilon = 0`.
71 inline Vector3D MakeUnitVector(
72 const float epsilon = 2.0f * std::numeric_limits<float>::epsilon()) const {
73 const float length = Length();
74 if (length <= std::max(epsilon, 0.0f)) {
75 return *this;
76 }
77 const float k = 1.0f / length;
78 return Vector3D(x * k, y * k, z * k);
79 }

80void AVehicleStatusSensor::CollectAndStream(float /*DeltaSeconds*/)
81{
82 const UCarlaEpisode *CurrentEpisode = UCarlaStatics::GetCurrentEpisode(GetWorld());
83 if (!OwningVehicle.IsValid() && (CurrentEpisode != nullptr) && !CurrentEpisode->bIsPrimaryServer)
84 {
85 // A Multi-GPU mirror is attached in the frame it is spawned, and streams
86 // from the primary sensor's first tick, not the timer's.
87 OwningVehicle = FindAttachmentParentVehicle();
88 }
89 if (!OwningVehicle.IsValid())
90 {
91 return;
92 }
93
94 ACarlaWheeledVehicle* Vehicle = OwningVehicle.Get();
95 if (!IsValid(Vehicle))
96 {
97 return;
98 }
99
100 // Update cached velocity info
101 SetVelocityInfoToLocal(Vehicle);
102
103 // Get Max Steering
104 const auto MaxSteerAngleInRadians = FMath::DegreesToRadians(Vehicle->GetMaximumSteerAngle());
105
106 // Control flags (b0 = reverse, b1 = manual gear shift)
107 const auto& Control = Vehicle->GetVehicleControl();
108 const auto Flags = (Control.bReverse ? 0x01 : 0) | (Control.bManualGearShift ? 0x02 : 0);
109
110 // Turn mask (b0 = left blinker, b1 = right blinker, b2 = hazards)
111 const auto& Lights = Vehicle->GetVehicleLightState();
112 const bool bHazard = Lights.LeftBlinker && Lights.RightBlinker;
113 const auto TurnMask = (Lights.LeftBlinker ? 0x01 : 0) | (Lights.RightBlinker ? 0x02 : 0) | (bHazard ? 0x04 : 0);
114
115 const FVehicleStatusData Msg
116 {
117 static_cast<double>(GetWorld()->GetTimeSeconds()),
118 static_cast<float>(VelocityInfo.GetSpeed()),
119 static_cast<float>(VelocityInfo.Velocity.X),
120 static_cast<float>(VelocityInfo.Velocity.Y),
121 static_cast<float>(VelocityInfo.Velocity.Z),
122 static_cast<float>(VelocityInfo.AngularVelocity.X),
123 static_cast<float>(VelocityInfo.AngularVelocity.Y),
124 static_cast<float>(VelocityInfo.AngularVelocity.Z),
125 static_cast<float>(Vehicle->GetActorRotation().Pitch),
126 static_cast<float>(Vehicle->GetActorRotation().Yaw),
127 static_cast<float>(Vehicle->GetActorRotation().Roll),
128 static_cast<float>(Vehicle->GetVehicleControl().Steer * MaxSteerAngleInRadians),
129 static_cast<int32_t>(Vehicle->GetVehicleCurrentGear()),
130 static_cast<uint8_t>(TurnMask),
131 static_cast<uint8_t>(Flags),
132 };
133
134 auto DataStream = GetDataStream(*this);
135
136 // ROS2 forwarding
137#if defined(WITH_ROS2)
138 if (auto ROS2 = carla::ros2::ROS2::GetInstance(); ROS2->IsEnabled())
139 {
140 TRACE_CPUPROFILER_EVENT_SCOPE_STR("ROS2 Send");
141 auto StreamId = carla::streaming::detail::token_type(GetToken()).get_stream_id();
142 ROS2->ProcessDataFromStatusSensor(
143 DataStream.GetSensorType(),
144 StreamId,
145 GetActorTransform(),
146 Msg,
147 Vehicle,
148 this);
149 }
150#endif
151
152 // UE client streaming (field-wise VehicleStatusSerializer overload)
153 {
154 TRACE_CPUPROFILER_EVENT_SCOPE_STR("AVehicleStatusSensor Stream Send");
155 DataStream.SerializeAndSend(
156 *this,
157 Msg.timestamp,
158 Msg.speed_mps,
159 Msg.vel_x_mps, Msg.vel_y_mps, Msg.vel_z_mps,
160 Msg.angVel_x_mps, Msg.angVel_y_mps, Msg.angVel_z_mps,
161 Msg.rot_pitch, Msg.rot_yaw, Msg.rot_roll,
162 Msg.steer,
163 Msg.gear,
164 Msg.turn_mask,
165 Msg.control_flags);
166 }
167}

169void AVehicleStatusSensor::SetVelocityInfoToLocal(const AActor* VehicleActor)
170{
171 if (!VehicleActor)
172 {
173 return;
174 }
175
176 // World linear velocity (cm/s)
177 const FVector WorldVel_cmps = VehicleActor->GetVelocity();
178
179 // Convert to m/s, then rotate into local space
180 const FQuat InvRot = VehicleActor->GetActorTransform().GetRotation().Inverse();
181 VelocityInfo.Velocity = InvRot.RotateVector(CmpsToMps(WorldVel_cmps));
182
183 // Physics angular velocity (rad/s, world space); a Multi-GPU mirror, which
184 // never simulates, has the primary's, when the primary's body simulates.
185 FVector WorldAngVel = FVector::ZeroVector;
186 if (const auto* MirroredVehicle = Cast<ACarlaWheeledVehicle>(VehicleActor);
187 (MirroredVehicle != nullptr) && MirroredVehicle->HasMirroredMotion())
188 {
189 if (MirroredVehicle->IsMirroredSimulatingPhysics())
190 {
191 WorldAngVel = CarlaGetActorAngularVelocityInRadians(*VehicleActor);
192 }
193 }
194 else if (UPrimitiveComponent* RootPrim = Cast<UPrimitiveComponent>(VehicleActor->GetRootComponent()))
195 {
196 if (RootPrim->IsSimulatingPhysics())
197 {
198 WorldAngVel = RootPrim->GetPhysicsAngularVelocityInRadians();
199 }
200 }
201
202 // Rotate into local space
203 VelocityInfo.AngularVelocity = InvRot.RotateVector(WorldAngVel);
204
205 // Convert angular velocity vector into a Rotator for convenience
206 VelocityInfo.RotationRate = VelocityInfo.AngularVelocity.Rotation();
207}

37void CarlaRecorderEventsDel::Write(std::ostream &OutFile)
38{
39 // write the packet id
40 WriteValue<char>(OutFile, static_cast<char>(CarlaRecorderPacketId::EventDel));
41
42 std::streampos PosStart = OutFile.tellp();
43
44 // write a dummy packet size
45 uint32_t Total = 0;
46 WriteValue<uint32_t>(OutFile, Total);
47
48 // write total records
49 Total = std::min<size_t>(Events.size(), std::numeric_limits<uint16_t>::max());
50 WriteValue<uint16_t>(OutFile, Total);
51
52 for (uint16_t i=0; i<Total; ++i)
53 {
54 Events[i].Write(OutFile);
55 }
56
57 // write the real packet size
58 std::streampos PosEnd = OutFile.tellp();
59 Total = PosEnd - PosStart - sizeof(uint32_t);
60 OutFile.seekp(PosStart, std::ios::beg);
61 WriteValue<uint32_t>(OutFile, Total);
62 OutFile.seekp(PosEnd, std::ios::beg);
63}

32// A one-lane road (0 to 100 m) that enters a junction, in which it either
33// goes straight on for 20 m into a second road or turns right into a third.
34// The InMemoryMap counts a junction only if a road has two paths into or out
35// of it, and the junction's id must not be a road id, or CARLA links the
36// incoming road straight to that road.

109// Localizes a vehicle standing just before the junction, facing it, then lets
110// `forget` drop the vehicle from the stage and destroys the map. Returns the
111// waypoints of the map that are still alive while the stage lives on.
112size_t WaypointsAliveAfter(
113 const std::function<void(ctm::LocalizationStage &, ctm::ActorId)> &forget) {
114 const ctm::ActorId vehicle = 1u;
115 const std::vector<ctm::ActorId> vehicle_id_list{vehicle};
116 ctm::BufferMap buffer_map;
117 ctm::SimulationState simulation_state;
118 ctm::TrackTraffic track_traffic;
119 std::unordered_map<ctm::ActorId, std::pair<float, bool>> large_vehicles;
120 ctm::Parameters parameters;
121 std::vector<ctm::ActorId> marked_for_removal;
122 ctm::LocalizationFrame output_array(1u);
123 ctm::RandomGenerator random_device(0u);
124
125 auto world_map = carla::MakeShared<carla::client::Map>("junction", RoadIntoJunction());
126 ctm::LocalMapPtr local_map = std::make_shared<ctm::InMemoryMap>(world_map);
127 local_map->SetUp();
128 std::vector<std::weak_ptr<ctm::SimpleWaypoint>> waypoints;
129 for (const auto &waypoint : local_map->GetDenseTopology()) {
130 waypoints.emplace_back(waypoint);
131 }
132
133 ctm::KinematicState state;
134 state.location = carla::geom::Location(97.0f, 1.75f, 0.0f);
135 state.rotation = carla::geom::Rotation(0.0f, 0.0f, 0.0f);
136 state.velocity = carla::geom::Vector3D(0.0f, 0.0f, 0.0f);
137 state.speed_limit = 50.0f;
138 state.physics_enabled = true;
139 state.is_dormant = false;
140 ctm::StaticAttributes attributes;
141 attributes.actor_type = ctm::ActorType::Vehicle;
142 attributes.half_length = 2.0f;
143 attributes.half_width = 1.0f;
144 attributes.half_height = 0.75f;
145 simulation_state.AddActor(vehicle, state, attributes,
146 ctm::TrafficLightState{ctm::TLS::Green, false});
147
148 ctm::LocalizationStage stage(vehicle_id_list, buffer_map, simulation_state,
149 track_traffic, local_map, large_vehicles,
150 parameters, marked_for_removal, output_array,
151 random_device);
152 // The first update fills the vehicle's waypoint buffer, the second finds the
153 // junction ahead in it.
154 stage.Update(0u);
155 stage.Update(0u);
156 EXPECT_TRUE(output_array.front().is_at_junction_entrance);
157 EXPECT_NE(output_array.front().junction_end_point, nullptr);
158 EXPECT_NE(output_array.front().safe_point, nullptr);
159
160 forget(stage, vehicle);
161 output_array.clear();
162 buffer_map.clear();
163 local_map.reset();
164
165 EXPECT_FALSE(waypoints.empty());
166 size_t alive = 0u;
167 for (const auto &waypoint : waypoints) {
168 alive += waypoint.expired() ? 0u : 1u;
169 }
170 return alive;
171}

175TEST(LocalizationStage, remove_actor_forgets_its_junction_entrance) {
176 EXPECT_EQ(WaypointsAliveAfter([](ctm::LocalizationStage &stage, ctm::ActorId vehicle) {
177 stage.RemoveActor(vehicle);
178 }), 0u);

33 // A vertex carrying the elevation the reference line was sampled at; the
34 // planar algorithms ignore z, the OpenDRIVE elevation profile reads it back.
35 struct Point3 {
36 double x = 0.0;
37 double y = 0.0;
38 double z = 0.0;
39 };

41 using Line3 = std::vector<Point3>;

43 inline Point To2D(const Point3 &p) {
44 return Point(p.x, p.y);
45 }

47 inline LineString To2D(const Line3 &line) {
48 LineString out;
49 out.reserve(line.size());
50 for (const auto &p : line) {
51 bg::append(out, To2D(p));
52 }
53 return out;
54 }

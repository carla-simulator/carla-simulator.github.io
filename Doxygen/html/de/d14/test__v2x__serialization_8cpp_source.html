22// The custom V2X sensor ships a fixed 100-byte binary blob across the RPC
23// boundary via MsgPack (client -> server "send" path). These tests pin the
24// blob contract, the in-memory message containers, and the byte-for-byte
25// fidelity of the V2X serializers that feed the sensor data stream.

124// The CAM wire type must carry the optional pathDeltaTime by value (with an
125// availability flag) instead of a raw pointer: CAMDataSerializer memcpys the
126// whole message onto the data stream, so a server-side pointer must never reach
127// the wire.
128TEST(PathPoint, delta_time_is_an_inline_value_not_a_pointer) {
129 static_assert(
130 std::is_same<decltype(ITSContainer::PathPoint_t::pathDeltaTime),
131 ITSContainer::PathDeltaTime_t>::value,
132 "pathDeltaTime must be stored by value so no server pointer reaches the wire");
133 static_assert(
134 std::is_trivially_copyable<ITSContainer::PathPoint_t>::value,
135 "PathPoint_t is copied byte-for-byte by the V2X serializer");
136
137 ITSContainer::PathPoint_t point{};
138 EXPECT_FALSE(point.pathDeltaTimeAvailable);
139 EXPECT_EQ(point.pathDeltaTime, 0);
140}

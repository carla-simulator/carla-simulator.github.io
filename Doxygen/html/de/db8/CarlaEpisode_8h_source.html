284 bool DestroyActor(carla::rpc::ActorId ActorId)
285 {
286 if (bIsPrimaryServer)
287 {
288 GetFrameData().AddEvent(
289 CarlaRecorderEventDel{ActorId});
290 }
291 if (Recorder->IsEnabled())
292 {
293 // recorder event
294 CarlaRecorderEventDel RecEvent{ActorId};
295 Recorder->AddEvent(std::move(RecEvent));
296 }
297
298 return ActorDispatcher->DestroyActor(ActorId);
299 }

59void CarlaRecorderWeathers::Write(std::ostream &OutFile)
60{
61 if (Weathers.size() == 0)
62 {
63 return;
64 }
65
66 // write the packet id
67 WriteValue<char>(OutFile, static_cast<char>(CarlaRecorderPacketId::Weather));
68
69 // write the packet size
70 uint32_t Total = 2 + Weathers.size() * sizeof(CarlaRecorderWeather);
71 WriteValue<uint32_t>(OutFile, Total);
72
73 // write total records
74 Total = Weathers.size();
75 WriteValue<uint16_t>(OutFile, Total);
76
77 // write records
78 for (auto& Weather : Weathers)
79 {
80 Weather.Write(OutFile);
81 }
82}

84void CarlaRecorderWeathers::Read(std::istream &InFile)
85{
86 uint16_t Total = 0;
87 ReadValue<uint16_t>(InFile, Total);
88 for (uint16_t i = 0; (i < Total) && InFile; ++i)
89 {
90 CarlaRecorderWeather Record;
91 Record.Read(InFile);
92 Add(Record);
93 }
94}

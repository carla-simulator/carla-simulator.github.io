48/// What a LOAD_MAP carries besides the map: for a world the primary
49/// generated from OpenDRIVE (generate_opendrive_world), the OpenDRIVE and the
50/// generation parameters (rpc::OpendriveGenerationParameters, msgpack).
51struct LoadMapExtras {
52 std::optional<std::string> opendrive;
53 std::optional<std::string> generation_parameters;

55 bool operator==(const LoadMapExtras &rhs) const {
56 return (opendrive == rhs.opendrive) && (generation_parameters == rhs.generation_parameters);
57 }

60/// LOAD_MAP payload: the NUL-terminated map path, the load id, then the
61/// extras as [u8 kind][u32 size][bytes] fields; a reader skips the kinds it
62/// does not know.
63inline std::string MakeLoadMapPayload(
64 std::string_view map,
65 load_map_id_type load_id,
66 const LoadMapExtras &extras = {}) {
67 std::string payload(map);
68 payload.push_back('\0');
69 payload.append(reinterpret_cast<const char *>(&load_id), sizeof(load_id));
70 const auto append_field = [&payload](uint8_t kind, const std::optional<std::string> &value) {
71 if (!value) {
72 return;
73 }
74 const uint32_t size = static_cast<uint32_t>(value->size());
75 payload.push_back(static_cast<char>(kind));
76 payload.append(reinterpret_cast<const char *>(&size), sizeof(size));
77 payload.append(*value);
78 };
79 append_field(1u, extras.opendrive);
80 append_field(2u, extras.generation_parameters);
81 return payload;
82}

90inline LoadMapRequest ParseLoadMapPayload(std::string_view payload) {
91 LoadMapRequest request;
92 const auto end_of_map = payload.find('\0');
93 request.map = std::string(payload.substr(0u, end_of_map));
94 if (
95 (end_of_map == std::string_view::npos) ||
96 (payload.size() - end_of_map - 1u < sizeof(load_map_id_type))) {
97 return request;
98 }
99 std::memcpy(&request.load_id, payload.data() + end_of_map + 1u, sizeof(request.load_id));
100 std::size_t offset = end_of_map + 1u + sizeof(load_map_id_type);
101 while (payload.size() - offset >= 1u + sizeof(uint32_t)) {
102 const uint8_t kind = static_cast<uint8_t>(payload[offset]);
103 uint32_t size = 0u;
104 std::memcpy(&size, payload.data() + offset + 1u, sizeof(size));
105 offset += 1u + sizeof(uint32_t);
106 if (size > payload.size() - offset) {
107 // Cut short: what it carries is not usable.
108 request.extras = {};
109 break;
110 }
111 const std::string value(payload.substr(offset, size));
112 offset += size;
113 if (kind == 1u) {
114 request.extras.opendrive = value;
115 } else if (kind == 2u) {
116 request.extras.generation_parameters = value;
117 }
118 }
119 return request;
120}

122/// Episode-ready message: the marker followed by the load id of the LOAD_MAP
123/// that started the episode, so the primary can ignore a stale one.
124inline std::string MakeEpisodeReadyMessage(load_map_id_type load_id) {
125 std::string message(kEpisodeReadyMarker);
126 message.append(reinterpret_cast<const char *>(&load_id), sizeof(load_id));
127 return message;
128}
129
130/// Load id of an episode-ready message (0 for a bare marker), or nullopt if
131/// @a payload is not one.
132inline std::optional<load_map_id_type> ParseEpisodeReadyMessage(std::string_view payload) {
133 if (payload.substr(0u, kEpisodeReadyMarker.size()) != kEpisodeReadyMarker) {
134 return std::nullopt;
135 }
136 const auto load_id_bytes = payload.substr(kEpisodeReadyMarker.size());
137 if (load_id_bytes.empty()) {
138 return load_map_id_type{0u};
139 }
140 if (load_id_bytes.size() != sizeof(load_map_id_type)) {
141 return std::nullopt;
142 }
143 load_map_id_type load_id = 0u;
144 std::memcpy(&load_id, load_id_bytes.data(), sizeof(load_id));
145 return load_id;
146}

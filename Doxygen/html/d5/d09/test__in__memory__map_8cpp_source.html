7// The Traffic Manager's InMemoryMap must free its waypoint graph when it is
8// destroyed. Its SimpleWaypoints link to each other (next, previous, left and
9// right) through shared pointers, so the graph is freed only if the map breaks

63TEST(InMemoryMap, frees_its_waypoints_when_destroyed) {
64 const auto world_map = carla::MakeShared<carla::client::Map>("two_lanes", TwoLaneRoad());
65
66 std::vector<std::weak_ptr<SimpleWaypoint>> waypoints;
67 bool linked_along = false;
68 bool linked_across = false;
69 {
70 auto local_map = std::make_shared<InMemoryMap>(world_map);
71 local_map->SetUp();
72 for (const auto &waypoint : local_map->GetDenseTopology()) {
73 linked_along = linked_along || !waypoint->GetNextWaypoint().empty();
74 linked_across = linked_across ||
75 waypoint->GetLeftWaypoint() != nullptr ||
76 waypoint->GetRightWaypoint() != nullptr;
77 waypoints.emplace_back(waypoint);
78 }
79 }
80
81 // The graph had the links that make cycles...
82 ASSERT_FALSE(waypoints.empty());
83 ASSERT_TRUE(linked_along);
84 ASSERT_TRUE(linked_across);
85 // ... and none of its waypoints outlived the map.
86 size_t alive = 0u;
87 for (const auto &waypoint : waypoints) {
88 alive += waypoint.expired() ? 0u : 1u;
89 }
90 EXPECT_EQ(alive, 0u) << alive << " of " << waypoints.size()
91 << " waypoints outlived their InMemoryMap";
92}

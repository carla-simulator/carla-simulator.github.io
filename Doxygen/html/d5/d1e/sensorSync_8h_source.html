22 /// One record of frame data packet 26: a sensor of the primary that ticks
23 /// in this frame, and the delta of that tick.
24 struct SensorTick {
25 uint32_t actor_id = 0u;
26 float delta_seconds = 0.0f;
27 };

29 /// Frame data packet 26: every sensor of the primary that ticks in this
30 /// frame. A secondary's mirrors tick on exactly these frames, with these
31 /// deltas, whatever their own tick interval says, so their sensor_tick
32 /// phase is the primary's even if they were created later. Written on every
33 /// frame, even when empty.
34 struct SensorTicks {
35
36 static constexpr uint8_t kPacketId = 26u;
37
38 static constexpr uint16_t kRecordSizeV1 = 8u;
39
40 std::vector<SensorTick> ticks;
41
42 void Write(std::ostream &out) const {
43 const uint32_t count = static_cast<uint32_t>(ticks.size());
44 WriteFrameDataHeader(out, kPacketId, sizeof(uint32_t) + sizeof(uint16_t) + count * kRecordSizeV1);
45 WriteFixedListHeader(out, count, kRecordSizeV1);
46 for (const SensorTick &tick : ticks) {
47 detail::WritePod(out, tick.actor_id);
48 detail::WritePod(out, tick.delta_seconds);
49 }
50 }
51
52 bool Read(std::istream &in, uint32_t size) {
53 ticks.clear();
54 BoundedReader reader(in, size);
55 uint32_t count = 0u;
56 uint16_t record_size = 0u;
57 if (!reader.ReadFixedListHeader(kRecordSizeV1, count, record_size)) {
58 return false;
59 }
60 ticks.reserve(count);
61 for (uint32_t i = 0u; i < count; ++i) {
62 SensorTick tick;
63 const bool ok = reader.ReadRecord(record_size, [&](BoundedReader &record) {
64 return record.Read(tick.actor_id) && record.Read(tick.delta_seconds);
65 });
66 if (!ok) {
67 return false;
68 }
69 ticks.push_back(tick);
70 }
71 return true;
72 }
73 };

75 /// An IMU's history: its two previous locations (world, cm) and the delta
76 /// between them; its accelerometer is a second difference over them.
77 struct ImuHistory {
78 std::array<double, 3> older_location = {0.0, 0.0, 0.0};
79 std::array<double, 3> last_location = {0.0, 0.0, 0.0};
80 float last_delta_seconds = 0.0f;
81 };

83 /// One record of frame data packet 27: the runtime state a sensor of the
84 /// primary starts this frame's tick from. A mirror created at another frame
85 /// starts from its own defaults; with this it computes exactly what the
86 /// primary's sensor would.
87 struct SensorState {
88 uint32_t actor_id = 0u;
89 /// URandomEngine (noise, lidar drop-off, radar ray directions).
90 std::optional<uint32_t> random_engine;
91 /// Ray-cast lidars: where this tick's sweep starts (radians).
92 std::optional<float> lidar_horizontal_angle;
93 std::optional<ImuHistory> imu_history;
94 /// Radar: its previous location (world, cm), for its own velocity.
95 std::optional<std::array<double, 3>> radar_previous_location;
96 /// IMU: its rotation relative to its parent (x, y, z, w), which turns the
97 /// parent's angular velocity into the gyroscope's. A mirror is attached
98 /// rigidly, so a spring arm's rotation would differ.
99 std::optional<std::array<double, 4>> imu_local_rotation;
100
101 bool empty() const {
102 return !random_engine && !lidar_horizontal_angle && !imu_history && !radar_previous_location &&
103 !imu_local_rotation;
104 }
105 };

107 /// Frame data packet 27, for the sensors of packet 26 that carry state.
108 struct SensorStates {
109
110 static constexpr uint8_t kPacketId = 27u;
111
112 /// Field kinds; a reader skips the kinds it does not know.
113 enum Field : uint8_t {
114 kRandomEngine = 1u,
115 kLidarHorizontalAngle = 2u,
116 kImuHistory = 3u,
117 kRadarPreviousLocation = 4u,
118 kImuLocalRotation = 5u
119 };
120
121 std::vector<SensorState> states;
122
123 void Write(std::ostream &out) const {
124 std::ostringstream payload;
125 detail::WritePod(payload, static_cast<uint32_t>(states.size()));
126 for (const SensorState &state : states) {
127 std::ostringstream record;
128 detail::WritePod(record, state.actor_id);
129 if (state.random_engine) {
130 WriteField(record, kRandomEngine, Bytes(*state.random_engine));
131 }
132 if (state.lidar_horizontal_angle) {
133 WriteField(record, kLidarHorizontalAngle, Bytes(*state.lidar_horizontal_angle));
134 }
135 if (state.imu_history) {
136 std::ostringstream value;
137 WriteVector(value, state.imu_history->older_location);
138 WriteVector(value, state.imu_history->last_location);
139 detail::WritePod(value, state.imu_history->last_delta_seconds);
140 WriteField(record, kImuHistory, value.str());
141 }
142 if (state.radar_previous_location) {
143 std::ostringstream value;
144 WriteVector(value, *state.radar_previous_location);
145 WriteField(record, kRadarPreviousLocation, value.str());
146 }
147 if (state.imu_local_rotation) {
148 std::ostringstream value;
149 for (double axis : *state.imu_local_rotation) {
150 detail::WritePod(value, axis);
151 }
152 WriteField(record, kImuLocalRotation, value.str());
153 }
154 WriteSizedBytes(payload, record.str());
155 }
156 WriteFrameDataPacket(out, kPacketId, payload.str());
157 }
158
159 bool Read(std::istream &in, uint32_t size) {
160 states.clear();
161 BoundedReader reader(in, size);
162 uint32_t count = 0u;
163 if (!reader.ReadVariableListCount(count)) {
164 return false;
165 }
166 for (uint32_t i = 0u; i < count; ++i) {
167 SensorState state;
168 const bool ok = reader.ReadSizedRecord([&](BoundedReader &record) {
169 return record.Read(state.actor_id) && record.ReadFields([&](uint8_t kind, BoundedReader &value) {
170 switch (kind) {
171 case kRandomEngine: {
172 uint32_t engine = 0u;
173 if (!value.Read(engine)) {
174 return false;
175 }
176 state.random_engine = engine;
177 return true;
178 }
179 case kLidarHorizontalAngle: {
180 float angle = 0.0f;
181 if (!value.Read(angle)) {
182 return false;
183 }
184 state.lidar_horizontal_angle = angle;
185 return true;
186 }
187 case kImuHistory: {
188 ImuHistory history;
189 if (!ReadVector(value, history.older_location) ||
190 !ReadVector(value, history.last_location) ||
191 !value.Read(history.last_delta_seconds)) {
192 return false;
193 }
194 state.imu_history = history;
195 return true;
196 }
197 case kRadarPreviousLocation: {
198 std::array<double, 3> location;
199 if (!ReadVector(value, location)) {
200 return false;
201 }
202 state.radar_previous_location = location;
203 return true;
204 }
205 case kImuLocalRotation: {
206 std::array<double, 4> rotation;
207 for (double &axis : rotation) {
208 if (!value.Read(axis)) {
209 return false;
210 }
211 }
212 state.imu_local_rotation = rotation;
213 return true;
214 }
215 default:
216 return true; // a kind of a newer build: skipped
217 }
218 });
219 });
220 if (!ok) {
221 return false;
222 }
223 states.push_back(std::move(state));
224 }
225 return true;
226 }

245 if (!reader.Read(value)) {
246 return false;
247 }
248 }
249 return true;

253 /// The state of a minstd_rand: the value its next draw is computed from.
254 inline uint32_t GetEngineState(const std::minstd_rand &engine) {
255 std::minstd_rand copy = engine;
256 // copy() returns 48271 * x mod (2^31 - 1); 1899818559 is 48271's
257 // inverse modulo 2^31 - 1, so this recovers x.
258 const uint64_t next = copy();
259 return static_cast<uint32_t>((next * uint64_t{1899818559u}) % uint64_t{2147483647u});
260 }
261
262 /// Makes @a engine continue from @a state, as GetEngineState returned it.
263 /// False for a value no minstd_rand can hold.
264 inline bool SetEngineState(std::minstd_rand &engine, uint32_t state) {
265 if ((state == 0u) || (state >= std::minstd_rand::modulus)) {
266 return false;
267 }
268 engine.seed(state);
269 return true;
270 }

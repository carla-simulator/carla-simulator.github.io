24 template <typename Packet>
25 bool RoundTrip(const Packet &sent, Packet &got) {
26 std::ostringstream out;
27 sent.Write(out);
28 const std::string bytes = out.str();
29 if (static_cast<uint8_t>(bytes[0]) != Packet::kPacketId) {
30 return false;
31 }
32 uint32_t size = 0u;
33 std::memcpy(&size, bytes.data() + 1, sizeof(size));
34 std::istringstream in(bytes.substr(carla::multigpu::kFrameDataHeaderSize));
35 return got.Read(in, size);
36 }

40TEST(MultiGpuWorldState, globals_round_trip) {

44 sent.day_night_cycle = false;

29FMeshDescription UMapGenFunctionLibrary::BuildMeshDescriptionFromData(
30 const FProceduralCustomMesh& Data,
31 const TArray<FProcMeshTangent>& ParamTangents,
32 UMaterialInstance* MaterialInstance )
33{
34
35 int32 VertexCount = Data.Vertices.Num();
36 int32 VertexInstanceCount = Data.Triangles.Num();
37 int32 PolygonCount = Data.Vertices.Num()/3;
38
39 FMeshDescription MeshDescription;
40 FStaticMeshAttributes AttributeGetter(MeshDescription);
41 AttributeGetter.Register();
42
43 auto PolygonGroupNames = AttributeGetter.GetPolygonGroupMaterialSlotNames();
44 auto VertexPositions = AttributeGetter.GetVertexPositions();
45 auto Tangents = AttributeGetter.GetVertexInstanceTangents();
46 auto BinormalSigns = AttributeGetter.GetVertexInstanceBinormalSigns();
47 auto Normals = AttributeGetter.GetVertexInstanceNormals();
48 auto Colors = AttributeGetter.GetVertexInstanceColors();
49 auto UVs = AttributeGetter.GetVertexInstanceUVs();
50
51 // Calculate the totals for each ProcMesh element type
52 FPolygonGroupID PolygonGroupForSection;
53 MeshDescription.ReserveNewVertices(VertexCount);
54 MeshDescription.ReserveNewVertexInstances(VertexInstanceCount);
55 MeshDescription.ReserveNewPolygons(PolygonCount);
56 MeshDescription.ReserveNewEdges(PolygonCount * 2);
57 UVs.SetNumChannels(4);
58
59 // Create Materials
60 TMap<UMaterialInterface*, FPolygonGroupID> UniqueMaterials;
61 const int32 NumSections = 1;
62 UniqueMaterials.Reserve(1);
63 FPolygonGroupID NewPolygonGroup = MeshDescription.CreatePolygonGroup();
64
65 if( MaterialInstance != nullptr ){
66 UMaterialInterface *Material = MaterialInstance;
67 UniqueMaterials.Add(Material, NewPolygonGroup);
68 PolygonGroupNames[NewPolygonGroup] = Material->GetFName();
69 }else{
70 UE_LOG(LogCarla, Error, TEXT("MaterialInstance is nullptr"));
71 }
72 PolygonGroupForSection = NewPolygonGroup;
73
74
75
76 // Create the vertex
77 int32 NumVertex = Data.Vertices.Num();
78 TMap<int32, FVertexID> VertexIndexToVertexID;
79 VertexIndexToVertexID.Reserve(NumVertex);
80 for (int32 VertexIndex = 0; VertexIndex < NumVertex; ++VertexIndex)
81 {
82 auto &Vert = Data.Vertices[VertexIndex];
83 auto VertexID = MeshDescription.CreateVertex();
84 VertexPositions[VertexID] = FVector3f(Vert);
85 VertexIndexToVertexID.Add(VertexIndex, VertexID);
86 }
87
88 // Create the VertexInstance
89 int32 NumIndices = Data.Triangles.Num();
90 int32 NumTri = NumIndices / 3;
91 TMap<int32, FVertexInstanceID> IndiceIndexToVertexInstanceID;
92 IndiceIndexToVertexInstanceID.Reserve(NumVertex);
93 for (int32 IndiceIndex = 0; IndiceIndex < NumIndices; IndiceIndex++)
94 {
95 const int32 VertexIndex = Data.Triangles[IndiceIndex];
96 const FVertexID VertexID = VertexIndexToVertexID[VertexIndex];
97 const FVertexInstanceID VertexInstanceID =
98 MeshDescription.CreateVertexInstance(VertexID);
99 IndiceIndexToVertexInstanceID.Add(IndiceIndex, VertexInstanceID);
100 Normals[VertexInstanceID] = FVector3f(Data.Normals[VertexIndex]);
101
102 if(ParamTangents.Num() == Data.Vertices.Num())
103 {
104 Tangents[VertexInstanceID] = FVector3f(ParamTangents[VertexIndex].TangentX);
105 BinormalSigns[VertexInstanceID] =
106 ParamTangents[VertexIndex].bFlipTangentY ? -1.f : 1.f;
107 }else{
108
109 }
110 Colors[VertexInstanceID] = FLinearColor(0,0,0);
111 if(Data.UV0.Num() == Data.Vertices.Num())
112 {
113 UVs.Set(VertexInstanceID, 0, FVector2f(Data.UV0[VertexIndex]));
114 }else{
115 UVs.Set(VertexInstanceID, 0, FVector2f(0,0));
116 }
117 UVs.Set(VertexInstanceID, 1, FVector2f(0,0));
118 UVs.Set(VertexInstanceID, 2, FVector2f(0,0));
119 UVs.Set(VertexInstanceID, 3, FVector2f(0,0));
120 }
121
122 for (int32 TriIdx = 0; TriIdx < NumTri; TriIdx++)
123 {
124 FVertexID VertexIndexes[3];
125 TArray<FVertexInstanceID> VertexInstanceIDs;
126 VertexInstanceIDs.SetNum(3);
127
128 for (int32 CornerIndex = 0; CornerIndex < 3; ++CornerIndex)
129 {
130 const int32 IndiceIndex = (TriIdx * 3) + CornerIndex;
131 const int32 VertexIndex = Data.Triangles[IndiceIndex];
132 VertexIndexes[CornerIndex] = VertexIndexToVertexID[VertexIndex];
133 VertexInstanceIDs[CornerIndex] =
134 IndiceIndexToVertexInstanceID[IndiceIndex];
135 }
136
137 // Insert a polygon into the mesh
138 MeshDescription.CreatePolygon(NewPolygonGroup, VertexInstanceIDs);
139
140 }
141
142 return MeshDescription;
143}

214FVector2D UMapGenFunctionLibrary::GetTransversemercProjection(float lat, float lon, float lat0, float lon0)
215{
216 // earth radius in m
217 const float R = 6373000.0f;
218 float latt = FMath::DegreesToRadians(lat);
219 float lonn = FMath::DegreesToRadians(lon - lon0);
220 float latt0 = FMath::DegreesToRadians(lat0);
221 float eps = atan(tan(latt)/cos(lonn));
222 float nab = asinh(sin(lonn)/sqrt(tan(latt)*tan(latt)+cos(lonn)*cos(lonn)));
223 float x = R*nab;
224 float y = R*eps;
225 float eps0 = atan(tan(latt0)/cos(0));
226 float nab0 = asinh(sin(0)/sqrt(tan(latt0)*tan(latt0)+cos(0)*cos(0)));
227 float x0 = R*nab0;
228 float y0 = R*eps0;
229 FVector2D Result = FVector2D(x, -(y - y0)) * OSMToCentimetersScaleFactor;
230
231 return Result;
232}

264 UGameplayStatics::GetAllActorsOfClass(World, AStaticMeshActor::StaticClass(), FoundActors);

279 {
280
281 if(CurrentPair.Value.Num() > MinNumOfInstancesToBeChanged)

283 TArray<FTransform> TransformsToBeInstanced;
284 for( AStaticMeshActor* SMActor : CurrentPair.Value )

291 if(InstancedStaticMeshActor)

293 InstancedStaticMeshActor->GetInstancedStaticMeshComponent()->SetStaticMesh(CurrentPair.Key);
294 InstancedStaticMeshActor->GetInstancedStaticMeshComponent()->AddInstances(TransformsToBeInstanced, false, true);
295 Result.Add(InstancedStaticMeshActor);
296 }
297 }
298 }

383FVector2D UMapGenFunctionLibrary::InverseTransverseMercatorProjection(float x, float y, float lat0, float lon0)
384{
385 const float R = 6373000.0f;
386
387 x /= OSMToCentimetersScaleFactor;
388 y = -y / OSMToCentimetersScaleFactor;
389
390 float latt0 = FMath::DegreesToRadians(lat0);
391 float eps0 = latt0; // atan(tan(latt0)/cos(0));
392 float y0 = R * eps0;
393
394 float eps = (y + y0) / R;
395 float nab = x / R;
396
397 float lat = FMath::RadiansToDegrees(atan(sin(eps)/sqrt(tan(nab)*tan(nab)+cos(eps)*cos(eps))));
398 float lon = lon0 + FMath::RadiansToDegrees(atan(sinh(nab) / cos(eps)));
399
400 return FVector2D(lat, lon);
401}

433void UMapGenFunctionLibrary::SmoothVerticesDeep(
434 TArray<FVector>& Vertices,
435 const TArray<int32>& Indices,
436 int Depth, // Number of neighbor levels
437 int NumIterations,
438 float SmoothingFactor // Blend between original and averaged
439)
440{
441 const int32 NumVertices = Vertices.Num();
442
443 // Step 1: Build adjacency map
444 TMap<int32, TSet<int32>> VertexNeighbors;
445 const int32 NumTriangles = Indices.Num() / 3;
446
447 for (int32 i = 0; i < NumTriangles; ++i)
448 {
449 int32 I0 = Indices[i * 3 + 0];
450 int32 I1 = Indices[i * 3 + 1];
451 int32 I2 = Indices[i * 3 + 2];
452
453 VertexNeighbors.FindOrAdd(I0).Add(I1);
454 VertexNeighbors.FindOrAdd(I0).Add(I2);
455
456 VertexNeighbors.FindOrAdd(I1).Add(I0);
457 VertexNeighbors.FindOrAdd(I1).Add(I2);
458
459 VertexNeighbors.FindOrAdd(I2).Add(I0);
460 VertexNeighbors.FindOrAdd(I2).Add(I1);
461 }
462
463 // Step 2: Iterative smoothing
464 for (int32 Iter = 0; Iter < NumIterations; ++Iter)
465 {
466 TArray<FVector> NewVertices = Vertices;
467 for (int32 i = 0; i < NumVertices; ++i)
468 {
469 TSet<int32> Visited;
470 TQueue<TPair<int32, int32>> Queue; // Pair of (vertex index, current depth)
471 Visited.Add(i);
472 Queue.Enqueue(TPair<int32, int32>(i, 0));
473 TArray<int32> CollectedNeighbors;
474
475 while (!Queue.IsEmpty())
476 {
477 TPair<int32, int32> Current;
478 Queue.Dequeue(Current);
479 int32 CurrentIndex = Current.Key;
480 int32 CurrentDepth = Current.Value;
481
482 if (CurrentDepth >= Depth)
483 continue;
484 const TSet<int32>* Neighbors = VertexNeighbors.Find(CurrentIndex);
485 if (!Neighbors) continue;
486
487 for (int32 NeighborIndex : *Neighbors)
488 {
489 if (!Visited.Contains(NeighborIndex))
490 {
491 Visited.Add(NeighborIndex);
492 CollectedNeighbors.Add(NeighborIndex);
493 Queue.Enqueue(TPair<int32, int32>(NeighborIndex, CurrentDepth + 1));
494 }
495 }
496 }
497
498 if (CollectedNeighbors.Num() > 0)
499 {
500 FVector Average = FVector::ZeroVector;
501 for (int32 NeighborIdx : CollectedNeighbors)
502 {
503 Average += Vertices[NeighborIdx];
504 }
505 Average /= CollectedNeighbors.Num();
506 NewVertices[i] = FMath::Lerp(Vertices[i], Average, SmoothingFactor);
507 }
508 }
509 Vertices = NewVertices;
510 }
511}

513float UMapGenFunctionLibrary::BicubicSampleG16(const TArrayView64<const uint16>& Pixels, int Width, int Height, float X, float Y)
514{
515 int ix = FMath::FloorToInt(X);
516 int iy = FMath::FloorToInt(Y);
517 float fx = X - ix;
518 float fy = Y - iy;
519
520 float patch[4][4];
521
522 // Fetch surrounding 4x4 pixel values
523 for (int m = -1; m <= 2; ++m)
524 {
525 for (int n = -1; n <= 2; ++n)
526 {
527 uint16 val = GetPixelG16(Pixels, Width, Height, ix + n, iy + m);
528 patch[m + 1][n + 1] = val / 65535.0f; // Normalize to [0,1]
529 }
530 }
531
532 float col[4];
533 for (int i = 0; i < 4; ++i)
534 {
535 col[i] = CubicHermite(patch[i][0], patch[i][1], patch[i][2], patch[i][3], fx);
536 }
537
538 float result = CubicHermite(col[0], col[1], col[2], col[3], fy);
539 return FMath::Clamp(result, 0.0f, 1.0f); // Final result in [0,1]

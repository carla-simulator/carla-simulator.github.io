75rpc::VehicleControl ConstantVelocityAgent::RunStep() {
76 if (!_is_constant_velocity_active) {
77 const double elapsed =
78 _world->GetSnapshot().GetTimestamp().elapsed_seconds;
79 if (_has_stop_timestamp &&
80 (elapsed - _constant_velocity_stop_time) >
81 static_cast<double>(_restart_time)) {
82 RestartConstantVelocity();
83 _is_constant_velocity_active = true;
84 } else if (_use_basic_behavior) {
85 return BasicAgent::RunStep();
86 } else {
87 return rpc::VehicleControl{};
88 }
89 }
90
91 bool hazard_detected = false;
92 float hazard_speed = 0.0f;
93
94 auto actor_list = _world->GetActors();
95 auto vehicle_list = actor_list->Filter("*vehicle*");
96 auto lights_list = actor_list->Filter("*traffic_light*");
97
98 const float vehicle_speed = _vehicle->GetVelocity().Length();
99
100 // Vehicle obstacle check.
101 const float max_vehicle_distance = _base_vehicle_threshold + vehicle_speed;
102 auto v_obs = VehicleObstacleDetected(vehicle_list, max_vehicle_distance);
103 if (v_obs.affected && v_obs.actor) {
104 const auto vehicle_velocity = _vehicle->GetVelocity();
105 const float v_len = vehicle_velocity.Length();
106 if (v_len == 0.0f) {
107 hazard_speed = 0.0f;
108 } else {
109 const auto adv_velocity = v_obs.actor->GetVelocity();
110 const float dot = vehicle_velocity.x * adv_velocity.x +
111 vehicle_velocity.y * adv_velocity.y +
112 vehicle_velocity.z * adv_velocity.z;
113 hazard_speed = dot / v_len;
114 }
115 hazard_detected = true;
116 }
117
118 // Red-light check (uses 0.3 * speed factor instead of _speed_ratio — match
119 // the Python source verbatim).
120 const float max_tlight_distance = _base_tlight_threshold + 0.3f * vehicle_speed;
121 auto tl_pair = AffectedByTrafficLight(lights_list, max_tlight_distance);
122 if (tl_pair.first) {
123 hazard_speed = 0.0f;
124 hazard_detected = true;
125 }
126
127 // Run the local planner so the wheels keep spinning even though the actual
128 // forward motion comes from EnableConstantVelocity.
129 rpc::VehicleControl control = _local_planner->RunStep();
130 if (hazard_detected) {
131 SetConstantVelocity(hazard_speed);
132 } else {
133 SetConstantVelocity(_target_speed_ms);
134 }
135 return control;
136}

138void ConstantVelocityAgent::SetCollisionSensor() {
139 auto blueprint_lib = _world->GetBlueprintLibrary();
140 const auto *blueprint = blueprint_lib->Find("sensor.other.collision");
141 if (!blueprint) {
142 return;
143 }
144
145 auto actor = _world->SpawnActor(*blueprint, geom::Transform{}, _vehicle.get());
146 _collision_sensor = std::dynamic_pointer_cast<client::Sensor>(actor);
147 if (!_collision_sensor) {
148 return;
149 }
150
151 // Capture `this` raw — the agent owns the sensor and tears it down in the
152 // destructor, so the callback can never fire on a destroyed agent.
153 _collision_sensor->Listen(
154 [this](SharedPtr<sensor::SensorData>) {
155 this->StopConstantVelocity();
156 });
157}

159void ConstantVelocityAgent::DestroySensor() {
160 if (_collision_sensor) {
161 if (_collision_sensor->IsListening()) {
162 _collision_sensor->Stop();
163 }
164 _collision_sensor->Destroy();
165 _collision_sensor.reset();
166 }

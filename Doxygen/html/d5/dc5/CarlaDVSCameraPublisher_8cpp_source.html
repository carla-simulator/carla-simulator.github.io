18bool CarlaDVSImagePublisher::WriteImageFromEvents(
19 std::int32_t seconds,
20 std::uint32_t nanoseconds,
21 std::uint32_t height,
22 std::uint32_t width,
23 const std::uint8_t *events,
24 std::size_t event_count,
25 std::size_t event_stride) {
26 auto encoded = EncodeDvsEventsToBgr(height, width, events, event_count, event_stride);
27 // CarlaCameraPublisher's public WriteImage takes a const uint8_t* and runs
28 // it through ComputeImage; the default ComputeImage just copies the bytes.
29 // For DVS we have already produced the dense BGR buffer, so route through
30 // the standard `data` pointer pathway.
31 return WriteImage(seconds, nanoseconds, height, width, encoded.data());
32}

46std::vector<std::uint8_t> CarlaDVSPointCloudPublisher::ComputePointCloud(
47 std::uint32_t height, std::uint32_t width, const std::uint8_t *data) const {
48 const std::size_t total_points =
49 static_cast<std::size_t>(height) * static_cast<std::size_t>(width);
50 // The int64 event timestamp is converted to a real double in the FLOAT64 `t`
51 // slot inside the seam (see EncodeDvsEventsToPointCloud); a raw byte copy
52 // would leave int64 bits in a field a subscriber decodes as a double.
53 return EncodeDvsEventsToPointCloud(
54 data, total_points, sizeof(sensor::data::DVSEvent));
55}

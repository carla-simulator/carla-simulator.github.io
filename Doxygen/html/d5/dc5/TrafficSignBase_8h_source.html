48/// km/h of a speed-limit state, -1 for any other state (or for the custom state, whose

74/// The dedicated state for a km/h value, ETrafficSignState::SpeedLimit when there is none
75/// (then store the value in ATrafficSignBase::SpeedLimitKmh), UNKNOWN for a non-positive value.
76inline ETrafficSignState SpeedLimitStateForKmh(int32 Kmh)
77{
78 if (Kmh <= 0)
79 {
80 return ETrafficSignState::UNKNOWN;
81 }
82 static const ETrafficSignState Dedicated[] = {
83 ETrafficSignState::SpeedLimit_5, ETrafficSignState::SpeedLimit_10, ETrafficSignState::SpeedLimit_15,
84 ETrafficSignState::SpeedLimit_20, ETrafficSignState::SpeedLimit_25, ETrafficSignState::SpeedLimit_30,
85 ETrafficSignState::SpeedLimit_40, ETrafficSignState::SpeedLimit_50, ETrafficSignState::SpeedLimit_60,
86 ETrafficSignState::SpeedLimit_70, ETrafficSignState::SpeedLimit_80, ETrafficSignState::SpeedLimit_90,
87 ETrafficSignState::SpeedLimit_100, ETrafficSignState::SpeedLimit_110, ETrafficSignState::SpeedLimit_120,
88 ETrafficSignState::SpeedLimit_130 };
89 for (ETrafficSignState State : Dedicated)
90 {
91 if (KmhForSpeedLimitState(State) == Kmh)
92 {
93 return State;
94 }
95 }
96 return ETrafficSignState::SpeedLimit;
97}

99inline bool IsSpeedLimitState(ETrafficSignState State)
100{
101 return State == ETrafficSignState::SpeedLimit || KmhForSpeedLimitState(State) > 0;
102}

95FVector UJsonFileManagerLibrary::JsonObjectToVector(const TSharedPtr<FJsonObject>& JsonObject)
96{
97 FVector Vector = FVector::ZeroVector;
98 if(!JsonObject.IsValid())
99 {
100 return Vector;
101 }
102
103 const FString VectorStringX = JsonObject->GetStringField(TEXT("X"));
104 const FString VectorStringY = JsonObject->GetStringField(TEXT("Y"));
105 const FString VectorStringZ = JsonObject->GetStringField(TEXT("Z"));
106 Vector = FVector{ FCString::Atof(*VectorStringX),FCString::Atof(*VectorStringY),FCString::Atof(*VectorStringZ) };
107
108 /* Same result than the previous.
109 float ValueParsedX, ValueParsedY, ValueParsedZ;
110 FDefaultValueHelper::ParseFloat(JsonObject->GetStringField(TEXT("X")), ValueParsedX);
111 Vector.X = ValueParsedX;
112 FDefaultValueHelper::ParseFloat(JsonObject->GetStringField(TEXT("Y")), ValueParsedY);
113 Vector.Y = ValueParsedY;
114 FDefaultValueHelper::ParseFloat(JsonObject->GetStringField(TEXT("Z")), ValueParsedZ);
115 Vector.Z = ValueParsedZ;*/
116
117 RoundVectorToTwoDecimals(Vector);
118
119 return Vector;

132bool UJsonFileManagerLibrary::SaveIMUDataToJson(const FString& JsonFilePath, const FVector& Accelerometer,
133 const FVector& Gyroscope, float Compass, const FString& FrameNumber)
134{
135 TArray<TSharedPtr<FJsonValue>> ArrayValue;
136 TSharedPtr<FJsonObject> JsonDataObject = GetSensorJsonData(JsonFilePath, ArrayValue);
137
138 // Accelerometer and Gyroscope
139 const TSharedPtr<FJsonObject> IMUDataObject = MakeShared<FJsonObject>();
140 IMUDataObject->SetStringField(TEXT("Frame"), FrameNumber);
141 IMUDataObject->SetObjectField(TEXT("Accelerometer"), VectorToJsonObject(Accelerometer));
142 IMUDataObject->SetObjectField( TEXT("Gyroscope"), VectorToJsonObject(Gyroscope));
143
144 // Compass
145 const TSharedPtr<FJsonObject> CompassJsonObject = MakeShared<FJsonObject>();
146 FString Heading = Compass > 270.5f || Compass < 89.5f ? "N" : "";
147 Heading += Compass > 90.5f && Compass < 269.5f ? "S" : "";
148 Heading += Compass > 0.5f && Compass < 179.5f ? "E" : "";
149 Heading += Compass > 180.5f && Compass < 359.5f ? "W" : "";
150 CompassJsonObject->SetStringField(TEXT("Degrees"), FString::SanitizeFloat(Compass));
151 CompassJsonObject->SetStringField(TEXT("Sign"), Heading);
152 IMUDataObject->SetObjectField(TEXT("Compass"), CompassJsonObject);
153
154 const bool bWriteSuccess = SaveSensorJsonData(JsonFilePath, IMUDataObject, JsonDataObject, ArrayValue);
155 return bWriteSuccess;
156}

175bool UJsonFileManagerLibrary::SaveLidarDataToPly(const FString& FilePath, const TArray<float>& PointArray, int ArrayElementSize)
176{
177 FString DataToWrite;
178 const bool bReadSuccess = UJsonFileManagerLibrary::ReadStringFromFile(FilePath, DataToWrite);
179 if(!bReadSuccess)
180 {
181 DataToWrite = "";
182 }
183
184 FString LineTerminator = "\n";
185
186#if PLATFORM_WINDOWS
187 LineTerminator = LINE_TERMINATOR;
188#endif
189
190 // Create header if no data exist
191 DataToWrite += FString("ply") + LineTerminator +
192 FString("format ascii 1.0") + LineTerminator +
193 FString("element vertex ") + FString::FromInt(PointArray.Num()/ArrayElementSize) + LineTerminator +
194 FString("property float32 x") + LineTerminator +
195 FString("property float32 y") + LineTerminator +
196 FString("property float32 z") + LineTerminator +
197 FString("property float32 I") + LineTerminator +
198 FString("end_header") + LineTerminator;
199
200 for(SIZE_T i = 0; i < PointArray.Num(); i+=4)
201 {
202 if(PointArray.IsValidIndex(i+3))
203 {
204 // As Unreal is right handed, some visualizers will see the PointCloud Y axis inverted. If we want to get a left handed mesh output, it is necessary to negate the Y.
205 DataToWrite += FString::SanitizeFloat(PointArray[i]) + " " + FString::SanitizeFloat(PointArray[i+1]) + " " + FString::SanitizeFloat(PointArray[i+2]) + " " + FString::SanitizeFloat(PointArray[i+3]) + LineTerminator;
206 }
207 }
208
209 const bool bWriteSuccess = UJsonFileManagerLibrary::WriteStringFromFile(FilePath, DataToWrite);
210 if(!bWriteSuccess)
211 {
212 return false;
213 }
214
215 return true;
216}

249bool UJsonFileManagerLibrary::SaveRadarDataToJson(const FString& JsonFilePath,
250 const FRadarData& Rays, const FString& FrameNumber)
251{
252 TArray<TSharedPtr<FJsonValue>> ArrayValue;
253 TSharedPtr<FJsonObject> JsonDataObject = GetSensorJsonData(JsonFilePath, ArrayValue);
254 bool bWriteSuccess = true;
255 for( auto& var : Rays.GetDetections() )
256 {
257 const TSharedPtr<FJsonObject> RadarDataObject = MakeShared<FJsonObject>();
258 RadarDataObject->SetStringField(TEXT("Frame"), FrameNumber);
259 RadarDataObject->SetStringField(TEXT("Velocity"), FString::SanitizeFloat(var.velocity));
260 RadarDataObject->SetStringField(TEXT("Azimuth"), FString::SanitizeFloat(var.azimuth));
261 RadarDataObject->SetStringField(TEXT("Altitude"), FString::SanitizeFloat(var.altitude));
262 RadarDataObject->SetStringField(TEXT("Depth"), FString::SanitizeFloat(var.depth));
263 bWriteSuccess &= SaveSensorJsonData(JsonFilePath, RadarDataObject, JsonDataObject, ArrayValue);
264 }
265
266 return bWriteSuccess;
267}

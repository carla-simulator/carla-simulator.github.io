56 /// True when the definition's class or any attribute/variation value
57 /// points at content under @a MountPoint.
58 static bool DefinitionReferencesPath(const FActorDefinition &Definition, const FString &MountPoint)
59 {
60 const UClass *Class = Definition.Class.Get();
61 if (IsValid(Class) && Class->GetPathName().StartsWith(MountPoint))
62 {
63 return true;
64 }
65 for (const FActorAttribute &Attribute : Definition.Attributes)
66 {
67 if (Attribute.Value.StartsWith(MountPoint))
68 {
69 return true;
70 }
71 }
72 for (const FActorVariation &Variation : Definition.Variations)
73 {
74 for (const FString &Value : Variation.RecommendedValues)
75 {
76 if (Value.StartsWith(MountPoint))
77 {
78 return true;
79 }
80 }
81 }
82 return false;
83 }

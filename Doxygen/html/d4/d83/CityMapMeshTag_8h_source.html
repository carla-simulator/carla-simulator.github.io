96 /// @name Tag conversions

98
99 /// Convert @a Tag to an unsigned integer type.

102 }
103
104 /// Convert an unsigned integer to a ECityMapMeshTag.
105 static ECityMapMeshTag FromUInt(uint8 Value) {
106 check(Value < GetNumberOfTags());

109
110 /// Get @a Tag name as FString.

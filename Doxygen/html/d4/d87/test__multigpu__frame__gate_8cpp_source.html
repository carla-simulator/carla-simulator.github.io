44TEST(MultiGpuFrameGate, never_waits_while_the_primary_waits_for_this_server) {
45 // The primary waits for EPISODE_READY, which only the travel sends: waiting
46 // here during a load would deadlock both servers until the timeout.
47 BeginFrameState travel = Steady();
48 travel.travel_pending = true;
49 EXPECT_EQ(DecideBeginFrame(travel), BeginFrameAction::DontWait);
50
51 BeginFrameState load_map = Steady();
52 load_map.load_map_pending = true;
53 EXPECT_EQ(DecideBeginFrame(load_map), BeginFrameAction::DontWait);
54
55 BeginFrameState no_episode = Steady();
56 no_episode.has_episode = false;
57 EXPECT_EQ(DecideBeginFrame(no_episode), BeginFrameAction::DontWait);
58
59 BeginFrameState exiting = Steady();
60 exiting.exit_requested = true;

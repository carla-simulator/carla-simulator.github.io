19// These checks lock the type partitioning that ROS2::GetOrCreateCameraSensor's
20// cache-hit guard relies on. The guard casts a cached publisher with
21// dynamic_pointer_cast<CameraT> to enforce the one-actor-one-camera-type
22// invariant. Its safety rests on two facts verified here:
23// 1. RGB/Depth/SS/IS/Normals all alias the SAME CarlaRGBCameraPublisher
24// (shared BGRA passthrough), so a cache hit across those types is intended
25// and the cast succeeds.
26// 2. CarlaOpticalFlowCameraPublisher is a DISTINCT sibling type; both derive
27// from CarlaCameraPublisher. An RGB <-> OpticalFlow cache hit therefore
28// fails the cast, which is exactly the dangerous mismatch the guard catches.
29// If anyone later promotes one of the BGRA aliases to a real subclass, these
30// static_asserts break loudly so the guard's assumption is revisited.
31
32namespace carla {
33namespace ros2 {
34
35static_assert(
36 std::is_same_v<CarlaDepthCameraPublisher, CarlaRGBCameraPublisher>,
37 "Depth camera must alias the RGB BGRA-passthrough publisher.");
38static_assert(
39 std::is_same_v<CarlaNormalsCameraPublisher, CarlaRGBCameraPublisher>,
40 "Normals camera must alias the RGB BGRA-passthrough publisher.");
41static_assert(
42 std::is_same_v<CarlaSSCameraPublisher, CarlaRGBCameraPublisher>,
43 "Semantic-segmentation camera must alias the RGB BGRA-passthrough publisher.");
44static_assert(
45 std::is_same_v<CarlaISCameraPublisher, CarlaRGBCameraPublisher>,
46 "Instance-segmentation camera must alias the RGB BGRA-passthrough publisher.");
47
48static_assert(
49 !std::is_same_v<CarlaOpticalFlowCameraPublisher, CarlaRGBCameraPublisher>,
50 "Optical-flow camera must stay a distinct type from the RGB publisher.");
51
52static_assert(
53 std::is_base_of_v<CarlaCameraPublisher, CarlaRGBCameraPublisher>,
54 "RGB publisher must derive from CarlaCameraPublisher.");
55static_assert(
56 std::is_base_of_v<CarlaCameraPublisher, CarlaOpticalFlowCameraPublisher>,
57 "Optical-flow publisher must derive from CarlaCameraPublisher.");
58
59static_assert(
60 std::is_polymorphic_v<CarlaCameraPublisher>,
61 "CarlaCameraPublisher must be polymorphic for the dynamic_pointer_cast guard.");

68TEST(CameraPublisherTypes, bgra_cameras_share_one_publisher_type) {
69 // The five BGRA-passthrough cameras collapse to a single publisher type, so a
70 // cache hit across them in GetOrCreateCameraSensor is intended and safe.
71 EXPECT_TRUE((std::is_same_v<carla::ros2::CarlaDepthCameraPublisher,
72 carla::ros2::CarlaRGBCameraPublisher>));
73 EXPECT_TRUE((std::is_same_v<carla::ros2::CarlaNormalsCameraPublisher,
74 carla::ros2::CarlaRGBCameraPublisher>));
75 EXPECT_TRUE((std::is_same_v<carla::ros2::CarlaSSCameraPublisher,
76 carla::ros2::CarlaRGBCameraPublisher>));
77 EXPECT_TRUE((std::is_same_v<carla::ros2::CarlaISCameraPublisher,
78 carla::ros2::CarlaRGBCameraPublisher>));
79}

81TEST(CameraPublisherTypes, optical_flow_is_a_distinct_sibling) {
82 // Optical flow is the one camera type the cache-hit guard must reject when it
83 // collides with an RGB-family publisher (and vice versa).
84 EXPECT_FALSE((std::is_same_v<carla::ros2::CarlaOpticalFlowCameraPublisher,
85 carla::ros2::CarlaRGBCameraPublisher>));
86 EXPECT_TRUE((std::is_base_of_v<carla::ros2::CarlaCameraPublisher,
87 carla::ros2::CarlaOpticalFlowCameraPublisher>));
88 EXPECT_TRUE((std::is_base_of_v<carla::ros2::CarlaCameraPublisher,
89 carla::ros2::CarlaRGBCameraPublisher>));
90}

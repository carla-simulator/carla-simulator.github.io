33TEST(RadarPolarToCartesian, positive_azimuth_flips_y_to_negative) {
34 // azimuth = +pi/2 with the per-axis sign flip means y component picks up
35 // sin(-pi/2) * cos(0) = -1. depth = 10 -> y = -10.
36 constexpr float kHalfPi = static_cast<float>(M_PI) / 2.0f;
37 const auto p = carla::ros2::RadarPolarToCartesian(10.0f, kHalfPi, 0.0f);

61TEST(RadarPolarToCartesian, combined_polar_matches_hand_computed) {
62 // depth = 2, azimuth = pi/4, altitude = pi/6.
63 constexpr float depth = 2.0f;
64 const float azimuth = static_cast<float>(M_PI) / 4.0f;
65 const float altitude = static_cast<float>(M_PI) / 6.0f;
66 const auto p = carla::ros2::RadarPolarToCartesian(depth, azimuth, altitude);
67
68 const float expected_x = depth * std::cos(azimuth) * std::cos(-altitude);
69 const float expected_y = depth * std::sin(-azimuth) * std::cos(altitude);
70 const float expected_z = depth * std::sin(altitude);
71
72 EXPECT_NEAR(p[0], expected_x, kEps);
73 EXPECT_NEAR(p[1], expected_y, kEps);
74 EXPECT_NEAR(p[2], expected_z, kEps);
75}

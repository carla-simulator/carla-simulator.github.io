21 // Sections ordered by their s coordinate, so "next" and "previous" mean
22 // along the road rather than by id.
23 static std::vector<const road::LaneSection *> SectionsByDistance(const road::Road &road) {
24 std::vector<const road::LaneSection *> sections;
25 for (const road::LaneSection &section : road.GetLaneSections()) {
26 sections.emplace_back(&section);
27 }
28 std::sort(sections.begin(), sections.end(),
29 [](const road::LaneSection *lhs, const road::LaneSection *rhs) {
30 return lhs->GetDistance() < rhs->GetDistance();
31 });
32 return sections;
33 }

53 std::vector<SharedPtr<Lane>> LaneSection::GetLanes(road::Lane::LaneType type) const {
54 std::vector<SharedPtr<Lane>> result;
55 const road::LaneSection &section =
56 _parent->GetMap().GetData().GetRoad(_road_id).GetLaneSectionById(_section_id);
57 for (const auto &pair : section.GetLanes()) {
58 const road::Lane &lane = pair.second;
59 // The centre lane has no width and cannot be driven or walked on.
60 if (lane.GetId() == 0) {
61 continue;
62 }
63 if ((static_cast<int32_t>(lane.GetType()) & static_cast<int32_t>(type)) == 0) {
64 continue;
65 }
66 result.emplace_back(_parent->GetLane(_road_id, _section_id, lane.GetId()));
67 }
68 return result;
69 }

80 SharedPtr<LaneSection> LaneSection::GetNextSection() const {
81 const road::Road &road = _parent->GetMap().GetData().GetRoad(_road_id);
82 const auto sections = SectionsByDistance(road);
83 for (size_t i = 0u; i < sections.size(); ++i) {
84 if (sections[i]->GetId() == _section_id) {
85 if (i + 1u < sections.size()) {
86 return _parent->GetLaneSection(_road_id, sections[i + 1u]->GetId());
87 }
88 break;
89 }
90 }
91 return nullptr;
92 }

94 SharedPtr<LaneSection> LaneSection::GetPreviousSection() const {
95 const road::Road &road = _parent->GetMap().GetData().GetRoad(_road_id);
96 const auto sections = SectionsByDistance(road);
97 for (size_t i = 0u; i < sections.size(); ++i) {
98 if (sections[i]->GetId() == _section_id) {
99 if (i > 0u) {
100 return _parent->GetLaneSection(_road_id, sections[i - 1u]->GetId());
101 }
102 break;
103 }
104 }
105 return nullptr;
106 }

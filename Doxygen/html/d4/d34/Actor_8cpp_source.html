171 bool Actor::Destroy() {
172 // The actor state known to the client comes from the last world snapshot
173 // received, which cannot list an actor spawned after it was taken (until
174 // the next frame in asynchronous mode, until the next tick in synchronous
175 // mode). A missing entry is therefore indistinguishable from a dead actor
176 // here, so the server decides: it rejects unknown ids with "unable to
177 // destroy actor: not found" and this returns false.
178 //
179 // A handle whose own Destroy() already succeeded has had its episode
180 // cleared (Simulator::DestroyActor); asking the server again is pointless
181 // and Lock() would throw, so a repeated Destroy() is simply false.
182 auto episode = GetEpisode().TryLock();
183 if (episode == nullptr) {
184 log_debug(GetDisplayId(), "already destroyed through this handle.");
185 return false;
186 }
187 return episode->DestroyActor(*this);
188 }

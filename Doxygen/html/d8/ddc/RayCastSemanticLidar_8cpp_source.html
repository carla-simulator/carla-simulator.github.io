76void ARayCastSemanticLidar::CreateLasers()
77{
78 const auto NumberOfLasers = Description.Channels;
79 check(NumberOfLasers > 0u);
80 const float DeltaAngle = NumberOfLasers == 1u ? 0.f :
81 (Description.UpperFovLimit - Description.LowerFovLimit) /
82 static_cast<float>(NumberOfLasers - 1);
83 LaserAngles.Empty(NumberOfLasers);
84 for(auto i = 0u; i < NumberOfLasers; ++i)
85 {
86 const float VerticalAngle =
87 Description.UpperFovLimit - static_cast<float>(i) * DeltaAngle;
88 LaserAngles.Emplace(VerticalAngle);
89 }
90}

122void ARayCastSemanticLidar::SimulateLidar(const float DeltaTime, bool bLockPhysics)
123{
124 TRACE_CPUPROFILER_EVENT_SCOPE(ARayCastSemanticLidar::SimulateLidar);
125 const uint32 ChannelCount = Description.Channels;
126 const uint32 PointsToScanWithOneLaser =
127 FMath::RoundHalfFromZero(
128 Description.PointsPerSecond * DeltaTime / float(ChannelCount));
129
130 if (PointsToScanWithOneLaser <= 0)
131 {
132 UE_LOG(
133 LogCarla,
134 Warning,
135 TEXT("%s: no points requested this frame, try increasing the number of points per second."),
136 *GetName());
137 return;
138 }
139
140 check(ChannelCount == LaserAngles.Num());
141
142 const float CurrentHorizontalAngle = carla::geom::Math::ToDegrees(
143 SemanticLidarData.GetHorizontalAngle());
144 const float AngleDistanceOfTick = Description.RotationFrequency * Description.HorizontalFov
145 * DeltaTime;
146 const float AngleDistanceOfLaserMeasure = AngleDistanceOfTick / PointsToScanWithOneLaser;
147 const float HalfHorizontalFov = Description.HorizontalFov / 2;
148
149 ResetDetections(ChannelCount, PointsToScanWithOneLaser);
150 PreprocessRays(ChannelCount, PointsToScanWithOneLaser);
151
152 const FTransform ActorTransform = GetTransform();
153 const FTransform InverseSensorTransform = ActorTransform.Inverse();
154 const FVector LidarBodyLocation = ActorTransform.GetLocation();
155 const FRotator LidarBodyRotation = ActorTransform.Rotator();
156
157 auto RunChannelRaycasts = [&]()
158 {
159 TRACE_CPUPROFILER_EVENT_SCOPE(ParallelFor);
160 ParallelFor(ChannelCount, [&](int32 idxChannel) {
161 TRACE_CPUPROFILER_EVENT_SCOPE(ParallelForTask);
162
163 FCollisionQueryParams TraceParams = FCollisionQueryParams(FName(TEXT("Laser_Trace")), true, this);
164 TraceParams.bTraceComplex = true;
165 TraceParams.bReturnPhysicalMaterial = false;
166
167 const float VertAngle = LaserAngles[idxChannel];
168 const std::vector<bool>& ChannelPreprocessCondition = RayPreprocessCondition[idxChannel];
169
170 for (auto idxPtsOneLaser = 0u; idxPtsOneLaser < PointsToScanWithOneLaser; idxPtsOneLaser++) {
171 // Rays dropped by a subclass's PreprocessRays.
172 if (!ChannelPreprocessCondition[idxPtsOneLaser])
173 {
174 continue;
175 }
176
177 FHitResult HitResult;
178 const float HorizAngle = std::fmod(CurrentHorizontalAngle + AngleDistanceOfLaserMeasure
179 * idxPtsOneLaser, Description.HorizontalFov) - HalfHorizontalFov;
180
181 if (ShootLaser(VertAngle, HorizAngle, HitResult, TraceParams, LidarBodyLocation, LidarBodyRotation)) {
182 WriteDetectionAsync(idxChannel, HitResult, InverseSensorTransform, LidarBodyLocation);
183 }
184 };
185 });
186 };
187
188 if (bLockPhysics)
189 {
190 auto LockedPhysObject = FPhysicsObjectExternalInterface::LockRead(GetWorld()->GetPhysicsScene());
191 RunChannelRaycasts();
192 LockedPhysObject.Release();
193 }
194 else
195 {
196 RunChannelRaycasts();
197 }
198
199 ComputeAndSaveDetections(ActorTransform);
200
201 const float HorizontalAngle = carla::geom::Math::ToRadians(
202 std::fmod(CurrentHorizontalAngle + AngleDistanceOfTick, Description.HorizontalFov));
203 SemanticLidarData.SetHorizontalAngle(HorizontalAngle);
204}

226void ARayCastSemanticLidar::ComputeAndSaveDetections(const FTransform& SensorTransform)
227{
228 TRACE_CPUPROFILER_EVENT_SCOPE_STR(__FUNCTION__);
229

231 {
232 PointsPerChannel[idxChannel] = SemanticDetections[idxChannel].size();

243void ARayCastSemanticLidar::ComputeRawDetection(const FHitResult& HitInfo, const FTransform& InverseSensorTransform, const FVector& SensorLocation, FSemanticDetection& Detection) const
244{
245 static const uint32_t TerrainTag = static_cast<uint32_t>(ATagger::GetTagFromString("Terrain"));
246
247 const FVector HitPoint = HitInfo.ImpactPoint;
248 Detection.point = InverseSensorTransform.TransformPosition(HitPoint);
249
250 const FVector VecInc = - (HitPoint - SensorLocation).GetSafeNormal();
251 Detection.cos_inc_angle = FVector::DotProduct(VecInc, HitInfo.ImpactNormal);
252
253 const FActorRegistry &Registry = GetEpisode().GetActorRegistry();
254
255 const AActor* actor = HitInfo.GetActor();
256 Detection.object_idx = 0;
257 Detection.object_tag = static_cast<uint32_t>(crp::CityObjectLabel::None);
258
259 // Given that landscapes do not have tags for now, assign it here if the
260 // actor is a landscape, otherwise get the component tag.
261 // GetTagOfTaggedComponent tolerates untagged components (World Partition
262 // streams in HLOD/proxy geometry that never went through ATagger) by
263 // falling back to the stencil value, where reading ComponentTags[0]
264 // directly asserts on an empty array.
265 if (actor != nullptr && actor->IsA<ALandscape>())
266 {
267 Detection.object_tag = TerrainTag;
268 }
269 else if (HitInfo.Component.IsValid())
270 {
271 Detection.object_tag = static_cast<uint32_t>(ATagger::GetTagOfTaggedComponent(*HitInfo.Component));
272 }
273
274 if (actor != nullptr)
275 {
276 const FCarlaActor* view = Registry.FindCarlaActor(actor);
277 if (view != nullptr)
278 {
279 // On a Multi-GPU secondary, the id the primary's clients know.
280 Detection.object_idx = view->GetPrimaryActorId();
281 }
282 }
283 else
284 {
285 UE_LOG(LogCarla, Warning, TEXT("Actor not valid %p!!!!"), actor);
286 }
287}

290bool ARayCastSemanticLidar::ShootLaser(const float VerticalAngle, const float HorizontalAngle, FHitResult& HitResult, FCollisionQueryParams& TraceParams,
291 const FVector& LidarBodyLocation, const FRotator& LidarBodyRotation) const
292{
293 TRACE_CPUPROFILER_EVENT_SCOPE_STR(__FUNCTION__);
294
295 FRotator LaserRotation(VerticalAngle, HorizontalAngle, 0); // float InPitch, float InYaw, float InRoll
296 FRotator ResultRotation = UKismetMathLibrary::ComposeRotators(
297 LaserRotation,
298 LidarBodyRotation
299 );
300
301 const auto Range = Description.Range;
302 FVector EndTrace = Range * UKismetMathLibrary::GetForwardVector(ResultRotation) + LidarBodyLocation;
303
304 return GetWorld()->ParallelLineTraceSingleByChannel(
305 HitResult,
306 LidarBodyLocation,
307 EndTrace,
308 ECC_GameTraceChannel2,
309 TraceParams,
310 FCollisionResponseParams::DefaultResponseParam
311 );
312}

81 std::vector<SharedPtr<Lane>> Road::GetLanes(double s, road::Lane::LaneType type) const {
82 std::vector<SharedPtr<Lane>> result;
83 const road::Road &road = GetRoadData(_parent, _road_id);
84 for (const road::Lane *lane : road.GetLanesByDistance(s)) {
85 if (lane == nullptr || lane->GetId() == 0 || lane->GetLaneSection() == nullptr) {
86 continue;
87 }
88 if ((static_cast<int32_t>(lane->GetType()) & static_cast<int32_t>(type)) == 0) {
89 continue;
90 }
91 result.emplace_back(_parent->GetLane(
92 _road_id, lane->GetLaneSection()->GetId(), lane->GetId()));
93 }
94 return result;
95 }

121 SharedPtr<LaneSection> Road::GetSectionAt(double s) const {
122 const road::Road &road = GetRoadData(_parent, _road_id);
123 // GetLaneSectionsAt yields every section starting at or before s; the last
124 // is the one that actually covers it.
125 const road::LaneSection *covering = nullptr;
126 for (const road::LaneSection &section : road.GetLaneSectionsAt(s)) {
127 covering = &section;
128 }
129 if (covering == nullptr) {
130 return nullptr;
131 }
132 return _parent->GetLaneSection(_road_id, covering->GetId());
133 }

142 geom::Transform Road::GetTransformAt(double s) const {
143 auto directed_point = GetRoadData(_parent, _road_id).GetDirectedPointIn(s);
144 // Same Unreal Y-axis flip road::Lane::ComputeTransform applies; without it
145 // the transform comes back in the OpenDRIVE frame and disagrees with every
146 // waypoint on the same road.
147 directed_point.location.y *= -1.0f;
148 directed_point.tangent *= -1.0;
149 return geom::Transform(
150 directed_point.location,
151 geom::Rotation(
152 geom::Math::ToDegrees(static_cast<float>(directed_point.pitch)),
153 geom::Math::ToDegrees(static_cast<float>(directed_point.tangent)),
154 0.0f));
155 }

157 std::vector<SharedPtr<Waypoint>> Road::GetWaypointsAt(
158 double s,
159 road::Lane::LaneType type) const {
160 std::vector<SharedPtr<Waypoint>> result;
161 for (const SharedPtr<Lane> &lane : GetLanes(s, type)) {
162 SharedPtr<Waypoint> waypoint = lane->GetWaypoint(s);
163 if (waypoint != nullptr) {
164 result.emplace_back(std::move(waypoint));

170 std::vector<SharedPtr<Waypoint>> Road::GetWaypoints(
171 double distance,
172 road::Lane::LaneType type) const {
173 std::vector<SharedPtr<Waypoint>> result;
174 if (distance <= 0.0) {
175 return result;
176 }
177 const double length = GetLength();
178 for (double s = 0.0; s < length; s += distance) {
179 for (SharedPtr<Waypoint> &waypoint : GetWaypointsAt(s, type)) {
180 result.emplace_back(std::move(waypoint));
181 }
182 }
183 return result;
184 }

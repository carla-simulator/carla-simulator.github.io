759#ifdef CARLA_HAS_GBUFFER_API
760 template <
761 typename SensorT,
762 typename CameraGBufferT>
763 static void SendGBuffer(
764 SensorT &Self,
765 CameraGBufferT &CameraGBuffer,
766 FGBufferRequest &GBufferData,
767 EGBufferTextureID TextureID)
768 {
769 using PixelType = typename std::conditional<
770 std::is_same<std::remove_reference_t<CameraGBufferT>, FCameraGBufferUint8>::value,
771 FColor,
772 FLinearColor>::type;
773 FIntPoint ViewSize;
774 TArray<PixelType> Pixels;
775 if (GBufferData.WaitForTextureTransfer(TextureID))
776 {
777 TRACE_CPUPROFILER_EVENT_SCOPE_STR("GBuffer Decode");
778 void *PixelData;
779 int32 SourcePitch;
780 FIntPoint SourceExtent;
781 GBufferData.MapTextureData(
782 TextureID,
783 PixelData,
784 SourcePitch,
785 SourceExtent);
786 auto Format = GBufferData.Readbacks[(size_t)TextureID]->GetFormat();
787 ViewSize = GBufferData.ViewRect.Size();
788 Pixels.AddUninitialized(ViewSize.X * ViewSize.Y);
789 FReadSurfaceDataFlags Flags = {};
790 Flags.SetLinearToGamma(true);
791 ImageUtil::DecodePixelsByFormat(
792 PixelData,
793 SourcePitch,
794 SourceExtent,
795 ViewSize,
796 Format,
797 Flags,
798 Pixels);
799 GBufferData.UnmapTextureData(TextureID);
800 }
801 else
802 {
803 ViewSize = GBufferData.ViewRect.Size();
804 Pixels.SetNum(ViewSize.X * ViewSize.Y);
805 for (auto &Pixel : Pixels)

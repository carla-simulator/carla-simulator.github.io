44AutowareVehicleStatusPublisher::AutowareVehicleStatusPublisher()
45 : BasePublisher("rt/vehicle/status", kFrameId),
46 _impl(std::make_shared<Impl>()) {
47 // QoS per tier4's AutowarePublisher: RELIABLE / VOLATILE / KEEP_LAST 1.
48 const QosProfile qos = QosProfile::ReliableVolatile();
49 bool ok = true;
50 ok &= _impl->velocity.Init("rt/vehicle/status/velocity_status", qos);
51 ok &= _impl->steering.Init("rt/vehicle/status/steering_status", qos);
52 ok &= _impl->control_mode.Init("rt/vehicle/status/control_mode", qos);
53 ok &= _impl->gear.Init("rt/vehicle/status/gear_status", qos);
54 ok &= _impl->turn_indicators.Init("rt/vehicle/status/turn_indicators_status", qos);
55 ok &= _impl->hazard_lights.Init("rt/vehicle/status/hazard_lights_status", qos);
56 if (!ok) {
57 log_error("AutowareVehicleStatusPublisher: failed to initialise one or more report writers");
58 }
59}

73void AutowareVehicleStatusPublisher::SetSteering(float steering_tire_angle) {
74 auto *report = _impl->steering.GetMessage();
75 // Report the *actual* tire angle: apply the inverse compensation LUT so the
76 // published state matches what the command-side GetSteeringInput mapping
77 // asked the vehicle to do (tier4 AutowarePublisher::SetSteering).
78 report->steering_tire_angle =
79 autoware_steering_compensation::GetSteeringOutput(steering_tire_angle);
80}

170bool AutowareVehicleStatusPublisher::Publish() {
171 // All reports are published at once with the same stamp based on the
172 // assumption that they share one frequency.
173 // https://github.com/tier4/AWSIM/blob/v1.3.1/docs/Components/ROS2/ROS2TopicAndServiceList/index.md
174 _impl->velocity.GetMessage()->header.stamp.sec = _seconds;
175 _impl->velocity.GetMessage()->header.stamp.nanosec = _nanoseconds;
176 _impl->steering.GetMessage()->stamp.sec = _seconds;
177 _impl->steering.GetMessage()->stamp.nanosec = _nanoseconds;
178 _impl->control_mode.GetMessage()->stamp.sec = _seconds;
179 _impl->control_mode.GetMessage()->stamp.nanosec = _nanoseconds;
180 _impl->gear.GetMessage()->stamp.sec = _seconds;
181 _impl->gear.GetMessage()->stamp.nanosec = _nanoseconds;
182 _impl->turn_indicators.GetMessage()->stamp.sec = _seconds;
183 _impl->turn_indicators.GetMessage()->stamp.nanosec = _nanoseconds;
184 _impl->hazard_lights.GetMessage()->stamp.sec = _seconds;
185 _impl->hazard_lights.GetMessage()->stamp.nanosec = _nanoseconds;
186
187 bool ok = true;
188 ok &= _impl->velocity.Publish();
189 ok &= _impl->steering.Publish();
190 ok &= _impl->control_mode.Publish();
191 ok &= _impl->gear.Publish();
192 ok &= _impl->turn_indicators.Publish();
193 ok &= _impl->hazard_lights.Publish();
194 return ok;
195}

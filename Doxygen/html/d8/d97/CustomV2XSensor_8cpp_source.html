47void ACustomV2XSensor::EndPlay(const EEndPlayReason::Type EndPlayReason)
48{
49 // Drop our queued messages from the shared static maps so a destroyed
50 // sensor cannot leave a dangling AActor* key for the surviving sensors.
51 {
52 std::lock_guard<std::mutex> lock(v2xDataLock);
53 gActorV2XDataMap.erase(GetSenderId());
54 gActorV2XDataMapNextFrame.erase(GetSenderId());
55 }
56
57 // PathLossModelObj is a std::unique_ptr released automatically in the
58 // destructor; not reset here so teardown paths never see a freed object.
59 Super::EndPlay(EndPlayReason);
60}

115 * Function stores the actor details in to the static list.
116 * Calls the CaService object to generate CAM message
117 * Stores the message in static map
118 */
119void ACustomV2XSensor::PrePhysTick(float DeltaSeconds)
120{
121 Super::PrePhysTick(DeltaSeconds);
122 if (GetOwner())
123 {
124 // Swap the messages with the ones sent out since last sim cycle
125 std::lock_guard<std::mutex> lock(v2xDataLock);
126 auto LastMessagesResult = ACustomV2XSensor::gActorV2XDataMap.find(GetSenderId());
127 auto NextMessagesResult = ACustomV2XSensor::gActorV2XDataMapNextFrame.find(GetSenderId());
128 if ( LastMessagesResult != ACustomV2XSensor::gActorV2XDataMap.end() ) {
129 if ( NextMessagesResult != ACustomV2XSensor::gActorV2XDataMapNextFrame.end() ) {
130 // swap the next frame messages to current frame
131 LastMessagesResult->second.swap(NextMessagesResult->second);
132 ACustomV2XSensor::gActorV2XDataMapNextFrame.erase(NextMessagesResult);
133 } else {
134 // no new messages, just clear
135 ACustomV2XSensor::gActorV2XDataMap.erase(LastMessagesResult);
136 }
137 }
138 else if ( NextMessagesResult != ACustomV2XSensor::gActorV2XDataMapNextFrame.end() ) {
139 // first time, just move next to current
140 ACustomV2XSensor::gActorV2XDataMap.insert({GetSenderId(), V2XDataList()}).first->second.swap(NextMessagesResult->second);
141 ACustomV2XSensor::gActorV2XDataMapNextFrame.erase(NextMessagesResult);
142 }
143 else {
144 // no messages at all
145 }
146 }
147}

150/*
151 * Function takes care of sending messages to the current actor.
152 * First simulates the communication by calling LOSComm object.
153 * If there is a list present then messages from those list are sent to the current actor
154 */
155void ACustomV2XSensor::PostPhysTick(UWorld *World, ELevelTick TickType, float DeltaTime)
156{
157 // tracing after we get the lock
158 TRACE_CPUPROFILER_EVENT_SCOPE(ACustomV2XSensor::PostPhysTick);
159 // Step 1: Create an actor list which has messages to send targeting this v2x sensor instance
160 std::vector<ActorPowerPair> ActorPowerList;
161 {
162 std::lock_guard<std::mutex> lock(v2xDataLock);
163 for (const auto &pair : gActorV2XDataMap)
164 {
165 // only different sensors with the same ChannelId talk to each other
166 if ((pair.first.Actor != this) && (mChannelId == pair.first.ChannelId))
167 {
168 // Skip senders with no queued message this tick; front() on an
169 // empty list is undefined behaviour.
170 if (pair.second.empty())
171 {
172 continue;
173 }
174 ActorPowerPair actor_power_pair;
175 actor_power_pair.first = pair.first.Actor;
176 // actor sending with transmit power (identical for all messages of this tick)
177 actor_power_pair.second = pair.second.front().Power;
178 ActorPowerList.push_back(actor_power_pair);
179 }
180 }
181 }
182
183 // Step 2: Simulate the communication for the actors in actor list to current actor.
184 FV2XData v2XData;
185 if (!ActorPowerList.empty())
186 {
187 UCarlaEpisode *carla_episode = UCarlaStatics::GetCurrentEpisode(GetWorld());
188 PathLossModelObj->Simulate(ActorPowerList, carla_episode, GetWorld());
189 // Step 3: Get the list of actors who can send message to current actor, and the receive power of their messages.
190 ActorPowerMap actor_receivepower_map = PathLossModelObj->GetReceiveActorPowerList();
191 // Step 4: Retrieve the messages of the actors that are received
192 {
193 std::lock_guard<std::mutex> lock(v2xDataLock);
194 for (const auto &pair : actor_receivepower_map)
195 {
196 // find + skip-on-miss: at() would throw std::out_of_range if the
197 // sender's queue was cleared between the two locked sections.
198 const auto sender_it = gActorV2XDataMap.find({pair.first, mChannelId});
199 if (sender_it == gActorV2XDataMap.end())
200 {
201 continue;
202 }
203 for ( const auto & send_msg_and_pw: sender_it->second)
204 {
205 carla::sensor::data::CustomV2XData received_msg_and_pw;
206 // sent CAM
207 received_msg_and_pw.Message = send_msg_and_pw.Message;
208 // receive power
209 received_msg_and_pw.Power = pair.second;
210
211 v2XData.WriteMessage(received_msg_and_pw);
212 }
213 }
214 }
215 }
216 // Step 5: Send message
217 if (v2XData.GetMessageCount() > 0)
218 {
219 auto DataStream = GetDataStream(*this);
220 DataStream.SerializeAndSend(*this, v2XData, DataStream.PopBufferFromPool());
221 }
222}

224void ACustomV2XSensor::Send(const carla::rpc::CustomV2XBytes &data)
225{
226 // We have to queue the data immediately otherwhise only one single message can be processed per frame!
227 std::lock_guard<std::mutex> lock(v2xDataLock);
228
229 UpdateStationId();
230
231 // make a pair of message and sending power
232 // if different v2x sensors send with different power, we need to store that
233 carla::sensor::data::CustomV2XData message_pw;
234 message_pw.Message.header.protocolVersion = mProtocolVersion;
235 message_pw.Message.header.messageID = mMessageId;
236 message_pw.Message.header.stationID = mStationId;
237 message_pw.Message.data = data;
238 message_pw.Power = PathLossModelObj->GetTransmitPower();
239 gActorV2XDataMapNextFrame.insert({GetSenderId(), V2XDataList()}).first->second.push_back(message_pw);
240}

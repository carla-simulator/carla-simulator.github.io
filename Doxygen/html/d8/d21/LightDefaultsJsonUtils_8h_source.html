68// One row of the Vehicles tab. Unlike every other group here, vehicles have
69// no single UCarlaLight to key a Color/Intensity/Emissive triad on -- a
70// Blueprint like BP_LincolnMKZ places 18 separate real light components
71// (front-low_beam-l-1, back-brake-l-1, ...) AND a shared "*_Emissive"
72// MaterialInstance exposing 8 named scalar parameters mirroring those same
73// groups (confirmed identical, same order, across Lincoln/MiniCooper/
74// Ambulance) plus one standalone global "Intensity" param that turned out to
75// do nothing visible on its own. Two controls, two different targets:
76// - Intensity (this struct's field): writes the SAME value into every
77// exposed group parameter on the Emissive material at once (Position,
78// Low Beam, ... all glow together) -- a "light them all up" shortcut,
79// not the material's own unused "Intensity" param.
80// - GroupIntensity (per column): the REAL Spot/Rect/Point light
81// components' physical Intensity, matched by name (see
82// CollectVehicleLightComponentsForGroup in the .cpp) -- actual scene
83// illumination, independent of the material glow above.
84USTRUCT(BlueprintType)
85struct FVehicleLightSummary
86{
87 GENERATED_BODY()
88
89 // The Blueprint class name (e.g. "BP_LincolnMKZ_C") -- apply/select key,
90 // matched against AActor::GetClass()->GetName() same as everywhere else.
91 UPROPERTY(BlueprintReadOnly)
92 FString ClassName;
93
94 // "Lincoln Mkz" -- Make + Model from VehicleParameters.json, shown
95 // instead of the raw class name (which the Class column still shows).
96 UPROPERTY(BlueprintReadOnly)
97 FString DisplayName;
98
99 // How many instances of this class are currently placed in the level --
100 // normally 0, since this tab lists the full catalog (VehicleParameters.json),
101 // not a level scan. Placed instances (or a live Preview spawn) still
102 // receive ApplyVehicleLightsLive same as everywhere else.
103 UPROPERTY(BlueprintReadOnly)
104 int32 InstanceCount = 0;
105
106 // Global "light every exposed Emissive group up to this value at once"
107 // control -- see the struct comment above. Sampled at scan time from the
108 // material's own (otherwise-unused) "Intensity" param as a starting
109 // value only; ApplyVehicleLightsLive gives it its real meaning.
110 UPROPERTY(BlueprintReadWrite)
111 float Intensity = 1.0f;
112
113 // Canonical group name (VehicleLightGroupNames) -> that group's real
114 // light components' physical Intensity, current/edited value -- scaled
115 // down by VehicleLightIntensityDisplayScale (10000) from the raw UE5
116 // photometric lux the component actually stores (real values run
117 // ~500000; awkward to type/read directly), so this field runs roughly
118 // 10-100 instead. Only groups this vehicle actually HAS matching
119 // components for are meaningful to apply; see GroupPresent for which
120 // those are.
121 UPROPERTY(BlueprintReadWrite)
122 TMap<FString, float> GroupIntensity;
123
124 // Canonical group name -> whether this vehicle has any real light
125 // component matching that group's naming convention (checked once at
126 // scan time, see CollectVehicleLightComponentsForGroup). The panel greys
127 // out/disables a group's field when this is false instead of showing a
128 // control that would silently do nothing.
129 UPROPERTY(BlueprintReadOnly)
130 TMap<FString, bool> GroupPresent;
131
132 UPROPERTY(BlueprintReadOnly)
133 bool bHasSavedDefault = false;
134};

254 // Buildings baked by the modular building tool (BP_BuildingConverterToCode)
255 // are placed as plain, non-Blueprint AActor instances referencing Content
256 // Browser HierarchicalInstancedStaticMeshComponent modules directly --
257 // there's no Blueprint class to key a per-class default on like every
258 // other group here. Detected purely by material instead: any HISM mesh
259 // slot whose static asset lives under /Game/Carla/Static/Building/ and
260 // whose name contains "FakeInterior" (the window/interior-illusion
261 // family -- the shared M_MaterialMaster base underneath is used by far
262 // more than windows, see ResolveStaticProceduralBuildingAsset). Exposed
263 // as a single synthetic "ProceduralBuildings" class covering every baked
264 // building in the level -- one global emissive control until per-style
265 // grouping (by material family) proves necessary.
266 //
267 // Only instanced (ISM/HISM) slots -- see ScanLegacyBuildingsInLevel for the same
268 // material family on plain (non-instanced) StaticMeshComponents, which
269 // can't be controlled the same way.
270 //
271 // A HISM component's material slot is shared by every physical instance
272 // it draws (some hold hundreds of windows at once -- confirmed no
273 // per-instance custom data exists to vary through C++), so per-window
274 // on/off can't be decided here. Instead M_FakeInterior's own graph gates
275 // its emissive output with a PerInstanceRandom node compared against a
276 // "PercentLitThreshold" parameter -- PercentLit (0..100) is just handed
277 // down as that threshold (divided by 100), and the renderer resolves the
278 // per-instance randomness deterministically per physical window.
279 // Color goes to M_FakeInterior's "Emissive Color" (the material ships
280 // white, which reads as cold fluorescent blocks at night).

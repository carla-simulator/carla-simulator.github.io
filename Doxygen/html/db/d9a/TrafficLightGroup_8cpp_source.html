44void ATrafficLightGroup::Tick(float DeltaTime)
45{
46 TRACE_CPUPROFILER_EVENT_SCOPE(ATrafficLightGroup::Tick);
47 Super::Tick(DeltaTime);
48
49 // Do not update if the replayer is replaying
50 auto* Episode = UCarlaStatics::GetCurrentEpisode(GetWorld());
51 if (Episode)
52 {
53 // A Multi-GPU secondary takes every light's state from the primary's
54 // frames: cycling here too would switch lights ahead of the primary.
55 if (!Episode->bIsPrimaryServer)
56 {
57 return;
58 }
59 auto* Replayer = Episode->GetReplayer();
60 if (Replayer)
61 {
62 if(Replayer->IsEnabled())
63 {
64 return;
65 }
66 }
67 }
68
69 if (bIsFrozen)
70 {
71 return;
72 }
73
74 if (Controllers.Num() == 0)
75 {
76 UE_LOG(LogCarla, Error, TEXT("TrafficLightGroup::Tick() JunctionId=%d has NO CONTROLLERS!"), JunctionId);
77 return;
78 }
79
80 UTrafficLightController* controller = Controllers[CurrentController];
81 if (controller->AdvanceTimeAndCycleFinished(DeltaTime))
82 {
83 NextController();
84 }
85}

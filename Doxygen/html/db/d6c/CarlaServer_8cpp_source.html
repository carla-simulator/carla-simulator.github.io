126/// True when the stream belongs to a camera and this process has no RHI
127/// (-nullrhi): a local token for it could never deliver a frame, so the
128/// camera must be owned by a rendering Multi-GPU secondary instead.
129static bool IsUnrenderableLocalSensorStream(
130 UCarlaEpisode &Episode,
131 carla::streaming::detail::stream_id_type StreamId)
132{
133 if (FApp::CanEverRender())
134 {
135 return false;
136 }
137 const FCarlaActor *CarlaActor = Episode.FindCarlaActorFromStream(StreamId);
138 if (CarlaActor == nullptr)
139 {
140 return false;
141 }
142 const AActor *Actor = CarlaActor->GetActor();
143 return Actor != nullptr &&
144 (Actor->IsA<ASceneCaptureSensor>() || Actor->IsA<ASceneCaptureSensor_WideAngleLens>());
145}

147/// Resolve the AWalkerController behind a walker-navigation RPC actor id.
148/// Accepts either the walker pawn itself or the controller.ai.walker handle
149/// actor (which clients spawn attached to the walker). Returns nullptr for
150/// unknown, dormant or dead actors -- the navigation RPCs answer false in
151/// that case instead of raising, per the phase-1 contract.
152static AWalkerController *ResolveWalkerNavController(FCarlaActor *CarlaActor)
153{
154 if (CarlaActor == nullptr || CarlaActor->IsDormant())
155 {
156 return nullptr;
157 }
158 AActor *Actor = CarlaActor->GetActor();
159 if (Actor == nullptr)
160 {
161 return nullptr;
162 }
163 if (Cast<AWalkerAIController>(Actor) != nullptr)
164 {
165 Actor = Actor->GetAttachParentActor();
166 if (Actor == nullptr)
167 {
168 return nullptr;
169 }
170 }
171 auto *Walker = Cast<AWalkerBase>(Actor);
172 if (Walker != nullptr && !Walker->bAlive)
173 {
174 return nullptr;
175 }
176 auto *Pawn = Cast<APawn>(Actor);
177 if (Pawn == nullptr)
178 {
179 return nullptr;
180 }
181 return Cast<AWalkerController>(Pawn->GetController());
182}

325
326 constexpr ServerBinder(const char *name, carla::rpc::Server &srv, bool sync)
327 : _name(name),
328 _server(srv),
329 _sync(sync) {}
330
331 template <typename FuncT>
332 auto operator<<(FuncT func)
333 {
334 if (_sync)
335 {
336 _server.BindSync(_name, func);
337 }
338 else
339 {
340 _server.BindAsync(_name, func);
341 }
342 return func;
343 }
344
345private:
346
347 const char *_name;
348
349 carla::rpc::Server &_server;
350
351 bool _sync;
352};

372 /// Gets a pair filled with the <IP, port> of the Trafic Manager running on port.
373 /// If there is no Traffic Manager running the pair will be ("", 0)
374 BIND_SYNC(get_traffic_manager_running) << [this] (uint16_t port) ->R<std::pair<std::string, uint16_t>>
375 {
376 auto it = TrafficManagerInfo.find(port);

378 return std::pair<std::string, uint16_t>(it->second, it->first);
379 }
380 return std::pair<std::string, uint16_t>("",0);
381 };

383 /// Add a new Traffic Manager running on <IP, port>
384 BIND_SYNC(add_traffic_manager_running) << [this] (std::pair<std::string, uint16_t> trafficManagerInfo) ->R<bool>

386 REQUIRE_PRIMARY_SERVER();
387 uint16_t port = trafficManagerInfo.second;
388 auto it = TrafficManagerInfo.find(port);
389 if(it == TrafficManagerInfo.end()) {
390 TrafficManagerInfo.insert(
391 std::pair<uint16_t, std::string>(port, trafficManagerInfo.first));
392 return true;
393 }
394 return false;

737 BIND_SYNC(get_required_files) << [this](std::string folder = "") -> R<std::vector<std::string>>
738 {
739 REQUIRE_CARLA_EPISODE();
740
741 // Check that the path ends in a slash, add it otherwise
742 if (!folder.empty() && folder[folder.size() - 1] != '/' && folder[folder.size() - 1] != '\\') {
743 folder += "/";
744 }
745
746 // Get the map's folder absolute path and check if it's in its own folder
747 ACarlaGameModeBase* GameMode = UCarlaStatics::GetGameMode(Episode->GetWorld());
748 const auto mapDir = GameMode->GetFullMapPath();
749 const auto folderDir = mapDir + "/" + folder.c_str();
750 const auto fileName = mapDir.EndsWith(Episode->GetMapName()) ? "*" : Episode->GetMapName();
751
752 // Find all the xodr and bin files from the map. A generated OpenDRIVE
753 // world only ever has current files in Saved/ (see
754 // UCarlaEpisode::LoadNewOpendriveEpisode); copies of OpenDriveMap.* in
755 // the content dir (cooked into the base pak) are stale by construction
756 // and must never be served.
757 const bool bGeneratedWorld = Episode->GetMapName().Equals(TEXT("OpenDriveMap"), ESearchCase::IgnoreCase);
758 TArray<FString> Files;
759 if (!bGeneratedWorld)
760 {
761 IFileManager::Get().FindFilesRecursive(Files, *folderDir, *(fileName + ".xodr"), true, false, false);
762 IFileManager::Get().FindFilesRecursive(Files, *folderDir, *(fileName + ".bin"), true, false, false);
763 }
764
765 // Generated OpenDRIVE worlds keep their xodr/nav in Saved/; they are
766 // served under the conventional Carla/Maps/{OpenDrive,Nav}/ names and
767 // filtered by the requested sub-folder like the content walk above.
768 const FString SavedDir = FPaths::ConvertRelativePathToFull(FPaths::ProjectSavedDir()) / TEXT("");
769 if (bGeneratedWorld)
770 {
771 const FString Folder = UTF8_TO_TCHAR(folder.c_str());
772 const bool bWantXODR = Folder.IsEmpty() || Folder.StartsWith(TEXT("OpenDrive"), ESearchCase::IgnoreCase);
773 const bool bWantNav = Folder.IsEmpty() || Folder.StartsWith(TEXT("Nav"), ESearchCase::IgnoreCase);
774 if (bWantNav)
775 {
776 // The client asks for the navmesh right after load_world; RecastBuilder
777 // may still be running (bounded wait, logs on failure).
778 UCarlaEpisode::WaitForPendingNavigationBuild(60.0);
779 }
780 // Where this server keeps the xodr: a Multi-GPU secondary has its own
781 // folder, and builds no navigation mesh.
782 const FString SavedXODR = UOpenDrive::GetSavedXODRPath(Episode->GetMapName());
783 const FString SavedNav = SavedDir / TEXT("Nav") / (Episode->GetMapName() + TEXT(".bin"));
784 if (bWantXODR && IFileManager::Get().FileExists(*SavedXODR)) Files.Add(SavedXODR);
785 if (bWantNav && !bIsSecondaryServer && IFileManager::Get().FileExists(*SavedNav)) Files.Add(SavedNav);
786 }
787
788 // Remove the start of the path until the content folder and put each file
789 // in the result. In a packaged build the file manager hands back
790 // pak-relative paths ("../../../<Project>/Content/..."), so normalize each
791 // file to an absolute path before stripping the absolute content dir;
792 // otherwise the raw relative path leaks to the client, which would try to
793 // resolve it against its cache folder and escape it. Files of a content
794 // pack are named Packs/<Pack>/<path under its Content dir>.
795 // Every prefix carries a trailing '/', so a sibling dir such as
796 // "ContentBackup" can never match.
797 const FString ContentDir = FPaths::ConvertRelativePathToFull(FPaths::ProjectContentDir()) / TEXT("");
798 const UCarlaContentPackManager *ContentPacks = UCarlaContentPackManager::Get();
799 std::vector<std::string> result;
800 for (auto File : Files) {
801 File = FPaths::ConvertRelativePathToFull(File);
802 if (File.StartsWith(SavedDir) && File.EndsWith(TEXT(".xodr")))
803 {
804 // Under its conventional name, whatever folder this server keeps it in.
805 File = TEXT("Carla/Maps/OpenDrive") / FPaths::GetCleanFilename(File);
806 }
807 else if (File.StartsWith(SavedDir))
808 {
809 File.RemoveFromStart(SavedDir);
810 File = TEXT("Carla/Maps") / File;
811 }
812 else if (File.StartsWith(ContentDir))
813 {
814 File.RemoveFromStart(ContentDir);
815 }
816 else if (ContentPacks != nullptr)
817 {
818 for (const FCarlaContentPack &Pack : ContentPacks->GetPacks())
819 {
820 const FString PackContentDir = Pack.ContentDir / TEXT("");
821 if (Pack.bMounted && File.StartsWith(PackContentDir))
822 {
823 File.RemoveFromStart(PackContentDir);
824 File = FString::Printf(TEXT("Packs/%s"), *Pack.GetName()) / File;
825 break;
826 }
827 }
828 }
829 result.emplace_back(TCHAR_TO_UTF8(*File));
830 }
831
832 return result;
833 };
834 BIND_SYNC(request_file) << [this](std::string name) -> R<std::vector<uint8_t>>
835 {
836 REQUIRE_CARLA_EPISODE();
837
838 // Get the absolute path of the file, refusing names that escape the
839 // content folder: the name comes from the client, and ".." segments would
840 // otherwise let it read arbitrary files on the server host.
841 const FString ContentDir = FPaths::ConvertRelativePathToFull(FPaths::ProjectContentDir());
842 FString path = ContentDir;
843 path.Append(name.c_str());
844 path = FPaths::ConvertRelativePathToFull(path);
845 if (!path.StartsWith(ContentDir))
846 {
847 RESPOND_ERROR("file requested outside the content folder");
848 }
849
850 // Names produced by get_required_files for a content pack
851 // (Packs/<Pack>/...) resolve inside that pack's Content dir. The
852 // generated OpenDRIVE world's files (Carla/Maps/{OpenDrive,Nav}/...) are
853 // served from Saved/ whenever a copy exists there: the content-dir copy
854 // cooked into the base pak is stale by construction.
855 {
856 FString Rest = path;
857 Rest.RemoveFromStart(ContentDir);
858 FString PackName, PackRelative;
859 FString SavedRelative = Rest;
860 if (Rest.RemoveFromStart(TEXT("Packs/")) && Rest.Split(TEXT("/"), &PackName, &PackRelative))
861 {
862 const UCarlaContentPackManager *ContentPacks = UCarlaContentPackManager::Get();
863 const FCarlaContentPack *Pack = ContentPacks ? ContentPacks->FindPack(PackName) : nullptr;
864 if (Pack != nullptr && Pack->bMounted)
865 {
866 const FString PackContentDir = Pack->ContentDir / TEXT("");
867 const FString PackPath = FPaths::ConvertRelativePathToFull(PackContentDir / PackRelative);
868 if (!PackPath.StartsWith(PackContentDir))
869 {
870 RESPOND_ERROR("file requested outside the content pack folder");
871 }
872 path = PackPath;
873 }
874 }
875 else if (SavedRelative.RemoveFromStart(TEXT("Carla/Maps/")) &&
876 (SavedRelative.StartsWith(TEXT("OpenDrive/")) || SavedRelative.StartsWith(TEXT("Nav/"))))
877 {
878 const FString SavedDir = FPaths::ConvertRelativePathToFull(FPaths::ProjectSavedDir()) / TEXT("");
879 // The xodr from where this server keeps it (a Multi-GPU secondary has
880 // a folder of its own); a secondary builds no navigation mesh, and the
881 // one in a shared Saved/ folder can be another run's.
882 const bool bOwnXODR = bIsSecondaryServer &&
883 (FPaths::GetPath(SavedRelative) == TEXT("OpenDrive")) &&
884 FPaths::GetExtension(SavedRelative).Equals(TEXT("xodr"), ESearchCase::IgnoreCase);
885 const FString SavedPath = bOwnXODR ?
886 UOpenDrive::GetSavedXODRPath(FPaths::GetBaseFilename(SavedRelative)) :
887 FPaths::ConvertRelativePathToFull(SavedDir / SavedRelative);
888 if (SavedPath.StartsWith(SavedDir) && IFileManager::Get().FileExists(*SavedPath) &&
889 !(bIsSecondaryServer && SavedRelative.StartsWith(TEXT("Nav/"))))
890 {
891 path = SavedPath;
892 }
893 }
894 }
895
896 // Copy the binary data of the file into the result and return it
897 TArray<uint8_t> Content;
898 FFileHelper::LoadFileToArray(Content, *path, 0);
899 std::vector<uint8_t> Result(Content.Num());
900 memcpy(&Result[0], Content.GetData(), Content.Num());
901
902 return Result;
903 };

1301 BIND_SYNC(get_sensor_token) << [this](carla::streaming::detail::stream_id_type sensor_id) ->
1302 R<carla::streaming::Token>
1303 {
1304 REQUIRE_CARLA_EPISODE();
1305 bool ForceInPrimary = false;
1306
1307 // check for the world observer (always in primary server)
1308 if (sensor_id == 1)
1309 {
1310 ForceInPrimary = true;
1311 }
1312
1313 // A Multi-GPU secondary serves its own world snapshots, which every client
1314 // of it subscribes to in get_world(); the sensor streams are the primary's
1315 // to route.
1316 const bool bOwnWorldObserver = (sensor_id ==
1317 carla::streaming::detail::token_type(BroadcastStream.token()).get_stream_id());
1318 if (!bOwnWorldObserver)
1319 {
1320 REQUIRE_PRIMARY_SERVER();
1321 }
1322
1323 // collision sensor always in primary server in multi-gpu
1324 FString Desc = Episode->GetActorDescriptionFromStream(sensor_id);
1325 if (Desc == "" || Desc == "sensor.other.collision")
1326 {
1327 ForceInPrimary = true;
1328 }
1329
1330 // V2X sensors too: they exchange messages between the primary's own
1331 // actors and keep per-process state, so a mirror could never match them.
1332 if (Desc == "sensor.other.v2x" || Desc == "sensor.other.v2x_custom")
1333 {
1334 ForceInPrimary = true;
1335 }
1336
1337 if (SecondaryServer->HasClientsConnected() && !ForceInPrimary)
1338 {
1339 // multi-gpu
1340 UE_LOG(LogCarla, Log, TEXT("Sensor %d '%s' created in secondary server"), sensor_id, *Desc);
1341 const FCarlaActor *SensorActor = Episode->FindCarlaActorFromStream(sensor_id);
1342 if (SensorActor == nullptr)
1343 {
1344 RESPOND_ERROR("sensor stream has no actor");
1345 }
1346 auto Token = SecondaryServer->GetCommander().GetToken(sensor_id, SensorActor->GetActorId());
1347 if (!Token)
1348 {
1349 RESPOND_ERROR("no secondary server could provide a token for this sensor");
1350 }
1351 return *Token;
1352 }
1353 else
1354 {
1355 // single-gpu
1356 if (IsUnrenderableLocalSensorStream(*Episode, sensor_id))
1357 {
1358 UE_LOG(LogCarla, Warning,
1359 TEXT("Sensor %d '%s' refused: this server cannot render (-nullrhi) and no secondary server is connected"),
1360 sensor_id, *Desc);
1361 RESPOND_ERROR("this server cannot render (-nullrhi) and no rendering secondary server is connected to own this camera sensor");
1362 }
1363 UE_LOG(LogCarla, Log, TEXT("Sensor %d '%s' created in primary server"), sensor_id, *Desc);
1364 return StreamingServer.GetToken(sensor_id);
1365 }

3682
3683 BIND_SYNC(spawn_custom_mesh) << [this](
3684 std::vector<float> vertices,
3685 std::vector<uint32_t> triangles,
3686 std::string material) -> R<void>
3687 {
3688 REQUIRE_CARLA_EPISODE();
3689 REQUIRE_PRIMARY_SERVER();
3690 auto *World = Episode->GetWorld();
3691 check(World != nullptr);
3692 if (vertices.empty() || (vertices.size() % 3u) != 0u)
3693 {
3694 RESPOND_ERROR("spawn_custom_mesh: vertices must be a non-empty flat (x, y, z) triple list");
3695 }
3696 if (triangles.empty() || (triangles.size() % 3u) != 0u)
3697 {
3698 RESPOND_ERROR("spawn_custom_mesh: triangles must be a non-empty index-triple list");
3699 }
3700 const uint32_t NumVertices = vertices.size() / 3u;
3701 for (const uint32_t Index : triangles)
3702 {
3703 if (Index >= NumVertices)
3704 {
3705 RESPOND_ERROR("spawn_custom_mesh: triangle index out of range");
3706 }
3707 }
3708
3709 // Client sends metres in the client (UE-handed) frame; UE wants cm.
3710 TArray<FVector> Verts;
3711 Verts.Reserve(NumVertices);
3712 for (uint32_t i = 0u; i < NumVertices; ++i)
3713 {
3714 Verts.Add(FVector(
3715 vertices[3u * i + 0u] * 100.0f,
3716 vertices[3u * i + 1u] * 100.0f,
3717 vertices[3u * i + 2u] * 100.0f));
3718 }
3719 TArray<int32> Tris;
3720 Tris.Reserve(triangles.size());
3721 for (const uint32_t Index : triangles)
3722 {
3723 Tris.Add(static_cast<int32>(Index));
3724 }
3725
3726 // Area-weighted vertex normals. UE's left-handed winding means the
3727 // front face is the one from which the vertices wind clockwise, so the
3728 // outward normal of triangle (V0, V1, V2) is (V2-V0) x (V1-V0).
3729 TArray<FVector> Normals;
3730 Normals.Init(FVector::ZeroVector, Verts.Num());
3731 for (int32 i = 0; i + 2 < Tris.Num(); i += 3)
3732 {
3733 const FVector &V0 = Verts[Tris[i]];
3734 const FVector &V1 = Verts[Tris[i + 1]];
3735 const FVector &V2 = Verts[Tris[i + 2]];
3736 const FVector FaceNormal = FVector::CrossProduct(V2 - V0, V1 - V0);
3737 Normals[Tris[i]] += FaceNormal;
3738 Normals[Tris[i + 1]] += FaceNormal;
3739 Normals[Tris[i + 2]] += FaceNormal;
3740 }
3741 for (FVector &Normal : Normals)
3742 {
3743 Normal = Normal.GetSafeNormal(1e-8, FVector::UpVector);
3744 }
3745
3746 // Planar world-space UVs so tiled materials repeat sensibly (5 m tile).
3747 TArray<FVector2D> UV0;
3748 UV0.Reserve(Verts.Num());
3749 for (const FVector &V : Verts)
3750 {
3751 UV0.Add(FVector2D(V.X / 500.0f, V.Y / 500.0f));
3752 }
3753
3754 ACarlaProceduralMeshActor *MeshActor = World->SpawnActor<ACarlaProceduralMeshActor>();
3755 if (MeshActor == nullptr)
3756 {
3757 RESPOND_ERROR("spawn_custom_mesh: failed to spawn mesh actor");
3758 }
3759 UProceduralMeshComponent *PMC = MeshActor->MeshComponent;
3760 PMC->bUseAsyncCooking = true;
3761 PMC->bUseComplexAsSimpleCollision = true;
3762 PMC->SetCollisionEnabled(ECollisionEnabled::QueryAndPhysics);
3763 PMC->CreateMeshSection_LinearColor(
3764 0,
3765 Verts,
3766 Tris,
3767 Normals,
3768 UV0,
3769 TArray<FLinearColor>(),
3770 TArray<FProcMeshTangent>(),
3771 true);
3772
3773 static const TMap<FString, FString> MaterialHints = {
3774 {TEXT("grass"), TEXT("/Game/Carla/Static/GenericMaterials/Ground/MI_LargeLandscape_Grass.MI_LargeLandscape_Grass")},
3775 {TEXT("dirt"), TEXT("/Game/Carla/Static/GenericMaterials/Ground/MI_Dirt.MI_Dirt")},
3776 {TEXT("road"), TEXT("/Game/Carla/Static/GenericMaterials/Roads/MI_RoadAsphalt_Town15.MI_RoadAsphalt_Town15")},
3777 {TEXT("sidewalk"), TEXT("/Game/Carla/Static/GenericMaterials/Sidewalk/MI_Sidewalk_Apartment.MI_Sidewalk_Apartment")}};
3778 const FString MaterialName = cr::ToFString(material);
3779 const FString *MaterialPath = MaterialHints.Find(MaterialName.ToLower());
3780 const FString ResolvedPath = (MaterialPath != nullptr) ? *MaterialPath : MaterialName;
3781 if (UMaterialInterface *Material = LoadObject<UMaterialInterface>(nullptr, *ResolvedPath))
3782 {
3783 PMC->SetMaterial(0, Material);
3784 }
3785
3786 // Tag as terrain for semantic segmentation, mirroring the generated
3787 // ground plane (OpenDriveGenerator's TagGeneratedComponent).
3788 ATagger::SetStencilValue(*PMC, MeshActor->GetUniqueID(), cr::CityObjectLabel::Terrain, true);
3789 PMC->ComponentTags.Add(FName(*ATagger::GetTagAsString(cr::CityObjectLabel::Terrain)));
3790
3791 return R<void>::Success();
3792 };

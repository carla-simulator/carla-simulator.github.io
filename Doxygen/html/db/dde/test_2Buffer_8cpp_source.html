14 shared_buffer make_random(size_t size) {
15 if (size == 0u)
16 return make_empty();
17 // independent_bits_engine requires an unsigned result type wider than
18 // char ([rand.req.genl]/1.6); MSVC enforces this with a static_assert.
19 using random_bytes_engine = std::independent_bits_engine<
20 std::random_device,
21 CHAR_BIT,
22 unsigned short>;
23 random_bytes_engine rbe;
24 auto buffer = make_empty(size);
25 std::generate(buffer->begin(), buffer->end(), [&rbe]() {
26 return static_cast<unsigned char>(rbe());
27 });
28 return buffer;
29 }

31 std::string to_hex_string(const Buffer &buf, size_t length) {
32 length = std::min(static_cast<size_t>(buf.size()), length);
33 auto buffer = std::make_unique<char[]>(2u * length + 1u);
34 for (auto i = 0u; i < length; ++i)

36 if (length < buf.size())
37 return std::string(buffer.get()) + std::string("...");
38 return std::string(buffer.get());

21 Server::Server(boost::asio::io_context &io_context, endpoint ep)

33 void Server::OpenSession(
34 time_duration timeout,
35 ServerSession::callback_function_type on_opened,
36 ServerSession::callback_function_type on_closed) {
37 using boost::system::error_code;
38
39 auto session = std::make_shared<ServerSession>(_io_context, timeout, *this);
40
41 auto handle_query = [on_opened, on_closed, session](const error_code &ec) {
42 if (!ec) {
43 session->Open(std::move(on_opened), std::move(on_closed));
44 } else {
45 log_error("tcp accept stream error:", ec.message());
46 }
47 };
48
49 _acceptor.async_accept(session->_socket, [=](error_code ec) {
50 // Handle query and open a new session immediately.
51 boost::asio::post(_io_context, [=, this]() { handle_query(ec); });
52 OpenSession(timeout, on_opened, on_closed);
53 });
54 }

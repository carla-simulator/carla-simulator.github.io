32// Registered by CarlaDeviceProfileSelectorModule, which loads first and sets it
33// per quality tier. -1 follows each sensor's bUseRayTracing, 0 forces ray
34// tracing off and 1 forces it on for every camera.
35static int32 GetCameraRayTracingOverride()
36{
37 static std::atomic<const TConsoleVariableData<int32> *> CVar{nullptr};
38 const TConsoleVariableData<int32> *Cached = CVar.load(std::memory_order_relaxed);
39 if (Cached == nullptr)
40 {
41 Cached = IConsoleManager::Get().FindTConsoleVariableDataInt(TEXT("carla.Camera.UseRayTracing"));
42 CVar.store(Cached, std::memory_order_relaxed);
43 }
44 return (Cached != nullptr) ? Cached->GetValueOnAnyThread() : -1;
45}

59// Exposure-linked motion blur for the RGB cameras. A real camera blurs what
60// moves during its exposure time, which its auto exposure lengthens in the
61// dark (a machine-vision AV camera: ~0.1 ms by day, capped around 20 ms at
62// night, gain above that). Every capture reads the exposure the sensor's own
63// eye adaptation settled on (EV100 = log2(1 / (1.2 * Exposure))), turns it into
64// an exposure time t = N^2 / 2^EV100 for an f/N lens at base ISO, clamps it and
65// sets MotionBlurAmount = t / capture interval (1 = 360 degree shutter). Night
66// lamps then streak along their motion instead of twinkling between pixels.
67static TAutoConsoleVariable<int32> CVarCarlaCameraExposureMotionBlur(
68 TEXT("carla.Camera.ExposureMotionBlur"),
69 1,
70 TEXT("1: RGB cameras derive their motion blur from the auto exposure time (overrides motion_blur_intensity).\n")
71 TEXT("0: use the motion_blur_* attributes / post-process profile as they are."),
72 ECVF_Default);
73
74static TAutoConsoleVariable<float> CVarCarlaCameraFNumber(
75 TEXT("carla.Camera.FNumber"),
76 2.0f,
77 TEXT("Lens f-number used to turn the auto exposure into an exposure time (carla.Camera.ExposureMotionBlur)."),
78 ECVF_Default);

92// A new camera has no eye adaptation history: left alone it starts from a
93// default exposure and takes seconds to adapt (a white sky flash every time a
94// client respawns a camera, e.g. manual_control's TAB). Its first captures are
95// rendered as camera cuts, which makes the eye adaptation jump to its target:
96// the first image is still metered from scratch, the second one is settled.
97// (Extra unsent captures in the same tick do not help: the exposure a capture
98// computes only reaches the captures of later ticks.)
99static TAutoConsoleVariable<int32> CVarCarlaCameraSpawnCameraCuts(
100 TEXT("carla.Camera.SpawnCameraCuts"),
101 2,
102 TEXT("Number of first captures of a new camera rendered as camera cuts, so its auto exposure settles ")
103 TEXT("from the second image on instead of adapting for seconds from a default exposure (0 disables)."),
104 ECVF_Default);

135ASceneCaptureSensor::ASceneCaptureSensor(const FObjectInitializer &ObjectInitializer)
136 : Super(ObjectInitializer)
137{
138 PrimaryActorTick.bCanEverTick = true;
139 PrimaryActorTick.TickGroup = TG_PrePhysics;
140
141 // Snapshot the counter once so the render target and capture component
142 // share the same suffix on this instance.
143 const int Index = SCENE_CAPTURE_COUNTER.fetch_add(1, std::memory_order_relaxed);
144
145 CaptureRenderTarget = CreateDefaultSubobject<UTextureRenderTarget2D>(
146 FName(*FString::Printf(TEXT("CaptureRenderTarget_d%d"), Index)));
147 CaptureRenderTarget->CompressionSettings = TextureCompressionSettings::TC_Default;
148 CaptureRenderTarget->SRGB = false;
149 CaptureRenderTarget->bAutoGenerateMips = false;
150 CaptureRenderTarget->AddressX = TextureAddress::TA_Clamp;
151 CaptureRenderTarget->AddressY = TextureAddress::TA_Clamp;
152
153 CaptureComponent2D = CreateDefaultSubobject<USceneCaptureComponent2D_CARLA>(
154 FName(*FString::Printf(TEXT("USceneCaptureComponent2D%d"), Index)));
155 check(CaptureComponent2D != nullptr);
156 CaptureComponent2D->ViewActor = this;
157 CaptureComponent2D->SetupAttachment(RootComponent);
158 CaptureComponent2D->PrimitiveRenderMode = ESceneCapturePrimitiveRenderMode::PRM_RenderScenePrimitives;
159 CaptureComponent2D->bCaptureOnMovement = false;
160 CaptureComponent2D->bCaptureEveryFrame = false;
161 // Persistent rendering state is the baseline for every scene-capture
162 // subclass. The encoding cameras (depth, semantic / instance
163 // segmentation, normals, DVS) require it for their post-process
164 // material pipeline; RGB and optical-flow inherit it without needing
165 // to re-assert in their own ctors.
166 CaptureComponent2D->bAlwaysPersistRenderingState = true;
167 ApplyRayTracingSetting();
168
169 SceneCaptureSensor_local_ns::SetCameraDefaultOverrides(*CaptureComponent2D);
170}

199bool ASceneCaptureSensor::ResolveCameraRayTracing(const bool bSensorEnabled, const bool bRequired)
200{
201 if (bRequired)

1007void ASceneCaptureSensor::BeginPlay()
1008{
1009 using namespace SceneCaptureSensor_local_ns;
1010
1011 // Determine the gamma of the player.
1012 const bool bInForceLinearGamma = !bEnablePostProcessingEffects;
1013
1014 CaptureRenderTarget->InitCustomFormat(
1015 ImageWidth,
1016 ImageHeight,
1017 bEnable16BitFormat ? PF_FloatRGBA : PF_B8G8R8A8,
1018 bInForceLinearGamma);
1019
1020 if (bEnablePostProcessingEffects)
1021 {
1022 CaptureRenderTarget->TargetGamma = TargetGamma;
1023 }
1024
1025 check(IsValid(CaptureComponent2D) && IsValidChecked(CaptureComponent2D));
1026
1027 CaptureComponent2D->Deactivate();
1028 CaptureComponent2D->TextureTarget = CaptureRenderTarget;
1029
1030 // Optional DLSS upscaling: render the capture at a fraction of the target
1031 // resolution and let the engine's scene-capture temporal upscaler (DLSS
1032 // Super Resolution, bilinear fallback on non-NVIDIA hardware) fill the
1033 // render target. The render target itself stays at ImageWidth x ImageHeight,
1034 // so client-visible output resolution is unchanged.
1035 CaptureComponent2D->CaptureScreenPercentage =
1036 bEnableDLSSUpscale ? FMath::Clamp(DLSSScreenPercentage, 25.0f, 100.0f) : 100.0f;
1037 if (bEnableDLSSUpscale && !bEnablePostProcessingEffects)
1038 {
1039 UE_LOG(LogCarla, Warning, TEXT(
1040 "%s: DLSS upscaling requested but post-processing effects are disabled; "
1041 "the capture has no temporal AA method, so the engine will spatially "
1042 "upscale instead of DLSS."), *GetName());
1043 }
1044
1045 // Call derived classes to set up their things.
1046 SetUpSceneCaptureComponent(*CaptureComponent2D);
1047
1048 // The capture component's own settings assert bOverride_ on every camera
1049 // field (SetCameraDefaultOverrides), so map post-process volumes never blend
1050 // into sensor renders and sensors don't match the main viewport. Absorb the
1051 // map's enabled unbound PostProcessVolume here; the cache logic inside
1052 // ApplyPostProcessVolumeToSensor keeps the camera attributes configured
1053 // through the sensor API (exposure, film curve, DoF, ...) authoritative.
1054 {
1055 int32 VolumeCount = 0;
1056 bool bApplied = false;
1057 for (TActorIterator<APostProcessVolume> It(GetWorld()); It; ++It)
1058 {
1059 APostProcessVolume *Volume = *It;
1060 ++VolumeCount;
1061 if (!bApplied && bEnablePostProcessingEffects &&
1062 IsValid(Volume) && Volume->bEnabled && Volume->bUnbound)
1063 {
1064 ApplyPostProcessVolumeToSensor(Volume, this, false);
1065 UE_LOG(LogCarla, Log, TEXT("%s: absorbed map post-process volume %s"),
1066 *GetName(), *Volume->GetName());
1067 bApplied = true;
1068 }
1069 }
1070 if (!bApplied && VolumeCount > 0)
1071 {
1072 UE_LOG(LogCarla, Verbose, TEXT(
1073 "%s: no map post-process volume absorbed (bEnablePP=%d, volumes in world=%d)"),
1074 *GetName(), bEnablePostProcessingEffects ? 1 : 0, VolumeCount);
1075 }
1076 }
1077
1078 CaptureComponent2D->CaptureSource = ESceneCaptureSource::SCS_FinalToneCurveHDR;
1079
1080 CaptureComponent2D->UpdateContent();
1081 CaptureComponent2D->Activate();
1082
1083 // Nudge texture streaming priority at the new sensor's location so
1084 // a freshly attached camera does not stall the first few captures
1085 // waiting for its local region to reach full mip. One-shot 2-second
1086 // duration fades out naturally; permanent hints would widen the
1087 // priority surface on multi-camera dataset runs (N sensors = N
1088 // permanent streaming sources).
1089 IStreamingManager::Get().AddViewLocation(
1090 GetActorLocation(),
1091 /*BoostFactor=*/ 1.5f,
1092 /*bOverrideLocation=*/ true,
1093 /*Duration=*/ 2.0f);
1094
1095 auto PostProcessConfig = FPostProcessConfig(
1096 CaptureComponent2D->PostProcessSettings,
1097 CaptureComponent2D->ShowFlags);
1098 PostProcessConfig.UpdateFromSceneCaptureComponent2D(*CaptureComponent2D);
1099 PostProcessConfig.EnablePostProcessingEffects(ArePostProcessingEffectsEnabled());
1100 UpdatePostProcessConfig(PostProcessConfig);
1101 CaptureComponent2D->ShowFlags = PostProcessConfig.EngineShowFlags;
1102 CaptureComponent2D->PostProcessSettings = PostProcessConfig.PostProcessSettings;
1103
1104 // Exposure-linked motion blur (UpdateExposureMotionBlur) manages the flag
1105 // itself; otherwise motion blur stays off below 1080p as before.
1106 if ((ImageWidth < 1920 || ImageHeight < 1080) &&
1107 (CVarCarlaCameraExposureMotionBlur.GetValueOnGameThread() == 0 || !ArePostProcessingEffectsEnabled()))
1108 CaptureComponent2D->ShowFlags.SetMotionBlur(false);
1109
1110 // This ensures the camera is always spawning the raindrops in case the
1111 // weather was previously set to have rain.
1112 auto Weather = GetEpisode().GetWeather();
1113 if (Weather != nullptr)
1114 Weather->NotifyWeather(this);
1115
1116 ReadbackPool = MakeShared<FRHIGPUReadbackPool, ESPMode::ThreadSafe>(
1117 TEXT("SceneCaptureReadback"));
1118
1119 Super::BeginPlay();
1120}

1198void ASceneCaptureSensor::UpdateExposureMotionBlur(UWorld *World, float DeltaTime)
1199{
1200 // Simulation time between two captures of this sensor: the motion vectors
1201 // span it (the view state keeps the previous capture's matrices).
1202 const double Now = World != nullptr ? World->GetTimeSeconds() : 0.0;
1203 const float Interval = LastCaptureTime >= 0.0 && Now > LastCaptureTime
1204 ? static_cast<float>(Now - LastCaptureTime)
1205 : FMath::Max(DeltaTime, 1e-3f);
1206 LastCaptureTime = Now;
1207
1208 if (!ArePostProcessingEffectsEnabled() || CVarCarlaCameraExposureMotionBlur.GetValueOnGameThread() == 0)
1209 {
1210 return;
1211 }
1212 FSceneViewStateInterface *ViewState = CaptureComponent2D->GetViewState(0);
1213 const float Exposure = ViewState != nullptr ? ViewState->GetLastEyeAdaptationExposure() : 0.0f;
1214 if (!(Exposure > 0.0f) || !FMath::IsFinite(Exposure))
1215 {
1216 return; // no eye adaptation readback yet (first frames)
1217 }
1218 const float FNumber = FMath::Max(CVarCarlaCameraFNumber.GetValueOnGameThread(), 0.5f);
1219 const float MinSeconds = FMath::Max(CVarCarlaCameraMinExposureMs.GetValueOnGameThread(), 0.0f) * 1e-3f;
1220 const float MaxSeconds = FMath::Max(CVarCarlaCameraMaxExposureMs.GetValueOnGameThread() * 1e-3f, MinSeconds);
1221 ExposureSeconds = FMath::Clamp(FNumber * FNumber * 1.2f * Exposure, MinSeconds, MaxSeconds);
1222
1223 FPostProcessSettings &Settings = CaptureComponent2D->PostProcessSettings;
1224 Settings.bOverride_MotionBlurAmount = true;
1225 Settings.MotionBlurAmount = FMath::Clamp(ExposureSeconds / FMath::Max(Interval, 1e-3f), 0.0f, 1.0f);
1226 // 0 = follow the render rate, so the blur length does not depend on how
1227 // fast the simulator runs in wall-clock time (a fixed target FPS scales it
1228 // by target / real delta, wrong in synchronous mode).
1229 Settings.bOverride_MotionBlurTargetFPS = true;
1230 Settings.MotionBlurTargetFPS = 0;
1231 CaptureComponent2D->ShowFlags.SetMotionBlur(Settings.MotionBlurAmount > 0.001f);
1232
1233 if (CVarCarlaCameraExposureLog.GetValueOnGameThread() != 0)
1234 {
1235 UE_LOG(LogCarla, Log, TEXT("%s exposure %.4g (EV100 %.2f) -> %.3f ms, interval %.1f ms, motion blur %.3f"),
1236 *GetName(), Exposure, FMath::Log2(1.0f / (1.2f * Exposure)), ExposureSeconds * 1e3f, Interval * 1e3f,
1237 Settings.MotionBlurAmount);
1238 }
1239}

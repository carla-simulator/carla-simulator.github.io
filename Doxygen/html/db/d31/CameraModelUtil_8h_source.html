44// Shared lens-model descriptor: single source of truth for lens forward/inverse
45// mapping math, mirrored by the GPU core in Engine/Shaders/Private/LensModels.ush.
46//
47// Coordinate convention for all CameraModelUtil forward/inverse ray functions
48// below (camera space, matches the GPU core, NOT Unreal's native +X-forward
49// camera convention -- callers must convert):
50// +X = right, +Y = down, +Z = forward.
51//
52// Coeffs packing:
53// KannalaBrandt : Coeffs[0..3] = k1..k4
54// BrownConrady : Coeffs[0..2] = k1,k2,k3 (radial), Coeffs[3..4] = p1,p2 (tangential), OpenCV convention
55// LUT1D : Coeffs unused; LUT holds the samples instead
56//
57// LUT1D: LUT[i] is R(Theta) uniformly sampled for Theta in [0, ThetaMax],
58// i = 0 .. LUT.Num() - 1, assumed monotonically non-decreasing.
59struct FLensModelDescriptor
60{
61 ECameraModel Model = ECameraModel::Perspective;
62 float FocalX = 1.f, FocalY = 1.f; // normalized focal lengths
63 float CenterX = 0.5f, CenterY = 0.5f; // normalized principal point
64 TArray<float> Coeffs; // KB: k1..k4; BC: k1,k2,k3,p1,p2
65 TArray<float> LUT; // LUT1D samples theta->r
66 float ThetaMax = PI/2; // max half-FOV angle (radians)
67 float CAScaleR = 1.f, CAScaleB = 1.f; // per-channel radial scale for chromatic aberration
68};

135 // Forward map: camera-space direction (need not be normalized; +X right,
136 // +Y down, +Z forward -- see FLensModelDescriptor) -> normalized
137 // image-plane point (Descriptor.CenterX/Y +/- Descriptor.FocalX/Y * extent).
138 FVector2f ComputeLensImagePoint(

140 FVector Direction,
141 ELensColorChannel Channel = ELensColorChannel::Green);

143 // Inverse map: normalized image-plane point -> camera-space unit ray
144 // direction (+X right, +Y down, +Z forward). Uses Newton-Raphson for
145 // KannalaBrandt and BrownConrady, closed form for the other radial models,
146 // and binary-search interpolation for LUT1D.
147 FVector ComputeLensRayDirection(
148 const FLensModelDescriptor& Descriptor,
149 FVector2f ImagePoint,
150 ELensColorChannel Channel = ELensColorChannel::Green);
151
152 // Convention adapters between the lens core's camera space (+X right,
153 // +Y down, +Z forward -- matches OpenCV, see FLensModelDescriptor) and
154 // Unreal's native camera space (+X forward, +Y right, +Z up). Pure axis
155 // relabeling: an isometry, and each is the other's exact inverse. Not a
156 // "rotation" in the proper sense -- the lens core's convention is
157 // right-handed and UE's is left-handed, so this necessarily flips
158 // handedness rather than preserving it.
159 FVector LensDirToUE(FVector LensDir);
160 FVector UEDirToLens(FVector UEDir);

17 /// Frame data packet 25: the primary's clocks and episode settings for one
18 /// frame. A secondary plays frame N with exactly this delta and these times,
19 /// so whatever it computes is stamped and integrated like on the primary.
20 struct FrameTime {
21
22 static constexpr uint8_t kPacketId = 25u;
23
24 /// Size of the version 1 payload; later versions only append fields.
25 static constexpr uint32_t kPayloadSizeV1 = 98u;
26
27 uint8_t protocol_version = kMultiGpuProtocolVersion;
28
29 /// Frame counter of frame N; equal to the frame's FrameCounter packet.
30 uint64_t frame = 0u;
31
32 /// The raw DeltaSeconds that UWorld::Tick(N) received, before time
33 /// dilation and clamping.
34 float delta_seconds = 0.0f;
35
36 /// UCarlaEpisode clocks at the end of frame N.
37 double elapsed_game_time = 0.0;
38 double visual_game_time = 0.0;
39
40 /// UWorld clocks at the start of frame N, before UWorld::Tick adds the
41 /// delta to them.
42 double world_time_seconds = 0.0;
43 double world_unpaused_time_seconds = 0.0;
44 double world_real_time_seconds = 0.0;
45 double world_audio_time_seconds = 0.0;
46
47 /// The primary's episode settings after frame N (FEpisodeSettings, in UE
48 /// units). no_rendering_mode is informational: secondaries keep their own.
49 bool synchronous_mode = false;
50 bool no_rendering_mode = false;
51 std::optional<double> fixed_delta_seconds;
52 bool substepping = true;
53 double max_substep_delta_time = 0.01;
54 int32_t max_substeps = 10;
55 float max_culling_distance = 0.0f;
56 bool deterministic_ragdolls = true;
57 float tile_streaming_distance = 300000.0f;
58 float actor_active_distance = 200000.0f;
59 bool spectator_as_ego = true;
60 bool adjust_signs_height_to_ground = false;
61
62 /// AWorldSettings::GetEffectiveTimeDilation() of frame N.
63 float time_dilation = 1.0f;
64
65 /// Set on the frame that rebuilds a secondary from scratch (a new or
66 /// reconnected secondary, or the first frame after a map load).
67 bool full_resync = false;
68
69 void Write(std::ostream &out) const {
70 WriteFrameDataHeader(out, kPacketId, kPayloadSizeV1);
71 detail::WritePod(out, protocol_version);
72 detail::WritePod(out, PackFlags());
73 detail::WritePod(out, frame);
74 detail::WritePod(out, delta_seconds);

79 detail::WritePod(out, world_real_time_seconds);
80 detail::WritePod(out, world_audio_time_seconds);
81 detail::WritePod(out, fixed_delta_seconds.value_or(0.0));
82 detail::WritePod(out, max_substep_delta_time);
83 detail::WritePod(out, max_substeps);
84 detail::WritePod(out, max_culling_distance);
85 detail::WritePod(out, tile_streaming_distance);
86 detail::WritePod(out, actor_active_distance);
87 detail::WritePod(out, time_dilation);
88 }

90 /// Reads a payload of @a size bytes. Fails (and fails @a in) on a payload
91 /// shorter than version 1; bytes beyond the fields known here are left for
92 /// the packet loop to skip.
93 bool Read(std::istream &in, uint32_t size) {
94 if (size < kPayloadSizeV1) {
95 in.setstate(std::ios_base::failbit);
96 return false;
97 }
98 BoundedReader reader(in, size);
99 uint8_t flags = 0u;
100 double fixed_delta = 0.0;
101 const bool ok =
102 reader.Read(protocol_version) &&
103 reader.Read(flags) &&
104 reader.Read(frame) &&
105 reader.Read(delta_seconds) &&
106 reader.Read(elapsed_game_time) &&
107 reader.Read(visual_game_time) &&
108 reader.Read(world_time_seconds) &&
109 reader.Read(world_unpaused_time_seconds) &&
110 reader.Read(world_real_time_seconds) &&
111 reader.Read(world_audio_time_seconds) &&
112 reader.Read(fixed_delta) &&
113 reader.Read(max_substep_delta_time) &&
114 reader.Read(max_substeps) &&
115 reader.Read(max_culling_distance) &&
116 reader.Read(tile_streaming_distance) &&
117 reader.Read(actor_active_distance) &&
118 reader.Read(time_dilation);
119 if (!ok) {
120 return false;
121 }
122 UnpackFlags(flags);
123 fixed_delta_seconds.reset();
124 if ((flags & kFixedDelta) != 0u) {
125 fixed_delta_seconds = fixed_delta;
126 }
127 return true;
128 }

132 enum Flag : uint8_t {
133 kSynchronousMode = 1u << 0u,
134 kFixedDelta = 1u << 1u,
135 kSubstepping = 1u << 2u,
136 kDeterministicRagdolls = 1u << 3u,
137 kSpectatorAsEgo = 1u << 4u,
138 kAdjustSignsHeight = 1u << 5u,
139 kNoRenderingMode = 1u << 6u,
140 kFullResync = 1u << 7u
141 };
142
143 uint8_t PackFlags() const {
144 uint8_t flags = 0u;
145 flags |= synchronous_mode ? kSynchronousMode : 0u;
146 flags |= fixed_delta_seconds.has_value() ? kFixedDelta : 0u;
147 flags |= substepping ? kSubstepping : 0u;
148 flags |= deterministic_ragdolls ? kDeterministicRagdolls : 0u;
149 flags |= spectator_as_ego ? kSpectatorAsEgo : 0u;
150 flags |= adjust_signs_height_to_ground ? kAdjustSignsHeight : 0u;
151 flags |= no_rendering_mode ? kNoRenderingMode : 0u;
152 flags |= full_resync ? kFullResync : 0u;
153 return flags;
154 }
155
156 void UnpackFlags(uint8_t flags) {
157 synchronous_mode = (flags & kSynchronousMode) != 0u;
158 substepping = (flags & kSubstepping) != 0u;
159 deterministic_ragdolls = (flags & kDeterministicRagdolls) != 0u;
160 spectator_as_ego = (flags & kSpectatorAsEgo) != 0u;
161 adjust_signs_height_to_ground = (flags & kAdjustSignsHeight) != 0u;
162 no_rendering_mode = (flags & kNoRenderingMode) != 0u;
163 full_resync = (flags & kFullResync) != 0u;
164 }
165 };

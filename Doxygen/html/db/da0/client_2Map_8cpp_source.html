76 Map::TopologyList Map::GetTopology() const {
77 namespace re = carla::road::element;
78 std::unordered_map<re::Waypoint, SharedPtr<Waypoint>> waypoints;
79
80 auto get_or_make_waypoint = [&](const auto &waypoint) {
81 auto it = waypoints.find(waypoint);
82 if (it == waypoints.end()) {
83 it = waypoints.emplace(
84 waypoint,
85 SharedPtr<Waypoint>(new Waypoint{shared_from_this(), waypoint})).first;
86 }
87 return it->second;
88 };
89
90 TopologyList result;
91 auto topology = _map.GenerateTopology();
92 result.reserve(topology.size());
93 for (const auto &pair : topology) {
94 result.emplace_back(
95 get_or_make_waypoint(pair.first),
96 get_or_make_waypoint(pair.second));
97 }
98 return result;
99 }

253 std::vector<SharedPtr<Landmark>>
254 Map::GetLandmarkGroup(const Landmark &landmark) const {
255 std::vector<SharedPtr<Landmark>> result;
256 auto &controllers = landmark._signal->GetSignal()->GetControllers();
257 for (auto& controller_id : controllers) {
258 const auto &controller = _map.GetControllers().at(controller_id);
259 for(auto& signal_id : controller->GetSignals()) {
260 auto& signal = _map.GetSignals().at(signal_id);
261 auto new_landmarks = GetLandmarksFromId(signal->GetSignalId());
262 result.insert(result.end(), new_landmarks.begin(), new_landmarks.end());
263 }
264 }
265 return result;

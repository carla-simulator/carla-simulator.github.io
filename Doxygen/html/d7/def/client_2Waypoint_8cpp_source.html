80 std::vector<SharedPtr<Waypoint>> Waypoint::GetNextUntilLaneEnd(double distance) const {
81 std::vector<SharedPtr<Waypoint>> result;
82 std::vector<SharedPtr<Waypoint>> next = GetNext(distance);
83
84 while (next.size() == 1 && next.front()->GetRoadId() == GetRoadId()) {
85 result.emplace_back(next.front());
86 next = result.back()->GetNext(distance);
87 }
88 double current_s = GetDistance();
89 if(result.size()) {
90 current_s = result.back()->GetDistance();
91 }
92 double remaining_length;
93 double road_length = _parent->GetMap().GetLane(_waypoint).GetRoad()->GetLength();
94 if(IsPositiveDirection()) {
95 remaining_length = road_length - current_s;
96 } else {
97 remaining_length = current_s;
98 }
99 remaining_length -= std::numeric_limits<double>::epsilon();
100 if(result.size()) {
101 result.emplace_back(result.back()->GetNext(remaining_length).front());
102 } else {
103 result.emplace_back(GetNext(remaining_length).front());
104 }
105
106 return result;
107 }

190 road::element::LaneMarking::LaneChange Waypoint::GetLaneChange() const {
191 using lane_change_type = road::element::LaneMarking::LaneChange;
192
193 auto road = _parent->GetMap().GetLane(_waypoint).GetRoad();
194
195 const auto lane_change_right_info = road->IsRHT() ? _mark_record.first : _mark_record.second;
196 lane_change_type c_right;
197 if (lane_change_right_info != nullptr) {
198 const auto lane_change_right = lane_change_right_info->GetLaneChange();
199 c_right = static_cast<lane_change_type>(lane_change_right);
200 } else {
201 c_right = lane_change_type::Both;
202 }
203
204 const auto lane_change_left_info = road->IsRHT() ? _mark_record.second : _mark_record.first;
205 lane_change_type c_left;
206 if (lane_change_left_info != nullptr) {
207 const auto lane_change_left = lane_change_left_info->GetLaneChange();
208 c_left = static_cast<lane_change_type>(lane_change_left);
209 } else {
210 c_left = lane_change_type::Both;
211 }
212
213 if (!IsPositiveDirection()) {
214 // if road goes backward
215 if (c_right == lane_change_type::Right) {
216 c_right = lane_change_type::Left;
217 } else if (c_right == lane_change_type::Left) {
218 c_right = lane_change_type::Right;
219 }
220 }
221
222 if (((!IsPositiveDirection()) ? _waypoint.lane_id - 1 : _waypoint.lane_id + 1) < 0) {
223 // if road goes backward
224 if (c_left == lane_change_type::Right) {
225 c_left = lane_change_type::Left;
226 } else if (c_left == lane_change_type::Left) {
227 c_left = lane_change_type::Right;
228 }
229 }
230
231 return (c_right & lane_change_type::Right) | (c_left & lane_change_type::Left);
232 }

234 std::vector<SharedPtr<Landmark>> Waypoint::GetAllLandmarksInDistance(
235 double distance, bool stop_at_junction) const {
236 std::vector<SharedPtr<Landmark>> result;
237 auto signals = _parent->GetMap().GetSignalsInDistance(
238 _waypoint, distance, stop_at_junction);
239 std::unordered_set<const road::element::RoadInfoSignal*> added_signals; // check for repeated signals
240 for(auto &signal_data : signals){
241 if(added_signals.count(signal_data.signal) > 0) {
242 continue;
243 }
244 added_signals.insert(signal_data.signal);
245 auto waypoint = SharedPtr<Waypoint>(new Waypoint(_parent, signal_data.waypoint));
246 result.emplace_back(
247 new Landmark(waypoint, _parent, signal_data.signal, signal_data.accumulated_s));
248 }
249 return result;
250 }

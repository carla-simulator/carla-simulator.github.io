23TArray<FEnvironmentObject> UObjectRegister::GetEnvironmentObjects(uint8 InTagQueried) const
24{
25 TArray<FEnvironmentObject> Result;
26
27 crp::CityObjectLabel TagQueried = (crp::CityObjectLabel)InTagQueried;
28 bool FilterByTagEnabled = (TagQueried != crp::CityObjectLabel::Any);
29
30 for(const FEnvironmentObject& It : EnvironmentObjects)
31 {
32 if(!FilterByTagEnabled || (It.ObjectLabel == TagQueried))

36 }
37

41void UObjectRegister::RegisterObjects(TArray<AActor*> Actors)
42{
43 // Empties the array but doesn't change memory allocations
44 EnvironmentObjects.Reset();
45 // Kept across registrations: a hidden traffic-light box registers no more
46 // (it has no bounds), and its id must still reach its components to show it
47 // again. Only components that are gone are dropped.
48 for (auto It = ObjectIdToComp.CreateIterator(); It; ++It)
49 {
50 if (!It.Value().IsValid())
51 {
52 It.RemoveCurrent();
53 }
54 }
55 // Foliage instance ids count instances from here: from zero on every
56 // registration, an instance keeps its id whenever the objects register
57 // again (a map layer loaded or unloaded), and two servers with the same
58 // levels loaded give it the same one.
59 FoliageActorInstanceCount = 0;
60
61 for(AActor* Actor : Actors)
62 {
63
64 FString ClassName = Actor->GetClass()->GetName();
65 // Discard Sky to not break the global ilumination
66 if(ClassName.Equals("BP_Sky_C")) continue;
67
68 ACarlaWheeledVehicle* Vehicle = Cast<ACarlaWheeledVehicle>(Actor);
69 if (Vehicle)
70 {
71 RegisterVehicle(Vehicle);
72 continue;
73 }
74
75 ACharacter* Character = Cast<ACharacter>(Actor);
76 if (Character)
77 {
78 RegisterCharacter(Character);
79 continue;
80 }
81
82 ATrafficLightBase* TrafficLight = Cast<ATrafficLightBase>(Actor);
83 if(TrafficLight)
84 {
85 RegisterTrafficLight(TrafficLight);
86 continue;
87 }
88
89 RegisterISMComponents(Actor);
90
91 RegisterSMComponents(Actor);
92
93 RegisterSKMComponents(Actor);
94 }
95
96#if WITH_EDITOR
97 // To help debug
98 FString FileContent;
99 FileContent += FString::Printf(TEXT("Num actors %d\n"), Actors.Num());
100 FileContent += FString::Printf(TEXT("Num registered objects %d\n\n"), EnvironmentObjects.Num());
101
102 for(const FEnvironmentObject& Object : EnvironmentObjects)
103 {
104 FileContent += FString::Printf(TEXT("%llu\t"), Object.Id);
105 FileContent += FString::Printf(TEXT("%s\t"), *Object.Name);
106 FileContent += FString::Printf(TEXT("%s\t"), *Object.IdStr);
107 FileContent += FString::Printf(TEXT("%d\n"), static_cast<int32>(Object.Type));
108 }
109
110
111 FString FilePath = FPaths::ProjectSavedDir() + "RegisteredObjects.txt";
112 FFileHelper::SaveStringToFile(
113 FileContent,
114 *FilePath,
115 FFileHelper::EEncodingOptions::AutoDetect,
116 &IFileManager::Get(),
117 EFileWrite::FILEWRITE_Silent);
118
119#endif // WITH_EDITOR
120
121}

183void UObjectRegister::RegisterEnvironmentObject(

220void UObjectRegister::RegisterTrafficLight(ATrafficLightBase* TrafficLight)
221{
222 check(TrafficLight);
223
224 TArray<FBoundingBox> BBs;
225 TArray<uint8> Tags;
226
227 UBoundingBoxCalculator::GetTrafficLightBoundingBox(TrafficLight, BBs, Tags);
228 check(BBs.Num() == Tags.Num());
229
230 const FTransform Transform = TrafficLight->GetTransform();
231 const FString ActorName = TrafficLight->GetName();
232 const bool IsActorTickEnabled = TrafficLight->IsActorTickEnabled();
233
234 for(int i = 0; i < BBs.Num(); i++)
235 {
236 const FBoundingBox& BB = BBs[i];
237 const uint8 Tag = Tags[i];
238
239 crp::CityObjectLabel ObjectLabel = static_cast<crp::CityObjectLabel>(Tag);
240
241 const FString TagString = ATagger::GetTagAsString(ObjectLabel);
242 const FString SMName = FString::Printf(TEXT("%s_%s_%d"), *ActorName, *TagString, i);
243
244 FEnvironmentObject EnvironmentObject;
245 EnvironmentObject.Transform = Transform;
246 EnvironmentObject.Id = CityHash64(TCHAR_TO_ANSI(*SMName), SMName.Len());
247 EnvironmentObject.Name = SMName;
248 EnvironmentObject.Actor = TrafficLight;
249 EnvironmentObject.CanTick = IsActorTickEnabled;
250 EnvironmentObject.BoundingBox = BB;
251 EnvironmentObject.Type = EnvironmentObjectType::TrafficLight;
252 EnvironmentObject.ObjectLabel = ObjectLabel;
253 EnvironmentObjects.Emplace(EnvironmentObject);
254
255 // Register components with its ID; it's not the best solution since we are recalculating the BBs
256 // But this is only calculated when the level is loaded
257 TArray<UStaticMeshComponent*> StaticMeshComps;
258 UBoundingBoxCalculator::GetMeshCompsFromActorBoundingBox(TrafficLight, BB, StaticMeshComps);
259 for(const UStaticMeshComponent* Comp : StaticMeshComps)
260 {
261 ObjectIdToComp.AddUnique(EnvironmentObject.Id, Comp);
262 }
263 }

266void UObjectRegister::RegisterISMComponents(AActor* Actor)
267{
268 check(Actor);
269
270 TArray<UInstancedStaticMeshComponent*> ISMComps;
271 Actor->GetComponents<UInstancedStaticMeshComponent>(ISMComps);
272
273 const FString ActorName = Actor->GetName();
274 int InstanceCount = 0;
275 bool IsActorTickEnabled = Actor->IsActorTickEnabled();
276
277 // Foliage actor is a special case, it can appear more than once while traversing the actor list
278 if(Cast<AInstancedFoliageActor>(Actor))
279 {
280 InstanceCount = FoliageActorInstanceCount;
281 }
282
283 for(UInstancedStaticMeshComponent* Comp : ISMComps)
284 {
285 const TArray<FInstancedStaticMeshInstanceData>& PerInstanceSMData = Comp->PerInstanceSMData;
286 const FTransform CompTransform = Comp->GetComponentTransform();
287
288 TArray<FBoundingBox> BoundingBoxes;
289 UBoundingBoxCalculator::GetISMBoundingBox(Comp, BoundingBoxes);
290
291 FString CompName = Comp->GetName();
292 const crp::CityObjectLabel Tag = ATagger::GetTagOfTaggedComponent(*Comp);
293
294 for(int i = 0; i < PerInstanceSMData.Num(); i++)
295 {
296 const FInstancedStaticMeshInstanceData& It = PerInstanceSMData[i];
297 const FTransform InstanceTransform = FTransform(It.Transform);
298 const FVector InstanceLocation = InstanceTransform.GetLocation();
299
300 // Discard decimal part
301 const int32 X = static_cast<int32>(InstanceLocation.X);
302 const int32 Y = static_cast<int32>(InstanceLocation.Y);
303 const int32 Z = static_cast<int32>(InstanceLocation.Z);
304
305 const FString InstanceName = FString::Printf(TEXT("%s_Inst_%d_%d"), *ActorName, InstanceCount, i);
306 const FString InstanceIdStr = FString::Printf(TEXT("%s_%s_%d_%d_%d_%d"), *ActorName, *CompName, X, Y, Z, InstanceCount);
307 uint64 InstanceId = CityHash64(TCHAR_TO_ANSI(*InstanceIdStr), InstanceIdStr.Len());
308
309 FEnvironmentObject EnvironmentObject;
310 EnvironmentObject.Transform = InstanceTransform * CompTransform;
311 EnvironmentObject.Id = InstanceId;
312 EnvironmentObject.Name = InstanceName;
313 EnvironmentObject.IdStr = InstanceIdStr;
314 EnvironmentObject.Actor = Actor;
315 EnvironmentObject.CanTick = IsActorTickEnabled;
316 if( i < BoundingBoxes.Num())
317 {
318 EnvironmentObject.BoundingBox = BoundingBoxes[i];
319 }
320
321 EnvironmentObject.Type = EnvironmentObjectType::ISMComp;
322 EnvironmentObject.ObjectLabel = static_cast<crp::CityObjectLabel>(Tag);
323 EnvironmentObjects.Emplace(EnvironmentObject);
324
325 ObjectIdToComp.AddUnique(InstanceId, Comp);
326 InstanceCount++;
327 }
328 }
329
330 if(Cast<AInstancedFoliageActor>(Actor))
331 {
332 FoliageActorInstanceCount = InstanceCount;
333 }
334}

445void UObjectRegister::EnableTrafficLight(FEnvironmentObject& EnvironmentObject, bool Enable)
446{
447 // We need to look for the component(s) that form the EnvironmentObject
448 // i.e.: The light box is composed by various SMComponents, one per light,
449 // we need to enable/disable all of them
450
451 TArray<TWeakObjectPtr<const UStaticMeshComponent>> ObjectComps;
452 ObjectIdToComp.MultiFind(EnvironmentObject.Id, ObjectComps);
453
454 for(const TWeakObjectPtr<const UStaticMeshComponent>& Comp : ObjectComps)
455 {
456 if (!Comp.IsValid())
457 {
458 continue;
459 }
460 UStaticMeshComponent* SMComp = const_cast<UStaticMeshComponent*>(Comp.Get());
461 SMComp->SetHiddenInGame(!Enable);
462 ECollisionEnabled::Type CollisionType = Enable ? ECollisionEnabled::Type::QueryAndPhysics : ECollisionEnabled::Type::NoCollision;
463 SMComp->SetCollisionEnabled(CollisionType);
464 }

468void UObjectRegister::EnableISMComp(FEnvironmentObject& EnvironmentObject, bool Enable)
469{
470 TArray<TWeakObjectPtr<const UStaticMeshComponent>> ObjectComps;
471 TArray<FString> InstanceName;
472 FTransform InstanceTransform = EnvironmentObject.Transform;
473
474 ObjectIdToComp.MultiFind(EnvironmentObject.Id, ObjectComps);
475 if (ObjectComps.Num() == 0 || !ObjectComps[0].IsValid())
476 {
477 return;
478 }
479 EnvironmentObject.Name.ParseIntoArray(InstanceName, TEXT("_"), false);
480
481 int Index = FCString::Atoi(*InstanceName[InstanceName.Num() - 1]);
482
483 if(!Enable)
484 {
485 InstanceTransform.SetTranslation(FVector(1000000.0f));
486 InstanceTransform.SetScale3D(FVector(0.0f));
487 }
488
489 UStaticMeshComponent* SMComp = const_cast<UStaticMeshComponent*>(ObjectComps[0].Get());
490 UInstancedStaticMeshComponent* ISMComp = Cast<UInstancedStaticMeshComponent>(SMComp);
491 if (ISMComp == nullptr)
492 {
493 return;
494 }
495 ISMComp->UpdateInstanceTransform(Index, InstanceTransform, true, true);

32bool UTrafficLightController::AdvanceTimeAndCycleFinished(float DeltaTime)
33{

102void UTrafficLightController::SetTrafficLightsState(ETrafficLightState NewState)
103{
104 SetCurrentLightState(NewState);
105 for(auto *Light : TrafficLights)
106 {
107 // World Partition can stream out the cell that owns a light while its
108 // controller keeps running; skip entries the GC has invalidated.
109 if (IsValid(Light))
110 {
111 Light->SetLightState(NewState);
112 }
113 }
114 for(FCarlaActor* Light : TrafficLightCarlaActors)
115 {
116 Light->SetTrafficLightState(NewState);
117 }

70 void EnableForROS(stream_id_type sensor_id) {
71 std::scoped_lock<std::mutex> lock(_mutex);
72 auto search = _stream_map.find(sensor_id);
73 if (search != _stream_map.end()) {
74 search->second->EnableForROS();
75 }
76 }
77
78 void DisableForROS(stream_id_type sensor_id) {
79 std::scoped_lock<std::mutex> lock(_mutex);
80 auto search = _stream_map.find(sensor_id);
81 if (search != _stream_map.end()) {
82 search->second->DisableForROS();
83 }
84 }
85
86 bool IsEnabledForROS(stream_id_type sensor_id) {
87 std::scoped_lock<std::mutex> lock(_mutex);
88 auto search = _stream_map.find(sensor_id);
89 if (search != _stream_map.end()) {
90 return search->second->IsEnabledForROS();
91 }
92 return false;
93 }

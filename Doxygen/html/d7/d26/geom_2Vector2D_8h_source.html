51 /// Return a unit-length copy of this vector. When the vector's length
52 /// is at or below `epsilon`, the input is returned unchanged (no
53 /// runtime assert, no NaN).
54 Vector2D MakeUnitVector(
55 const float epsilon = 2.0f * std::numeric_limits<float>::epsilon()) const {
56 const float len = Length();
57 if (len <= std::max(epsilon, 0.0f)) {
58 return *this;
59 }
60 const float k = 1.0f / len;
61 return Vector2D(x * k, y * k);
62 }

58 uint64_t World::ApplySettings(const rpc::EpisodeSettings &settings, time_duration timeout) {
59 rpc::EpisodeSettings new_settings = settings;
60 uint64_t id = _episode.Lock()->SetEpisodeSettings(settings);
61
62 time_duration local_timeout = timeout.milliseconds() == 0 ?
63 _episode.Lock()->GetNetworkingTimeout() : timeout;
64
65 if (settings.fixed_delta_seconds.has_value()) {
66 using namespace std::literals::chrono_literals;
67
68 const auto number_of_attemps = 30u;
69 uint64_t tics_correct = 0;
70 for (auto i = 0u; i < number_of_attemps; i++) {
71 const auto curr_snapshot = GetSnapshot();
72
73 const double error = abs(new_settings.fixed_delta_seconds.value() - curr_snapshot.GetTimestamp().delta_seconds);
74 if (error < std::numeric_limits<float>::epsilon())
75 tics_correct++;
76
77 if (tics_correct >= 2)
78 return id;
79
80 if (settings.synchronous_mode) {
81 // only drive the simulation forward in synchronous mode; in
82 // asynchronous mode wait for the server to advance on its own
83 Tick(local_timeout);
84 } else {
85 WaitForTick(local_timeout);
86 }
87 }
88
89 log_warning("World::ApplySettings: After", number_of_attemps, " attemps, the settings were not correctly set. Please check that everything is consistent.");
90 }
91 return id;
92 }

187 SharedPtr<Actor> World::GetTrafficSign(const Landmark& landmark) const {
188 SharedPtr<ActorList> actors = GetActors();
189 SharedPtr<TrafficSign> result;
190 std::string landmark_id = landmark.GetId();
191 for (size_t i = 0; i < actors->size(); i++) {
192 SharedPtr<Actor> actor = actors->at(i);
193 if (StringUtil::Match(actor->GetTypeId(), "*traffic.*")) {
194 TrafficSign* sign = static_cast<TrafficSign*>(actor.get());
195 if(sign && (sign->GetSignId() == landmark_id)) {
196 return actor;
197 }
198 }
199 }
200 return nullptr;
201 }

287 std::vector<SharedPtr<Actor>> World::GetTrafficLightsFromWaypoint(
288 const Waypoint& waypoint, double distance) const {
289 std::vector<SharedPtr<Actor>> Result;
290 std::vector<SharedPtr<Landmark>> landmarks =
291 waypoint.GetAllLandmarksInDistance(distance);
292 std::set<std::string> added_signals;
293 for (auto& landmark : landmarks) {
294 if (road::SignalType::IsTrafficLight(landmark->GetType())) {
295 SharedPtr<Actor> traffic_light = GetTrafficLight(*(landmark.get()));
296 if (traffic_light) {
297 if(added_signals.count(landmark->GetId()) == 0) {
298 Result.emplace_back(traffic_light);
299 added_signals.insert(landmark->GetId());
300 }
301 }
302 }
303 }
304 return Result;
305 }

307 std::vector<SharedPtr<Actor>> World::GetTrafficLightsInJunction(
308 const road::JuncId junc_id) const {
309 std::vector<SharedPtr<Actor>> Result;
310 SharedPtr<Map> map = GetMap();
311 const road::Junction* junction = map->GetMap().GetJunction(junc_id);
312 for (const road::ContId& cont_id : junction->GetControllers()) {
313 const std::unique_ptr<road::Controller>& controller =
314 map->GetMap().GetControllers().at(cont_id);
315 for (road::SignId sign_id : controller->GetSignals()) {
316 SharedPtr<Actor> traffic_light = GetTrafficLightFromOpenDRIVE(sign_id);
317 if (traffic_light) {
318 Result.emplace_back(traffic_light);
319 }
320 }
321 }
322 return Result;
323 }

384 void World::ApplyTexturesToObject(
385 const std::string &object_name,
386 const rpc::TextureColor& diffuse_texture,
387 const rpc::TextureFloatColor& emissive_texture,
388 const rpc::TextureFloatColor& normal_texture,
389 const rpc::TextureFloatColor& ao_roughness_metallic_emissive_texture)
390 {
391 if (diffuse_texture.GetWidth() && diffuse_texture.GetHeight()) {
392 ApplyColorTextureToObject(
393 object_name, rpc::MaterialParameter::Tex_Diffuse, diffuse_texture);
394 }
395 if (normal_texture.GetWidth() && normal_texture.GetHeight()) {
396 ApplyFloatColorTextureToObject(
397 object_name, rpc::MaterialParameter::Tex_Normal, normal_texture);
398 }
399 if (ao_roughness_metallic_emissive_texture.GetWidth() &&
400 ao_roughness_metallic_emissive_texture.GetHeight()) {
401 ApplyFloatColorTextureToObject(
402 object_name,
403 rpc::MaterialParameter::Tex_Ao_Roughness_Metallic_Emissive,
404 ao_roughness_metallic_emissive_texture);
405 }
406 if (emissive_texture.GetWidth() && emissive_texture.GetHeight()) {
407 ApplyFloatColorTextureToObject(
408 object_name, rpc::MaterialParameter::Tex_Emissive, emissive_texture);
409 }
410 }

21// Image-side companion publisher: writes the dense visualisation of the DVS
22// event stream as a 3-channel BGR sensor_msgs/Image to the existing wire
23// format ROS 2 subscribers depend on. The encoding (positive polarity =
24// red, negative polarity = blue) lives in DvsEventEncoding.h so a GTest
25// pins it without instantiating FastDDS.
26class CarlaDVSImagePublisher final : public CarlaCameraPublisher {
27public:
28 CarlaDVSImagePublisher(std::string base_topic_name, std::string frame_id)
29 : CarlaCameraPublisher(std::move(base_topic_name), std::move(frame_id)) {}
30
31 // Encode-from-events overload: walks the DVSEvent stream and writes the
32 // dense image via the seam.
33 bool WriteImageFromEvents(
34 std::int32_t seconds,
35 std::uint32_t nanoseconds,
36 std::uint32_t height,
37 std::uint32_t width,
38 const std::uint8_t *events,
39 std::size_t event_count,
40 std::size_t event_stride);
41
42protected:
43 [[nodiscard]] std::uint8_t GetChannels() const override { return 3u; }
44 [[nodiscard]] std::string GetEncoding() const override { return "bgr8"; }
45};

26struct ddsi_serdata* carla_cdr_alloc_serdata(
27 const struct ddsi_sertype* type,
28 enum ddsi_serdata_kind kind,
29 uint32_t cdr_size)
30{
31 void* mem = malloc(sizeof(struct carla_cdr_serdata) + static_cast<size_t>(cdr_size));
32 if (!mem) { return nullptr; }
33 struct carla_cdr_serdata* csd = reinterpret_cast<struct carla_cdr_serdata*>(mem);
34 ddsi_serdata_init(&csd->sd, type, kind);
35 csd->cdr_size = cdr_size;
36 return &csd->sd;
37}

54static struct ddsi_serdata* carla_cdr_from_ser(
55 const struct ddsi_sertype* type,
56 enum ddsi_serdata_kind kind,
57 const struct nn_rdata* fragchain,
58 size_t size)
59{
60 struct ddsi_serdata* sd =
61 carla_cdr_alloc_serdata(type, kind, static_cast<uint32_t>(size));
62 if (!sd) { return nullptr; }
63 uint8_t* dest = reinterpret_cast<uint8_t*>(
64 reinterpret_cast<struct carla_cdr_serdata*>(sd) + 1);
65 const size_t gathered = carla_cdr_gather_fragments(
66 dest, size, fragchain,
67 [](const struct nn_rdata* f) -> const uint8_t* {
68 return NN_RMSG_PAYLOADOFF(f->rmsg, NN_RDATA_PAYLOAD_OFF(f));
69 });
70 if (gathered != size) {
71 log_error(
72 "carla_cdr_from_ser: incomplete fragment chain (",
73 gathered, " of ", size, " bytes); dropping sample");
74 ddsi_serdata_unref(sd);
75 return nullptr;
76 }
77 return sd;
78}

80static struct ddsi_serdata* carla_cdr_from_ser_iov(
81 const struct ddsi_sertype* type,
82 enum ddsi_serdata_kind kind,
83 ddsrt_msg_iovlen_t niov,
84 const ddsrt_iovec_t* iov,
85 size_t size)
86{
87 struct ddsi_serdata* sd =
88 carla_cdr_alloc_serdata(type, kind, static_cast<uint32_t>(size));
89 if (!sd) { return nullptr; }
90 uint8_t* dest = reinterpret_cast<uint8_t*>(
91 reinterpret_cast<struct carla_cdr_serdata*>(sd) + 1);
92 size_t off = 0u;
93 for (ddsrt_msg_iovlen_t i = 0; i < niov && off < size; ++i) {
94 const size_t avail = static_cast<size_t>(iov[i].iov_len);
95 const size_t copy = (off + avail > size) ? (size - off) : avail;
96 memcpy(dest + off, iov[i].iov_base, copy);
97 off += copy;
98 }
99 return sd;
100}

292dds_entity_t carla_cdr_get_participant() {
293 // Function-local static: created once on first call, thread-safe in C++11.
294 // The participant lives until process exit; CycloneDDS cleans up internally.
295 // Use the effective domain id resolved by the abstraction layer
296 // (--ros-domain-id, else ROS_DOMAIN_ID, else the default domain 0), so the
297 // domain selection matches FastDDS instead of relying on DDS_DOMAIN_DEFAULT.
298 static dds_entity_t participant = []() {
299 const dds_domainid_t domain_id =
300 static_cast<dds_domainid_t>(MiddlewareConfig::GetEffectiveDomainId());
301 return dds_create_participant(domain_id, nullptr, nullptr);
302 }();
303 return participant;
304}

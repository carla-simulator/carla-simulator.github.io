78 /// Method that computes the label corresponding to an specific object
79 /// using the folder path in which it is stored
80 template <typename T>
81 static crp::CityObjectLabel GetLabelByPath(const T* Object)
82 {
83 if (Object == nullptr)
84 {
85 // A component whose mesh is not assigned yet (props get their mesh
86 // after SpawnActor, i.e. after the OnActorSpawned tagger delegate).
87 return crp::CityObjectLabel::None;
88 }
89 const FString Path = Object->GetPathName();
90 TArray<FString> StringArray;
91 Path.ParseIntoArray(StringArray, TEXT("/"), false);
92 if (!Path.StartsWith(TEXT("/Game/")))
93 {
94 // Content pack (or any other mount root): same folder rule as the base
95 // content, /<Pack>/Static/<Tag>/..., i.e. the folder right after the
96 // first "Static" folder names the label. Plugin content without a
97 // "Static" folder falls through to the nearest-folder fallback below.
98 for (int32 i = 1; i + 1 < StringArray.Num(); ++i)
99 {
100 if (StringArray[i] == TEXT("Static"))
101 {
102 const crp::CityObjectLabel PackLabel = GetLabelByFolderName(StringArray[i + 1]);
103 if (PackLabel != crp::CityObjectLabel::None)
104 {
105 return PackLabel;
106 }
107 break;
108 }
109 }
110 }
111 const int32 Primary = Path.Contains("UE5UseOnly") ? 5 : 4;
112 crp::CityObjectLabel Label =
113 (StringArray.Num() > Primary ? GetLabelByFolderName(StringArray[Primary]) : crp::CityObjectLabel::None);
114 // Fallback for content mounted outside /Game/Carla/Static/<Label>/... (e.g. plugin
115 // content such as /CarlaDigitalTwinsTool/Static/Building/...): the labelled folder
116 // nearest the asset wins. Runs only when the fixed-index lookup found nothing, so
117 // classic content keeps its exact previous labels.
118 for (int32 i = StringArray.Num() - 2; i >= 0 && Label == crp::CityObjectLabel::None; --i)
119 {
120 Label = GetLabelByFolderName(StringArray[i]);
121 }
122 return Label;
123 }
124
125 /// Method that computes the label corresponding to an specific object
126 /// using the folder path in which it is stored
127 template <typename T>
128 static crp::CityObjectLabel GetLabelByPath(TObjectPtr<T> Object)
129 {
130 return GetLabelByPath(Object.Get());
131 }

26 /// Snapshot of a vehicle's kinematic + control state, streamed each tick by
27 /// the Autoware VehicleStatusSensor (UE side, next phase) and decoded by
28 /// ROS2::ProcessDataFromStatusSensor into the six /vehicle/status/* report
29 /// publishers. Ported from tier4/autoware-support.
30 ///
31 /// Bit masks:
32 /// control_flags: b0 = reverse, b1 = manual gear shift
33 /// turn_mask: b0 = left blinker, b1 = right blinker, b2 = hazards
34 struct VehicleStatusData {
35 double timestamp;
36 float speed_mps;
37 float vel_x_mps, vel_y_mps, vel_z_mps; // local
38 float angVel_x_mps, angVel_y_mps, angVel_z_mps; // local
39 float rot_pitch, rot_yaw, rot_roll; // relative
40 float steer;
41 int32_t gear;
42 uint8_t turn_mask;
43 uint8_t control_flags;
44
45 MSGPACK_DEFINE_ARRAY(timestamp,
46 speed_mps,
47 vel_x_mps, vel_y_mps, vel_z_mps,
48 angVel_x_mps, angVel_y_mps, angVel_z_mps,
49 rot_pitch, rot_yaw, rot_roll,
50 steer,
51 gear,
52 turn_mask,
53 control_flags)
54 };

97 /// Overload for the Unreal sensor streaming a packed struct. The UE-side
98 /// packed layout carries two trailing pad bytes (4-byte struct alignment);
99 /// everything before them matches VehicleStatusData byte-for-byte.
100 template <typename SensorT>
101 static Buffer Serialize(const SensorT &, const Buffer &buffer) {
102 constexpr size_t packed_size = sizeof(VehicleStatusData) + 2u; // + _pad0/_pad1
103 if (buffer.size() != packed_size && buffer.size() != sizeof(VehicleStatusData)) {
104 throw std::runtime_error("Invalid buffer size for VehicleStatusSensor");
105 }
106 VehicleStatusData msg;
107 std::memcpy(&msg, buffer.data(), sizeof(VehicleStatusData));
108 return MsgPack::Pack(msg);
109 }

47/// One head of a baked Traffic Light Tool rig, and the OpenDRIVE signal it shows.
48///
49/// A mast arm can carry a protected-turn arrow beside the through head, and the two are red
50/// and green at the same instant. ATrafficLightBase holds exactly one UTrafficLightComponent
51/// and one ETrafficLightState (TrafficLightBase.h), and get_state / get_opendrive_id /
52/// get_stop_waypoints / the ALSM's IsAtTrafficLight are all per *actor*, so one actor cannot
53/// express two states. The rig is therefore split into one ADigitalTwinsTrafficLight per
54/// distinct SignalID, and this is the explicit binding that says which components go where --
55/// ComponentPrefix is the name ATrafficLightActor::Build gives the head's components
56/// ("Pole_02_Head_03"), which Bake preserves.
57USTRUCT()
58struct FTrafficLightHead
59{
60 GENERATED_BODY()
61
62 UPROPERTY()
63 FString ComponentPrefix;
64
65 UPROPERTY()
66 FString SignalID;
67
68 UPROPERTY()
69 TArray<int32> LaneIds;
70};

24 /// A thread pool based on Boost.Asio's io context.
25 class ThreadPool : private NonCopyable {
26 public:
27
28 ThreadPool() : _work_to_do(boost::asio::make_work_guard(_io_context)) {}
29
30 /// Stops the ThreadPool and joins all its threads.
31 ~ThreadPool() {
32 Stop();
33 }
34
35 /// Return the underlying io_context.
36 auto &io_context() {
37 return _io_context;
38 }
39
40 /// Post a task to the pool.
41 template <
42 typename FunctorT,
43 typename ResultT = typename std::invoke_result_t<FunctorT()>>
44 std::future<ResultT> Post(FunctorT &&functor) {
45 auto task = std::packaged_task<ResultT()>(std::forward<FunctorT>(functor));
46 auto future = task.get_future();
47 boost::asio::post(_io_context, carla::MoveHandler(task));
48 return future;
49 }
50
51 /// Launch threads to run tasks asynchronously. Launch specific number of
52 /// threads if @a worker_threads is provided, otherwise use all available
53 /// hardware concurrency.
54 void AsyncRun(size_t worker_threads) {
55 _workers.CreateThreads(worker_threads, [this]() { Run(); });
56 }
57
58 /// @copydoc AsyncRun(size_t)
59 void AsyncRun() {
60 AsyncRun(std::thread::hardware_concurrency());
61 }
62
63 /// Run tasks in this thread.
64 ///
65 /// @warning This function blocks until the ThreadPool has been stopped.
66 void Run() {
67 _io_context.run();
68 }
69
70 /// Run tasks in this thread for an specific @a duration.
71 ///
72 /// @warning This function blocks until the ThreadPool has been stopped, or
73 /// until the specified time duration has elapsed.
74 void RunFor(time_duration duration) {
75 _io_context.run_for(duration.to_chrono());
76 }
77
78 /// Stop the ThreadPool and join all its threads.
79 void Stop() {
80 _work_to_do.reset();
81 _io_context.stop();
82 _workers.JoinAll();
83 }
84
85 private:
86
87 boost::asio::io_context _io_context;
88
89 boost::asio::executor_work_guard<boost::asio::io_context::executor_type> _work_to_do;
90
91 ThreadGroup _workers;
92 };

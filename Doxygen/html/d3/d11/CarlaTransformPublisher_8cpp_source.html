28bool IsSameTransform(
29 const CarlaTransformPublisher::CachedTransform &cached,
30 float tx, float ty, float tz,
31 float pitch, float yaw, float roll) noexcept {
32 return std::abs(cached.tx - tx) < kEpsilon
33 && std::abs(cached.ty - ty) < kEpsilon
34 && std::abs(cached.tz - tz) < kEpsilon
35 && std::abs(cached.pitch - pitch) < kEpsilon
36 && std::abs(cached.yaw - yaw) < kEpsilon
37 && std::abs(cached.roll - roll) < kEpsilon;
38}

56bool CarlaTransformPublisher::Write(
57 std::int32_t seconds,
58 std::uint32_t nanoseconds,
59 const std::string &parent_frame_id,
60 const std::string &child_frame_id,
61 float tx, float ty, float tz,
62 float pitch_deg, float yaw_deg, float roll_deg) {
63 std::array<float, 3> translation;
64 std::array<float, 4> rotation;
65
66 auto it = _last_transforms.find(child_frame_id);
67 if (it != _last_transforms.end()
68 && IsSameTransform(it->second, tx, ty, tz, pitch_deg, yaw_deg, roll_deg)) {
69 translation = it->second.translation;
70 rotation = it->second.rotation;
71 } else {
72 // Quaternion conversion lives in TransformQuaternion.h (header-only
73 // seam, pinned by test_transform_quaternion.cpp).
74 const auto computed =
75 TransformFromCarlaRotation(tx, ty, tz, pitch_deg, yaw_deg, roll_deg);
76 translation = computed.translation;
77 rotation = computed.rotation;
78 _last_transforms.insert_or_assign(
79 child_frame_id,
80 CachedTransform{
81 tx, ty, tz, pitch_deg, yaw_deg, roll_deg, translation, rotation});
82 }
83
84 msg::TransformStamped ts{};
85 ts.header.stamp.sec = seconds;
86 ts.header.stamp.nanosec = nanoseconds;
87 ts.header.frame_id = parent_frame_id;
88 ts.child_frame_id = child_frame_id;
89
90 ts.transform.translation.x = translation[0];
91 ts.transform.translation.y = translation[1];
92 ts.transform.translation.z = translation[2];
93 ts.transform.rotation.w = rotation[0];
94 ts.transform.rotation.x = rotation[1];
95 ts.transform.rotation.y = rotation[2];
96 ts.transform.rotation.z = rotation[3];
97
98 _impl->GetMessage()->transforms = {ts};
99 return true;
100}

27 FShapeVisitor(UWorld &InWorld, FColor InColor, float InLifeTime, bool bInPersistentLines)
28 : World(&InWorld),
29 Color(InColor.ReinterpretAsLinear() * BrightMultiplier),
30 LifeTime(InLifeTime),

36 void operator()(const Shape::Point &Point) const
37 {
38 FVector Location = FVector(Point.location);
39 ALargeMapManager* LargeMap = UCarlaStatics::GetLargeMapManager(World);
40 if (LargeMap)
41 {
42 Location = LargeMap->GlobalToLocalLocation(Location);
43 }
44 World->GetLineBatcher(UWorld::ELineBatcherType::WorldPersistent)->DrawPoint(
45 Location,
46 Color,
47 1e2f * Point.size,
48 DepthPriority,
49 LifeTime);

52 void operator()(const Shape::Line &Line) const
53 {

71 void operator()(const Shape::Arrow &Arrow) const
72 {
73 FVector Begin = FVector(Arrow.line.begin);
74 FVector End = FVector(Arrow.line.end);
75 ALargeMapManager* LargeMap = UCarlaStatics::GetLargeMapManager(World);
76 if (LargeMap)
77 {
78 Begin = LargeMap->GlobalToLocalLocation(Begin);
79 End = LargeMap->GlobalToLocalLocation(End);
80 }
81 const auto Diff = End - Begin;
82 const FRotator LookAt = FRotationMatrix::MakeFromX(Diff).Rotator();
83 const FTransform Transform = {LookAt, Begin};
84
85 // Everything in centimeters
86 const auto Dist = Diff.Size();
87 const auto ArrowSize = 1e2f * Arrow.arrow_size;
88 const auto ArrowTipDist = Dist - ArrowSize;
89 const auto Thickness = 1e2f * Arrow.line.thickness;
90
91 FBatchedLine BatchedLines[] =
92 {
93 FBatchedLine(
94 Begin,
95 End,
96 Color,
97 LifeTime,
98 Thickness,
99 DepthPriority),
100 FBatchedLine(
101 Transform.TransformPosition(FVector(ArrowTipDist, +ArrowSize, +ArrowSize)),
102 End,
103 Color,
104 LifeTime,
105 Thickness,
106 DepthPriority),
107 FBatchedLine(
108 Transform.TransformPosition(FVector(ArrowTipDist, +ArrowSize, -ArrowSize)),
109 End,
110 Color,
111 LifeTime,
112 Thickness,
113 DepthPriority),
114 FBatchedLine(
115 Transform.TransformPosition(FVector(ArrowTipDist, -ArrowSize, +ArrowSize)),
116 End,
117 Color,
118 LifeTime,
119 Thickness,
120 DepthPriority),
121 FBatchedLine(
122 Transform.TransformPosition(FVector(ArrowTipDist, -ArrowSize, -ArrowSize)),
123 End,
124 Color,
125 LifeTime,
126 Thickness,
127 DepthPriority)
128 };
129
130 World->GetLineBatcher(UWorld::ELineBatcherType::WorldPersistent)->DrawLines(TArrayView<FBatchedLine>(BatchedLines, 5));
131 }

133 void operator()(const Shape::Box &Box) const
134 {
135 const FVector Extent = 1e2f * FVector{Box.box.extent.x, Box.box.extent.y, Box.box.extent.z};
136 FTransform Transform = {FRotator(Box.rotation), Box.box.location};
137 const auto Thickness = 1e2f * Box.thickness;
138
139 ALargeMapManager* LargeMap = UCarlaStatics::GetLargeMapManager(World);
140 if (LargeMap)
141 {
142 Transform = LargeMap->GlobalToLocalTransform(Transform);
143 }
144
145 FVector B[2], P, Q;
146 B[0] = -Extent;
147 B[1] = Extent;
148
149 for(int32 i = 0; i < 2; ++i)
150 {
151 for(int32 j = 0; j < 2; ++j)
152 {
153 P.X=B[i].X;
154 Q.X=B[i].X;
155 P.Y=B[j].Y;
156 Q.Y=B[j].Y;
157 P.Z=B[0].Z;
158 Q.Z=B[1].Z;
159 World->GetLineBatcher(UWorld::ELineBatcherType::WorldPersistent)->DrawLine(
160 Transform.TransformPosition(P),
161 Transform.TransformPosition(Q),
162 Color,
163 DepthPriority,
164 Thickness,
165 LifeTime);
166
167 P.Y=B[i].Y;
168 Q.Y=B[i].Y;
169 P.Z=B[j].Z;
170 Q.Z=B[j].Z;
171 P.X=B[0].X;
172 Q.X=B[1].X;
173 World->GetLineBatcher(UWorld::ELineBatcherType::WorldPersistent)->DrawLine(
174 Transform.TransformPosition(P),
175 Transform.TransformPosition(Q),
176 Color,
177 DepthPriority,
178 Thickness,
179 LifeTime);
180
181 P.Z=B[i].Z;
182 Q.Z=B[i].Z;
183 P.X=B[j].X;
184 Q.X=B[j].X;
185 P.Y=B[0].Y;
186 Q.Y=B[1].Y;
187 World->GetLineBatcher(UWorld::ELineBatcherType::WorldPersistent)->DrawLine(
188 Transform.TransformPosition(P),
189 Transform.TransformPosition(Q),
190 Color,
191 DepthPriority,
192 Thickness,
193 LifeTime);
194 }
195 }

198 void operator()(const Shape::String &Str) const
199 {
200 auto PlayerController = UGameplayStatics::GetPlayerController(World, 0);
201 if (PlayerController == nullptr)
202 {
203 UE_LOG(LogCarla, Error, TEXT("Can't find player controller!"));
204 return;
205 }
206 FVector Location = FVector(Str.location);
207 ALargeMapManager* LargeMap = UCarlaStatics::GetLargeMapManager(World);
208 if (LargeMap)
209 {
210 Location = LargeMap->GlobalToLocalLocation(Location);
211 }
212 ACarlaHUD *Hud = Cast<ACarlaHUD>(PlayerController->GetHUD());
213 Hud->AddHUDString(carla::rpc::ToFString(Str.text), Location, Color.QuantizeRound(), LifeTime);
214 }

66// ==========================================================================
67// Internal CDR helpers — one overload pair per message type.
68// Ordered from least-dependent to most-dependent so each helper's body
69// can call the helpers for its nested types without forward declarations.
70//
71// Wire format: OMG DDSI-RTPS v2.5 Section 10 + DDS-XTypes 1.3 clause 7.4.1.1
72// (Classic CDR, encoding version 1, little-endian). Sequences are encoded
73// as a uint32_t length followed by elements; strings as a uint32_t length
74// (including the terminating NUL) followed by bytes; bool as a single octet.
75// ==========================================================================
76
77/// Sanity cap for the length field of a CDR sequence read from the wire.
78/// Protects against malformed/hostile payloads claiming a multi-GB sequence,
79/// which would otherwise OOM-abort the process inside std::vector::resize().
80/// 1,048,576 elements is far above any realistic ROS2 message (PointCloud2
81/// rarely has more than a dozen fields; TFMessage rarely has more than a few
82/// hundred transforms) while bounding the worst-case allocation to ~80 MiB.
83static constexpr uint32_t kMaxCdrSequenceElements = 1u << 20;

834/// PointCloud2::fields is a sequence of structs; FastCDR's generic vector
835/// operator calls serialize() on each element, which our POD types don't
836/// provide. Write length + elements manually.
837inline void serialize_cdr(
838 eprosima::fastcdr::Cdr& cdr, const msg::PointCloud2& m) {
839 serialize_cdr(cdr, m.header);
840 cdr << m.height;
841 cdr << m.width;
842 // CDR sequence length is uint32_t per DDS-XTypes 1.3 clause 7.4.1.1.
843 cdr << static_cast<uint32_t>(m.fields.size());
844 for (const auto& f : m.fields) {
845 serialize_cdr(cdr, f);
846 }
847 cdr << m.is_bigendian;
848 cdr << m.point_step;
849 cdr << m.row_step;
850 cdr << m.data;
851 cdr << m.is_dense;
852}

854inline void deserialize_cdr(
855 eprosima::fastcdr::Cdr& cdr, msg::PointCloud2& m) {
856 deserialize_cdr(cdr, m.header);
857 cdr >> m.height;
858 cdr >> m.width;
859 uint32_t fields_size{0u};
860 cdr >> fields_size;
861 if (fields_size > kMaxCdrSequenceElements) {
862 throw eprosima::fastcdr::exception::BadParamException(
863 "PointCloud2::fields length exceeds sane CDR sequence cap");
864 }
865 m.fields.resize(static_cast<size_t>(fields_size));
866 for (auto& f : m.fields) {
867 deserialize_cdr(cdr, f);
868 }
869 cdr >> m.is_bigendian;
870 cdr >> m.point_step;
871 cdr >> m.row_step;
872 cdr >> m.data;
873 cdr >> m.is_dense;
874}

907/// Serialize a msg::X to a CDR byte buffer including the DDS encapsulation
908/// header (Classic CDR, encoding version 1, little-endian). The returned
909/// buffer is wire-compatible with all ROS2 distros and can be passed to
910/// FastDDS write_serialized_payload() or CycloneDDS dds_writecdr().
911/// Returns an empty vector if Fast-CDR raises an exception (e.g. out of
912/// memory while growing the internal FastBuffer).
913template<typename T>
914std::vector<uint8_t> serialize_to_cdr(const T& msg) {
915 eprosima::fastcdr::FastBuffer fb;
916 // Force LITTLE_ENDIANNESS so the encapsulation header is CDR_LE
917 // ({0x00, 0x01}) per DDSI-RTPS v2.5 Table 10.3, regardless of host
918 // endianness. ROS2 ecosystems test against CDR_LE.
919 eprosima::fastcdr::Cdr cdr{
920 fb,
921 eprosima::fastcdr::Cdr::LITTLE_ENDIANNESS,
922 eprosima::fastcdr::CdrVersion::XCDRv1};
923 try {
924 cdr.serialize_encapsulation();
925 serialize_cdr(cdr, msg);
926 } catch (const eprosima::fastcdr::exception::Exception&) {
927 return std::vector<uint8_t>{};
928 }
929 const char* buf{fb.getBuffer()};
930 const size_t len{cdr.get_serialized_data_length()};
931 return std::vector<uint8_t>{
932 reinterpret_cast<const uint8_t*>(buf),
933 reinterpret_cast<const uint8_t*>(buf) + len};
934}
935
936/// Return the exact CDR-serialized size in bytes for a message instance,
937/// including the 4-byte DDS encapsulation header. Used by GenericCdrPubSubType
938/// to tell FastDDS the actual payload size before write(), so the payload
939/// buffer is sized correctly for variable-length fields (e.g. Image::data,
940/// PointCloud2::data). The result is provably equal to serialize_to_cdr(msg).size().
941template<typename T>
942uint32_t cdr_serialized_size(const T& msg) {
943 eprosima::fastcdr::FastBuffer fb;
944 eprosima::fastcdr::Cdr cdr{
945 fb,
946 eprosima::fastcdr::Cdr::LITTLE_ENDIANNESS,
947 eprosima::fastcdr::CdrVersion::XCDRv1};
948 cdr.serialize_encapsulation();
949 serialize_cdr(cdr, msg);
950 return static_cast<uint32_t>(cdr.get_serialized_data_length());
951}
952
953/// Deserialize a msg::X from a CDR byte buffer that was produced by
954/// serialize_to_cdr() or by any ROS2-compatible DDS middleware.
955/// Returns true on success, false on any Fast-CDR error (truncated buffer,
956/// malformed encapsulation header, sequence length exceeding the sanity
957/// cap, etc.). The buffer must include the 4-byte DDS encapsulation header.
958template<typename T>
959bool deserialize_from_cdr(
960 const uint8_t* data, size_t size, T& msg) {
961 eprosima::fastcdr::FastBuffer fb{
962 // FastBuffer requires a non-const pointer; the buffer is only read.
963 reinterpret_cast<char*>(const_cast<uint8_t*>(data)),
964 size};
965 eprosima::fastcdr::Cdr cdr{
966 fb,
967 eprosima::fastcdr::Cdr::LITTLE_ENDIANNESS,
968 eprosima::fastcdr::CdrVersion::XCDRv1};
969 try {
970 cdr.read_encapsulation();
971 deserialize_cdr(cdr, msg);
972 } catch (const eprosima::fastcdr::exception::Exception&) {
973 return false;
974 }
975 return true;
976}

27void UCarlaLightSubsystem::RegisterLight(UCarlaLight* CarlaLight)
28{
29 if(CarlaLight)
30 {
31 // OnRegister and BeginPlay both register: the same light coming back
32 // keeps its entry. Taken for a clone below, it got a second id on every
33 // call and the stale entries were never switched (World Partition maps
34 // ended up with ~3 entries per lamp).
35 if (const int* ExistingId = Lights.FindKey(CarlaLight))
36 {
37 CarlaLight->SetId(*ExistingId);
38 }
39 auto LightId = CarlaLight->GetId();
40 if (Lights.Contains(LightId) && Lights[LightId] != CarlaLight)
41 {
42 // Runtime-spawned lights cloned from a template (PCG Spawn Actor,
43 // duplicated actors...) all arrive carrying the template's id;
44 // dropping them here silently disconnected every clone from the
45 // subsystem (day/night events, client light API). Assign the next
46 // free id instead.
47 while (Lights.Contains(LightId))
48 {
49 ++LightId;
50 }
51 CarlaLight->SetId(LightId);
52 }
53 Lights.Add(LightId, CarlaLight);
54 DayTimeChangeEvent.AddUniqueDynamic(CarlaLight, &UCarlaLight::HandleDayTimeChanged);
55 // The day/night state is only broadcast when the weather changes, so a
56 // light that registers later (a World Partition cell streamed in at
57 // night) stayed in its default, day, state until the next change. Applied
58 // on the next tick: from inside registration the lamp blueprint has not
59 // run its BeginPlay yet, and its SetLight looped forever (PIE stopped).
60 if (bHasDayTimeState && bDayNightCycle)
61 {
62 ScheduleDayTimeState(CarlaLight);
63 }
64 }
65 SetClientStatesdirty("");
66}

68void UCarlaLightSubsystem::ScheduleDayTimeState(UCarlaLight* CarlaLight)
69{
70 UWorld* World = GetWorld();
71 if (World == nullptr || !World->IsGameWorld())
72 {
73 return;
74 }
75 const bool bFirst = PendingDayTimeLights.Num() == 0;
76 PendingDayTimeLights.Add(CarlaLight);
77 if (!bFirst)
78 {
79 return;
80 }
81 World->GetTimerManager().SetTimerForNextTick(FTimerDelegate::CreateWeakLambda(this, [this]()
82 {
83 TSet<TWeakObjectPtr<UCarlaLight>> Pending = MoveTemp(PendingDayTimeLights);
84 PendingDayTimeLights.Reset();
85 for (const TWeakObjectPtr<UCarlaLight>& Weak : Pending)
86 {
87 UCarlaLight* Light = Weak.Get();
88 // Only lights still registered here: one streamed out meanwhile is skipped.
89 if (IsValid(Light) && Lights.FindKey(Light) != nullptr)
90 {
91 Light->HandleDayTimeChanged(bLastIsDay);
92 Light->ApplyLegacyComponentConversion();
93 }
94 }
95 SetClientStatesdirty("");
96 }));
97}

109void UCarlaLightSubsystem::NotifyDayTimeChange(bool bIsDay)
110{
111 // Every weather push notifies (AWeather and the sky rig, twice per
112 // set_weather), but only an actual day/night change needs the broadcast:
113 // it walks every registered light (blueprint handlers, emissive material
114 // instances, intensity conversion) and cost ~270 ms per set_weather on
115 // Town12 with ~4400 lights loaded. Lights registering later get the
116 // current state through ScheduleDayTimeState.
117 // The sky rig notifies too, so the client's set_day_night_cycle(False) is
118 // enforced here rather than only on AWeather.
119 if (!bDayNightCycle || (bHasDayTimeState && bLastIsDay == bIsDay))
120 {
121 return;
122 }
123 bHasDayTimeState = true;
124 bLastIsDay = bIsDay;
125 DayTimeChangeEvent.Broadcast(bIsDay);
126 // Blueprints bind this event too (BlueprintAssignable) and their handlers
127 // push raw authored UE4-era intensities into the light components; a
128 // per-light conversion inside the broadcast gets overwritten by whichever
129 // handler runs later. Convert once the whole broadcast is done.
130 for (auto& LightPair : Lights)
131 {
132 if (UCarlaLight* CarlaLight = LightPair.Value)
133 {
134 CarlaLight->ApplyLegacyComponentConversion();
135 }
136 }
137 // A day/night change flips light states server-side; flag every connected
138 // client so their LightManager re-queries instead of serving stale is_on
139 // values from its local cache.
140 SetClientStatesdirty("");
141}

155std::vector<carla::rpc::LightState> UCarlaLightSubsystem::GetLights(FString Client)
156{
157 std::vector<carla::rpc::LightState> result;
158
159 ClientStates.FindOrAdd(Client) = false;
160
161 for(auto& Light : Lights)
162 {
163 UCarlaLight* CarlaLight = Light.Value;
164 // World Partition can stream out a lamp's owner between the light's GC
165 // death and its EndPlay unregistration; this RPC runs on the server
166 // thread and must not touch such carcasses.
167 if (!IsValid(CarlaLight) || !IsValid(CarlaLight->GetOwner()))
168 {
169 continue;
170 }
171 result.push_back(CarlaLight->GetLightState());
172 }
173 return result;
174}

209void UCarlaLightSubsystem::SetDayNightCycle(const bool active) {
210 bDayNightCycle = active;
211 // Re-enabled: the next weather push broadcasts again, even with the same
212 // state, so lights the client changed meanwhile follow the cycle again.
213 if (active)
214 {
215 bHasDayTimeState = false;
216 }
217 TArray<AActor*> WeatherActors;
218 UGameplayStatics::GetAllActorsOfClass(GetWorld(), AWeather::StaticClass(), WeatherActors);
219 if (WeatherActors.Num())
220 {
221 if (AWeather* WeatherActor = Cast<AWeather>(WeatherActors[0]))
222 {
223 WeatherActor->SetDayNightCycle(active);
224 }
225 }

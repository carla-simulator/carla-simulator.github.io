17LightManager::~LightManager(){
18 // A destructor must not throw (that calls std::terminate), and the RPC below
19 // times out if the server is unreachable or stalled.
20#ifndef LIBCARLA_NO_EXCEPTIONS
21 try {
22#endif // LIBCARLA_NO_EXCEPTIONS
23 if(_episode.IsValid()) {
24 _episode.Lock()->RemoveOnTickEvent(_on_tick_register_id);
25 _episode.Lock()->RemoveLightUpdateChangeEvent(_on_light_update_register_id);
26 }
27 UpdateServerLightsState(true);
28#ifndef LIBCARLA_NO_EXCEPTIONS
29 } catch (const std::exception &e) {
30 log_error("exception in the light manager destructor:", e.what());
31 }
32#endif // LIBCARLA_NO_EXCEPTIONS
33}

35void LightManager::SetEpisode(detail::WeakEpisodeProxy episode) {
36
37 _episode = episode;
38
39 _on_tick_register_id = _episode.Lock()->RegisterOnTickEvent(
40 [&](const WorldSnapshot&) {
41 UpdateServerLightsState();
42 });
43
44 _on_light_update_register_id = _episode.Lock()->RegisterLightUpdateChangeEvent(
45 [&](const WorldSnapshot& ) {

48 });
49
50 QueryLightsStateToServer();
51}

283void LightManager::QueryLightsStateToServer() {
284 std::scoped_lock<std::mutex> lock(_mutex);
285 // Send blocking query
286 std::vector<rpc::LightState> lights_snapshot = _episode.Lock()->QueryLightsStateToServer();
287
288 // Update lights
289 SharedPtr<LightManager> lm = _episode.Lock()->GetLightManager();
290
291 // The snapshot is the whole set of lights the server has now: lights of
292 // World Partition cells that streamed out are gone from it, and their ids
293 // can come back for other lights. Merging into the cache kept every light
294 // ever seen, with its first location.
295 std::unordered_map<LightId, LightState> lights_state;
296 std::unordered_map<LightId, Light> lights;
297 for(const auto& it : lights_snapshot) {
298 lights_state[it._id] = LightState(
299 it._intensity,
300 Color(it._color.r, it._color.g, it._color.b),
301 static_cast<LightState::LightGroup>(it._group),
302 it._active
303 );
304 lights[it._id] = Light(lm, it._location, it._id);
305 }
306 _lights_state = std::move(lights_state);
307 _lights = std::move(lights);
308}

310void LightManager::UpdateServerLightsState(bool discard_client) {
311 std::scoped_lock<std::mutex> lock(_mutex);
312
313 if(_dirty) {
314 std::vector<rpc::LightState> message;
315 for(auto it : _lights_changes) {
316 auto it_light = _lights.find(it.first);
317 if(it_light != _lights.end()) {
318 rpc::LightState state(
319 it_light->second.GetLocation(),
320 it.second._intensity,
321 it.second._group,
322 rpc::Color(it.second._color.r, it.second._color.g, it.second._color.b),
323 it.second._active
324 );
325 state._id = it.first;
326 // Add to command
327 message.push_back(state);
328 }
329 }
330 _episode.Lock()->UpdateServerLightsState(message, discard_client);
331
332 _lights_changes.clear();
333 _dirty = false;
334 }
335}

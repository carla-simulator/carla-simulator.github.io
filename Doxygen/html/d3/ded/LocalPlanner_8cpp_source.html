28/// Compute the topological connection between @a current and @a next_wp,
29/// matching `_compute_connection` in local_planner.py.
30RoadOption ComputeConnection(
31 SharedPtr<client::Waypoint> current,
32 SharedPtr<client::Waypoint> next_wp,
33 float threshold = 35.0f) {
34 // Python uses `% 360` and `% 180` on yaws to wrap them. Replicate with
35 // `std::fmod` and a positive-fixup, since C++ `%` doesn't accept floats.
36 auto wrap = [](float v, float mod) {
37 float r = std::fmod(v, mod);
38 if (r < 0.0f) r += mod;
39 return r;
40 };
41
42 const float n = wrap(next_wp->GetTransform().rotation.yaw, 360.0f);
43 const float c = wrap(current->GetTransform().rotation.yaw, 360.0f);
44 const float diff = wrap(n - c, 180.0f);
45
46 if (diff < threshold || diff > (180.0f - threshold)) {
47 return RoadOption::Straight;
48 }
49 if (diff > 90.0f) {
50 return RoadOption::Left;
51 }
52 return RoadOption::Right;
53}

55/// Mirror of `_retrieve_options` in local_planner.py.
56std::vector<RoadOption> RetrieveOptions(
57 const std::vector<SharedPtr<client::Waypoint>> &candidates,
58 SharedPtr<client::Waypoint> current) {
59 std::vector<RoadOption> options;
60 options.reserve(candidates.size());
61 for (const auto &next_wp : candidates) {
62 // The Python code probes 3 m further to disambiguate intersections that
63 // share an entry waypoint.
64 auto next_next = next_wp->GetNext(3.0);
65 if (next_next.empty()) {
66 options.push_back(ComputeConnection(current, next_wp));
67 } else {
68 options.push_back(ComputeConnection(current, next_next.front()));
69 }
70 }
71 return options;
72}

141void LocalPlanner::ComputeNextWaypoints(std::size_t k) {
142 // Bound k by remaining queue capacity, like Python's
143 // `available_entries = self._waypoints_queue.maxlen - len(...)`.
144 const std::size_t available = (_waypoints_queue.size() < _max_queue_size)
145 ? (_max_queue_size - _waypoints_queue.size())
146 : 0;
147 k = std::min(k, available);
148
149 for (std::size_t i = 0; i < k; ++i) {
150 if (_waypoints_queue.empty()) break;
151 auto last_waypoint = _waypoints_queue.back().first;
152 if (!last_waypoint) break;
153 auto next_waypoints = last_waypoint->GetNext(_sampling_radius);
154
155 if (next_waypoints.empty()) {
156 break;
157 }
158
159 SharedPtr<client::Waypoint> next_waypoint;
160 RoadOption road_option;
161 if (next_waypoints.size() == 1) {
162 next_waypoint = next_waypoints.front();
163 road_option = RoadOption::LaneFollow;
164 } else {
165 // Random choice among the available branches.
166 auto road_options_list = RetrieveOptions(next_waypoints, last_waypoint);
167 std::uniform_int_distribution<std::size_t> dist(
168 0, road_options_list.size() - 1);
169 const std::size_t idx = dist(_rng);
170 road_option = road_options_list[idx];
171 // Python pulls `next_waypoints[road_options_list.index(road_option)]`,
172 // i.e. the *first* candidate that shares the chosen RoadOption. With a
173 // pre-randomised idx that's just the same as picking idx directly,
174 // because there's a 1:1 alignment between the lists (the first option
175 // matching the random pick is the random pick itself).
176 next_waypoint = next_waypoints[idx];
177 }
178
179 _waypoints_queue.emplace_back(next_waypoint, road_option);
180 }
181}

183void LocalPlanner::SetGlobalPlan(
184 const std::vector<PlanItem> &plan,
185 bool stop_waypoint_creation,
186 bool clean_queue) {
187 if (clean_queue) {
188 _waypoints_queue.clear();
189 }
190
191 // The Python implementation grows the deque's `maxlen` whenever the new
192 // plan exceeds it; we keep a soft cap and just raise it as needed.
193 const std::size_t new_plan_length =
194 plan.size() + _waypoints_queue.size();
195 if (new_plan_length > _max_queue_size) {
196 _max_queue_size = new_plan_length;
197 }
198
199 for (const auto &elem : plan) {
200 _waypoints_queue.push_back(elem);
201 }
202 _stop_waypoint_creation = stop_waypoint_creation;
203}

212rpc::VehicleControl LocalPlanner::RunStep(bool debug) {
213 if (_follow_speed_limits && _vehicle) {
214 _target_speed = _vehicle->GetSpeedLimit();
215 }
216
217 if (!_stop_waypoint_creation
218 && _waypoints_queue.size() < _min_queue_length) {
219 ComputeNextWaypoints(_min_queue_length);
220 }
221
222 // Purge waypoints we've already passed.
223 if (_vehicle) {
224 const auto veh_location = _vehicle->GetLocation();
225 const float vehicle_speed_ms = GetSpeed(_vehicle) * kKmhToMs;
226 const float min_distance =
227 _base_min_distance + _distance_ratio * vehicle_speed_ms;
228
229 std::size_t num_removed = 0;
230 for (const auto &item : _waypoints_queue) {
231 // Don't remove the very last waypoint until we're almost on top of it
232 // (mirrors Python's `min_distance = 1` in the same situation).
233 const float threshold =
234 (_waypoints_queue.size() - num_removed == 1) ? 1.0f : min_distance;
235
236 const auto &wp_loc = item.first->GetTransform().location;
237 if (veh_location.Distance(wp_loc) < threshold) {
238 ++num_removed;
239 } else {
240 break;
241 }
242 }
243 for (std::size_t i = 0; i < num_removed; ++i) {
244 _waypoints_queue.pop_front();
245 }
246 }
247
248 rpc::VehicleControl control;
249 if (_waypoints_queue.empty()) {
250 control.steer = 0.0f;
251 control.throttle = 0.0f;
252 control.brake = 1.0f;
253 control.hand_brake = false;
254 control.manual_gear_shift = false;
255 } else {
256 _target_waypoint = _waypoints_queue.front().first;
257 _target_road_option = _waypoints_queue.front().second;
258 control = _vehicle_controller->RunStep(_target_speed, _target_waypoint);
259 }
260
261 if (debug && _vehicle && _target_waypoint) {
262 auto world = std::make_shared<client::World>(_vehicle->GetWorld());
263 DrawWaypoints(world, {_target_waypoint}, 1.0f);
264 }
265
266 return control;
267}

269std::pair<SharedPtr<client::Waypoint>, RoadOption>
270LocalPlanner::GetIncomingWaypointAndDirection(int steps) const {
271 if (steps >= 0
272 && static_cast<std::size_t>(steps) < _waypoints_queue.size()) {
273 return _waypoints_queue[static_cast<std::size_t>(steps)];
274 }
275 if (!_waypoints_queue.empty()) {
276 return _waypoints_queue.back();
277 }
278 return {nullptr, RoadOption::Void};

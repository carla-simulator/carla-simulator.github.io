32 class Ellipsoid {
33 public:
34
35 Ellipsoid() = default;
36
37 Ellipsoid(double a, double f_inv):
38 a(a), f_inv(f_inv) {}
39
40 double a = 6378137.0;
41 double f_inv = std::numeric_limits<double>::infinity(); // Default to sphere
42
43 double f() const {
44 return 1.0 / f_inv;
45 }
46 double b() const {
47 return a * (1.0 - f());
48 }
49
50 double e2() const {
51 return f() * (2.0 - f());
52 }
53 double ep2() const{
54 return e2() / (1.0 - e2());
55 }
56
57 void fromb(double b) {
58 f_inv = 1.0 / (1.0 - b/a);
59 }
60 void fromf(double f) {
61 f_inv = 1.0 / f;
62 }
63
64 bool operator==(const Ellipsoid &rhs) const {
65 return (a == rhs.a) && (f_inv == rhs.f_inv);
66 }
67
68 bool operator!=(const Ellipsoid &rhs) const {
69 return (a != rhs.a) || (f_inv != rhs.f_inv);
70 }
71 };

73 class OffsetTransform {
74 public:
75
76 OffsetTransform() = default;
77
78 OffsetTransform(double offset_x, double offset_y, double offset_z, double offset_hdg):
79 offset_x(offset_x), offset_y(offset_y), offset_z(offset_z), offset_cos_h(std::cos(-offset_hdg)), offset_sin_h(std::sin(-offset_hdg)) {}
80
81 double offset_x = 0.0;
82 double offset_y = 0.0;
83 double offset_z = 0.0;
84 double offset_cos_h = 1.0;
85 double offset_sin_h = 0.0;
86
87 Location ApplyTransformation(const Location& location) const{
88
89 double tx = location.x + offset_x;
90 double ty = -location.y + offset_y;
91
92 double x_rot = (tx * offset_cos_h) - (ty * offset_sin_h);
93 double y_rot = (tx * offset_sin_h) + (ty * offset_cos_h);
94
95 return Location{static_cast<float>(x_rot), static_cast<float>(y_rot), static_cast<float>(location.z + offset_z)};
96 }
97
98 /// Inverse of @ref ApplyTransformation: recovers the pre-offset location
99 /// so that `ApplyTransformation(ApplyInverseTransformation(l)) == l`.
100 Location ApplyInverseTransformation(const Location& location) const{
101
102 double tx = (location.x * offset_cos_h) + (location.y * offset_sin_h);
103 double ty = -(location.x * offset_sin_h) + (location.y * offset_cos_h);
104
105 double x = tx - offset_x;
106 double y = offset_y - ty;
107
108 return Location{static_cast<float>(x), static_cast<float>(y), static_cast<float>(location.z - offset_z)};
109 }
110
111 bool operator==(const OffsetTransform& rhs) const {
112 return (offset_x == rhs.offset_x) && (offset_y == rhs.offset_y) && (offset_z == rhs.offset_z) && (offset_cos_h == rhs.offset_cos_h) && (offset_sin_h == rhs.offset_sin_h);
113 }
114
115 bool operator!=(const OffsetTransform& rhs) const {
116 return !(*this == rhs);
117 }
118 };

27void UTrafficLightComponent::InitializeSign(const carla::road::Map &Map)
28{
29 const double epsilon = 0.00001;
30
31 auto References = GetAllReferencesToThisSignal(Map);
32
33 for (auto& Reference : References)
34 {
35 auto RoadId = Reference.first;
36 const auto* SignalReference = Reference.second;
37 for(auto &validity : SignalReference->GetValidities())
38 {
39 for(auto lane : carla::geom::Math::GenerateRange(validity._from_lane, validity._to_lane))
40 {
41 if(lane == 0)
42 continue;
43
44 carla::road::element::Waypoint signal_waypoint;
45 std::optional<carla::road::element::Waypoint> opt_signal_waypoint = Map.GetWaypoint(
46 RoadId, lane, SignalReference->GetS());
47 if(opt_signal_waypoint){
48 signal_waypoint = opt_signal_waypoint.value();
49 }else{
50 continue;
51 }
52
53 if (Map.IsJunction(RoadId)) {
54 auto predecessors = Map.GetPredecessors(signal_waypoint);
55 if (predecessors.size() == 1) {
56 auto predecessor = predecessors.front();
57 if (!Map.IsJunction(predecessor.road_id)) {
58 signal_waypoint = predecessor;
59 }
60 }
61 }
62
63 if(Map.GetLane(signal_waypoint).GetType() != cr::Lane::LaneType::Driving)
64 continue;
65
66 float BoxWidth = static_cast<float>(0.5f*Map.GetLaneWidth(signal_waypoint)*0.5);
67 float BoxLength = 1.5f;
68 float BoxHeight = 1.0f;
69
70 BoxWidth = std::max(0.01f, BoxWidth);
71 double LaneLength = Map.GetLane(signal_waypoint).GetLength();
72 double LaneDistance = Map.GetLane(signal_waypoint).GetDistance();
73 float AdditionalDistance = 1.5f;
74 if(lane < 0)
75 {
76 signal_waypoint.s = FMath::Clamp(signal_waypoint.s - (BoxLength + AdditionalDistance),
77 LaneDistance + epsilon, LaneDistance + LaneLength - epsilon);
78 }
79 else
80 {
81 signal_waypoint.s = FMath::Clamp(signal_waypoint.s + (BoxLength + AdditionalDistance),
82 LaneDistance + epsilon, LaneDistance + LaneLength - epsilon);
83 }
84 FTransform BoxTransform = Map.ComputeTransform(signal_waypoint);
85 ALargeMapManager* LargeMapManager = UCarlaStatics::GetLargeMapManager(GetWorld());
86 if (LargeMapManager)
87 {
88 BoxTransform = LargeMapManager->GlobalToLocalTransform(BoxTransform);
89 }
90 GenerateTrafficLightBox(BoxTransform, FVector(100*BoxLength, 100*BoxWidth, 100*BoxHeight));
91 }
92 }
93 }
94}

105void UTrafficLightComponent::SetLightState(ETrafficLightState NewState)
106{
107 LightState = NewState;
108 LightChangeDispatcher.Broadcast();
109 if (GetOwner() == nullptr)
110 {
111 return;
112 }
113 if (GetOwner()->Implements<UTrafficLightInterface>())
114 {
115 ITrafficLightInterface::Execute_LightChanged(GetOwner(), LightState);
116 }
117 ATrafficLightBase* OldTrafficLight = Cast<ATrafficLightBase>(GetOwner());
118 if (OldTrafficLight)
119 {
120 OldTrafficLight->LightChangedCompatibility(NewState);
121 }
122
123 for (auto Controller : Vehicles)
124 {
125 if (Controller != nullptr)
126 {
127 Controller->SetTrafficLightState(LightState);
128 }
129 }
130}

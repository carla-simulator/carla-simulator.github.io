18 using Tags = std::map<std::string, std::string>;
19
20 inline const std::string *FindTag(const Tags &tags, const std::string &key) {
21 const auto it = tags.find(key);
22 return (it == tags.end()) ? nullptr : &it->second;
23 }
24
25 inline bool HasTag(const Tags &tags, const std::string &key, const std::string &value) {
26 const std::string *found = FindTag(tags, key);
27 return (found != nullptr) && (*found == value);
28 }
